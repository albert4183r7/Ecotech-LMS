     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import {
     5→  ZoomIn,
     6→  ZoomOut,
     7→  Maximize2,
     8→  RotateCcw,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  X,
    12→  GraduationCap,
    13→  Loader2,
    14→  StickyNote,
    15→  Bookmark,
    16→  BookmarkCheck,
    17→  Trash2,
    18→  Send,
    19→} from "lucide-react";
    20→import { toast } from "sonner";
    21→import { Button } from "@/components/ui/button";
    22→import { Separator } from "@/components/ui/separator";
    23→import { Tooltip, TooltipTrigger, TooltipContent } from "@/components/ui/tooltip";
    24→import {
    25→  Sheet,
    26→  SheetContent,
    27→  SheetHeader,
    28→  SheetTitle,
    29→  SheetDescription,
    30→} from "@/components/ui/sheet";
    31→import { ScrollArea } from "@/components/ui/scroll-area";
    32→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    33→import type { SlideContent, ClassroomState } from "@/types/lms";
    34→
    35→const MIN_ZOOM = 50;
    36→const MAX_ZOOM = 200;
    37→const ZOOM_STEP = 25;
    38→
    39→export function ClassroomPage() {
    40→  const { classroomState, goBack } = useNavigationStore();
    41→  const userId = useUserStore((s) => s.currentUserId);
    42→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    43→  const [zoom, setZoom] = useState(100);
    44→  const [loadingSlides, setLoadingSlides] = useState(false);
    45→  const [slideDirection, setSlideDirection] = useState<"forward" | "back">("forward");
    46→  const [isAnimating, setIsAnimating] = useState(false);
    47→  const hasFetchedRef = useRef(false);
    48→  const confettiShownRef = useRef(false);
    49→  const [showConfetti, setShowConfetti] = useState(false);
    50→
    51→  // ─── Notes Sidebar State ──────────────────────
    52→  interface Note {
    53→    id: string;
    54→    content: string;
    55→    slideNumber: number;
    56→    bookmarked: boolean;
    57→    createdAt?: string;
    58→  }
    59→  const [notes, setNotes] = useState<Note[]>([]);
    60→  const [notesSidebarOpen, setNotesSidebarOpen] = useState(false);
    61→  const [newNoteContent, setNewNoteContent] = useState("");
    62→  const [savingNote, setSavingNote] = useState(false);
    63→  const notesEndRef = useRef<HTMLDivElement>(null);
    64→
    65→  // ─── Keyboard Hint Fade ────────────────────────
    66→  const [showKeyboardHint, setShowKeyboardHint] = useState(true);
    67→  const [isMobile, setIsMobile] = useState(false);
    68→
    69→  useEffect(() => {
    70→    const mq = window.matchMedia("(max-width: 1023px)");
    71→    setIsMobile(mq.matches);
    72→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
    73→    mq.addEventListener("change", handler);
    74→    return () => mq.removeEventListener("change", handler);
    75→  }, []);
    76→
    77→  // Keep a working copy of classroomState so we can mutate currentSlide
    78→  useEffect(() => {
    79→    if (classroomState) {
    80→      setLocalState({ ...classroomState });
    81→      setZoom(100);
    82→      hasFetchedRef.current = false;
    83→    }
    84→  }, [classroomState]);
    85→
    86→  // If slides were empty initially, try fetching them
    87→  useEffect(() => {
    88→    if (!localState || localState.slides.length > 0 || hasFetchedRef.current) return;
    89→    hasFetchedRef.current = true;
    90→    const fetchSlides = async () => {
    91→      setLoadingSlides(true);
    92→      try {
    93→        const res = await fetch(`/api/sections/${localState.sectionId}`);
    94→        if (!res.ok) return;
    95→        const json = await res.json();
    96→        if (json.success && json.data.content) {
    97→          const slides: SlideContent[] = Array.isArray(json.data.content)
    98→            ? json.data.content
    99→            : [];
   100→          setLocalState((prev) =>
   101→            prev
   102→              ? {
   103→                  ...prev,
   104→                  slides,
   105→                  totalPages: slides.length || prev.totalPages,
   106→                }
   107→              : prev
   108→          );
   109→        }
   110→      } catch {
   111→        // Silently fail
   112→      } finally {
   113→        setLoadingSlides(false);
   114→      }
   115→    };
   116→    fetchSlides();
   117→  }, [localState?.sectionId, localState?.slides.length]);
   118→
   119→  /** Navigate slides with transition animation */
   120→  const goToSlide = useCallback(
   121→    (index: number) => {
   122→      if (!localState || isAnimating) return;
   123→      const next = Math.max(0, Math.min(index, localState.totalPages - 1));
   124→      if (next === localState.currentSlide) return;
   125→      setSlideDirection(next > localState.currentSlide ? "forward" : "back");
   126→      setIsAnimating(true);
   127→      // Small delay for exit animation, then update slide
   128→      setTimeout(() => {
   129→        setLocalState((prev) =>
   130→          prev ? { ...prev, currentSlide: next } : prev
   131→        );
   132→        setIsAnimating(false);
   133→      }, 150);
   134→    },
   135→    [localState, isAnimating]
   136→  );
   137→
   138→  /** Persist progress to API when slide changes */
   139→  useEffect(() => {
   140→    if (!localState || !userId) return;
   141→    const saveProgress = async () => {
   142→      try {
   143→        // Get enrollment ID from the API
   144→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   145→        const enrollJson = await enrollRes.json();
   146→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   147→          const enrollment = enrollJson.data.find(
   148→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   149→          );
   150→          if (enrollment) {
   151→            const completed = localState.currentSlide >= localState.totalPages - 1;
   152→            await fetch("/api/progress", {
   153→              method: "POST",
   154→              headers: { "Content-Type": "application/json" },
   155→              body: JSON.stringify({
   156→                enrollmentId: enrollment.id,
   157→                sectionId: localState.sectionId,
   158→                currentPage: localState.currentSlide + 1,
   159→                completed,
   160→              }),
   161→            });
   162→          }
   163→        }
   164→      } catch {
   165→        // Progress saving is best-effort
   166→      }
   167→    };
   168→    // Debounce to avoid excessive API calls
   169→    const timer = setTimeout(saveProgress, 500);
   170→    return () => clearTimeout(timer);
   171→  }, [localState?.currentSlide]);
   172→
   173→  const goPrev = () => goToSlide((localState?.currentSlide ?? 0) - 1);
   174→  const goNext = () => goToSlide((localState?.currentSlide ?? 0) + 1);
   175→
   176→  // ─── Keyboard shortcuts ──────────────────────────────────────
   177→  useEffect(() => {
   178→    function handleKeyDown(e: KeyboardEvent) {
   179→      const tag = (e.target as HTMLElement)?.tagName;
   180→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   181→
   182→      switch (e.key) {
   183→        case "Escape":
   184→          e.preventDefault();
   185→          goBack();
   186→          break;
   187→        case "ArrowLeft":
   188→          e.preventDefault();
   189→          goPrev();
   190→          break;
   191→        case "ArrowRight":
   192→          e.preventDefault();
   193→          goNext();
   194→          break;
   195→        case " ":
   196→          e.preventDefault();
   197→          goNext();
   198→          break;
   199→      }
   200→    }
   201→
   202→    document.addEventListener("keydown", handleKeyDown);
   203→
   204→    // Listen for custom event from KeyboardShortcuts component
   205→    function handleCustomNextSlide() {
   206→      goNext();
   207→    }
   208→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   209→
   210→    return () => {
   211→      document.removeEventListener("keydown", handleKeyDown);
   212→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   213→    };
   214→  }, [goBack, goPrev, goNext]);
   215→
   216→  /** Zoom controls */
   217→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   218→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   219→  const zoomFit = () => setZoom(100);
   220→
   221→  // ─── Confetti on section completion ───────────
   222→  useEffect(() => {
   223→    if (!localState) return;
   224→    const isOnLastSlide = localState.currentSlide === localState.totalPages - 1;
   225→    if (isOnLastSlide && !confettiShownRef.current) {
   226→      confettiShownRef.current = true;
   227→      setShowConfetti(true);
   228→      toast.success("🎉 You completed the course! Great job!");
   229→      // Auto-cleanup after 3.5s (animation lasts ~3s + buffer)
   230→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   231→      return () => clearTimeout(timer);
   232→    }
   233→  }, [localState?.currentSlide]);
   234→
   235→  // Reset confetti flag when classroom state changes (new section)
   236→  useEffect(() => {
   237→    confettiShownRef.current = false;
   238→    setShowConfetti(false);
   239→  }, [classroomState?.sectionId]);
   240→
   241→  // ─── Keyboard Hint Auto-Fade ────────────────────
   242→  useEffect(() => {
   243→    setShowKeyboardHint(true);
   244→    const timer = setTimeout(() => setShowKeyboardHint(false), 5000);
   245→    return () => clearTimeout(timer);
   246→  }, [localState?.sectionId]);
   247→
   248→  // ─── Notes CRUD ─────────────────────────────────
   249→  const fetchNotes = useCallback(async () => {
   250→    if (!localState || !userId) return;
   251→    try {
   252→      const res = await fetch(
   253→        `/api/notes?userId=${userId}&courseId=${localState.courseId}&sectionId=${localState.sectionId}`
   254→      );
   255→      if (!res.ok) return;
   256→      const json = await res.json();
   257→      if (json.success && Array.isArray(json.data)) {
   258→        setNotes(json.data);
   259→      }
   260→    } catch {
   261→      // Silently fail
   262→    }
   263→  }, [localState, userId]);
   264→
   265→  // Fetch notes when section changes or sidebar opens
   266→  useEffect(() => {
   267→    if (notesSidebarOpen) fetchNotes();
   268→  }, [notesSidebarOpen, fetchNotes, localState?.sectionId]);
   269→
   270→  const createNote = useCallback(async () => {
   271→    if (!localState || !userId || !newNoteContent.trim()) return;
   272→    setSavingNote(true);
   273→    try {
   274→      const res = await fetch("/api/notes", {
   275→        method: "POST",
   276→        headers: { "Content-Type": "application/json" },
   277→        body: JSON.stringify({
   278→          userId,
   279→          courseId: localState.courseId,
   280→          sectionId: localState.sectionId,
   281→          content: newNoteContent.trim(),
   282→          slideNumber: localState.currentSlide + 1,
   283→        }),
   284→      });
   285→      if (!res.ok) return;
   286→      const json = await res.json();
   287→      if (json.success && json.data) {
   288→        setNotes((prev) => [...prev, json.data]);
   289→        setNewNoteContent("");
   290→        setTimeout(() => notesEndRef.current?.scrollIntoView({ behavior: "smooth" }), 50);
   291→      }
   292→    } catch {
   293→      toast.error("Failed to save note");
   294→    } finally {
   295→      setSavingNote(false);
   296→    }
   297→  }, [localState, userId, newNoteContent]);
   298→
   299→  const deleteNote = useCallback(async (noteId: string) => {
   300→    try {
   301→      const res = await fetch(`/api/notes?id=${noteId}`, { method: "DELETE" });
   302→      if (!res.ok) return;
   303→      const json = await res.json();
   304→      if (json.success) {
   305→        setNotes((prev) => prev.filter((n) => n.id !== noteId));
   306→      }
   307→    } catch {
   308→      // Silently fail
   309→    }
   310→  }, []);
   311→
   312→  const toggleBookmark = useCallback((noteId: string) => {
   313→    setNotes((prev) =>
   314→      prev.map((n) =>
   315→        n.id === noteId ? { ...n, bookmarked: !n.bookmarked } : n
   316→      )
   317→    );
   318→  }, []);
   319→
   320→  // ─── No state guard ─────────────────────────────
   321→  if (!localState) {
   322→    return (
   323→      <div className="flex h-screen items-center justify-center bg-muted/30">
   324→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   325→      </div>
   326→    );
   327→  }
   328→
   329→  const currentSlide =
   330→    localState.slides[localState.currentSlide] ?? null;
   331→  const isFirst = localState.currentSlide === 0;
   332→  const isLast = localState.currentSlide >= localState.totalPages - 1;
   333→  const progressPercent = localState.totalPages > 1
   334→    ? Math.round(((localState.currentSlide + 1) / localState.totalPages) * 100)
   335→    : 100;
   336→
   337→  return (
   338→    <div className="flex h-screen flex-col bg-muted/30 relative">
   339→      {/* ─── Mobile Notes Sheet ──────────────────── */}
   340→      <Sheet open={notesSidebarOpen && isMobile} onOpenChange={(open) => { if (!open) setNotesSidebarOpen(false); }}>
   341→        <SheetContent side="right" className="w-full sm:max-w-sm p-0">
   342→          <SheetHeader className="px-4 pt-4 pb-0">
   343→            <SheetTitle className="flex items-center gap-2 text-base">
   344→              <StickyNote className="h-4 w-4" />
   345→              Notes
   346→            </SheetTitle>
   347→            <SheetDescription>
   348→              Notes for slide {localState.currentSlide + 1} of {localState.totalPages}
   349→            </SheetDescription>
   350→          </SheetHeader>
   351→          <NotesSidebarContent
   352→            notes={notes}
   353→            currentSlide={localState.currentSlide}
   354→            newNoteContent={newNoteContent}
   355→            savingNote={savingNote}
   356→            onContentChange={setNewNoteContent}
   357→            onCreate={createNote}
   358→            onDelete={deleteNote}
   359→            onToggleBookmark={toggleBookmark}
   360→            endRef={notesEndRef}
   361→          />
   362→        </SheetContent>
   363→      </Sheet>
   364→      {/* ─── Confetti Celebration Overlay ─────── */}
   365→      {showConfetti && <ConfettiCelebration />}
   366→      {/* ─── Top Progress Bar ─────────────────────── */}
   367→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   368→        <div
   369→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   370→          style={{ width: `${progressPercent}%` }}
   371→        />
   372→      </div>
   373→
   374→      {/* ─── Top Bar ──────────────────────────────── */}
   375→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   376→        {/* Logo, Section Title, Course Title */}
   377→        <div className="flex items-center gap-3 min-w-0">
   378→          <div className="flex items-center gap-2 shrink-0">
   379→            <div className="flex h-7 w-7 items-center justify-center rounded-md bg-primary">
   380→              <GraduationCap className="h-4 w-4 text-primary-foreground" />
   381→            </div>
   382→            <span className="hidden sm:inline text-sm font-bold text-primary">
   383→              OpenClass
   384→            </span>
   385→          </div>
   386→          <Separator orientation="vertical" className="h-5" />
   387→          <div className="min-w-0">
   388→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   389→              {localState.sectionTitle}
   390→            </h2>
   391→            <p className="text-xs text-muted-foreground truncate leading-tight">
   392→              {localState.courseTitle}
   393→            </p>
   394→          </div>
   395→        </div>
   396→
   397→        {/* Slide Counter - Enhanced */}
   398→        <span className="shrink-0 inline-flex items-center gap-1.5 rounded-full bg-primary/10 px-3 py-1 text-sm text-primary tabular-nums">
   399→          <span className="font-semibold">Slide {localState.currentSlide + 1}</span>
   400→          <span className="text-primary/40 font-normal">of</span>
   401→          <span className="font-semibold">{localState.totalPages}</span>
   402→        </span>
   403→      </header>
   404→
   405→      {/* ─── Main Layout: Slide + Desktop Notes Sidebar ── */}
   406→      <div className="flex flex-1 overflow-hidden">
   407→      {/* ─── Slide Content Area ────────────────────── */}
   408→      <div className="flex-1 overflow-auto flex flex-col items-center py-8 px-4 sm:px-6">
   409→        <div
   410→          className={`
   411→            bg-card rounded-2xl shadow-lg border w-full max-w-3xl
   412→            ring-1 ring-black/5
   413→            dark:ring-white/5
   414→            paper-texture
   415→            ${isAnimating
   416→              ? (slideDirection === "forward" ? "slide-exit" : "opacity-0 -translate-x-2 transition-all duration-150")
   417→              : "slide-enter"
   418→            }
   419→          `}
   420→          style={{
   421→            transform: `scale(${zoom / 100})`,
   422→            transformOrigin: "top center",
   423→          }}
   424→        >
   425→          {loadingSlides ? (
   426→            <div className="flex items-center justify-center py-32">
   427→              <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   428→            </div>
   429→          ) : currentSlide ? (
   430→            <div className="p-8 sm:p-12 max-w-2xl mx-auto">
   431→              <SlideRenderer slide={currentSlide} />
   432→            </div>
   433→          ) : (
   434→            <div className="flex items-center justify-center py-32 text-muted-foreground">
   435→              <p>No slide content available.</p>
   436→            </div>
   437→          )}
   438→        </div>
   439→
   440→        {/* ─── Slide Progress Dots ────────────────── */}
   441→        <div className="slide-progress-indicator mt-6 shrink-0">
   442→          {Array.from({ length: localState.totalPages }, (_, i) => {
   443→            const isCompleted = i < localState.currentSlide;
   444→            const isActive = i === localState.currentSlide;
   445→            return (
   446→              <button
   447→                key={i}
   448→                className={`dot ${isActive ? "active" : ""} ${isCompleted ? "completed" : ""}`}
   449→                onClick={() => goToSlide(i)}
   450→                aria-label={`Go to slide ${i + 1}`}
   451→              />
   452→            );
   453→          })}
   454→        </div>
   455→      </div>
   456→
   457→      {/* ─── Desktop Notes Sidebar (slide-in panel) ── */}
   458→      <aside
   459→        className={`hidden lg:flex flex-col shrink-0 border-l bg-card transition-all duration-300 ease-in-out overflow-hidden ${
   460→          notesSidebarOpen ? "w-80 opacity-100" : "w-0 opacity-0 border-l-0"
   461→        }`}
   462→      >
   463→        <div className="flex flex-col h-full w-80">
   464→          {/* Sidebar Header */}
   465→          <div className="flex items-center justify-between px-4 py-3 border-b">
   466→            <div className="flex items-center gap-2">
   467→              <StickyNote className="h-4 w-4 text-primary" />
   468→              <span className="text-sm font-semibold">Notes</span>
   469→            </div>
   470→            <Button
   471→              variant="ghost"
   472→              size="icon"
   473→              className="h-7 w-7"
   474→              onClick={() => setNotesSidebarOpen(false)}
   475→            >
   476→              <X className="h-4 w-4" />
   477→            </Button>
   478→          </div>
   479→          <NotesSidebarContent
   480→            notes={notes}
   481→            currentSlide={localState.currentSlide}
   482→            newNoteContent={newNoteContent}
   483→            savingNote={savingNote}
   484→            onContentChange={setNewNoteContent}
   485→            onCreate={createNote}
   486→            onDelete={deleteNote}
   487→            onToggleBookmark={toggleBookmark}
   488→            endRef={notesEndRef}
   489→          />
   490→        </div>
   491→      </aside>
   492→      </div>
   493→
   494→      {/* ─── Keyboard Shortcuts Hint (auto-fades) ── */}
   495→      <div
   496→        className={`shrink-0 frosted-glass border-t px-4 py-1.5 text-center transition-opacity duration-700 ${
   497→          showKeyboardHint ? "opacity-100" : "opacity-0 pointer-events-none h-0 py-0 overflow-hidden"
   498→        }`}
   499→      >
   500→        <p className="text-xs text-muted-foreground/70 tracking-wide">
   501→          ← → Navigate&nbsp;&nbsp;|&nbsp;&nbsp;Space: Next&nbsp;&nbsp;|&nbsp;&nbsp;Esc: Exit
   502→        </p>
   503→      </div>
   504→
   505→      {/* ─── Bottom Controls (Frosted Glass) ───────── */}
   506→      <footer className="shrink-0 border-t frosted-glass">
   507→        <div className="mx-auto flex h-16 max-w-5xl items-center justify-between px-4 sm:px-6">
   508→          {/* Zoom Controls */}
   509→          <div className="flex items-center gap-1">
   510→            <Tooltip>
   511→              <TooltipTrigger asChild>
   512→                <Button
   513→                  variant="ghost"
   514→                  size="icon"
   515→                  className="h-8 w-8"
   516→                  onClick={zoomOut}
   517→                  disabled={zoom <= MIN_ZOOM}
   518→                >
   519→                  <ZoomOut className="h-4 w-4" />
   520→                </Button>
   521→              </TooltipTrigger>
   522→              <TooltipContent>Zoom Out</TooltipContent>
   523→            </Tooltip>
   524→
   525→            <span className="w-12 text-center text-xs font-medium text-muted-foreground tabular-nums">
   526→              {zoom}%
   527→            </span>
   528→
   529→            <Tooltip>
   530→              <TooltipTrigger asChild>
   531→                <Button
   532→                  variant="ghost"
   533→                  size="icon"
   534→                  className="h-8 w-8"
   535→                  onClick={zoomIn}
   536→                  disabled={zoom >= MAX_ZOOM}
   537→                >
   538→                  <ZoomIn className="h-4 w-4" />
   539→                </Button>
   540→              </TooltipTrigger>
   541→              <TooltipContent>Zoom In</TooltipContent>
   542→            </Tooltip>
   543→
   544→            <Tooltip>
   545→              <TooltipTrigger asChild>
   546→                <Button
   547→                  variant="ghost"
   548→                  size="icon"
   549→                  className="h-8 w-8"
   550→                  onClick={zoomFit}
   551→                >
   552→                  <Maximize2 className="h-4 w-4" />
   553→                </Button>
   554→              </TooltipTrigger>
   555→              <TooltipContent>Fit Width</TooltipContent>
   556→            </Tooltip>
   557→
   558→            <Tooltip>
   559→              <TooltipTrigger asChild>
   560→                <Button
   561→                  variant="ghost"
   562→                  size="icon"
   563→                  className="h-8 w-8"
   564→                  onClick={zoomFit}
   565→                >
   566→                  <RotateCcw className="h-4 w-4" />
   567→                </Button>
   568→              </TooltipTrigger>

  1045→function QuizSlide({ slide }: { slide: SlideContent }) {

  1049→  // Determine correct answer: look for item with icon === "check" or first item

  1051→  const isCorrect = selectedIdx === correctIdx;

  1067→              else if (isSelected && !isCorrect) borderColor = "border-red-400 bg-red-400/10";
  1068→              else borderColor = "border-border opacity-60";
  1069→            } else if (isSelected) {
  1070→              borderColor = "border-primary bg-primary/10";

  1072→
  1073→            return (
  1074→              <button
  1075→                key={i}

  1077→                className={`w-full rounded-lg border p-4 text-left transition-all duration-200 ${borderColor}`}
  1078→                onClick={() => {
  1079→                  if (!hasSubmitted) setSelectedIdx(i);
  1080→                }}
  1081→                disabled={hasSubmitted}
  1082→              >
  1083→                <div className="flex items-start gap-3">
  1084→                  <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold ${

  1087→                      : hasSubmitted && isSelected && !isCorrect
  1088→                        ? "bg-red-400 text-white"
  1089→                        : "bg-primary/10 text-primary"
  1090→                  }`}>

  1094→                    <p className="text-sm font-medium text-foreground">

  1099→                    )}
  1100→                  </div>
  1101→                </div>
  1102→              </button>
  1103→            );
  1104→          })}
  1105→
  1106→          {/* Feedback after selection */}
  1107→          {hasSubmitted && (
  1108→            <div className={`mt-4 rounded-lg border p-4 ${
