     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import {
     5→  ZoomIn,
     6→  ZoomOut,
     7→  Maximize2,
     8→  RotateCcw,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  X,
    12→  GraduationCap,
    13→  Loader2,
    14→  StickyNote,
    15→  Bookmark,
    16→  BookmarkCheck,
    17→  Trash2,
    18→  Send,
    19→} from "lucide-react";
    20→import { toast } from "sonner";
    21→import { Button } from "@/components/ui/button";
    22→import { Separator } from "@/components/ui/separator";
    23→import { Tooltip, TooltipTrigger, TooltipContent } from "@/components/ui/tooltip";
    24→import {
    25→  Sheet,
    26→  SheetContent,
    27→  SheetHeader,
    28→  SheetTitle,
    29→  SheetDescription,
    30→} from "@/components/ui/sheet";
    31→import { ScrollArea } from "@/components/ui/scroll-area";
    32→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    33→import type { SlideContent, ClassroomState } from "@/types/lms";
    34→import { StudyTimer } from "@/components/lms/study-timer";
    35→
    36→const MIN_ZOOM = 50;
    37→const MAX_ZOOM = 200;
    38→const ZOOM_STEP = 25;
    39→
    40→export function ClassroomPage() {
    41→  const { classroomState, goBack } = useNavigationStore();
    42→  const userId = useUserStore((s) => s.currentUserId);
    43→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    44→  const [zoom, setZoom] = useState(100);
    45→  const [loadingSlides, setLoadingSlides] = useState(false);
    46→  const [slideDirection, setSlideDirection] = useState<"forward" | "back">("forward");
    47→  const [isAnimating, setIsAnimating] = useState(false);
    48→  const hasFetchedRef = useRef(false);
    49→  const confettiShownRef = useRef(false);
    50→  const [showConfetti, setShowConfetti] = useState(false);
    51→
    52→  // ─── Notes Sidebar State ──────────────────────
    53→  interface Note {
    54→    id: string;
    55→    content: string;
    56→    slideNumber: number;
    57→    bookmarked: boolean;
    58→    createdAt?: string;
    59→  }
    60→  const [notes, setNotes] = useState<Note[]>([]);
    61→  const [notesSidebarOpen, setNotesSidebarOpen] = useState(false);
    62→  const [newNoteContent, setNewNoteContent] = useState("");
    63→  const [savingNote, setSavingNote] = useState(false);
    64→  const notesEndRef = useRef<HTMLDivElement>(null);
    65→
    66→  // ─── Keyboard Hint Fade ────────────────────────
    67→  const [showKeyboardHint, setShowKeyboardHint] = useState(true);
    68→  const [isMobile, setIsMobile] = useState(false);
    69→
    70→  useEffect(() => {
    71→    const mq = window.matchMedia("(max-width: 1023px)");
    72→    setIsMobile(mq.matches);
    73→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
    74→    mq.addEventListener("change", handler);
    75→    return () => mq.removeEventListener("change", handler);
    76→  }, []);
    77→
    78→  // Keep a working copy of classroomState so we can mutate currentSlide
    79→  useEffect(() => {
    80→    if (classroomState) {
    81→      setLocalState({ ...classroomState });
    82→      setZoom(100);
    83→      hasFetchedRef.current = false;
    84→    }
    85→  }, [classroomState]);
    86→
    87→  // If slides were empty initially, try fetching them
    88→  useEffect(() => {
    89→    if (!localState || localState.slides.length > 0 || hasFetchedRef.current) return;
    90→    hasFetchedRef.current = true;
    91→    const fetchSlides = async () => {
    92→      setLoadingSlides(true);
    93→      try {
    94→        const res = await fetch(`/api/sections/${localState.sectionId}`);
    95→        if (!res.ok) return;
    96→        const json = await res.json();
    97→        if (json.success && json.data.content) {
    98→          const slides: SlideContent[] = Array.isArray(json.data.content)
    99→            ? json.data.content
   100→            : [];
   101→          setLocalState((prev) =>
   102→            prev
   103→              ? {
   104→                  ...prev,
   105→                  slides,
   106→                  totalPages: slides.length || prev.totalPages,
   107→                }
   108→              : prev
   109→          );
   110→        }
   111→      } catch {
   112→        // Silently fail
   113→      } finally {
   114→        setLoadingSlides(false);
   115→      }
   116→    };
   117→    fetchSlides();
   118→  }, [localState?.sectionId, localState?.slides.length]);
   119→
   120→  /** Navigate slides with transition animation */
   121→  const goToSlide = useCallback(
   122→    (index: number) => {
   123→      if (!localState || isAnimating) return;
   124→      const next = Math.max(0, Math.min(index, localState.totalPages - 1));
   125→      if (next === localState.currentSlide) return;
   126→      setSlideDirection(next > localState.currentSlide ? "forward" : "back");
   127→      setIsAnimating(true);
   128→      // Small delay for exit animation, then update slide
   129→      setTimeout(() => {
   130→        setLocalState((prev) =>
   131→          prev ? { ...prev, currentSlide: next } : prev
   132→        );
   133→        setIsAnimating(false);
   134→      }, 150);
   135→    },
   136→    [localState, isAnimating]
   137→  );
   138→
   139→  /** Persist progress to API when slide changes */
   140→  useEffect(() => {
   141→    if (!localState || !userId) return;
   142→    const saveProgress = async () => {
   143→      try {
   144→        // Get enrollment ID from the API
   145→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   146→        const enrollJson = await enrollRes.json();
   147→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   148→          const enrollment = enrollJson.data.find(
   149→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   150→          );
   151→          if (enrollment) {
   152→            const completed = localState.currentSlide >= localState.totalPages - 1;
   153→            await fetch("/api/progress", {
   154→              method: "POST",
   155→              headers: { "Content-Type": "application/json" },
   156→              body: JSON.stringify({
   157→                enrollmentId: enrollment.id,
   158→                sectionId: localState.sectionId,
   159→                currentPage: localState.currentSlide + 1,
   160→                completed,
   161→              }),
   162→            });
   163→          }
   164→        }
   165→      } catch {
   166→        // Progress saving is best-effort
   167→      }
   168→    };
   169→    // Debounce to avoid excessive API calls
   170→    const timer = setTimeout(saveProgress, 500);
   171→    return () => clearTimeout(timer);
   172→  }, [localState?.currentSlide]);
   173→
   174→  const goPrev = () => goToSlide((localState?.currentSlide ?? 0) - 1);
   175→  const goNext = () => goToSlide((localState?.currentSlide ?? 0) + 1);
   176→
   177→  // ─── Keyboard shortcuts ──────────────────────────────────────
   178→  useEffect(() => {
   179→    function handleKeyDown(e: KeyboardEvent) {
   180→      const tag = (e.target as HTMLElement)?.tagName;
   181→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   182→
   183→      switch (e.key) {
   184→        case "Escape":
   185→          e.preventDefault();
   186→          goBack();
   187→          break;
   188→        case "ArrowLeft":
   189→          e.preventDefault();
   190→          goPrev();
   191→          break;
   192→        case "ArrowRight":
   193→          e.preventDefault();
   194→          goNext();
   195→          break;
   196→        case " ":
   197→          e.preventDefault();
   198→          goNext();
   199→          break;
   200→      }
   201→    }
   202→
   203→    document.addEventListener("keydown", handleKeyDown);
   204→
   205→    // Listen for custom event from KeyboardShortcuts component
   206→    function handleCustomNextSlide() {
   207→      goNext();
   208→    }
   209→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   210→
   211→    return () => {
   212→      document.removeEventListener("keydown", handleKeyDown);
   213→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   214→    };
   215→  }, [goBack, goPrev, goNext]);
   216→
   217→  /** Zoom controls */
   218→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   219→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   220→  const zoomFit = () => setZoom(100);
   221→
   222→  // ─── Confetti on section completion ───────────
   223→  useEffect(() => {
   224→    if (!localState) return;
   225→    const isOnLastSlide = localState.currentSlide === localState.totalPages - 1;
   226→    if (isOnLastSlide && !confettiShownRef.current) {
   227→      confettiShownRef.current = true;
   228→      setShowConfetti(true);
   229→      toast.success("🎉 You completed the course! Great job!");
   230→      // Auto-cleanup after 3.5s (animation lasts ~3s + buffer)
   231→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   232→      return () => clearTimeout(timer);
   233→    }
   234→  }, [localState?.currentSlide]);
   235→
   236→  // Reset confetti flag when classroom state changes (new section)
   237→  useEffect(() => {
   238→    confettiShownRef.current = false;
   239→    setShowConfetti(false);
   240→  }, [classroomState?.sectionId]);
   241→
   242→  // ─── Keyboard Hint Auto-Fade ────────────────────
   243→  useEffect(() => {
   244→    setShowKeyboardHint(true);
   245→    const timer = setTimeout(() => setShowKeyboardHint(false), 5000);
   246→    return () => clearTimeout(timer);
   247→  }, [localState?.sectionId]);
   248→
   249→  // ─── Notes CRUD ─────────────────────────────────
   250→  const fetchNotes = useCallback(async () => {
   251→    if (!localState || !userId) return;
   252→    try {
   253→      const res = await fetch(
   254→        `/api/notes?userId=${userId}&courseId=${localState.courseId}&sectionId=${localState.sectionId}`
   255→      );
   256→      if (!res.ok) return;
   257→      const json = await res.json();
   258→      if (json.success && Array.isArray(json.data)) {
   259→        setNotes(json.data);
   260→      }
   261→    } catch {
   262→      // Silently fail
   263→    }
   264→  }, [localState, userId]);
   265→
   266→  // Fetch notes when section changes or sidebar opens
   267→  useEffect(() => {
   268→    if (notesSidebarOpen) fetchNotes();
   269→  }, [notesSidebarOpen, fetchNotes, localState?.sectionId]);
   270→
   271→  const createNote = useCallback(async () => {
   272→    if (!localState || !userId || !newNoteContent.trim()) return;
   273→    setSavingNote(true);
   274→    try {
   275→      const res = await fetch("/api/notes", {
   276→        method: "POST",
   277→        headers: { "Content-Type": "application/json" },
   278→        body: JSON.stringify({
   279→          userId,
   280→          courseId: localState.courseId,
   281→          sectionId: localState.sectionId,
   282→          content: newNoteContent.trim(),
   283→          slideNumber: localState.currentSlide + 1,
   284→        }),
   285→      });
   286→      if (!res.ok) return;
   287→      const json = await res.json();
   288→      if (json.success && json.data) {
   289→        setNotes((prev) => [...prev, json.data]);
   290→        setNewNoteContent("");
   291→        setTimeout(() => notesEndRef.current?.scrollIntoView({ behavior: "smooth" }), 50);
   292→      }
   293→    } catch {
   294→      toast.error("Failed to save note");
   295→    } finally {
   296→      setSavingNote(false);
   297→    }
   298→  }, [localState, userId, newNoteContent]);
   299→
   300→  const deleteNote = useCallback(async (noteId: string) => {
   301→    try {
   302→      const res = await fetch(`/api/notes?id=${noteId}`, { method: "DELETE" });
   303→      if (!res.ok) return;
   304→      const json = await res.json();
   305→      if (json.success) {
   306→        setNotes((prev) => prev.filter((n) => n.id !== noteId));
   307→      }
   308→    } catch {
   309→      // Silently fail
   310→    }
   311→  }, []);
   312→
   313→  const toggleBookmark = useCallback((noteId: string) => {
   314→    setNotes((prev) =>
   315→      prev.map((n) =>
   316→        n.id === noteId ? { ...n, bookmarked: !n.bookmarked } : n
   317→      )
   318→    );
   319→  }, []);
   320→
   321→  // ─── No state guard ─────────────────────────────
   322→  if (!localState) {
   323→    return (
   324→      <div className="flex h-screen items-center justify-center bg-muted/30">
   325→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   326→      </div>
   327→    );
   328→  }
   329→
   330→  const currentSlide =
   331→    localState.slides[localState.currentSlide] ?? null;
   332→  const isFirst = localState.currentSlide === 0;
   333→  const isLast = localState.currentSlide >= localState.totalPages - 1;
   334→  const progressPercent = localState.totalPages > 1
   335→    ? Math.round(((localState.currentSlide + 1) / localState.totalPages) * 100)
   336→    : 100;
   337→
   338→  return (
   339→    <div className="flex h-screen flex-col bg-muted/30 relative">
   340→      {/* ─── Mobile Notes Sheet ──────────────────── */}
   341→      <Sheet open={notesSidebarOpen && isMobile} onOpenChange={(open) => { if (!open) setNotesSidebarOpen(false); }}>
   342→        <SheetContent side="right" className="w-full sm:max-w-sm p-0">
   343→          <SheetHeader className="px-4 pt-4 pb-0">
   344→            <SheetTitle className="flex items-center gap-2 text-base">
   345→              <StickyNote className="h-4 w-4" />
   346→              Notes
   347→            </SheetTitle>
   348→            <SheetDescription>
   349→              Notes for slide {localState.currentSlide + 1} of {localState.totalPages}
   350→            </SheetDescription>
   351→          </SheetHeader>
   352→          <NotesSidebarContent
   353→            notes={notes}
   354→            currentSlide={localState.currentSlide}
   355→            newNoteContent={newNoteContent}
   356→            savingNote={savingNote}
   357→            onContentChange={setNewNoteContent}
   358→            onCreate={createNote}
   359→            onDelete={deleteNote}
   360→            onToggleBookmark={toggleBookmark}
   361→            endRef={notesEndRef}
   362→          />
   363→        </SheetContent>
   364→      </Sheet>
   365→      {/* ─── Confetti Celebration Overlay ─────── */}
   366→      {showConfetti && <ConfettiCelebration />}
   367→      {/* ─── Top Progress Bar ─────────────────────── */}
   368→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   369→        <div
   370→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   371→          style={{ width: `${progressPercent}%` }}
   372→        />
   373→      </div>
   374→
   375→      {/* ─── Top Bar ──────────────────────────────── */}
   376→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   377→        {/* Logo, Section Title, Course Title */}
   378→        <div className="flex items-center gap-3 min-w-0">
   379→          <div className="flex items-center gap-2 shrink-0">
   380→            <div className="flex h-7 w-7 items-center justify-center rounded-md bg-primary">
   381→              <GraduationCap className="h-4 w-4 text-primary-foreground" />
   382→            </div>
   383→            <span className="hidden sm:inline text-sm font-bold text-primary">
   384→              OpenClass
   385→            </span>
   386→          </div>
   387→          <Separator orientation="vertical" className="h-5" />
   388→          <div className="min-w-0">
   389→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   390→              {localState.sectionTitle}
   391→            </h2>
   392→            <p className="text-xs text-muted-foreground truncate leading-tight">
   393→              {localState.courseTitle}
   394→            </p>
   395→          </div>
   396→        </div>
   397→
   398→        {/* Slide Counter - Enhanced */}
   399→        <span className="shrink-0 inline-flex items-center gap-1.5 rounded-full bg-primary/10 px-3 py-1 text-sm text-primary tabular-nums">
   400→          <span className="font-semibold">Slide {localState.currentSlide + 1}</span>
   401→          <span className="text-primary/40 font-normal">of</span>
   402→          <span className="font-semibold">{localState.totalPages}</span>
   403→        </span>
   404→      </header>
   405→
   406→      {/* ─── Main Layout: Slide + Desktop Notes Sidebar ── */}
   407→      <div className="flex flex-1 overflow-hidden">
   408→      {/* ─── Slide Content Area ────────────────────── */}
   409→      <div className="flex-1 overflow-auto flex flex-col items-center py-8 px-4 sm:px-6">
   410→        <div
   411→          className={`
   412→            bg-card rounded-2xl shadow-lg border w-full max-w-3xl
   413→            ring-1 ring-black/5
   414→            dark:ring-white/5
   415→            paper-texture
   416→            ${isAnimating
   417→              ? (slideDirection === "forward" ? "slide-exit" : "opacity-0 -translate-x-2 transition-all duration-150")
   418→              : "slide-enter"
   419→            }
   420→          `}
   421→          style={{
   422→            transform: `scale(${zoom / 100})`,
   423→            transformOrigin: "top center",
   424→          }}
   425→        >
   426→          {loadingSlides ? (
   427→            <div className="flex items-center justify-center py-32">
   428→              <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   429→            </div>
   430→          ) : currentSlide ? (
   431→            <div className="p-8 sm:p-12 max-w-2xl mx-auto">
   432→              <SlideRenderer slide={currentSlide} />
   433→            </div>
   434→          ) : (
   435→            <div className="flex items-center justify-center py-32 text-muted-foreground">
   436→              <p>No slide content available.</p>
   437→            </div>
   438→          )}
   439→        </div>
   440→
   441→        {/* ─── Slide Progress Dots ────────────────── */}
   442→        <div className="slide-progress-indicator mt-6 shrink-0">
   443→          {Array.from({ length: localState.totalPages }, (_, i) => {
   444→            const isCompleted = i < localState.currentSlide;
   445→            const isActive = i === localState.currentSlide;
   446→            return (
   447→              <button
   448→                key={i}
   449→                className={`dot ${isActive ? "active" : ""} ${isCompleted ? "completed" : ""}`}
   450→                onClick={() => goToSlide(i)}
   451→                aria-label={`Go to slide ${i + 1}`}
   452→              />
   453→            );
   454→          })}
   455→        </div>
   456→      </div>
   457→
   458→      {/* ─── Desktop Notes Sidebar (slide-in panel) ── */}
   459→      <aside
   460→        className={`hidden lg:flex flex-col shrink-0 border-l bg-card transition-all duration-300 ease-in-out overflow-hidden ${
   461→          notesSidebarOpen ? "w-80 opacity-100" : "w-0 opacity-0 border-l-0"
   462→        }`}
   463→      >
   464→        <div className="flex flex-col h-full w-80">
   465→          {/* Sidebar Header */}
   466→          <div className="flex items-center justify-between px-4 py-3 border-b">
   467→            <div className="flex items-center gap-2">
   468→              <StickyNote className="h-4 w-4 text-primary" />
   469→              <span className="text-sm font-semibold">Notes</span>
   470→            </div>
   471→            <Button
   472→              variant="ghost"
   473→              size="icon"
   474→              className="h-7 w-7"
   475→              onClick={() => setNotesSidebarOpen(false)}
   476→            >
   477→              <X className="h-4 w-4" />
   478→            </Button>
   479→          </div>
   480→          <NotesSidebarContent
   481→            notes={notes}
   482→            currentSlide={localState.currentSlide}
   483→            newNoteContent={newNoteContent}
   484→            savingNote={savingNote}
   485→            onContentChange={setNewNoteContent}
   486→            onCreate={createNote}
   487→            onDelete={deleteNote}
   488→            onToggleBookmark={toggleBookmark}
   489→            endRef={notesEndRef}
   490→          />
   491→        </div>
   492→      </aside>
   493→      </div>
   494→
   495→      {/* ─── Keyboard Shortcuts Hint (auto-fades) ── */}
   496→      <div
   497→        className={`shrink-0 frosted-glass border-t px-4 py-1.5 text-center transition-opacity duration-700 ${
   498→          showKeyboardHint ? "opacity-100" : "opacity-0 pointer-events-none h-0 py-0 overflow-hidden"
   499→        }`}
   500→      >
   501→        <p className="text-xs text-muted-foreground/70 tracking-wide">
   502→          ← → Navigate&nbsp;&nbsp;|&nbsp;&nbsp;Space: Next&nbsp;&nbsp;|&nbsp;&nbsp;Esc: Exit
   503→        </p>
   504→      </div>
   505→
   506→      {/* ─── Bottom Controls (Frosted Glass) ───────── */}
   507→      <footer className="shrink-0 border-t frosted-glass">
   508→        <div className="mx-auto flex h-16 max-w-5xl items-center justify-between px-4 sm:px-6">
   509→          {/* Zoom Controls */}
   510→          <div className="flex items-center gap-1">
   511→            <Tooltip>
   512→              <TooltipTrigger asChild>
   513→                <Button
   514→                  variant="ghost"
   515→                  size="icon"
   516→                  className="h-8 w-8"
   517→                  onClick={zoomOut}
   518→                  disabled={zoom <= MIN_ZOOM}
   519→                >
   520→                  <ZoomOut className="h-4 w-4" />
   521→                </Button>
   522→              </TooltipTrigger>
   523→              <TooltipContent>Zoom Out</TooltipContent>
   524→            </Tooltip>
   525→
   526→            <span className="w-12 text-center text-xs font-medium text-muted-foreground tabular-nums">
   527→              {zoom}%
   528→            </span>
   529→
   530→            <Tooltip>
   531→              <TooltipTrigger asChild>
   532→                <Button
   533→                  variant="ghost"
   534→                  size="icon"
   535→                  className="h-8 w-8"
   536→                  onClick={zoomIn}
   537→                  disabled={zoom >= MAX_ZOOM}
   538→                >
   539→                  <ZoomIn className="h-4 w-4" />
   540→                </Button>
   541→              </TooltipTrigger>
   542→              <TooltipContent>Zoom In</TooltipContent>
   543→            </Tooltip>
   544→
   545→            <Tooltip>
   546→              <TooltipTrigger asChild>
   547→                <Button
   548→                  variant="ghost"
   549→                  size="icon"
   550→                  className="h-8 w-8"
   551→                  onClick={zoomFit}
   552→                >
   553→                  <Maximize2 className="h-4 w-4" />
   554→                </Button>
   555→              </TooltipTrigger>
   556→              <TooltipContent>Fit Width</TooltipContent>
   557→            </Tooltip>
   558→
   559→            <Tooltip>
   560→              <TooltipTrigger asChild>
   561→                <Button
   562→                  variant="ghost"
   563→                  size="icon"
   564→                  className="h-8 w-8"
   565→                  onClick={zoomFit}
   566→                >

  1049→function QuizSlide({ slide }: { slide: SlideContent }) {

  1053→  // Determine correct answer: look for item with icon === "check" or first item

  1070→              if (i === correctIdx) borderColor = "border-emerald-500 bg-emerald-500/10";
  1071→              else if (isSelected && !isCorrect) borderColor = "border-red-400 bg-red-400/10";
  1072→              else borderColor = "border-border opacity-60";
  1073→            } else if (isSelected) {
  1074→              borderColor = "border-primary bg-primary/10";
  1075→            }
  1076→
  1077→            return (
  1078→              <button
  1079→                key={i}
  1080→                type="button"
  1081→                className={`w-full rounded-lg border p-4 text-left transition-all duration-200 ${borderColor}`}
  1082→                onClick={() => {
  1083→                  if (!hasSubmitted) setSelectedIdx(i);
  1084→                }}
  1085→                disabled={hasSubmitted}
  1086→              >
  1087→                <div className="flex items-start gap-3">
  1088→                  <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold ${
  1089→                    hasSubmitted && i === correctIdx
  1090→                      ? "bg-emerald-500 text-white"
  1091→                      : hasSubmitted && isSelected && !isCorrect
  1092→                        ? "bg-red-400 text-white"
  1093→                        : "bg-primary/10 text-primary"
  1094→                  }`}>
  1095→                    {hasSubmitted && i === correctIdx ? "✓" : hasSubmitted && isSelected && !isCorrect ? "✗" : letter}
  1096→                  </span>
  1097→                  <div className="flex-1">
  1098→                    <p className="text-sm font-medium text-foreground">
  1099→                      {item.heading || item.text}
  1100→                    </p>
  1101→                    {item.heading && item.text !== item.heading && (
  1102→                      <p className="text-xs text-muted-foreground mt-1">{item.text}</p>
  1103→                    )}
  1104→                  </div>
  1105→                </div>
  1106→              </button>
  1107→            );
  1108→          })}
  1109→
  1110→          {/* Feedback after selection */}
