   249→// ============================================
   250→// Create Course Page Component
   251→// ============================================
   252→
   253→const MAX_SECTIONS = 10;
   254→const MAX_TITLE_LENGTH = 100;
   255→const MAX_DESC_LENGTH = 3000;
   256→const MAX_COURSE_DESC_LENGTH = 500;
   257→
   258→export function CreateCoursePage() {
   259→  const { goBack } = useNavigationStore();
   260→  const { currentUserId } = useUserStore();
   261→  const { createPrompt, setCreatePrompt } = useCourseStore();
   262→
   263→  // ---- Form state ----
   264→  const [title, setTitle] = useState("");
   265→  const [description, setDescription] = useState("");
   266→  const [categoryId, setCategoryId] = useState("");
   267→  const [language, setLanguage] = useState("english");
   268→  const [coverImage, setCoverImage] = useState("");
   269→  const [coverPreview, setCoverPreview] = useState<string | null>(null);
   270→  const [sections, setSections] = useState<SectionDraft[]>([]);
   271→
   272→  // ---- UI state ----
   273→  const [categories, setCategories] = useState<CategoryItem[]>([]);
   274→  const [loading, setLoading] = useState(false);
   275→  const [saving, setSaving] = useState(false);
   276→  const [generating, setGenerating] = useState(false);
   277→  const [modalOpen, setModalOpen] = useState(false);
   278→  const [urlInputOpen, setUrlInputOpen] = useState(false);
   279→  const [imageUrl, setImageUrl] = useState("");
   280→
   281→  // ---- Modal state ----
   282→  const [sectionName, setSectionName] = useState("");
   283→  const [sectionPrompt, setSectionPrompt] = useState("");
   284→  const [sectionLanguage, setSectionLanguage] = useState("english");
   285→  const [sectionPdfName, setSectionPdfName] = useState("");
   286→
   287→  // ---- Pre-fill title from hero prompt ----
   288→  useEffect(() => {
   289→    if (createPrompt) {
   290→      setTitle(createPrompt);
   291→      setCreatePrompt("");
   292→    }
   293→  }, [createPrompt, setCreatePrompt]);
   294→
   295→  // ---- Refs ----
   296→  const fileInputRef = useRef<HTMLInputElement>(null);
   297→  const sectionFileInputRef = useRef<HTMLInputElement>(null);
   298→
   299→  // ---- Fetch categories ----
   300→  useEffect(() => {
   301→    async function fetchCategories() {
   302→      setLoading(true);
   303→      try {
   304→        const res = await fetch("/api/categories");
   305→        const json = await res.json();
   306→        if (json.success) {
   307→          setCategories(json.data);
   308→        }
   309→      } catch {
   310→        toast.error("Failed to load categories");
   311→      } finally {
   312→        setLoading(false);
   313→      }
   314→    }
   315→    fetchCategories();
   316→  }, []);
   317→
   318→  // ---- Cover image handlers ----
   319→  const handleCoverUpload = () => {
   320→    // In sandbox, simulate file selection
   321→    setUrlInputOpen(true);
   322→  };
   323→
   324→  const handleUrlConfirm = () => {
   325→    if (imageUrl.trim()) {
   326→      setCoverImage(imageUrl.trim());
   327→      setCoverPreview(imageUrl.trim());
   328→      setUrlInputOpen(false);
   329→      setImageUrl("");
   330→    }
   331→  };
   332→
   333→  const handleRemoveCover = () => {
   334→    setCoverImage("");
   335→    setCoverPreview(null);
   336→  };
   337→
   338→  // ---- Section management ----
   339→  const handleOpenModal = () => {
   340→    setSectionName("");
   341→    setSectionPrompt("");
   342→    setSectionLanguage(language);
   343→    setSectionPdfName("");
   344→    setModalOpen(true);
   345→  };
   346→
   347→  const handleGenerateSection = useCallback(() => {
   348→    if (!sectionName.trim()) {
   349→      toast.error("Section name is required");
   350→      return;
   351→    }
   352→    if (!sectionPrompt.trim()) {
   353→      toast.error("Course prompt is required");
   354→      return;
   355→    }
   356→    if (sections.length >= MAX_SECTIONS) {
   357→      toast.error(`Maximum ${MAX_SECTIONS} sections allowed`);
   358→      return;
   359→    }
   360→
   361→    setGenerating(true);
   362→
   363→    // Simulate generation delay
   364→    setTimeout(() => {
   365→      const slides = generatePlaceholderSlides(
   366→        sectionName.trim(),
   367→        sectionLanguage
   368→      );
   369→      const newSection: SectionDraft = {
   370→        id: `sec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   371→        title: sectionName.trim(),
   372→        totalPages: slides.length,
   373→        content: JSON.stringify(slides),
   374→        language: sectionLanguage,
   375→      };
   376→
   377→      setSections((prev) => [...prev, newSection]);
   378→      setGenerating(false);
   379→      setModalOpen(false);
   380→      toast.success(`Section "${sectionName.trim()}" generated successfully`);
   381→    }, 1200);
   382→  }, [sectionName, sectionPrompt, sectionLanguage, sections.length]);
   383→
   384→  const handleDeleteSection = (id: string) => {
   385→    setSections((prev) => prev.filter((s) => s.id !== id));
   386→  };
   387→
   388→  const handleMoveSection = (index: number, direction: "up" | "down") => {
   389→    const newSections = [...sections];
   390→    const targetIndex = direction === "up" ? index - 1 : index + 1;
   391→    if (targetIndex < 0 || targetIndex >= newSections.length) return;
   392→    [newSections[index], newSections[targetIndex]] = [
   393→      newSections[targetIndex],
   394→      newSections[index],
   395→    ];
   396→    setSections(newSections);
   397→  };
   398→
   399→  // ---- Form submission ----
   400→  const handleSave = async () => {
   401→    if (!title.trim()) {
   402→      toast.error("Course title is required");
   403→      return;
   404→    }
   405→
   406→    setSaving(true);
   407→    try {
   408→      const payload = {
   409→        title: title.trim(),
   410→        description: description.trim() || null,
   411→        categoryId: categoryId || null,
   412→        language,
   413→        creatorId: currentUserId,
   414→        coverImage: coverImage || null,
   415→        sections: sections.map((sec, index) => ({
   416→          title: sec.title,
   417→          content: sec.content,
   418→          totalPages: sec.totalPages,
   419→          order: index,
   420→        })),
   421→      };
   422→
   423→      const res = await fetch("/api/courses", {
   424→        method: "POST",
   425→        headers: { "Content-Type": "application/json" },
   426→        body: JSON.stringify(payload),
   427→      });
   428→
   429→      const json = await res.json();
   430→
   431→      if (json.success) {
   432→        toast.success("Course created successfully!");
   433→        goBack();
   434→      } else {
   435→        toast.error(json.error || "Failed to create course");
   436→      }
   437→    } catch {
   438→      toast.error("Failed to create course. Please try again.");
   439→    } finally {
   440→      setSaving(false);
   441→    }
   442→  };
   443→
   444→  // ---- Section PDF upload simulation ----
   445→  const handleSectionPdfUpload = () => {
   446→    sectionFileInputRef.current?.click();
   447→  };
   448→
   449→  const handleSectionFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   450→    const file = e.target.files?.[0];
   451→    if (file) {
   452→      setSectionPdfName(file.name);
   453→      // If no prompt yet, use the filename
   454→      if (!sectionPrompt.trim()) {
   455→        const nameWithoutExt = file.name.replace(/\.pdf$/i, "");
   456→        setSectionPrompt(
   457→          `Generate course content based on the uploaded PDF: ${nameWithoutExt}`
   458→        );
   459→        if (!sectionName.trim()) {
   460→          setSectionName(nameWithoutExt);
   461→        }
   462→      }
   463→    }
   464→  };
   465→
   466→  // ============================================
   467→  // Render
   468→  // ============================================
   469→
   470→  return (
   471→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   472→      {/* ---- Breadcrumbs ---- */}
   473→      <div className="mb-6 flex items-center gap-3">
   474→        <button
   475→          onClick={goBack}
   476→          className="flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-foreground"
   477→        >
   478→          <ArrowLeft className="h-4 w-4" />
   479→          Back
   480→        </button>
   481→        <Separator orientation="vertical" className="h-4" />
   482→        <h1 className="text-lg font-semibold text-foreground">Create Course</h1>
   483→      </div>
   484→
   485→      {/* ---- Two-Column Layout ---- */}
   486→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-5">
   487→        {/* ======== Left Panel: Course Metadata (3 cols) ======== */}
   488→        <div className="lg:col-span-3">
   489→          <Card className="border-border/50">
   490→            <CardContent className="space-y-6 p-6">
   491→              {/* Cover Image Upload */}
   492→              <div className="space-y-2">
   493→                <Label className="text-sm font-medium">Cover Image</Label>
   494→                {coverPreview ? (
   495→                  <div className="relative overflow-hidden rounded-lg border border-border">
   496→                    <img
   497→                      src={coverPreview}
   498→                      alt="Course cover"
   499→                      className="h-48 w-full object-cover"
   500→                      onError={() => setCoverPreview(null)}
   501→                    />
   502→                    <button
   503→                      onClick={handleRemoveCover}
   504→                      className="absolute top-2 right-2 flex h-7 w-7 items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80"
   505→                    >
   506→                      <X className="h-3.5 w-3.5" />
   507→                    </button>
   508→                  </div>
   509→                ) : (
   510→                  <button
   511→                    type="button"
   512→                    onClick={handleCoverUpload}
   513→                    className="flex h-48 w-full flex-col items-center justify-center gap-3 rounded-lg border-2 border-dashed border-border/70 bg-muted/30 transition-colors hover:border-primary/50 hover:bg-muted/50"
   514→                  >
   515→                    <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted">
   516→                      <ImageIcon className="h-5 w-5 text-muted-foreground" />
   517→                    </div>
   518→                    <div className="text-center">
   519→                      <p className="text-sm font-medium text-foreground">
   520→                        Upload cover image
   521→                      </p>
   522→                      <p className="mt-0.5 text-xs text-muted-foreground">
   523→                        Recommended 1920x1080
   524→                      </p>
   525→                    </div>
   526→                  </button>
   527→                )}
   528→
   529→                {/* URL Input */}
   530→                {urlInputOpen && (
   531→                  <div className="flex items-center gap-2">
   532→                    <Input
   533→                      placeholder="Paste image URL here..."
   534→                      value={imageUrl}
   535→                      onChange={(e) => setImageUrl(e.target.value)}
   536→                      onKeyDown={(e) => e.key === "Enter" && handleUrlConfirm()}
   537→                      className="flex-1"
   538→                    />
   539→                    <Button size="sm" onClick={handleUrlConfirm}>
   540→                      Apply
   541→                    </Button>
   542→                    <Button
   543→                      size="sm"
   544→                      variant="ghost"
   545→                      onClick={() => {
   546→                        setUrlInputOpen(false);
   547→                        setImageUrl("");
   548→                      }}
   549→                    >
   550→                      Cancel
   551→                    </Button>
   552→                  </div>
   553→                )}
   554→                <input
   555→                  ref={fileInputRef}
   556→                  type="file"
   557→                  accept="image/*"
   558→                  className="hidden"
   559→                />
   560→              </div>
   561→
   562→              <Separator />
   563→
   564→              {/* Course Title */}
   565→              <div className="space-y-2">
   566→                <div className="flex items-center justify-between">
   567→                  <Label htmlFor="course-title" className="text-sm font-medium">
   568→                    Course Title <span className="text-destructive">*</span>
   569→                  </Label>
   570→                  <span className="text-xs text-muted-foreground">
   571→                    {title.length}/100
   572→                  </span>
   573→                </div>
   574→                <Input
   575→                  id="course-title"
   576→                  placeholder="Enter course title"
   577→                  value={title}
   578→                  onChange={(e) => {
   579→                    if (e.target.value.length <= MAX_TITLE_LENGTH) {
   580→                      setTitle(e.target.value);
   581→                    }
   582→                  }}
   583→                  className="h-10"
   584→                />
   585→              </div>
   586→
   587→              {/* Course Description */}
   588→              <div className="space-y-2">
   589→                <div className="flex items-center justify-between">
   590→                  <Label
   591→                    htmlFor="course-description"
   592→                    className="text-sm font-medium"
   593→                  >
   594→                    Description
   595→                  </Label>
   596→                  <span className="text-xs text-muted-foreground">
   597→                    {description.length}/{MAX_COURSE_DESC_LENGTH}
   598→                  </span>
   599→                </div>
   600→                <Textarea
   601→                  id="course-description"
   602→                  placeholder="Describe what this course is about..."
   603→                  value={description}
   604→                  onChange={(e) => {
   605→                    if (e.target.value.length <= MAX_COURSE_DESC_LENGTH) {
   606→                      setDescription(e.target.value);
   607→                    }
   608→                  }}
   609→                  rows={4}
   610→                  className="resize-none"
   611→                />
   612→              </div>
   613→
   614→              <Separator />
   615→
   616→              {/* Category */}
   617→              <div className="space-y-2">
   618→                <Label className="text-sm font-medium">Category</Label>
   619→                <Select value={categoryId} onValueChange={setCategoryId}>
   620→                  <SelectTrigger className="w-full h-10">
   621→                    <SelectValue placeholder="Select a category" />
   622→                  </SelectTrigger>
   623→                  <SelectContent>
   624→                    {categories.map((cat) => (
   625→                      <SelectItem key={cat.id} value={cat.id}>
   626→                        {cat.name}
   627→                      </SelectItem>
   628→                    ))}
   629→                  </SelectContent>
   630→                </Select>
   631→              </div>
   632→
   633→              {/* Language */}
   634→              <div className="space-y-3">
   635→                <Label className="text-sm font-medium">Language</Label>
   636→                <RadioGroup
   637→                  value={language}
   638→                  onValueChange={setLanguage}
   639→                  className="flex gap-6"
   640→                >
   641→                  <div className="flex items-center gap-2">
   642→                    <RadioGroupItem value="chinese" id="lang-zh" />
   643→                    <Label
   644→                      htmlFor="lang-zh"
