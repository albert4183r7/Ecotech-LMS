   495→
   496→  // ============================================
   497→  // GENERATE SLIDES — Streaming HTML for all slides
   498→  // ============================================
   499→
   500→  const handleGenerateSlides = useCallback(
   501→    async (lessonId: string) => {
   502→      const lesson = outlineLessons.find((l) => l.id === lessonId);
   503→      if (!lesson) return;
   504→      if (lesson.slides.length === 0) {
   505→        toast.error("No slides to generate");
   506→        return;
   507→      }
   508→
   509→      // Initialize per-slide gen states
   510→      const initialStates: Record<string, SlideGenState> = {};
   511→      lesson.slides.forEach((s) => {
   512→        if (s.slideId) {
   513→          initialStates[s.slideId] = { status: "pending" };
   514→        }
   515→      });
   516→      setSlideGenStates(initialStates);
   517→      setGeneratingLessonId(lessonId);
   518→      setCurrentGenSlideId(null);
   519→      setStreamingHtml("");
   520→      setGenProgress({ current: 0, total: lesson.slides.length });
   521→      setExpandedOutlineLessonId(lessonId);
   522→
   523→      // Close modal if open
   524→      setModalOpen(false);
   525→
   526→      const abort = new AbortController();
   527→      abortGenRef.current = abort;
   528→
   529→      try {
   530→        const res = await fetch("/api/lessons/generate-slides", {
   531→          method: "POST",
   532→          headers: { "Content-Type": "application/json" },
   533→          body: JSON.stringify({
   534→            lessonId,
   535→            language: lesson.language,
   536→          }),
   537→          signal: abort.signal,
   538→        });
   539→
   540→        if (!res.ok || !res.body) {
   541→          toast.error("Failed to start slide generation");
   542→          setGeneratingLessonId(null);
   543→          return;
   544→        }
   545→
   546→        const reader = res.body.getReader();
   547→        const decoder = new TextDecoder();
   548→        let buffer = "";
   549→        let slidesCompleted = 0;
   550→        let rawHtmlAccum = "";
   551→
   552→        while (true) {
   553→          const { done, value } = await reader.read();
   554→          if (done) break;
   555→          buffer += decoder.decode(value, { stream: true });
   556→
   557→          const lines = buffer.split("\n");
   558→          buffer = lines.pop() || "";
   559→
   560→          for (const line of lines) {
   561→            if (!line.startsWith("data: ")) continue;
   562→            try {
   563→              const data = JSON.parse(line.slice(6));
   564→
   565→              if (data.slideId && data.slideTitle && data.totalSlides !== undefined) {
   566→                // slide_start event
   567→                setCurrentGenSlideId(data.slideId);
   568→                setSlideGenStates((prev) => ({
   569→                  ...prev,
   570→                  [data.slideId]: { status: "generating" },
   571→                }));
   572→                rawHtmlAccum = "";
   573→                setGenProgress({ current: data.slideIndex + 1, total: data.totalSlides });
   574→              }
   575→
   576→              if (data.slideId && data.html) {
   577→                // chunk event
   578→                rawHtmlAccum += data.html;
   579→                setStreamingHtml(buildLivePreviewDoc(rawHtmlAccum, data.slideId));
   580→              }
   581→
   582→              if (data.slideId && data.htmlBody) {
   583→                // slide_complete event
   584→                slidesCompleted++;
   585→                setSlideGenStates((prev) => ({
   586→                  ...prev,
   587→                  [data.slideId]: { status: "complete", htmlBody: data.htmlBody },
   588→                }));
   589→                setOutlineLessons((prev) =>
   590→                  prev.map((ol) =>
   591→                    ol.id === lessonId ? { ...ol, allReady: slidesCompleted === ol.slides.length } : ol
   592→                  )
   593→                );
   594→              }
   595→
   596→              if (data.slideId && data.error) {
   597→                // slide_error event
   598→                slidesCompleted++;
   599→                setSlideGenStates((prev) => ({
   600→                  ...prev,
   601→                  [data.slideId]: { status: "error", error: data.error },
   602→                }));
   603→              }
   604→
   605→              if (data.lessonId && data.slidesGenerated !== undefined) {
   606→                // all_complete event
   607→                toast.success(`All ${data.slidesGenerated} slides generated!`);
   608→              }
   609→            } catch {
   610→              // skip malformed JSON

  1241→
