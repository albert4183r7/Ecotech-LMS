     4→     4→import {

    23→    23→import { Separator } from "@/components/ui/separator";
    24→    24→import { Tooltip, TooltipTrigger, TooltipContent } from "@/components/ui/tooltip";
    25→    25→import {

    30→    30→  SheetDescription,
    31→    31→} from "@/components/ui/sheet";
    32→    32→import { ScrollArea } from "@/components/ui/scroll-area";
    33→    33→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    34→    34→import type { SlideContent, ClassroomState } from "@/types/lms";
    35→    35→import { StudyTimer } from "@/components/lms/study-timer";
    36→    36→
    37→    37→const MIN_ZOOM = 50;
    38→    38→const MAX_ZOOM = 200;
    39→    39→const ZOOM_STEP = 25;
    40→    40→
    41→    41→export function ClassroomPage() {
    42→    42→  const { classroomState, goBack } = useNavigationStore();
    43→    43→  const userId = useUserStore((s) => s.currentUserId);
    44→    44→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    45→    45→  const [zoom, setZoom] = useState(100);
    46→    46→  const [loadingSlides, setLoadingSlides] = useState(false);
    47→    47→  const [slideDirection, setSlideDirection] = useState<"forward" | "back">("forward");
    48→    48→  const [isAnimating, setIsAnimating] = useState(false);
    49→    49→  const hasFetchedRef = useRef(false);
    50→    50→  const confettiShownRef = useRef(false);
    51→    51→  const [showConfetti, setShowConfetti] = useState(false);
    52→    52→
    53→    53→  // ─── Notes Sidebar State ──────────────────────
    54→    54→  interface Note {
    55→    55→    id: string;
    56→    56→    content: string;
    57→    57→    slideNumber: number;
    58→    58→    bookmarked: boolean;
    59→    59→    createdAt?: string;
    60→    60→  }
    61→    61→  const [notes, setNotes] = useState<Note[]>([]);
    62→    62→  const [notesSidebarOpen, setNotesSidebarOpen] = useState(false);
    63→    63→  const [newNoteContent, setNewNoteContent] = useState("");
    64→    64→  const [savingNote, setSavingNote] = useState(false);
    65→    65→  const notesEndRef = useRef<HTMLDivElement>(null);
    66→    66→
    67→    67→  // ─── Keyboard Hint Fade ────────────────────────
    68→    68→  const [showKeyboardHint, setShowKeyboardHint] = useState(true);
    69→    69→  const [isMobile, setIsMobile] = useState(false);
    70→    70→  const [downloadingPptx, setDownloadingPptx] = useState(false);
    71→    71→
    72→    72→  useEffect(() => {
    73→    73→    const mq = window.matchMedia("(max-width: 1023px)");
    74→    74→    setIsMobile(mq.matches);
    75→    75→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
    76→    76→    mq.addEventListener("change", handler);
    77→    77→    return () => mq.removeEventListener("change", handler);
    78→    78→  }, []);
    79→    79→
    80→    80→  // Keep a working copy of classroomState so we can mutate currentSlide
    81→    81→  useEffect(() => {
    82→    82→    if (classroomState) {
    83→    83→      setLocalState({ ...classroomState });
    84→    84→      setZoom(100);
    85→    85→      hasFetchedRef.current = false;
    86→    86→    }
    87→    87→  }, [classroomState]);
    88→    88→
    89→    89→  // If slides were empty initially, try fetching them
    90→    90→  useEffect(() => {
    91→    91→    if (!localState || localState.slides.length > 0 || hasFetchedRef.current) return;
    92→    92→    hasFetchedRef.current = true;
    93→    93→    const fetchSlides = async () => {
    94→    94→      setLoadingSlides(true);
    95→    95→      try {
    96→    96→        const res = await fetch(`/api/sections/${localState.sectionId}`);
    97→    97→        if (!res.ok) return;
    98→    98→        const json = await res.json();
    99→    99→        if (json.success && json.data.content) {
   100→   100→          const slides: SlideContent[] = Array.isArray(json.data.content)
   101→   101→            ? json.data.content
   102→   102→            : [];
   103→   103→          setLocalState((prev) =>
   104→   104→            prev
   105→   105→              ? {
   106→   106→                  ...prev,
   107→   107→                  slides,
   108→   108→                  totalPages: slides.length || prev.totalPages,
   109→   109→                }
   110→   110→              : prev
   111→   111→          );
   112→   112→        }
   113→   113→      } catch {
   114→   114→        // Silently fail
   115→   115→      } finally {
   116→   116→        setLoadingSlides(false);
   117→   117→      }
   118→   118→    };
   119→   119→    fetchSlides();
   120→   120→  }, [localState?.sectionId, localState?.slides.length]);
   121→   121→
   122→   122→  /** Navigate slides with transition animation */
   123→   123→  const goToSlide = useCallback(
   124→   124→    (index: number) => {
   125→   125→      if (!localState || isAnimating) return;
   126→   126→      const next = Math.max(0, Math.min(index, localState.totalPages - 1));
   127→   127→      if (next === localState.currentSlide) return;
   128→   128→      setSlideDirection(next > localState.currentSlide ? "forward" : "back");
   129→   129→      setIsAnimating(true);
   130→   130→      // Small delay for exit animation, then update slide
   131→   131→      setTimeout(() => {
   132→   132→        setLocalState((prev) =>
   133→   133→          prev ? { ...prev, currentSlide: next } : prev
   134→   134→        );
   135→   135→        setIsAnimating(false);
   136→   136→      }, 150);
   137→   137→    },
   138→   138→    [localState, isAnimating]
   139→   139→  );
   140→   140→
   141→   141→  /** Persist progress to API when slide changes */
   142→   142→  useEffect(() => {
   143→   143→    if (!localState || !userId) return;
   144→   144→    const saveProgress = async () => {
   145→   145→      try {
   146→   146→        // Get enrollment ID from the API
   147→   147→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   148→   148→        const enrollJson = await enrollRes.json();
   149→   149→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   150→   150→          const enrollment = enrollJson.data.find(
   151→   151→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   152→   152→          );
   153→   153→          if (enrollment) {
   154→   154→            const completed = localState.currentSlide >= localState.totalPages - 1;
   155→   155→            await fetch("/api/progress", {
   156→   156→              method: "POST",
   157→   157→              headers: { "Content-Type": "application/json" },
   158→   158→              body: JSON.stringify({
   159→   159→                enrollmentId: enrollment.id,
   160→   160→                sectionId: localState.sectionId,
   161→   161→                currentPage: localState.currentSlide + 1,
   162→   162→                completed,
   163→   163→              }),
   164→   164→            });
   165→   165→          }
   166→   166→        }
   167→   167→      } catch {
   168→   168→        // Progress saving is best-effort
   169→   169→      }
   170→   170→    };
   171→   171→    // Debounce to avoid excessive API calls
   172→   172→    const timer = setTimeout(saveProgress, 500);
   173→   173→    return () => clearTimeout(timer);
   174→   174→  }, [localState?.currentSlide]);
   175→   175→
   176→   176→  const goPrev = () => goToSlide((localState?.currentSlide ?? 0) - 1);
   177→   177→  const goNext = () => goToSlide((localState?.currentSlide ?? 0) + 1);
   178→   178→
   179→   179→  // ─── Keyboard shortcuts ──────────────────────────────────────
   180→   180→  useEffect(() => {
   181→   181→    function handleKeyDown(e: KeyboardEvent) {
   182→   182→      const tag = (e.target as HTMLElement)?.tagName;
   183→   183→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   184→   184→
   185→   185→      switch (e.key) {
   186→   186→        case "Escape":
   187→   187→          e.preventDefault();
   188→   188→          goBack();
   189→   189→          break;
   190→   190→        case "ArrowLeft":
   191→   191→          e.preventDefault();
   192→   192→          goPrev();
   193→   193→          break;
   194→   194→        case "ArrowRight":
   195→   195→          e.preventDefault();
   196→   196→          goNext();
   197→   197→          break;
   198→   198→        case " ":
   199→   199→          e.preventDefault();
   200→   200→          goNext();
   201→   201→          break;
   202→   202→      }
   203→   203→    }
   204→   204→
   205→   205→    document.addEventListener("keydown", handleKeyDown);
   206→   206→
   207→   207→    // Listen for custom event from KeyboardShortcuts component
   208→   208→    function handleCustomNextSlide() {
   209→   209→      goNext();
   210→   210→    }
   211→   211→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   212→   212→
   213→   213→    return () => {
   214→   214→      document.removeEventListener("keydown", handleKeyDown);
   215→   215→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   216→   216→    };
   217→   217→  }, [goBack, goPrev, goNext]);
   218→   218→
   219→   219→  /** Zoom controls */
   220→   220→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   221→   221→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   222→   222→  const zoomFit = () => setZoom(100);
   223→   223→
   224→   224→  // ─── Confetti on section completion ───────────
   225→   225→  useEffect(() => {
   226→   226→    if (!localState) return;
   227→   227→    const isOnLastSlide = localState.currentSlide === localState.totalPages - 1;
   228→   228→    if (isOnLastSlide && !confettiShownRef.current) {
   229→   229→      confettiShownRef.current = true;
   230→   230→      setShowConfetti(true);
   231→   231→      toast.success("🎉 You completed the course! Great job!");
   232→   232→      // Auto-cleanup after 3.5s (animation lasts ~3s + buffer)
   233→   233→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   234→   234→      return () => clearTimeout(timer);
   235→   235→    }
   236→   236→  }, [localState?.currentSlide]);
   237→   237→
   238→   238→  // Reset confetti flag when classroom state changes (new section)
   239→   239→  useEffect(() => {
   240→   240→    confettiShownRef.current = false;
   241→   241→    setShowConfetti(false);
   242→   242→  }, [classroomState?.sectionId]);
   243→   243→
   244→   244→  // ─── Keyboard Hint Auto-Fade ────────────────────
   245→   245→  useEffect(() => {
   246→   246→    setShowKeyboardHint(true);
   247→   247→    const timer = setTimeout(() => setShowKeyboardHint(false), 5000);
   248→   248→    return () => clearTimeout(timer);
   249→   249→  }, [localState?.sectionId]);
   250→   250→
   251→   251→  // ─── Notes CRUD ─────────────────────────────────
   252→   252→  const fetchNotes = useCallback(async () => {
   253→   253→    if (!localState || !userId) return;
   254→   254→    try {
   255→   255→      const res = await fetch(
   256→   256→        `/api/notes?userId=${userId}&courseId=${localState.courseId}&sectionId=${localState.sectionId}`
   257→   257→      );
   258→   258→      if (!res.ok) return;
   259→   259→      const json = await res.json();
   260→   260→      if (json.success && Array.isArray(json.data)) {
   261→   261→        setNotes(json.data);
   262→   262→      }
   263→   263→    } catch {
   264→   264→      // Silently fail
   265→   265→    }
   266→   266→  }, [localState, userId]);
   267→   267→
   268→   268→  // Fetch notes when section changes or sidebar opens
   269→   269→  useEffect(() => {
   270→   270→    if (notesSidebarOpen) fetchNotes();
   271→   271→  }, [notesSidebarOpen, fetchNotes, localState?.sectionId]);
   272→   272→
   273→   273→  const createNote = useCallback(async () => {
   274→   274→    if (!localState || !userId || !newNoteContent.trim()) return;
   275→   275→    setSavingNote(true);
   276→   276→    try {
   277→   277→      const res = await fetch("/api/notes", {
   278→   278→        method: "POST",
   279→   279→        headers: { "Content-Type": "application/json" },
   280→   280→        body: JSON.stringify({
   281→   281→          userId,
   282→   282→          courseId: localState.courseId,
   283→   283→          sectionId: localState.sectionId,
   284→   284→          content: newNoteContent.trim(),
   285→   285→          slideNumber: localState.currentSlide + 1,
   286→   286→        }),
   287→   287→      });
   288→   288→      if (!res.ok) return;
   289→   289→      const json = await res.json();
   290→   290→      if (json.success && json.data) {
   291→   291→        setNotes((prev) => [...prev, json.data]);
   292→   292→        setNewNoteContent("");
   293→   293→        setTimeout(() => notesEndRef.current?.scrollIntoView({ behavior: "smooth" }), 50);
   294→   294→      }
   295→   295→    } catch {
   296→   296→      toast.error("Failed to save note");
   297→   297→    } finally {
   298→   298→      setSavingNote(false);
   299→   299→    }
   300→   300→  }, [localState, userId, newNoteContent]);
   301→   301→
   302→   302→  const deleteNote = useCallback(async (noteId: string) => {
   303→   303→    try {
   304→   304→      const res = await fetch(`/api/notes?id=${noteId}`, { method: "DELETE" });
   305→   305→      if (!res.ok) return;
   306→   306→      const json = await res.json();
   307→   307→      if (json.success) {
   308→   308→        setNotes((prev) => prev.filter((n) => n.id !== noteId));
   309→   309→      }
   310→   310→    } catch {
   311→   311→      // Silently fail
   312→   312→    }
   313→   313→  }, []);
   314→   314→
   315→   315→  const toggleBookmark = useCallback((noteId: string) => {
   316→   316→    setNotes((prev) =>
   317→   317→      prev.map((n) =>
   318→   318→        n.id === noteId ? { ...n, bookmarked: !n.bookmarked } : n
   319→   319→      )
   320→   320→    );
   321→   321→  }, []);
   322→   322→
   323→   323→  // ─── Download as PPTX ─────────────────────────────
   324→   324→  const handleDownloadPptx = useCallback(async () => {
   325→   325→    if (!localState || localState.slides.length === 0 || downloadingPptx) return;
   326→   326→    setDownloadingPptx(true);
   327→   327→    try {
   328→   328→      const res = await fetch('/api/generate-pptx', {
   329→   329→        method: 'POST',
   330→   330→        headers: { 'Content-Type': 'application/json' },
   331→   331→        body: JSON.stringify({
   332→   332→          slides: localState.slides,
   333→   333→          courseName: localState.courseTitle,
   334→   334→        }),
   335→   335→      });
   336→   336→      if (!res.ok) {
   337→   337→        toast.error('Failed to generate PPT');
   338→   338→        return;
   339→   339→      }
   340→   340→      const blob = await res.blob();
   341→   341→      const url = URL.createObjectURL(blob);
   342→   342→      const a = document.createElement('a');
   343→   343→      a.href = url;
   344→   344→      const safeName = localState.courseTitle
   345→   345→        .replace(/[^a-zA-Z0-9\s-]/g, '')
   346→   346→        .replace(/\s+/g, '-')
   347→   347→        .toLowerCase()
   348→   348→        .slice(0, 60);
   349→   349→      a.download = `${safeName}.pptx`;
   350→   350→      document.body.appendChild(a);
   351→   351→      a.click();
   352→   352→      a.remove();
   353→   353→      URL.revokeObjectURL(url);
   354→   354→      toast.success('PPT downloaded successfully!');
   355→   355→    } catch {
   356→   356→      toast.error('Failed to download PPT');
   357→   357→    } finally {
   358→   358→      setDownloadingPptx(false);
   359→   359→    }
   360→   360→  }, [localState, downloadingPptx]);
   361→   361→
   362→   362→  // ─── No state guard ─────────────────────────────
   363→   363→  if (!localState) {
   364→   364→    return (
   365→   365→      <div className="flex h-screen items-center justify-center bg-muted/30">
   366→   366→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   367→   367→      </div>
   368→   368→    );
   369→   369→  }
   370→   370→
   371→   371→  const currentSlide =
   372→   372→    localState.slides[localState.currentSlide] ?? null;
   373→   373→  const isFirst = localState.currentSlide === 0;
   374→   374→  const isLast = localState.currentSlide >= localState.totalPages - 1;
   375→   375→  const progressPercent = localState.totalPages > 1
   376→   376→    ? Math.round(((localState.currentSlide + 1) / localState.totalPages) * 100)
   377→   377→    : 100;
   378→   378→
   379→   379→  return (
   380→   380→    <div className="flex h-screen flex-col bg-muted/30 relative">
   381→   381→      {/* ─── Mobile Notes Sheet ──────────────────── */}
   382→   382→      <Sheet open={notesSidebarOpen && isMobile} onOpenChange={(open) => { if (!open) setNotesSidebarOpen(false); }}>
   383→   383→        <SheetContent side="right" className="w-full sm:max-w-sm p-0">
   384→   384→          <SheetHeader className="px-4 pt-4 pb-0">
   385→   385→            <SheetTitle className="flex items-center gap-2 text-base">
   386→   386→              <StickyNote className="h-4 w-4" />
   387→   387→              Notes
   388→   388→            </SheetTitle>
   389→   389→            <SheetDescription>
   390→   390→              Notes for slide {localState.currentSlide + 1} of {localState.totalPages}
   391→   391→            </SheetDescription>
   392→   392→          </SheetHeader>
   393→   393→          <NotesSidebarContent
   394→   394→            notes={notes}
   395→   395→            currentSlide={localState.currentSlide}
   396→   396→            newNoteContent={newNoteContent}
   397→   397→            savingNote={savingNote}
   398→   398→            onContentChange={setNewNoteContent}
   399→   399→            onCreate={createNote}
   400→   400→            onDelete={deleteNote}
   401→   401→            onToggleBookmark={toggleBookmark}
   402→   402→            endRef={notesEndRef}
   403→   403→          />
   404→   404→        </SheetContent>
   405→   405→      </Sheet>
   406→   406→      {/* ─── Confetti Celebration Overlay ─────── */}
   407→   407→      {showConfetti && <ConfettiCelebration />}
   408→   408→      {/* ─── Top Progress Bar ─────────────────────── */}
   409→   409→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   410→   410→        <div
   411→   411→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   412→   412→          style={{ width: `${progressPercent}%` }}
   413→   413→        />
   414→   414→      </div>
   415→   415→
   416→   416→      {/* ─── Top Bar ──────────────────────────────── */}
   417→   417→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   418→   418→        {/* Logo, Section Title, Course Title */}
   419→   419→        <div className="flex items-center gap-3 min-w-0">
   420→   420→          <div className="flex items-center gap-2 shrink-0">
   421→   421→            <img src="/ecotech-logo.png" alt="Ecotech" className="h-7 w-7 rounded-md object-contain" />
   422→   422→            <img src="/ecotech-name.png" alt="Ecotech" className="h-5 w-auto hidden sm:inline object-contain" />
   423→   423→          </div>
   424→   424→          <Separator orientation="vertical" className="h-5" />
   425→   425→          <div className="min-w-0">
   426→   426→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   427→   427→              {localState.sectionTitle}
   428→   428→            </h2>
   429→   429→            <p className="text-xs text-muted-foreground truncate leading-tight">
   430→   430→              {localState.courseTitle}
   431→   431→            </p>
   432→   432→          </div>
   433→   433→        </div>
   434→   434→
   435→   435→        {/* Slide Counter - Enhanced */}
   436→   436→        <span className="shrink-0 inline-flex items-center gap-1.5 rounded-full bg-primary/10 px-3 py-1 text-sm text-primary tabular-nums">
   437→   437→          <span className="font-semibold">Slide {localState.currentSlide + 1}</span>
   438→   438→          <span className="text-primary/40 font-normal">of</span>
   439→   439→          <span className="font-semibold">{localState.totalPages}</span>
   440→   440→        </span>
   441→   441→      </header>
   442→   442→
   443→   443→      {/* ─── Main Layout: Slide + Desktop Notes Sidebar ── */}
   444→   444→      <div className="flex flex-1 overflow-hidden">
   445→   445→      {/* ─── Slide Content Area ────────────────────── */}
   446→   446→      <div className="flex-1 overflow-auto flex flex-col items-center py-8 px-4 sm:px-6">
   447→   447→        <div
   448→   448→          className={`
   449→   449→            bg-card rounded-2xl shadow-lg border w-full max-w-3xl
   450→   450→            ring-1 ring-black/5
   451→   451→            dark:ring-white/5
   452→   452→            paper-texture
   453→   453→            ${isAnimating
   454→   454→              ? (slideDirection === "forward" ? "slide-exit" : "opacity-0 -translate-x-2 transition-all duration-150")
   455→   455→              : "slide-enter"
   456→   456→            }
   457→   457→          `}
   458→   458→          style={{
   459→   459→            transform: `scale(${zoom / 100})`,
   460→   460→            transformOrigin: "top center",
   461→   461→          }}
   462→   462→        >
   463→   463→          {loadingSlides ? (

   843→   843→    const result: ConfettiParticle[] = [];
   844→   844→    const shapes: ParticleShape[] = ["circle", "rectangle", "triangle"];
   845→   845→    const count = 80;

   898→   898→                "--spin": p.spin,
   899→   899→              } as React.CSSProperties}
   900→   900→            />
   901→   901→          );
   902→   902→        })}
   903→   903→      </div>
   904→   904→
   905→   905→      {/* Congratulations message */}
   906→   906→      <div className="confetti-message" role="status" aria-label="Congratulations!">
   907→   907→        <div className="confetti-message-text flex flex-col items-center gap-3">
   908→   908→          <div className="flex h-16 w-16 items-center justify-center rounded-2xl bg-white/90 dark:bg-zinc-800/90 shadow-lg">
