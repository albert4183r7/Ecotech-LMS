   182→/* ------------------------------------------------------------------ */
   183→/*  Circular Mini Progress (for course rows)                           */
   184→/* ------------------------------------------------------------------ */
   185→
   186→function CircularMiniProgress({ percentage }: { percentage: number }) {
   187→  const size = 40;
   188→  const stroke = 3.5;
   189→  const r = (size - stroke) / 2;
   190→  const circ = r * 2 * Math.PI;
   191→  const offset = circ - (percentage / 100) * circ;
   192→
   193→  return (
   194→    <div className="relative inline-flex shrink-0 items-center justify-center">
   195→      <svg width={size} height={size} className="-rotate-90">
   196→        <circle
   197→          cx={size / 2}
   198→          cy={size / 2}
   199→          r={r}
   200→          fill="none"
   201→          stroke="currentColor"
   202→          strokeWidth={stroke}
   203→          className="text-muted/30"
   204→        />
   205→        <circle
   206→          cx={size / 2}
   207→          cy={size / 2}
   208→          r={r}
   209→          fill="none"
   210→          stroke="url(#mini-grad)"
   211→          strokeWidth={stroke}
   212→          strokeLinecap="round"
   213→          strokeDasharray={circ}
   214→          strokeDashoffset={offset}
   215→          className="transition-all duration-500 ease-out"
   216→        />
   217→        <defs>
   218→          <linearGradient id="mini-grad" x1="0%" y1="0%" x2="100%" y2="0%">
   219→            <stop offset="0%" stopColor="#0891b2" />
   220→            <stop offset="100%" stopColor="#14b8a6" />
   221→          </linearGradient>
   222→        </defs>
   223→      </svg>
   224→      <span className="absolute text-[9px] font-bold text-foreground">
   225→        {percentage}%
   226→      </span>
   227→    </div>
   228→  );
   229→}

   597→
   598→  const [stats, setStats] = useState<UserStats | null>(null);
   599→  const [enrollments, setEnrollmentsState] = useState<EnrollmentItem[]>([]);
   600→  const [favorites, setFavoritesState] = useState<FavoriteItem[]>([]);
   601→  const [loadingStats, setLoadingStats] = useState(true);
   602→  const [loadingEnrollments, setLoadingEnrollments] = useState(true);
   603→  const [loadingFavorites, setLoadingFavorites] = useState(true);
   604→
   605→  const fetchStats = useCallback(async () => {
   606→    setLoadingStats(true);
   607→    try {
   608→      const res = await fetch(`/api/user/${currentUserId}`);
   609→      const json = await res.json();
   610→      if (json.success) {
   611→        setStats(json.data.stats);
   612→      }
   613→    } catch {
   614→      /* silent */
   615→    } finally {
   616→      setLoadingStats(false);
   617→    }
   618→  }, [currentUserId]);
   619→
   620→  const fetchEnrollments = useCallback(async () => {
   621→    setLoadingEnrollments(true);
   622→    try {
   623→      const res = await fetch(
   624→        `/api/enrollments?userId=${currentUserId}`
   625→      );
   626→      const json = await res.json();
   627→      if (json.success) {
   628→        setEnrollmentsState(json.data);
   629→        setEnrollments(
   630→          json.data.map((e: EnrollmentItem) => enrollmentToCourseItem(e))
   631→        );
   632→      }
   633→    } catch {
   634→      /* silent */
   635→    } finally {
   636→      setLoadingEnrollments(false);
   637→    }
   638→  }, [currentUserId, setEnrollments]);
   639→
   640→  const fetchFavorites = useCallback(async () => {
   641→    setLoadingFavorites(true);
   642→    try {
   643→      const res = await fetch(
   644→        `/api/favorites?userId=${currentUserId}`
   645→      );
   646→      const json = await res.json();
   647→      if (json.success) {
   648→        setFavoritesState(json.data);
   649→        setFavorites(
   650→          json.data.map((f: FavoriteItem) => favoriteToCourseItem(f))
   651→        );
   652→      }
   653→    } catch {
   654→      /* silent */
   655→    } finally {
   656→      setLoadingFavorites(false);
   657→    }
   658→  }, [currentUserId, setFavorites]);
   659→
   660→  useEffect(() => {
   661→    fetchStats();
   662→    fetchEnrollments();
   663→    fetchFavorites();
   664→  }, [fetchStats, fetchEnrollments, fetchFavorites]);
   665→
   666→  /* Derived lists */
   667→  const inProgressList = enrollments.filter(
   668→    (e) => e.status === "in_progress"
   669→  );
   670→  const completedList = enrollments.filter(
   671→    (e) => e.status === "completed"
   672→  );
   673→
   674→  /* Tab change handler */
   675→  const handleTabChange = (value: string) => {
   676→    setTab(value as MyLearningTab);
   677→  };
   678→
   679→  return (
   680→    <div className="space-y-6">
   681→      {/* Page Header */}
   682→      <div>
   683→        <h1 className="text-2xl font-bold text-foreground">My Learning</h1>
   684→        <p className="mt-1 text-sm text-muted-foreground">
   685→          Track your learning progress and manage your courses
   686→        </p>
   687→      </div>
   688→
   689→      {/* Stats Row */}
   690→      {loadingStats ? (
   691→        <StatsSkeleton />
   692→      ) : stats ? (
   693→        <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   694→          <StatCard
   695→            icon={BookOpen}
   696→            label="Total Courses"
   697→            value={stats.totalCourses}
   698→            gradient="bg-gradient-to-br from-blue-500 to-blue-700"
   699→            trend="up"
   700→            loading={false}
   701→          />
   702→          <StatCard
   703→            icon={Clock}
   704→            label="In Progress"
   705→            value={stats.inProgress}
   706→            gradient="bg-gradient-to-br from-teal-500 to-teal-700"
   707→            loading={false}
   708→          />
   709→          <StatCard
   710→            icon={CheckCircle}
   711→            label="Completed"
   712→            value={stats.completed}
   713→            gradient="bg-gradient-to-br from-emerald-500 to-emerald-700"
   714→            trend="up"
   715→            loading={false}
   716→          />
   717→          <StatCard
   718→            icon={TrendingUp}
   719→            label="Avg Progress"
   720→            value={`${stats.avgProgress}%`}
   721→            gradient="bg-gradient-to-br from-amber-500 to-amber-700"
   722→            trend="up"
   723→            loading={false}
   724→          />
   725→        </div>
   726→      ) : null}
   727→
   728→      {/* Continue Learning Widget */}
   729→      <ContinueLearningWidget
   730→        enrollment={
   731→          loadingEnrollments
   732→            ? undefined
   733→            : inProgressList.length > 0
   734→              ? inProgressList[0]
   735→              : null
   736→        }
   737→      />
   738→
   739→      {/* Tabs */}
   740→      <Tabs
   741→        value={tab}
   742→        onValueChange={handleTabChange}
   743→        className="w-full"
   744→      >
   745→        {/* Underline-style tab list with animated indicator */}
   746→        <TabsList className="relative h-auto w-full justify-start gap-6 rounded-none border-b border-border bg-transparent p-0">
   747→          <TabsTrigger
   748→            value="in-progress"
   749→            className="rounded-none border-b-2 border-transparent px-1 pb-3 pt-1 text-sm font-medium text-muted-foreground transition-all duration-300 data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:font-semibold data-[state=active]:shadow-none data-[state=active]:text-foreground"
   750→          >
   751→            <span className="flex items-center gap-1.5">
   752→              In Progress
   753→              {!loadingEnrollments && inProgressList.length > 0 && (
   754→                <Badge className="ml-0.5 h-5 min-w-5 rounded-full bg-primary/10 px-1.5 text-[10px] font-bold text-primary">
   755→                  {inProgressList.length}
   756→                </Badge>
   757→              )}
   758→            </span>
   759→          </TabsTrigger>
   760→          <TabsTrigger
   761→            value="completed"
   762→            className="rounded-none border-b-2 border-transparent px-1 pb-3 pt-1 text-sm font-medium text-muted-foreground transition-all duration-300 data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:font-semibold data-[state=active]:shadow-none data-[state=active]:text-foreground"
   763→          >
   764→            <span className="flex items-center gap-1.5">
   765→              Completed
   766→              {!loadingEnrollments && completedList.length > 0 && (
   767→                <Badge className="ml-0.5 h-5 min-w-5 rounded-full bg-primary/10 px-1.5 text-[10px] font-bold text-primary">
   768→                  {completedList.length}
   769→                </Badge>
   770→              )}
   771→            </span>
   772→          </TabsTrigger>
   773→          <TabsTrigger
   774→            value="favorites"
   775→            className="rounded-none border-b-2 border-transparent px-1 pb-3 pt-1 text-sm font-medium text-muted-foreground transition-all duration-300 data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:font-semibold data-[state=active]:shadow-none data-[state=active]:text-foreground"
   776→          >
   777→            <span className="flex items-center gap-1.5">
   778→              Favorites
   779→              {!loadingFavorites && favorites.length > 0 && (
   780→                <Badge className="ml-0.5 h-5 min-w-5 rounded-full bg-primary/10 px-1.5 text-[10px] font-bold text-primary">
   781→                  {favorites.length}
   782→                </Badge>
   783→              )}
   784→            </span>
   785→          </TabsTrigger>
   786→        </TabsList>
   787→
   788→        {/* In Progress Tab */}
   789→        <TabsContent value="in-progress" className="mt-6">
   790→          {loadingEnrollments ? (
   791→            <ListSkeleton />
   792→          ) : inProgressList.length === 0 ? (
   793→            <EmptyState
   794→              icon={BookOpen}
   795→              title="No courses in progress"
   796→              description="Browse our course catalog and enroll in courses that interest you. Your active courses will appear here with progress tracking."
   797→            />
   798→          ) : (
   799→            <div className="space-y-3">
   800→              {inProgressList.map((enrollment) => (
   801→                <InProgressRow
   802→                  key={enrollment.id}
   803→                  enrollment={enrollment}
   804→                />
   805→              ))}
   806→            </div>
   807→          )}
   808→        </TabsContent>
   809→
   810→        {/* Completed Tab */}
   811→        <TabsContent value="completed" className="mt-6">
   812→          {loadingEnrollments ? (
   813→            <ListSkeleton />
   814→          ) : completedList.length === 0 ? (
   815→            <EmptyState
   816→              icon={CheckCircle}
   817→              title="No completed courses yet"
   818→              description="Keep learning! Once you finish all sections of an enrolled course, it will be moved here to celebrate your achievement."
   819→            />
   820→          ) : (
   821→            <div className="space-y-3">
   822→              {completedList.map((enrollment) => (
   823→                <CompletedRow
   824→                  key={enrollment.id}
   825→                  enrollment={enrollment}
   826→                />
   827→              ))}
   828→            </div>
   829→          )}
   830→        </TabsContent>
   831→
   832→        {/* Favorites Tab */}
   833→        <TabsContent value="favorites" className="mt-6">
   834→          {loadingFavorites ? (
   835→            <GridSkeleton />
   836→          ) : favorites.length === 0 ? (
   837→            <EmptyState
   838→              icon={Heart}
   839→              title="No favorites yet"
   840→              description="Click the heart icon on any course to save it to your favorites. You can quickly access them from here anytime."
   841→            />
   842→          ) : (
   843→            <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
   844→              {favorites.map((favorite) => (
   845→                <CourseCard
   846→                  key={favorite.id}
   847→                  course={favoriteToCourseItem(favorite)}
   848→                  showFavorite
   849→                  isFavorited
   850→                />
   851→              ))}
   852→            </div>
   853→          )}
   854→        </TabsContent>
   855→      </Tabs>
   856→    </div>
   857→  );
   858→}
