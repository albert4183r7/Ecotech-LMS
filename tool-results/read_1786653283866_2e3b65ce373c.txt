    68→function CircularProgress({ percentage }: { percentage: number }) {
    69→  const radius = 70;
    70→  const stroke = 10;
    71→  const normalizedRadius = radius - stroke / 2;
    72→  const circumference = normalizedRadius * 2 * Math.PI;
    73→  const strokeDashoffset =
    74→    circumference - (percentage / 100) * circumference;
    75→
    76→  return (
    77→    <div className="relative inline-flex items-center justify-center">
    78→      <svg
    79→        width={radius * 2}
    80→        height={radius * 2}
    81→        className="-rotate-90"
    82→      >
    83→        {/* Background track */}
    84→        <circle
    85→          cx={radius}
    86→          cy={radius}
    87→          r={normalizedRadius}
    88→          fill="none"
    89→          stroke="currentColor"
    90→          strokeWidth={stroke}
    91→          className="text-muted/30"
    92→        />
    93→        {/* Progress arc with gradient stroke */}
    94→        <circle
    95→          cx={radius}
    96→          cy={radius}
    97→          r={normalizedRadius}
    98→          fill="none"
    99→          stroke="url(#progress-gradient)"
   100→          strokeWidth={stroke}
   101→          strokeLinecap="round"
   102→          strokeDasharray={circumference}
   103→          strokeDashoffset={strokeDashoffset}
   104→          className="transition-all duration-700 ease-out"
   105→        />
   106→        <defs>
   107→          <linearGradient
   108→            id="progress-gradient"
   109→            x1="0%"
   110→            y1="0%"
   111→            x2="100%"
   112→            y2="100%"
   113→          >
   114→            <stop offset="0%" stopColor="#06b6d4" />
   115→            <stop offset="50%" stopColor="#0891b2" />
   116→            <stop offset="100%" stopColor="#0d9488" />
   117→          </linearGradient>
   118→        </defs>
   119→      </svg>
   120→      <div className="absolute flex flex-col items-center justify-center">
   121→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   122→          {percentage}%
   123→        </span>
   124→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   125→          completed
   126→        </span>
   127→      </div>
   128→    </div>
   129→  );
   130→}

   132→/* ------------------------------------------------------------------ */
   133→/*  Animated Counter Hook                                              */
   134→/* ------------------------------------------------------------------ */
   135→
   136→function useAnimatedCounter(target: number, duration = 800) {
   137→  const [count, setCount] = useState(target);
   138→
   139→  useEffect(() => {
   140→    if (target === 0) return;
   141→    let start = 0;
   142→    const startTime = performance.now();
   143→    const step = (now: number) => {
   144→      const elapsed = now - startTime;
   145→      const progress = Math.min(elapsed / duration, 1);
   146→      const eased = 1 - Math.pow(1 - progress, 3);
   147→      start = Math.round(eased * target);
   148→      setCount(start);
   149→      if (progress < 1) {
   150→        frameRef.current = requestAnimationFrame(step);
   151→      }
   152→    };
   153→    frameRef.current = requestAnimationFrame(step);
   154→    return () => cancelAnimationFrame(frameRef.current);
   155→  }, [target, duration]);
   156→
   157→  return count;
   158→}

   257→function StreakCalendar({ streak, bestStreak }: { streak: number; bestStreak: number }) {
   258→  const [activityMap, setActivityMap] = useState<Map<string, number>>(new Map());
   259→  const [days, setDays] = useState<Array<{ date: string; day: string; isToday: boolean }>>([]);
   260→  const [maxMinutes, setMaxMinutes] = useState(1);
   261→
   262→  useEffect(() => {
   263→    async function fetchActivity() {
   264→      try {
   265→        const res = await fetch('/api/activity?userId=user_demo_001&weeks=5');
   266→        const json = await res.json();
   267→        if (json.success) {
   268→          // Build 30-day grid from the response
   269→          const allDays: ActivityDayEntry[] = [];
   270→          const today = new Date();
   271→          today.setHours(0, 0, 0, 0);
   272→          const start = new Date(today);
   273→          start.setDate(start.getDate() - 29);
   274→
   275→          const resMap = new Map<string, number>();
   276→          for (const d of (json.data as Activity30Data).dailyData) {
   277→            resMap.set(d.date, d.minutes);
   278→          }
   279→
   280→          for (let i = 0; i < 30; i++) {
   281→            const d = new Date(start);
   282→            d.setDate(d.getDate() + i);
   283→            const dateStr = d.toISOString().split('T')[0];
   284→            allDays.push({
   285→              date: dateStr,
   286→              day: d.toLocaleDateString('en-US', { weekday: 'narrow' }),
   287→              isToday: dateStr === today.toISOString().split('T')[0],
   288→            });
   289→          }
   290→
   291→          setDays(allDays);
   292→          setActivityMap(resMap);
   293→          const max = Math.max(...Array.from(resMap.values()), 1);
   294→          setMaxMinutes(max);
   295→        }
   296→      } catch { /* silent */ }
   297→    }
   298→    fetchActivity();
   299→  }, []);
   300→
   301→  function getIntensity(date: string): number {
   302→    const mins = activityMap.get(date) || 0;
   303→    if (mins === 0) return 0;
   304→    return Math.min(1, mins / maxMinutes);
   305→  }
   306→

   308→    if (intensity === 0) return 'bg-muted/40';
   309→    if (intensity <= 0.25) return 'bg-cyan-300/40 dark:bg-cyan-700/40';
   310→    if (intensity <= 0.5) return 'bg-cyan-400/60 dark:bg-cyan-600/50';
   311→    if (intensity <= 0.75) return 'bg-cyan-500/70 dark:bg-cyan-500/60';
   312→    return 'bg-cyan-600 dark:bg-cyan-400';
   313→  }
   314→
   315→  return (
   316→    <Card className="border-border/50">
   317→      <CardHeader className="pb-3">
   318→        <div className="flex items-center justify-between">
   319→          <CardTitle className="flex items-center gap-2 text-base font-semibold">
   320→            <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-gradient-to-br from-orange-500 to-amber-500">
   321→              <Flame className="h-4 w-4 text-white" />
   322→            </div>
   323→            Learning Streak
   324→          </CardTitle>
   325→          <div className="flex items-center gap-3">
   326→            <div className="flex items-center gap-1.5 text-sm font-bold text-orange-600 dark:text-orange-400">
   327→              <span className="text-base">🔥</span>
   328→              <span className="count-up">{streak}</span>
   329→              <span className="text-xs font-medium text-muted-foreground">days</span>
   330→            </div>
   331→          </div>
   332→        </div>
   333→      </CardHeader>
   334→      <CardContent className="space-y-4">
   335→        {/* Streak stats row */}
   336→        <div className="flex items-center justify-between">
