     1→"use client";
     2→
     3→import { useCallback, useEffect, useState } from "react";
     4→import {
     5→  Star,
     6→  Heart,
     7→  Share2,
     8→  BookOpen,
     9→  Users,
    10→  Calendar,
    11→  ChevronLeft,
    12→  AlertTriangle,
    13→  FileText,
    14→  Loader2,
    15→  CheckCircle2,
    16→  Clock,
    17→  PlayCircle,
    18→  FileDown,
    19→} from "lucide-react";
    20→import { Button } from "@/components/ui/button";
    21→import { Badge } from "@/components/ui/badge";
    22→import { Separator } from "@/components/ui/separator";
    23→import { Skeleton } from "@/components/ui/skeleton";
    24→import {
    25→  Breadcrumb,
    26→  BreadcrumbItem,
    27→  BreadcrumbLink,
    28→  BreadcrumbList,
    29→  BreadcrumbPage,
    30→  BreadcrumbSeparator,
    31→} from "@/components/ui/breadcrumb";
    32→import {
    33→  Accordion,
    34→  AccordionContent,
    35→  AccordionItem,
    36→  AccordionTrigger,
    37→} from "@/components/ui/accordion";
    38→import { Progress } from "@/components/ui/progress";
    39→import { toast } from "sonner";
    40→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    41→import { DiscussionPanel } from "@/components/lms/discussion-panel";
    42→import { ProgressTimeline } from "@/components/lms/progress-timeline";
    43→import { StarRating } from "@/components/lms/star-rating";
    44→import type { CourseItem, SectionItem, ClassroomState } from "@/types/lms";
    45→
    46→type SectionProgress = {
    47→  sectionId: string;
    48→  currentPage: number;
    49→  completed: boolean;
    50→};
    51→
    52→type CourseDetailData = CourseItem & {
    53→  isEnrolled: boolean;
    54→  isFavorited: boolean;
    55→  enrollmentsCount?: number;
    56→  favoritesCount?: number;
    57→};
    58→
    59→export function CourseDetailPage() {
    60→  const { selectedCourseId, goBack, openClassroom, navigateTo } =
    61→    useNavigationStore();
    62→  const userId = useUserStore((s) => s.currentUserId);
    63→  const currentRole = useUserStore((s) => s.currentRole);
    64→
    65→  const [course, setCourse] = useState<CourseDetailData | null>(null);
    66→  const [loading, setLoading] = useState(true);
    67→  const [error, setError] = useState<string | null>(null);
    68→  const [enrolling, setEnrolling] = useState(false);
    69→  const [togglingFav, setTogglingFav] = useState(false);
    70→  const [imgError, setImgError] = useState(false);
    71→  const [sectionProgress, setSectionProgress] = useState<SectionProgress[]>([]);
    72→  const [userRating, setUserRating] = useState<number | null>(null);
    73→  const [ratingCount, setRatingCount] = useState(0);
    74→  const [downloadingPptx, setDownloadingPptx] = useState(false);
    75→
    76→  /** Fetch course detail from API */
    77→  const fetchCourse = useCallback(async () => {
    78→    if (!selectedCourseId) return;
    79→    setLoading(true);
    80→    setError(null);
    81→    try {
    82→      const res = await fetch(
    83→        `/api/courses/${selectedCourseId}?userId=${userId}`
    84→      );
    85→      if (!res.ok) throw new Error("Failed to load course");
    86→      const json = await res.json();
    87→      if (json.success) {
    88→        setCourse(json.data);
    89→      } else {
    90→        throw new Error(json.error || "Unknown error");
    91→      }
    92→    } catch (err) {
    93→      setError(err instanceof Error ? err.message : "Failed to load course");
    94→    } finally {
    95→      setLoading(false);
    96→    }
    97→  }, [selectedCourseId, userId]);
    98→
    99→  useEffect(() => {
   100→    fetchCourse();
   101→  }, [fetchCourse]);
   102→
   103→  /** Fetch rating data */
   104→  const fetchRatingData = useCallback(async () => {
   105→    if (!selectedCourseId) return;
   106→    try {
   107→      const res = await fetch(
   108→        `/api/ratings?courseId=${selectedCourseId}&userId=${userId}`
   109→      );
   110→      if (res.ok) {
   111→        const json = await res.json();
   112→        if (json.success) {
   113→          setUserRating(json.data.userRating);
   114→          setRatingCount(json.data.count);
   115→          setCourse((prev) =>
   116→            prev ? { ...prev, rating: json.data.average } : prev
   117→          );
   118→        }
   119→      }
   120→    } catch {
   121→      // Best-effort
   122→    }
   123→  }, [selectedCourseId, userId]);
   124→
   125→  useEffect(() => {
   126→    fetchRatingData();
   127→  }, [fetchRatingData]);
   128→
   129→  /** Fetch section progress if enrolled */
   130→  const fetchSectionProgress = useCallback(async () => {
   131→    if (!userId || !course?.isEnrolled) return;
   132→    try {
   133→      const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   134→      const enrollJson = await enrollRes.json();
   135→      if (enrollJson.success && Array.isArray(enrollJson.data)) {
   136→        const enrollment = enrollJson.data.find(
   137→          (e: Record<string, unknown>) => e.courseId === selectedCourseId
   138→        );
   139→        if (enrollment) {
   140→          const progRes = await fetch(`/api/progress?enrollmentId=${enrollment.id}`);
   141→          const progJson = await progRes.json();
   142→          if (progJson.success && Array.isArray(progJson.data)) {
   143→            const mapped: SectionProgress[] = progJson.data.map((p: Record<string, unknown>) => ({
   144→              sectionId: (p.section as Record<string, unknown>)?.id ?? p.sectionId,
   145→              currentPage: (p.currentPage as number) || 0,
   146→              completed: (p.completed as boolean) || false,
   147→            }));
   148→            setSectionProgress(mapped);
   149→          }
   150→        }
   151→      }
   152→    } catch {
   153→      // Progress fetch is best-effort
   154→    }
   155→  }, [userId, course?.isEnrolled, selectedCourseId]);
   156→
   157→  useEffect(() => {
   158→    fetchSectionProgress();
   159→  }, [fetchSectionProgress]);
   160→
   161→  /** Handle enrollment */
   162→  const handleEnroll = async () => {
   163→    if (!selectedCourseId || enrolling) return;
   164→    setEnrolling(true);
   165→    try {
   166→      const res = await fetch("/api/enrollments", {
   167→        method: "POST",
   168→        headers: { "Content-Type": "application/json" },
   169→        body: JSON.stringify({ userId, courseId: selectedCourseId }),
   170→      });
   171→      if (res.ok) {
   172→        setCourse((prev) =>
   173→          prev
   174→            ? { ...prev, isEnrolled: true, studentCount: prev.studentCount + 1 }
   175→            : prev
   176→        );
   177→        toast.success("You're enrolled! Let's start learning.");
   178→      } else {
   179→        toast.error("Failed to enroll. Please try again.");
   180→      }
   181→    } catch {
   182→      toast.error("Network error. Please try again.");
   183→    } finally {
   184→      setEnrolling(false);
   185→    }
   186→  };
   187→
   188→  /** Handle favorite toggle */
   189→  const handleToggleFavorite = async (e: React.MouseEvent) => {
   190→    e.stopPropagation();
   191→    if (!selectedCourseId || togglingFav) return;
   192→    setTogglingFav(true);
   193→    try {
   194→      const res = await fetch("/api/favorites", {
   195→        method: "POST",
   196→        headers: { "Content-Type": "application/json" },
   197→        body: JSON.stringify({ userId, courseId: selectedCourseId }),
   198→      });
   199→      if (res.ok) {
   200→        const json = await res.json();
   201→        setCourse((prev) =>
   202→          prev
   203→            ? { ...prev, isFavorited: json.favorited }
   204→            : prev
   205→        );
   206→      } else {
   207→        toast.error("Failed to update favorite.");
   208→      }
   209→    } catch {
   210→      toast.error("Network error. Please try again.");
   211→    } finally {
   212→      setTogglingFav(false);
   213→    }
   214→  };
   215→
   216→  /** Download entire course as PPTX */
   217→  const handleDownloadCoursePptx = useCallback(async () => {
   218→    if (!course || downloadingPptx) return;
   219→    setDownloadingPptx(true);
   220→    try {
   221→      // Fetch all sections' HTML content
   222→      const sectionHtmlBodies: { title: string; htmlBody: string }[] = [];
   223→      for (const section of course.sections) {
   224→        try {
   225→          const res = await fetch(`/api/sections/${section.id}`);
   226→          if (!res.ok) continue;
   227→          const json = await res.json();
   228→          if (json.success && json.data.htmlBody) {
   229→            sectionHtmlBodies.push({ title: section.title, htmlBody: json.data.htmlBody });
   230→          }
   231→        } catch {
   232→          // Skip sections that fail to load
   233→        }
   234→      }
   235→      if (sectionHtmlBodies.length === 0) {
   236→        toast.error('No content available for download.');
   237→        return;
   238→      }
   239→      const res = await fetch('/api/generate-pptx', {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify({ sections: sectionHtmlBodies, courseName: course.title }),
   243→      });
   244→      if (!res.ok) {
   245→        toast.error('Failed to generate PPT');
   246→        return;
   247→      }
   248→      const blob = await res.blob();
   249→      const url = URL.createObjectURL(blob);
   250→      const a = document.createElement('a');
   251→      a.href = url;
   252→      const safeName = course.title
   253→        .replace(/[^a-zA-Z0-9\s-]/g, '')
   254→        .replace(/\s+/g, '-')
   255→        .toLowerCase()
   256→        .slice(0, 60);
   257→      a.download = `${safeName}.pptx`;
   258→      document.body.appendChild(a);
   259→      a.click();
   260→      a.remove();
   261→      URL.revokeObjectURL(url);
   262→      toast.success('Course PPT downloaded successfully!');
   263→    } catch {
   264→      toast.error('Failed to download course PPT.');
   265→    } finally {
   266→      setDownloadingPptx(false);
   267→    }
   268→  }, [course, downloadingPptx]);
   269→
   270→  /** Handle share button */
   271→  const handleShare = async () => {
   272→    if (navigator.share) {
   273→      try {
   274→        await navigator.share({ title: course?.title, url: window.location.href });
   275→      } catch {
   276→        // User cancelled share
   277→      }
   278→    } else {
   279→      await navigator.clipboard.writeText(window.location.href);
   280→    }
   281→  };
   282→
   283→  /** Handle section click → open classroom */
   284→  const handleSectionClick = async (section: SectionItem) => {
   285→    if (!course) return;
   286→    try {
   287→      const res = await fetch(`/api/sections/${section.id}`);
   288→      if (!res.ok) return;
   289→      const json = await res.json();
   290→      const htmlBody = json.success && json.data.htmlBody
   291→        ? json.data.htmlBody
   292→        : '<div class="flex items-center justify-center h-full"><p class="text-gray-500">No content available.</p></div>';
   293→      const allSectionIds = course.sections.map((s) => s.id);
   294→      const currentSectionIndex = allSectionIds.indexOf(section.id);
   295→      const classroomState: ClassroomState = {
   296→        courseId: course.id,
   297→        courseTitle: course.title,
   298→        sectionId: section.id,
   299→        sectionTitle: section.title,
   300→        htmlBody,
   301→        allSectionIds,
   302→        currentSectionIndex,
   303→      };
   304→      openClassroom(classroomState);
   305→    } catch {
   306→      // Silently fail
   307→    }
   308→  };
   309→
   310→  /** Render star rating */
   311→  const renderStars = (rating: number) => {
   312→    const stars = [];
   313→    for (let i = 1; i <= 5; i++) {
   314→      stars.push(
   315→        <Star
   316→          key={i}
   317→          className={`h-4 w-4 ${
   318→            i <= Math.round(rating)
   319→              ? "fill-amber-400 text-amber-400"
   320→              : "text-muted-foreground/30"
   321→          }`}
   322→        />
   323→      );
   324→    }
   325→    return stars;
   326→  };
   327→
   328→  /** Generate gradient fallback for cover image */
   329→  const getGradientStyle = (): React.CSSProperties => {
   330→    const colors = [
   331→      "linear-gradient(135deg, #0891b2 0%, #06b6d4 100%)",
   332→      "linear-gradient(135deg, #0d9488 0%, #14b8a6 100%)",
   333→      "linear-gradient(135deg, #059669 0%, #10b981 100%)",
   334→      "linear-gradient(135deg, #d97706 0%, #f59e0b 100%)",
   335→      "linear-gradient(135deg, #dc2626 0%, #ef4444 100%)",
   336→      "linear-gradient(135deg, #0891b2 0%, #0d9488 100%)",
   337→    ];
   338→    const index = course ? course.title.charCodeAt(0) % colors.length : 0;
   339→    return { background: colors[index] };
   340→  };
   341→
   342→  /** Format date */
   343→  const formatDate = (dateStr: string) => {
   344→    const d = new Date(dateStr);
   345→    return d.toLocaleDateString("en-US", {
   346→      year: "numeric",
   347→      month: "short",
   348→      day: "numeric",
   349→    });
   350→  };
   351→
   352→  // ─── Loading Skeleton ─────────────────────────────────
   353→  if (loading) {
   354→    return (
   355→      <div className="mx-auto max-w-5xl px-4 py-6 sm:px-6 lg:px-8">
   356→        {/* Hero skeleton (full-width banner) */}
   357→        <Skeleton className="w-full aspect-[21/9] sm:aspect-[3/1] rounded-2xl mb-6" />
   358→
   359→        {/* Description skeleton */}
   360→        <div className="flex items-start gap-3 mb-4">
   361→          <Skeleton className="h-5 w-16 rounded-full" />
   362→          <Skeleton className="h-4 w-2/3" />
   363→        </div>
   364→
   365→        {/* Action bar skeleton */}
   366→        <div className="flex items-center gap-3 mb-8">
   367→          <Skeleton className="h-10 w-10 rounded-lg" />
   368→          <Skeleton className="h-10 w-10 rounded-lg" />
   369→          <Skeleton className="ml-auto h-12 w-44 rounded-lg" />
   370→        </div>
   371→
   372→        {/* Curriculum skeleton */}
   373→        <Skeleton className="h-6 w-32 mb-4" />
   374→        <div className="space-y-3">
   375→          {[1, 2, 3].map((i) => (
   376→            <Skeleton key={i} className="h-16 w-full rounded-lg" />
   377→          ))}
   378→        </div>
   379→      </div>
   380→    );
   381→  }
   382→
   383→  // ─── Error State ───────────────────────────────────────
   384→  if (error || !course) {
   385→    return (
   386→      <div className="mx-auto max-w-5xl px-4 py-16 sm:px-6 lg:px-8 text-center">
   387→        <p className="text-muted-foreground text-lg mb-4">{error || "Course not found"}</p>
   388→        <Button variant="outline" onClick={goBack}>
   389→          <ChevronLeft className="h-4 w-4" />
   390→          Go Back
   391→        </Button>
   392→      </div>
   393→    );
   394→  }
   395→
   396→  const totalLessons = course.sections.reduce((sum, s) => sum + s.totalPages, 0);
   397→  const estimatedMinutes = Math.max(5, Math.round(totalLessons * 1.5));
   398→
   399→  /** Get progress info for a section */
   400→  const getSectionProgress = (sectionId: string) => {
   401→    return sectionProgress.find((p) => p.sectionId === sectionId);
   402→  };
   403→
   404→  /** Get section status badge */
   405→  const getSectionStatus = (section: SectionItem) => {
   406→    if (!course.isEnrolled) return null;
   407→    const prog = getSectionProgress(section.id);
   408→    if (!prog) return "not-started";
   409→    if (prog.completed) return "completed";
   410→    return "in-progress";
   411→  };
   412→
   413→  return (
   414→    <main className="mx-auto max-w-5xl px-4 py-6 sm:px-6 lg:px-8">
   415→      {/* ─── Course Hero (Full-Width Image + Overlay) ── */}
   416→      <div className="relative w-full aspect-[21/9] sm:aspect-[3/1] rounded-2xl overflow-hidden mb-6">
   417→        {/* Cover Image / Gradient Fallback */}
   418→        <div className="absolute inset-0">
   419→          {!imgError && course.coverImage ? (
   420→            <img
   421→              src={course.coverImage}
   422→              alt={course.title}
   423→              className="h-full w-full object-cover"
   424→              onError={() => setImgError(true)}
   425→            />
   426→          ) : (
   427→            <div
   428→              className="h-full w-full"
   429→              style={getGradientStyle()}
   430→            />
   431→          )}
   432→        </div>
   433→
   434→        {/* Gradient Overlay (bottom → transparent) */}
   435→        <div className="absolute inset-0 bg-gradient-to-t from-black/70 via-black/30 to-transparent" />
   436→
   437→        {/* Content overlayed on image */}
   438→        <div className="relative h-full flex flex-col justify-end p-5 sm:p-8">
   439→          {/* Breadcrumb overlayed on image */}
   440→          <Breadcrumb className="mb-3 [&_ol]:flex-nowrap">
   441→            <BreadcrumbList className="text-white/80">
   442→              <BreadcrumbItem>
   443→                <BreadcrumbLink
   444→                  className="cursor-pointer hover:text-white transition-colors text-white/80 hover:text-white"
   445→                  onClick={() => navigateTo("home")}
   446→                >
   447→                  Home
   448→                </BreadcrumbLink>
   449→              </BreadcrumbItem>
   450→              <BreadcrumbSeparator className="text-white/50" />
   451→              <BreadcrumbItem>
   452→                <BreadcrumbLink
   453→                  className="cursor-pointer hover:text-white transition-colors text-white/80 hover:text-white"
   454→                  onClick={() => navigateTo("courses")}
   455→                >
   456→                  All Courses
   457→                </BreadcrumbLink>
   458→              </BreadcrumbItem>
   459→              <BreadcrumbSeparator className="text-white/50" />
   460→              <BreadcrumbItem>
   461→                <BreadcrumbPage className="font-medium truncate max-w-[180px] sm:max-w-[280px] text-white">
   462→                  {course.title}
   463→                </BreadcrumbPage>
   464→              </BreadcrumbItem>
   465→            </BreadcrumbList>
   466→          </Breadcrumb>
   467→
   468→          {/* Title on image */}
   469→          <h1 className="text-2xl sm:text-3xl lg:text-4xl font-bold text-white leading-tight mb-3 [text-shadow:0_2px_8px_rgba(0,0,0,0.3)]">
   470→            {course.title}
   471→          </h1>
   472→
   473→          {/* Rating + Stats on image */}
   474→          <div className="flex flex-wrap items-center gap-x-4 gap-y-2 text-sm text-white/90">
   475→            {/* Rating with prominent number */}
   476→            <div className="flex items-center gap-1.5">
   477→              <span className="text-xl sm:text-2xl font-bold text-white [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   478→                {course.rating.toFixed(1)}
   479→              </span>
   480→              <div className="flex items-center gap-0.5">
   481→                {renderStars(course.rating)}
   482→              </div>
   483→              {ratingCount > 0 && (
   484→                <span className="text-xs text-white/60 ml-0.5">
   485→                  ({ratingCount})
   486→                </span>
   487→              )}
   488→            </div>
   489→
   490→            <span className="w-px h-4 bg-white/30" />
   491→
   492→            {/* Students count */}
   493→            <span className="inline-flex items-center gap-1.5 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   494→              <Users className="h-4 w-4" />
   495→              <span className="font-medium">{course.studentCount.toLocaleString()}</span>
   496→              <span className="text-white/70">students</span>
   497→            </span>
   498→
   499→            <span className="w-px h-4 bg-white/30" />
   500→
   501→            {/* Lesson count */}
   502→            <span className="inline-flex items-center gap-1.5 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   503→              <BookOpen className="h-4 w-4" />
   504→              <span className="font-medium">{totalLessons}</span>
   505→              <span className="text-white/70">lessons</span>
   506→            </span>
   507→
   508→            <span className="w-px h-4 bg-white/30" />
   509→
   510→            {/* Estimated duration */}
   511→            <span className="inline-flex items-center gap-1.5 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   512→              <Clock className="h-4 w-4" />
   513→              <span className="font-medium">~{estimatedMinutes} min</span>
   514→              <span className="text-white/70">estimated</span>
   515→            </span>
   516→
   517→            {/* Updated date */}
   518→            <span className="inline-flex items-center gap-1.5 text-white/70 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   519→              <Calendar className="h-4 w-4" />
   520→              {formatDate(course.updatedAt)}
   521→            </span>
   522→          </div>
   523→        </div>
   524→      </div>
   525→
   526→      {/* ─── Description + Language ─────────────────── */}
   527→      <div className="flex items-start gap-3 mb-2">
   528→        {course.language && (
   529→          <Badge variant="secondary" className="shrink-0 mt-0.5">
   530→            {course.language}
   531→          </Badge>
   532→        )}
   533→        {course.description && (
   534→          <p className="text-sm text-muted-foreground leading-relaxed">
   535→            {course.description}
   536→          </p>
   537→        )}
   538→      </div>
   539→
   540→      {/* ─── Interactive Star Rating ─────────────── */}
   541→      {userId && (
   542→        <div className="mb-4">
   543→          <StarRating
   544→            courseId={course.id}
   545→            userId={userId}
   546→            currentRating={userRating}
   547→            averageRating={course.rating}
   548→            ratingCount={ratingCount}
   549→            onRate={(data) => {
   550→              setUserRating(data.score);
   551→              setRatingCount(data.count);
   552→              setCourse((prev) =>
   553→                prev ? { ...prev, rating: data.average } : prev
   554→              );
   555→            }}
   556→          />
   557→        </div>
   558→      )}
   559→
   560→      {/* ─── Action Bar ──────────────────────────── */}
   561→      <Separator className="my-4" />
   562→      <div className="flex flex-wrap items-center gap-3 mb-8">
   563→        {/* Grouped: Favorite + Share */}
   564→        <div className="flex items-center gap-2">
   565→          <Button
   566→            variant="outline"
   567→            size="icon"
   568→            className="h-10 w-10"
   569→            onClick={handleToggleFavorite}
   570→            disabled={togglingFav}
   571→            aria-label={course.isFavorited ? "Remove from favorites" : "Add to favorites"}

   573→            <Heart
   574→              className={`h-4 w-4 transition-colors ${
   575→                course.isFavorited
   576→                  ? "fill-red-500 text-red-500"
   577→                  : "text-muted-foreground"
   578→              }`}
   579→            />
   580→          </Button>
   581→          <Button
   582→            variant="outline"
   583→            size="icon"
   584→            className="h-10 w-10"
   585→            onClick={handleShare}
   586→            aria-label="Share course"

   590→        </div>

   592→        {/* Download as PPT - Outline Button */}
   593→        <Button
   594→          variant="outline"
   595→          size="lg"
   596→          className="gap-2 font-medium text-sm px-5 py-6"
   597→          onClick={handleDownloadCoursePptx}
   598→          disabled={downloadingPptx || course.sections.length === 0}

   601→            <Loader2 className="h-4 w-4 animate-spin" />

   603→            <FileDown className="h-4 w-4" />

   611→            size="lg"

   614→                ? "bg-primary hover:bg-primary/90 text-primary-foreground"
   615→                : "bg-gradient-to-r from-primary to-accent hover:opacity-90 text-primary-foreground pulse-glow"

   617→            onClick={

   619→                ? () => {
   620→                    if (course.sections.length > 0) {
   621→                      handleSectionClick(course.sections[0]);

   623→                  }
   624→                : handleEnroll
   625→            }
   626→            disabled={enrolling}

   629→              <Loader2 className="h-5 w-5 animate-spin" />

   631→              <PlayCircle className="h-5 w-5" />
   632→            ) : (

   635→            {course.isEnrolled ? "Continue Learning" : "Start Learning"}

   709→                          <span className="text-xs text-muted-foreground">
   710→                            {totalP} pages
   711→                          </span>
   712→                          {/* Progress indicator if enrolled */}
   713→                          {course.isEnrolled && status && status !== "not-started" && (
   714→                            <span className="text-xs text-muted-foreground">
   715→                              ✓ {completedPages}/{totalP} pages
   716→                            </span>
   717→                          )}
   718→                          {course.isEnrolled && status === "not-started" && (
   719→                            <span className="text-xs text-muted-foreground">
   720→                              0/{totalP} pages
   721→                            </span>
   722→                          )}
   723→                        </div>
   724→                        {/* Mini progress bar if enrolled */}
   725→                        {course.isEnrolled && (
   726→                          <Progress
   727→                            value={progressPct}
   728→                            className="mt-1.5 h-1"
   729→                          />
   730→                        )}
   731→                      </div>
   732→                    </div>
   733→                  </AccordionTrigger>
   734→                  <AccordionContent className="pb-4">
   735→                    <p className="text-sm text-muted-foreground pl-11">
   736→                      Click &quot;{section.title}&quot; to open the lesson viewer and start learning.
   737→                    </p>
   738→                  </AccordionContent>
   739→                </AccordionItem>
   740→              );
   741→            })}
   742→          </Accordion>
   743→        )}
   744→      </section>
   745→
   746→      {/* ─── Progress Timeline ──────────────────── */}
   747→      {course.isEnrolled && (
   748→        <section className="mt-8">
   749→          <ProgressTimeline
   750→            courseId={course.id}
   751→            courseTitle={course.title}
   752→            userId={userId}
   753→            isEnrolled={course.isEnrolled}
   754→            sections={course.sections}
   755→            openClassroom={openClassroom}
   756→            course={course}
   757→          />
   758→        </section>
   759→      )}
   760→
   761→      <section className="mt-8">
   762→        <DiscussionPanel
   763→          courseId={course.id}
   764→          userId={userId}
   765→        />
   766→      </section>
   767→      {/* ─── Browser Warning ─────────────────────── */}
   768→      <div className="mt-8 flex items-start gap-3 rounded-lg border border-amber-200 bg-amber-50 p-3 text-sm text-amber-800 dark:border-amber-800/60 dark:bg-amber-950/40 dark:text-amber-200">
   769→        <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400" />
   770→        <p>
   771→          For the best learning experience, we recommend using the{" "}
   772→          <span className="font-semibold">Google Chrome</span> browser.
   773→        </p>
   774→      </div>
   775→    </main>
   776→  );
