   253→const MAX_SECTIONS = 10;
   254→const MAX_TITLE_LENGTH = 100;
   255→const MAX_DESC_LENGTH = 3000;
   256→const MAX_COURSE_DESC_LENGTH = 500;
   257→
   258→export function CreateCoursePage() {
   259→  const { goBack } = useNavigationStore();
   260→  const { currentUserId } = useUserStore();
   261→  const { createPrompt, setCreatePrompt } = useCourseStore();
   262→
   263→  // ---- Form state ----
   264→  const [title, setTitle] = useState("");
   265→  const [description, setDescription] = useState("");
   266→  const [categoryId, setCategoryId] = useState("");
   267→  const [language, setLanguage] = useState("english");
   268→  const [coverImage, setCoverImage] = useState("");
   269→  const [coverPreview, setCoverPreview] = useState<string | null>(null);
   270→  const [sections, setSections] = useState<SectionDraft[]>([]);
   271→
   272→  // ---- UI state ----
   273→  const [categories, setCategories] = useState<CategoryItem[]>([]);
   274→  const [loading, setLoading] = useState(false);
   275→  const [saving, setSaving] = useState(false);
   276→  const [generating, setGenerating] = useState(false);
   277→  const [modalOpen, setModalOpen] = useState(false);
   278→  const [urlInputOpen, setUrlInputOpen] = useState(false);
   279→  const [imageUrl, setImageUrl] = useState("");
   280→
   281→  // ---- Modal state ----
   282→  const [sectionName, setSectionName] = useState("");
   283→  const [sectionPrompt, setSectionPrompt] = useState("");
   284→  const [sectionLanguage, setSectionLanguage] = useState("english");
   285→  const [sectionPdfName, setSectionPdfName] = useState("");
   286→
   287→  // ---- Pre-fill title from hero prompt ----
   288→  useEffect(() => {
   289→    if (createPrompt) {
   290→      setTitle(createPrompt);
   291→      setCreatePrompt("");
   292→    }
   293→  }, [createPrompt, setCreatePrompt]);
   294→
   295→  // ---- Refs ----
   296→  const fileInputRef = useRef<HTMLInputElement>(null);
   297→  const sectionFileInputRef = useRef<HTMLInputElement>(null);
   298→
   299→  // ---- Fetch categories ----
   300→  useEffect(() => {
   301→    async function fetchCategories() {
   302→      setLoading(true);
   303→      try {
   304→        const res = await fetch("/api/categories");
   305→        const json = await res.json();
   306→        if (json.success) {
   307→          setCategories(json.data);
   308→        }
   309→      } catch {
   310→        toast.error("Failed to load categories");
   311→      } finally {
   312→        setLoading(false);
   313→      }
   314→    }
   315→    fetchCategories();
   316→  }, []);
   317→
   318→  // ---- Cover image handlers ----
   319→  const handleCoverUpload = () => {
   320→    // In sandbox, simulate file selection
   321→    setUrlInputOpen(true);
   322→  };
   323→
   324→  const handleUrlConfirm = () => {
   325→    if (imageUrl.trim()) {
   326→      setCoverImage(imageUrl.trim());
   327→      setCoverPreview(imageUrl.trim());
   328→      setUrlInputOpen(false);
   329→      setImageUrl("");
   330→    }
   331→  };
   332→
   333→  const handleRemoveCover = () => {
   334→    setCoverImage("");
   335→    setCoverPreview(null);
   336→  };
   337→
   338→  // ---- Section management ----
   339→  const handleOpenModal = () => {
   340→    setSectionName("");
   341→    setSectionPrompt("");
   342→    setSectionLanguage(language);
   343→    setSectionPdfName("");
   344→    setModalOpen(true);
   345→  };
   346→
   347→  const handleGenerateSection = useCallback(async () => {
   348→    if (!sectionName.trim()) {
   349→      toast.error("Section name is required");
   350→      return;
   351→    }
   352→    if (!sectionPrompt.trim()) {
   353→      toast.error("Course prompt is required");
   354→      return;
   355→    }
   356→    if (sections.length >= MAX_SECTIONS) {
   357→      toast.error(`Maximum ${MAX_SECTIONS} sections allowed`);
   358→      return;
   359→    }
   360→
   361→    setGenerating(true);
   362→
   363→    try {
   364→      const res = await fetch("/api/generate-content", {
   365→        method: "POST",
   366→        headers: { "Content-Type": "application/json" },
   367→        body: JSON.stringify({
   368→          topic: sectionName.trim(),
   369→          prompt: sectionPrompt.trim(),
   370→          language: sectionLanguage,
   371→        }),
   372→      });
   373→
   374→      const json = await res.json();
   375→
   376→      if (!json.success) {
   377→        toast.error(json.error || "Failed to generate section content");
   378→        setGenerating(false);
   379→        return;
   380→      }
   381→
   382→      const slides: SlideContent[] = json.data;
   383→      const newSection: SectionDraft = {
   384→        id: `sec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   385→        title: sectionName.trim(),
   386→        totalPages: slides.length,
   387→        content: JSON.stringify(slides),
   388→        language: sectionLanguage,
   389→      };
   390→
   391→      setSections((prev) => [...prev, newSection]);
   392→      setModalOpen(false);
   393→      toast.success(`Section "${sectionName.trim()}" generated successfully`);
   394→    } catch {
   395→      toast.error("Failed to generate section. Please try again.");
   396→    } finally {
   397→      setGenerating(false);
   398→    }
   399→  }, [sectionName, sectionPrompt, sectionLanguage, sections.length]);
   400→
   401→  const handleDeleteSection = (id: string) => {
   402→    setSections((prev) => prev.filter((s) => s.id !== id));
   403→  };
   404→
   405→  const handleMoveSection = (index: number, direction: "up" | "down") => {
   406→    const newSections = [...sections];
   407→    const targetIndex = direction === "up" ? index - 1 : index + 1;
   408→    if (targetIndex < 0 || targetIndex >= newSections.length) return;
   409→    [newSections[index], newSections[targetIndex]] = [
   410→      newSections[targetIndex],
   411→      newSections[index],
   412→    ];
   413→    setSections(newSections);
   414→  };
   415→
   416→  // ---- Form submission ----
   417→  const handleSave = async () => {
   418→    if (!title.trim()) {
   419→      toast.error("Course title is required");
   420→      return;
   421→    }
   422→
   423→    setSaving(true);
   424→    try {
   425→      const payload = {
   426→        title: title.trim(),
   427→        description: description.trim() || null,
   428→        categoryId: categoryId || null,
   429→        language,
   430→        creatorId: currentUserId,
   431→        coverImage: coverImage || null,
   432→        sections: sections.map((sec, index) => ({
   433→          title: sec.title,
   434→          content: sec.content,
   435→          totalPages: sec.totalPages,
   436→          order: index,
   437→        })),
   438→      };
   439→
   440→      const res = await fetch("/api/courses", {
   441→        method: "POST",
   442→        headers: { "Content-Type": "application/json" },
   443→        body: JSON.stringify(payload),
   444→      });
   445→
   446→      const json = await res.json();
   447→
   448→      if (json.success) {
   449→        toast.success("Course created successfully!");
   450→        goBack();
   451→      } else {
   452→        toast.error(json.error || "Failed to create course");
   453→      }
   454→    } catch {
   455→      toast.error("Failed to create course. Please try again.");
   456→    } finally {
   457→      setSaving(false);
   458→    }
   459→  };
   460→
   461→  // ---- Section PDF upload simulation ----
   462→  const handleSectionPdfUpload = () => {
   463→    sectionFileInputRef.current?.click();
   464→  };
   465→
   466→  const handleSectionFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   467→    const file = e.target.files?.[0];
   468→    if (file) {
   469→      setSectionPdfName(file.name);
   470→      // If no prompt yet, use the filename
   471→      if (!sectionPrompt.trim()) {
   472→        const nameWithoutExt = file.name.replace(/\.pdf$/i, "");
   473→        setSectionPrompt(
   474→          `Generate course content based on the uploaded PDF: ${nameWithoutExt}`
   475→        );
   476→        if (!sectionName.trim()) {
   477→          setSectionName(nameWithoutExt);
   478→        }
   479→      }
   480→    }
   481→  };
   482→
   483→  // ============================================
   484→  // Render
   485→  // ============================================
   486→
   487→  return (
   488→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   489→      {/* ---- Breadcrumbs ---- */}
   490→      <div className="mb-6 flex items-center gap-3">
   491→        <button
   492→          onClick={goBack}
   493→          className="flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-foreground"
   494→        >
   495→          <ArrowLeft className="h-4 w-4" />
   496→          Back
   497→        </button>
   498→        <Separator orientation="vertical" className="h-4" />
   499→        <h1 className="text-lg font-semibold text-foreground">Create Course</h1>
   500→      </div>
   501→
   502→      {/* ---- Two-Column Layout ---- */}
   503→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-5">
   504→        {/* ======== Left Panel: Course Metadata (3 cols) ======== */}
   505→        <div className="lg:col-span-3">
   506→          <Card className="border-border/50">
   507→            <CardContent className="space-y-6 p-6">
   508→              {/* Cover Image Upload */}
   509→              <div className="space-y-2">
   510→                <Label className="text-sm font-medium">Cover Image</Label>
   511→                {coverPreview ? (
   512→                  <div className="relative overflow-hidden rounded-lg border border-border">
   513→                    <img
   514→                      src={coverPreview}
   515→                      alt="Course cover"
   516→                      className="h-48 w-full object-cover"
   517→                      onError={() => setCoverPreview(null)}

   520→                      onClick={handleRemoveCover}
   521→                      className="absolute top-2 right-2 flex h-7 w-7 items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80"

   526→                ) : (
   527→                  <button
   528→                    type="button"
   529→                    onClick={handleCoverUpload}
   530→                    className="flex h-48 w-full flex-col items-center justify-center gap-3 rounded-lg border-2 border-dashed border-border/70 bg-muted/30 transition-colors hover:border-primary/50 hover:bg-muted/50"
   531→                  >
   532→                    <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted">
   533→                      <ImageIcon className="h-5 w-5 text-muted-foreground" />
   534→                    </div>
   535→                    <div className="text-center">
   536→                      <p className="text-sm font-medium text-foreground">
   537→                        Upload cover image
   538→                      </p>
   539→                      <p className="mt-0.5 text-xs text-muted-foreground">
   540→                        Recommended 1920x1080
   541→                      </p>
   542→                    </div>
   543→                  </button>

   548→                  <div className="flex items-center gap-2">

   555→                    />
   556→                    <Button size="sm" onClick={handleUrlConfirm}>
   557→                      Apply
   558→                    </Button>
   559→                    <Button
   560→                      size="sm"
   561→                      variant="ghost"
   562→                      onClick={() => {
   563→                        setUrlInputOpen(false);
   564→                        setImageUrl("");
   565→                      }}
   566→                    >

   579→              <Separator />

   689→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />

   693→                  "Save Course"

   696→            </CardContent>

   700→        {/* ======== Right Panel: Sections (2 cols) ======== */}
   701→        <div className="lg:col-span-2">
   702→          <Card className="border-border/50">

   704→              {/* Header */}
   705→              <div className="mb-4 flex items-center justify-between">
   706→                <div className="flex items-center gap-2">

   714→                    {sections.length}/{MAX_SECTIONS}

   718→                  size="sm"
   719→                  variant="outline"

   729→              {/* Section List */}

   818→          <div className="space-y-5">

   854→                onChange={(e) => {

   874→                  <span className="text-foreground font-medium">
   875→                    {sectionPdfName}

   889→
   890→            {/* Language */}
   891→            <div className="space-y-3">
   892→              <Label className="text-sm font-medium">
   893→                Generated Content Language
   894→              </Label>
   895→              <RadioGroup
   896→                value={sectionLanguage}
   897→                onValueChange={setSectionLanguage}
   898→                className="flex gap-6"
   899→              >
   900→                <div className="flex items-center gap-2">
   901→                  <RadioGroupItem
   902→                    value="chinese"
   903→                    id="sec-lang-zh"
   904→                  />
   905→                  <Label
   906→                    htmlFor="sec-lang-zh"
   907→                    className="cursor-pointer text-sm font-normal"
   908→                  >
   909→                    Chinese
   910→                  </Label>
   911→                </div>
   912→                <div className="flex items-center gap-2">
   913→                  <RadioGroupItem
   914→                    value="english"
   915→                    id="sec-lang-en"
   916→                  />
   917→                  <Label
   918→                    htmlFor="sec-lang-en"
   919→                    className="cursor-pointer text-sm font-normal"
   920→                  >
   921→                    English
   922→                  </Label>

   926→          </div>

   928→          <DialogFooter className="mt-2">

   930→              variant="ghost"

   934→              Cancel
   935→            </Button>
   936→            <Button
   937→              onClick={handleGenerateSection}
   938→              disabled={generating || !sectionName.trim() || !sectionPrompt.trim()}
   939→              className="gap-2"
   940→            >
   941→              {generating ? (
   942→                <>
   943→                  <Loader2 className="h-4 w-4 animate-spin" />
   944→                  Generating...
   945→                </>
   946→              ) : (
   947→                "Generate"
   948→              )}
   949→            </Button>
   950→          </DialogFooter>
   951→        </DialogContent>
   952→      </Dialog>
   953→    </div>
   954→  );
