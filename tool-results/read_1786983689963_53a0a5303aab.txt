   655→function PasswordStrength({ password }: { password: string }) {
   656→  const score = getStrengthScore(password);
   657→  const label = STRENGTH_LEVELS[score].label;
   658→  const color = STRENGTH_LEVELS[score].color;
   659→
   660→  return (
   661→    <div className="space-y-1.5">
   662→      <div className="flex gap-1">
   663→        {Array.from({ length: 4 }).map((_, i) => (
   664→          <div
   665→            key={i}
   666→            className="h-1 flex-1 rounded-full transition-colors duration-300"
   667→            style={{
   668→              backgroundColor: i < score ? color : "hsl(var(--muted))",
   669→            }}
   670→          />
   671→        ))}
   672→      </div>
   673→      <p className="text-xs text-muted-foreground">{label}</p>
   674→    </div>
   675→  );
   676→}

   678→/* -------------------------------------------------------------------------- */
   679→/*  Utilities                                                                 */
   680→/* -------------------------------------------------------------------------- */
   681→
   682→function getStrengthScore(pw: string): number {
   683→  let score = 0;
   684→  if (pw.length >= 8) score++;
   685→  if (/[A-Z]/.test(pw) && /[a-z]/.test(pw)) score++;
   686→  if (/[0-9]/.test(pw)) score++;
   687→  if (/[^A-Za-z0-9]/.test(pw)) score++;
   688→  return score;
   689→}
   690→
   691→const STRENGTH_LEVELS: Record<
   692→  number,
   693→  { label: string; color: string }
   694→> = {
   695→  0: { label: "Too short", color: "hsl(var(--destructive))" },
   696→  1: { label: "Weak", color: "hsl(var(--destructive))" },
   697→  2: { label: "Fair", color: "hsl(38 92% 50%)" },
   698→  3: { label: "Good", color: "hsl(var(--primary))" },
   699→  4: { label: "Strong", color: "hsl(142 71% 45%)" },
   700→};
