   648→              size="sm"

   650→                isLast || navigating

   657→              <span className="hidden sm:inline">Next</span>

   693→      {/* ─── Study Timer (floating panel) ────────── */}

   831→// Confetti Celebration Component

   833→
   834→/** LMS color palette for confetti particles */
   835→const CONFETTI_COLORS = [
   836→  "#3882f6", // blue
   837→  "#14b8a6", // teal
   838→  "#22d3ee", // cyan
   839→  "#10b981", // emerald
   840→  "#f59e0b", // amber
   841→  "#6366f1", // indigo accent
   842→  "#8b5cf6", // violet accent
   843→  "#ec4899", // pink accent
   844→];
   845→
   846→type ParticleShape = "circle" | "rectangle" | "triangle";
   847→
   848→interface ConfettiParticle {
   849→  id: number;
   850→  left: number;
   851→  color: string;
   852→  shape: ParticleShape;
   853→  size: number;
   854→  fallDuration: string;
   855→  fallDelay: string;
   856→  drift: string;
   857→  spin: string;
   858→}
   859→
   860→function ConfettiCelebration() {
   861→  const particles = useMemo<ConfettiParticle[]>(() => {
   862→    const result: ConfettiParticle[] = [];
   863→    const shapes: ParticleShape[] = ["circle", "rectangle", "triangle"];
   864→    const count = 80;
   865→
   866→    for (let i = 0; i < count; i++) {
   867→      const shape =
   868→        shapes[i % 3 === 0 ? 0 : i % 3 === 1 ? 1 : 2];
   869→      result.push({
   870→        id: i,
   871→        left: Math.random() * 100,
   872→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   873→        shape,
   874→        size: Math.floor(Math.random() * 8) + 6, // 6-14px
   875→        fallDuration: `${(Math.random() * 1.5 + 2).toFixed(2)}s`, // 2-3.5s
   876→        fallDelay: `${(Math.random() * 0.8).toFixed(2)}s`, // 0-0.8s
   877→        drift: `${(Math.random() * 120 - 60).toFixed(0)}px`, // -60 to +60px
   878→        spin: `${Math.floor(Math.random() * 720 + 360)}deg`, // 360-1080deg
   879→      });
   880→    }
   881→    return result;
   882→  }, []);
   883→
   884→  return (
   885→    <>
   886→      {/* Confetti particles layer */}
   887→      <div className="confetti-container" aria-hidden="true">
   888→        {particles.map((p) => {
   889→          if (p.shape === "triangle") {
   890→            return (
   891→              <div
   892→                key={p.id}
   893→                className="confetti-particle confetti-triangle"
   894→                style={{
   895→                  left: `${p.left}%`,
   896→                  "--confetti-color": p.color,
   897→                  "--tri-size": `${p.size}px`,
   898→                  "--fall-duration": p.fallDuration,
   899→                  "--fall-delay": p.fallDelay,
   900→                  "--drift": p.drift,
   901→                  "--spin": p.spin,
   902→                } as React.CSSProperties}
   903→              />
   904→            );
   905→          }
   906→          return (
   907→            <div
   908→              key={p.id}
   909→              className={`confetti-particle ${
   910→                p.shape === "circle"
   911→                  ? "confetti-circle"
   912→                  : "confetti-rectangle"
   913→              }`}
   914→              style={{
   915→                left: `${p.left}%`,
   916→                width:
   917→                  p.shape === "rectangle"
   918→                    ? `${p.size * 1.4}px`
   919→                    : `${p.size}px`,
   920→                height: `${p.size}px`,
   921→                backgroundColor: p.color,
   922→                "--fall-duration": p.fallDuration,
   923→                "--fall-delay": p.fallDelay,
   924→                "--drift": p.drift,
   925→                "--spin": p.spin,
   926→              } as React.CSSProperties}
   927→            />
   928→          );
   929→        })}
   930→      </div>
   931→
   932→      {/* Congratulations message */}
   933→      <div
   934→        className="confetti-message"
   935→        role="status"
   936→        aria-label="Congratulations!"
   937→      >
   938→        <div className="confetti-message-text flex flex-col items-center gap-3">
   939→          <div className="flex h-16 w-16 items-center justify-center rounded-2xl bg-white/90 dark:bg-zinc-800/90 shadow-lg">
   940→            <span className="text-4xl">🎉</span>
   941→          </div>
   942→          <div className="rounded-2xl bg-white/90 dark:bg-zinc-800/90 px-8 py-5 shadow-xl text-center">
   943→            <h2 className="text-2xl sm:text-3xl font-bold text-foreground">
   944→              🎉 Congratulations!
   945→            </h2>
   946→            <p className="mt-2 text-sm text-muted-foreground">
   947→              You&apos;ve completed this section!
   948→            </p>
   949→          </div>
   950→        </div>
   951→      </div>
   952→    </>
   953→  );
   954→}
