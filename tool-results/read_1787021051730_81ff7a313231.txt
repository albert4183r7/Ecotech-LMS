     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import {
     5→  ZoomIn,
     6→  ZoomOut,
     7→  Maximize2,
     8→  RotateCcw,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  X,
    12→  Loader2,
    13→  StickyNote,
    14→  Bookmark,
    15→  BookmarkCheck,
    16→  Trash2,
    17→  Send,
    18→  FileDown,
    19→  Wand2,
    20→} from "lucide-react";
    21→import { toast } from "sonner";
    22→import { Button } from "@/components/ui/button";
    23→import { Separator } from "@/components/ui/separator";
    24→import { Tooltip, TooltipTrigger, TooltipContent } from "@/components/ui/tooltip";
    25→import {
    26→  Sheet,
    27→  SheetContent,
    28→  SheetHeader,
    29→  SheetTitle,
    30→  SheetDescription,
    31→} from "@/components/ui/sheet";
    32→import { ScrollArea } from "@/components/ui/scroll-area";
    33→import { Textarea } from "@/components/ui/textarea";
    34→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    35→import type { ClassroomState } from "@/types/lms";
    36→import { StudyTimer } from "@/components/lms/study-timer";
    37→
    38→const MIN_ZOOM = 50;
    39→const MAX_ZOOM = 200;
    40→const ZOOM_STEP = 25;
    41→
    42→export function ClassroomPage() {
    43→  const { classroomState, goBack } = useNavigationStore();
    44→  const userId = useUserStore((s) => s.currentUserId);
    45→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    46→  const [zoom, setZoom] = useState(100);
    47→  const [navigating, setNavigating] = useState(false);
    48→  const confettiShownRef = useRef(false);
    49→  const [showConfetti, setShowConfetti] = useState(false);
    50→
    51→  // ─── Notes Sidebar State ──────────────────────
    52→  interface Note {
    53→    id: string;
    54→    content: string;
    55→    slideNumber: number;
    56→    bookmarked: boolean;
    57→    createdAt?: string;
    58→  }
    59→  const [notes, setNotes] = useState<Note[]>([]);
    60→  const [notesSidebarOpen, setNotesSidebarOpen] = useState(false);
    61→  const [newNoteContent, setNewNoteContent] = useState("");
    62→  const [savingNote, setSavingNote] = useState(false);
    63→  const notesEndRef = useRef<HTMLDivElement>(null);
    64→
    65→  // ─── Keyboard Hint Fade ────────────────────────
    66→  const [showKeyboardHint, setShowKeyboardHint] = useState(true);
    67→  const [isMobile, setIsMobile] = useState(false);
    68→  const [downloadingPptx, setDownloadingPptx] = useState(false);
    69→
    70→  // ─── AI Edit State ─────────────────────────────
    71→  const [showAiEdit, setShowAiEdit] = useState(false);
    72→  const [aiEditInstruction, setAiEditInstruction] = useState("");
    73→  const [aiEditLoading, setAiEditLoading] = useState(false);
    74→
    75→  useEffect(() => {
    76→    const mq = window.matchMedia("(max-width: 1023px)");
    77→    setIsMobile(mq.matches);
    78→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
    79→    mq.addEventListener("change", handler);
    80→    return () => mq.removeEventListener("change", handler);
    81→  }, []);
    82→
    83→  // Keep a working copy of classroomState
    84→  useEffect(() => {
    85→    if (classroomState) {
    86→      setLocalState({ ...classroomState });
    87→      setZoom(100);
    88→      confettiShownRef.current = false;
    89→      setShowConfetti(false);
    90→    }
    91→  }, [classroomState]);
    92→
    93→  // ─── Section Navigation ─────────────────────────
    94→  const goToSection = useCallback(
    95→    async (index: number) => {
    96→      if (!localState || index < 0 || index >= localState.allSectionIds.length) return;
    97→      const sectionId = localState.allSectionIds[index];
    98→      try {
    99→        setNavigating(true);
   100→        const res = await fetch(`/api/sections/${sectionId}`);
   101→        if (!res.ok) return;
   102→        const json = await res.json();
   103→        const htmlBody =
   104→          json.success && json.data.htmlBody
   105→            ? json.data.htmlBody
   106→            : '<div class="flex items-center justify-center h-full"><p class="text-gray-500">No content available.</p></div>';
   107→        const sectionTitle =
   108→          json.success && json.data.title
   109→            ? json.data.title
   110→            : `Section ${index + 1}`;
   111→        setLocalState((prev) =>
   112→          prev
   113→            ? {
   114→                ...prev,
   115→                sectionId,
   116→                sectionTitle,
   117→                htmlBody,
   118→                currentSectionIndex: index,
   119→              }
   120→            : prev
   121→        );
   122→      } catch {
   123→        /* silently fail */
   124→      } finally {
   125→        setNavigating(false);
   126→      }
   127→    },
   128→    [localState]
   129→  );
   130→
   131→  /** Mark section as completed and save progress */
   132→  const markSectionCompleted = useCallback(
   133→    async (sectionId: string) => {
   134→      if (!userId || !localState) return;
   135→      try {
   136→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   137→        const enrollJson = await enrollRes.json();
   138→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   139→          const enrollment = enrollJson.data.find(
   140→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   141→          );
   142→          if (enrollment) {
   143→            await fetch("/api/progress", {
   144→              method: "POST",
   145→              headers: { "Content-Type": "application/json" },
   146→              body: JSON.stringify({
   147→                enrollmentId: enrollment.id,
   148→                sectionId,
   149→                currentPage: 1,
   150→                completed: true,
   151→              }),
   152→            });
   153→          }
   154→        }
   155→      } catch {
   156→        /* best-effort */
   157→      }
   158→    },
   159→    [userId, localState]
   160→  );
   161→
   162→  // Save progress when section changes
   163→  useEffect(() => {
   164→    if (!localState || !userId) return;
   165→    const timer = setTimeout(() => {
   166→      markSectionCompleted(localState.sectionId);
   167→    }, 500);
   168→    return () => clearTimeout(timer);
   169→  }, [localState?.currentSectionIndex, userId, markSectionCompleted]);
   170→
   171→  const goPrev = () => {
   172→    if (!localState) return;
   173→    goToSection(localState.currentSectionIndex - 1);
   174→  };
   175→  const goNext = () => {
   176→    if (!localState) return;
   177→    goToSection(localState.currentSectionIndex + 1);
   178→  };
   179→
   180→  // ─── Keyboard shortcuts ──────────────────────────────────────
   181→  useEffect(() => {
   182→    function handleKeyDown(e: KeyboardEvent) {
   183→      const tag = (e.target as HTMLElement)?.tagName;
   184→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   185→
   186→      switch (e.key) {
   187→        case "Escape":
   188→          e.preventDefault();
   189→          goBack();
   190→          break;
   191→        case "ArrowLeft":
   192→          e.preventDefault();
   193→          goPrev();
   194→          break;
   195→        case "ArrowRight":
   196→          e.preventDefault();
   197→          goNext();
   198→          break;
   199→        case " ":
   200→          e.preventDefault();
   201→          goNext();
   202→          break;
   203→      }
   204→    }
   205→
   206→    document.addEventListener("keydown", handleKeyDown);
   207→
   208→    // Listen for custom event from KeyboardShortcuts component
   209→    function handleCustomNextSlide() {
   210→      goNext();
   211→    }
   212→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   213→
   214→    return () => {
   215→      document.removeEventListener("keydown", handleKeyDown);
   216→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   217→    };
   218→  }, [goBack, goPrev, goNext]);
   219→
   220→  /** Zoom controls */
   221→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   222→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   223→  const zoomFit = () => setZoom(100);
   224→
   225→  // ─── Confetti on last section ─────────────────
   226→  useEffect(() => {
   227→    if (!localState) return;
   228→    const totalSections = localState.allSectionIds.length;
   229→    const isOnLastSection =
   230→      totalSections > 0 &&
   231→      localState.currentSectionIndex === totalSections - 1;
   232→    if (isOnLastSection && !confettiShownRef.current) {
   233→      confettiShownRef.current = true;
   234→      setShowConfetti(true);
   235→      toast.success("🎉 You completed the course! Great job!");
   236→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   237→      return () => clearTimeout(timer);
   238→    }
   239→  }, [localState?.currentSectionIndex]);
   240→
   241→  // Reset confetti flag when classroom state changes (new course)
   242→  useEffect(() => {
   243→    confettiShownRef.current = false;
   244→    setShowConfetti(false);
   245→  }, [classroomState?.courseId]);
   246→
   247→  // ─── Keyboard Hint Auto-Fade ────────────────────
   248→  useEffect(() => {
   249→    setShowKeyboardHint(true);
   250→    const timer = setTimeout(() => setShowKeyboardHint(false), 5000);
   251→    return () => clearTimeout(timer);
   252→  }, [localState?.sectionId]);
   253→
   254→  // ─── Notes CRUD ─────────────────────────────────
   255→  const fetchNotes = useCallback(async () => {
   256→    if (!localState || !userId) return;
   257→    try {
   258→      const res = await fetch(
   259→        `/api/notes?userId=${userId}&courseId=${localState.courseId}&sectionId=${localState.sectionId}`
   260→      );
   261→      if (!res.ok) return;
   262→      const json = await res.json();
   263→      if (json.success && Array.isArray(json.data)) {
   264→        setNotes(json.data);
   265→      }
   266→    } catch {
   267→      // Silently fail
   268→    }
   269→  }, [localState, userId]);
   270→
   271→  // Fetch notes when section changes or sidebar opens
   272→  useEffect(() => {
   273→    if (notesSidebarOpen) fetchNotes();
   274→  }, [notesSidebarOpen, fetchNotes, localState?.sectionId]);
   275→
   276→  const createNote = useCallback(async () => {
   277→    if (!localState || !userId || !newNoteContent.trim()) return;
   278→    setSavingNote(true);
   279→    try {
   280→      const res = await fetch("/api/notes", {
   281→        method: "POST",
   282→        headers: { "Content-Type": "application/json" },
   283→        body: JSON.stringify({
   284→          userId,
   285→          courseId: localState.courseId,
   286→          sectionId: localState.sectionId,
   287→          content: newNoteContent.trim(),
   288→          slideNumber: localState.currentSectionIndex + 1,
   289→        }),
   290→      });
   291→      if (!res.ok) return;
   292→      const json = await res.json();
   293→      if (json.success && json.data) {
   294→        setNotes((prev) => [...prev, json.data]);
   295→        setNewNoteContent("");
   296→        setTimeout(
   297→          () => notesEndRef.current?.scrollIntoView({ behavior: "smooth" }),
   298→          50
   299→        );
   300→      }
   301→    } catch {
   302→      toast.error("Failed to save note");
   303→    } finally {
   304→      setSavingNote(false);
   305→    }
   306→  }, [localState, userId, newNoteContent]);
   307→
   308→  const deleteNote = useCallback(async (noteId: string) => {
   309→    try {
   310→      const res = await fetch(`/api/notes?id=${noteId}`, {
   311→        method: "DELETE",
   312→      });
   313→      if (!res.ok) return;
   314→      const json = await res.json();
   315→      if (json.success) {
   316→        setNotes((prev) => prev.filter((n) => n.id !== noteId));
   317→      }
   318→    } catch {
   319→      // Silently fail
   320→    }
   321→  }, []);
   322→
   323→  const toggleBookmark = useCallback((noteId: string) => {
   324→    setNotes((prev) =>
   325→      prev.map((n) =>
   326→        n.id === noteId ? { ...n, bookmarked: !n.bookmarked } : n
   327→      )
   328→    );
   329→  }, []);
   330→
   331→  // ─── Download as PPTX ─────────────────────────────
   332→  const handleDownloadPptx = useCallback(async () => {
   333→    if (!localState || downloadingPptx) return;
   334→    setDownloadingPptx(true);
   335→    try {
   336→      const res = await fetch("/api/generate-pptx", {
   337→        method: "POST",
   338→        headers: { "Content-Type": "application/json" },
   339→        body: JSON.stringify({
   340→          courseId: localState.courseId,
   341→          courseName: localState.courseTitle,
   342→        }),
   343→      });
   344→      if (!res.ok) {
   345→        toast.error("Failed to generate PPT");
   346→        return;
   347→      }
   348→      const blob = await res.blob();
   349→      const url = URL.createObjectURL(blob);
   350→      const a = document.createElement("a");
   351→      a.href = url;
   352→      const safeName = localState.courseTitle
   353→        .replace(/[^a-zA-Z0-9\s-]/g, "")
   354→        .replace(/\s+/g, "-")
   355→        .toLowerCase()
   356→        .slice(0, 60);
   357→      a.download = `${safeName}.pptx`;
   358→      document.body.appendChild(a);
   359→      a.click();
   360→      a.remove();
   361→      URL.revokeObjectURL(url);
   362→      toast.success("PPT downloaded successfully!");
   363→    } catch {
   364→      toast.error("Failed to download PPT");
   365→    } finally {
   366→      setDownloadingPptx(false);
   367→    }
   368→  }, [localState, downloadingPptx]);
   369→
   370→  // ─── AI Inline Edit ──────────────────────────────
   371→  const handleAiEdit = useCallback(async () => {
   372→    if (!localState || aiEditLoading || !aiEditInstruction.trim()) return;
   373→    setAiEditLoading(true);
   374→    try {
   375→      const res = await fetch("/api/generate-slide-inline-edit", {
   376→        method: "POST",
   377→        headers: { "Content-Type": "application/json" },
   378→        body: JSON.stringify({
   379→          htmlBody: localState.htmlBody,
   380→          instruction: aiEditInstruction.trim(),
   381→          slideTitle: localState.sectionTitle,
   382→        }),
   383→      });
   384→      if (!res.ok) {
   385→        toast.error("AI edit failed");
   386→        return;
   387→      }
   388→      const json = await res.json();
   389→      if (json.success && json.data?.htmlBody) {
   390→        const newHtmlBody = json.data.htmlBody;
   391→        setLocalState((prev) =>
   392→          prev ? { ...prev, htmlBody: newHtmlBody } : prev
   393→        );
   394→        toast.success("AI edit applied");
   395→        // Persist to database
   396→        try {
   397→          await fetch(`/api/sections/${localState.sectionId}`, {
   398→            method: "PUT",
   399→            headers: { "Content-Type": "application/json" },
   400→            body: JSON.stringify({ htmlBody: newHtmlBody }),
   401→          });
   402→        } catch {
   403→          /* best-effort save */
   404→        }
   405→      } else {
   406→        toast.error("AI edit returned no content");
   407→      }
   408→    } catch {
   409→      toast.error("AI edit request failed");
   410→    } finally {
   411→      setAiEditLoading(false);
   412→      setShowAiEdit(false);
   413→      setAiEditInstruction("");
   414→    }
   415→  }, [localState, aiEditLoading, aiEditInstruction]);
   416→
   417→  // ─── No state guard ─────────────────────────────
   418→  if (!localState) {
   419→    return (
   420→      <div className="flex h-screen items-center justify-center bg-muted/30">
   421→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   422→      </div>
   423→    );
   424→  }
   425→
   426→  const totalSections = localState.allSectionIds.length;
   427→  const currentIdx = localState.currentSectionIndex;
   428→  const isFirst = currentIdx === 0;
   429→  const isLast = totalSections > 0 && currentIdx >= totalSections - 1;
   430→  const progressPercent =
   431→    totalSections > 1
   432→      ? Math.round(((currentIdx + 1) / totalSections) * 100)
   433→      : 100;
   434→
   435→  return (
   436→    <div className="flex h-screen flex-col bg-muted/30 relative">
   437→      {/* ─── Mobile Notes Sheet ──────────────────── */}
   438→      <Sheet
   439→        open={notesSidebarOpen && isMobile}
   440→        onOpenChange={(open) => {
   441→          if (!open) setNotesSidebarOpen(false);
   442→        }}
   443→      >
   444→        <SheetContent side="right" className="w-full sm:max-w-sm p-0">
   445→          <SheetHeader className="px-4 pt-4 pb-0">
   446→            <SheetTitle className="flex items-center gap-2 text-base">
   447→              <StickyNote className="h-4 w-4" />
   448→              Notes
   449→            </SheetTitle>
   450→            <SheetDescription>
   451→              Notes for section {currentIdx + 1} of {totalSections}
   452→            </SheetDescription>
   453→          </SheetHeader>
   454→          <NotesSidebarContent
   455→            notes={notes}
   456→            currentSectionIndex={currentIdx}
   457→            newNoteContent={newNoteContent}
   458→            savingNote={savingNote}
   459→            onContentChange={setNewNoteContent}
   460→            onCreate={createNote}
   461→            onDelete={deleteNote}
   462→            onToggleBookmark={toggleBookmark}
   463→            endRef={notesEndRef}
   464→          />
   465→        </SheetContent>
   466→      </Sheet>
   467→
   468→      {/* ─── Confetti Celebration Overlay ─────── */}
   469→      {showConfetti && <ConfettiCelebration />}
   470→
   471→      {/* ─── Top Progress Bar ─────────────────────── */}
   472→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   473→        <div
   474→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   475→          style={{ width: `${progressPercent}%` }}
   476→        />
   477→      </div>
   478→
   479→      {/* ─── Top Bar ──────────────────────────────── */}
   480→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   481→        {/* Logo, Section Title, Course Title */}
   482→        <div className="flex items-center gap-3 min-w-0">
   483→          <div className="flex items-center gap-2 shrink-0">
   484→            <img
   485→              src="/ecotech-logo.png"
   486→              alt="Ecotech"
   487→              className="h-7 w-7 rounded-md object-contain"
   488→            />
   489→            <img
   490→              src="/ecotech-name.png"
   491→              alt="Ecotech"
   492→              className="h-5 w-auto hidden sm:inline object-contain"
   493→            />
   494→          </div>
   495→          <Separator orientation="vertical" className="h-5" />
   496→          <div className="min-w-0">
   497→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   498→              {localState.sectionTitle}
   499→            </h2>
   500→            <p className="text-xs text-muted-foreground truncate leading-tight">
   501→              {localState.courseTitle}
   502→            </p>
   503→          </div>
   504→        </div>
   505→
   506→        {/* Section Counter */}
   507→        <span className="shrink-0 inline-flex items-center gap-1.5 rounded-full bg-primary/10 px-3 py-1 text-sm text-primary tabular-nums">
   508→          <span className="font-semibold">
   509→            Section {currentIdx + 1}
   510→          </span>
   511→          <span className="text-primary/40 font-normal">of</span>
   512→          <span className="font-semibold">{totalSections}</span>
   513→        </span>
   514→      </header>
   515→
   516→      {/* ─── Main Layout: Content + Desktop Notes Sidebar ── */}
   517→      <div className="flex flex-1 overflow-hidden">
   518→        {/* ─── Section Content Area ────────────────── */}
   519→        <div className="flex-1 overflow-auto flex flex-col items-center py-8 px-4 sm:px-6">
   520→          <div
   521→            className="bg-card rounded-2xl shadow-lg border w-full max-w-3xl ring-1 ring-black/5 dark:ring-white/5 paper-texture overflow-hidden"
   522→            style={{
   523→              transform: `scale(${zoom / 100})`,
   524→              transformOrigin: "top center",
   525→            }}
   526→          >
   527→            {navigating ? (
   528→              <div className="flex items-center justify-center py-32">
   529→                <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   530→              </div>
   531→            ) : (
   532→              <iframe
   533→                srcDoc={localState.htmlBody || ""}
   534→                sandbox="allow-same-origin"
   535→                className="w-full rounded-lg border-0"
   536→                style={{ aspectRatio: "16/9" }}
   537→                title={`${localState.sectionTitle || "Slide"} content`}
   538→              />
   539→            )}
   540→          </div>
   541→        </div>
   542→
   543→        {/* ─── Desktop Notes Sidebar (slide-in panel) ── */}
   544→        <aside
   545→          className={`hidden lg:flex flex-col shrink-0 border-l bg-card transition-all duration-300 ease-in-out overflow-hidden ${
   546→            notesSidebarOpen
   547→              ? "w-80 opacity-100"
   548→              : "w-0 opacity-0 border-l-0"
   549→          }`}
   550→        >
   551→          <div className="flex flex-col h-full w-80">
   552→            {/* Sidebar Header */}
   553→            <div className="flex items-center justify-between px-4 py-3 border-b">
   554→              <div className="flex items-center gap-2">
   555→                <StickyNote className="h-4 w-4 text-primary" />
   556→                <span className="text-sm font-semibold">Notes</span>
   557→              </div>
   558→              <Button
   559→                variant="ghost"
   560→                size="icon"
   561→                className="h-7 w-7"
   562→                onClick={() => setNotesSidebarOpen(false)}
   563→              >
   564→                <X className="h-4 w-4" />
   565→              </Button>
   566→            </div>
   567→            <NotesSidebarContent
   568→              notes={notes}
   569→              currentSectionIndex={currentIdx}
   570→              newNoteContent={newNoteContent}
   571→              savingNote={savingNote}
   572→              onContentChange={setNewNoteContent}
   573→              onCreate={createNote}
   574→              onDelete={deleteNote}
   575→              onToggleBookmark={toggleBookmark}
   576→              endRef={notesEndRef}
   577→            />
   578→          </div>
   579→        </aside>
   580→      </div>
   581→
   582→      {/* ─── Keyboard Shortcuts Hint (auto-fades) ── */}
   583→      <div
   584→        className={`shrink-0 frosted-glass border-t px-4 py-1.5 text-center transition-opacity duration-700 ${
   585→          showKeyboardHint
   586→            ? "opacity-100"
   587→            : "opacity-0 pointer-events-none h-0 py-0 overflow-hidden"
   588→        }`}
   589→      >

   948→type ParticleShape = "circle" | "rectangle" | "triangle";
   949→
   950→interface ConfettiParticle {
   951→  id: number;
   952→  left: number;
   953→  color: string;
   954→  shape: ParticleShape;
   955→  size: number;
   956→  fallDuration: string;
   957→  fallDelay: string;
   958→  drift: string;
   959→  spin: string;
   960→}
   961→
   962→function ConfettiCelebration() {
   963→  const particles = useMemo<ConfettiParticle[]>(() => {
   964→    const result: ConfettiParticle[] = [];
   965→    const shapes: ParticleShape[] = ["circle", "rectangle", "triangle"];
   966→    const count = 80;
   967→
   968→    for (let i = 0; i < count; i++) {
   969→      const shape =
   970→        shapes[i % 3 === 0 ? 0 : i % 3 === 1 ? 1 : 2];
   971→      result.push({
   972→        id: i,
   973→        left: Math.random() * 100,
   974→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   975→        shape,
   976→        size: Math.floor(Math.random() * 8) + 6, // 6-14px
   977→        fallDuration: `${(Math.random() * 1.5 + 2).toFixed(2)}s`, // 2-3.5s
   978→        fallDelay: `${(Math.random() * 0.8).toFixed(2)}s`, // 0-0.8s
   979→        drift: `${(Math.random() * 120 - 60).toFixed(0)}px`, // -60 to +60px
   980→        spin: `${Math.floor(Math.random() * 720 + 360)}deg`, // 360-1080deg
   981→      });
   982→    }
   983→    return result;
   984→  }, []);
   985→
   986→  return (
   987→    <>
   988→      {/* Confetti particles layer */}
   989→      <div className="confetti-container" aria-hidden="true">
   990→        {particles.map((p) => {
   991→          if (p.shape === "triangle") {
   992→            return (
   993→              <div
   994→                key={p.id}
   995→                className="confetti-particle confetti-triangle"
   996→                style={{
   997→                  left: `${p.left}%`,
   998→                  "--confetti-color": p.color,
   999→                  "--tri-size": `${p.size}px`,
  1000→                  "--fall-duration": p.fallDuration,
  1001→                  "--fall-delay": p.fallDelay,
  1002→                  "--drift": p.drift,
  1003→                  "--spin": p.spin,
  1004→                } as React.CSSProperties}
  1005→              />
  1006→            );
  1007→          }

  1009→            <div
  1010→              key={p.id}
  1011→              className={`confetti-particle ${
  1012→                p.shape === "circle"
  1013→                  ? "confetti-circle"
  1014→                  : "confetti-rectangle"

  1019→                  p.shape === "rectangle"

  1024→                "--fall-duration": p.fallDuration,
  1025→                "--fall-delay": p.fallDelay,
