    63→function timeAgo(dateStr: string): string {
    64→  const now = new Date();
    65→  const date = new Date(dateStr);
    66→  const diffMs = now.getTime() - date.getTime();
    67→  const diffSec = Math.floor(diffMs / 1000);
    68→  const diffMin = Math.floor(diffSec / 60);
    69→  const diffHr = Math.floor(diffMin / 60);
    70→  const diffDay = Math.floor(diffHr / 24);
    71→  const diffWeek = Math.floor(diffDay / 7);
    72→  const diffMonth = Math.floor(diffDay / 30);
    73→
    74→  if (diffSec < 60) return "just now";
    75→  if (diffMin < 60) return `${diffMin}m ago`;
    76→  if (diffHr < 24) return `${diffHr}h ago`;
    77→  if (diffDay < 7) return `${diffDay}d ago`;
    78→  if (diffWeek < 5) return `${diffWeek}w ago`;
    79→  return `${diffMonth}mo ago`;
    80→}

    82→/* ------------------------------------------------------------------ */
    83→/*  Seeded random for consistent sparkline data                        */
    84→/* ------------------------------------------------------------------ */
    85→
    86→function seededRandom(seed: number) {
    87→  let s = seed;
    88→  return () => {
    89→    s = (s * 16807 + 0) % 2147483647;
    90→    return (s - 1) / 2147483646;
    91→  };
    92→}

    94→/* ------------------------------------------------------------------ */
    95→/*  Sparkline Mini Chart                                               */
    96→/* ------------------------------------------------------------------ */
    97→
    98→function SparklineChart({
    99→  data,
   100→  gradientFrom,
   101→  gradientTo,
   102→}: {
   103→  data: number[];
   104→  gradientFrom: string;
   105→  gradientTo: string;
   106→}) {
   107→  const max = Math.max(...data, 1);
   108→  return (
   109→    <div className="flex items-end gap-[3px] h-8">
   110→      {data.map((val, i) => {
   111→        const height = Math.max(4, (val / max) * 100);
   112→        return (
   113→          <div
   114→            key={i}
   115→            className={cn(
   116→              "w-[6px] rounded-full transition-all duration-500",
   117→              `bg-gradient-to-t ${gradientFrom} ${gradientTo}`
   118→            )}
   119→            style={{
   120→              height: `${height}%`,
   121→              opacity: 0.5 + (i / data.length) * 0.5,
   122→              animationDelay: `${i * 60}ms`,
   123→            }}
   124→          />
   125→        );
   126→      })}
   127→    </div>
   128→  );
   129→}

   146→function StatCard({

   149→  value,
   150→  gradient,

   153→  delay,

   156→  const sparkData = useMemo(() => {

   162→    <Card
   163→      className="glass-card stat-pop hover-lift overflow-hidden border-border/50 transition-shadow hover:shadow-lg"
   164→      style={{ animationDelay: `${delay}ms` }}
   165→    >
   166→      <CardContent className="p-4 sm:p-5">
   167→        <div className="flex items-center gap-3.5">
   168→          <div
   169→            className={cn(
   170→              "flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br shadow-md transition-transform duration-200 hover:scale-110",
   171→              gradient
   172→            )}
   173→          >
   174→            {icon}
   175→          </div>
   176→          <div className="min-w-0 flex-1">
   177→            <p className="text-[11px] font-medium text-muted-foreground uppercase tracking-wider">
   178→              {label}
   179→            </p>
   180→            <p className="text-2xl font-bold tracking-tight text-foreground leading-tight">
   181→              {value}
   182→            </p>
   183→          </div>
   184→          <div className="shrink-0 opacity-60">
   185→            <SparklineChart
   186→              data={sparkData}
   187→              gradientFrom={gradientFrom}
