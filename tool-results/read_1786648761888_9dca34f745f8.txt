   464→   514→function SlideRenderer({ slide }: { slide: SlideContent }) {
   465→   515→  switch (slide.type) {
   466→   516→    case "title":
   467→   517→      return <TitleSlide slide={slide} />;
   468→   518→    case "content":
   469→   519→      return <ContentSlide slide={slide} />;
   470→   520→    case "table":
   471→   521→      return <TableSlide slide={slide} />;
   472→   522→    case "list":

   476→   526→    case "quiz":

   640→   690→function QuizSlide({ slide }: { slide: SlideContent }) {
   641→   691→  const [selectedIdx, setSelectedIdx] = useState<number | null>(null);
   642→   692→  const hasSubmitted = selectedIdx !== null;
   643→   693→
   644→   694→  // Determine correct answer: look for item with icon === "check" or first item
   645→   695→  const correctIdx = slide.items?.findIndex((item) => item.icon === "check") ?? 0;
   646→   696→  const isCorrect = selectedIdx === correctIdx;
   647→   697→
   648→   698→  return (
   649→   699→    <div>
   650→   700→      <div className="mb-6 flex items-center gap-3">
   651→   701→        <div className="h-1 w-8 rounded-full bg-gradient-to-r from-primary to-accent" />
   652→   702→        <h2 className="text-2xl font-bold text-foreground">{slide.title}</h2>
   653→   703→      </div>
   654→   704→      {slide.items && slide.items.length > 0 && (
   655→   705→        <div className="space-y-3">
   656→   706→          {slide.items.map((item, i) => {
   657→   707→            const letter = String.fromCharCode(65 + i);
   658→   708→            const isSelected = selectedIdx === i;
   659→   709→            let borderColor = "border-border hover:border-primary/30 hover:bg-primary/5";
   660→   710→            if (hasSubmitted) {
   661→   711→              if (i === correctIdx) borderColor = "border-emerald-500 bg-emerald-500/10";
   662→   712→              else if (isSelected && !isCorrect) borderColor = "border-red-400 bg-red-400/10";
   663→   713→              else borderColor = "border-border opacity-60";
   664→   714→            } else if (isSelected) {
   665→   715→              borderColor = "border-primary bg-primary/10";
   666→   716→            }
   667→   717→
   668→   718→            return (
   669→   719→              <button
   670→   720→                key={i}
   671→   721→                type="button"
   672→   722→                className={`w-full rounded-lg border p-4 text-left transition-all duration-200 ${borderColor}`}
   673→   723→                onClick={() => {
   674→   724→                  if (!hasSubmitted) setSelectedIdx(i);
   675→   725→                }}
   676→   726→                disabled={hasSubmitted}
   677→   727→              >
   678→   728→                <div className="flex items-start gap-3">
   679→   729→                  <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold ${
   680→   730→                    hasSubmitted && i === correctIdx
   681→   731→                      ? "bg-emerald-500 text-white"
   682→   732→                      : hasSubmitted && isSelected && !isCorrect
   683→   733→                        ? "bg-red-400 text-white"
   684→   734→                        : "bg-primary/10 text-primary"
   685→   735→                  }`}>
   686→   736→                    {hasSubmitted && i === correctIdx ? "✓" : hasSubmitted && isSelected && !isCorrect ? "✗" : letter}
   687→   737→                  </span>
   688→   738→                  <div className="flex-1">
   689→   739→                    <p className="text-sm font-medium text-foreground">
   690→   740→                      {item.heading || item.text}
   691→   741→                    </p>
   692→   742→                    {item.heading && item.text !== item.heading && (
   693→   743→                      <p className="text-xs text-muted-foreground mt-1">{item.text}</p>
   694→   744→                    )}
   695→   745→                  </div>
   696→   746→                </div>
   697→   747→              </button>
   698→   748→            );
   699→   749→          })}
   700→   750→
   701→   751→          {/* Feedback after selection */}
   702→   752→          {hasSubmitted && (
   703→   753→            <div className={`mt-4 rounded-lg border p-4 ${
   704→   754→              isCorrect
   705→   755→                ? "border-emerald-200 bg-emerald-50 dark:border-emerald-800 dark:bg-emerald-950/50"
   706→   756→                : "border-amber-200 bg-amber-50 dark:border-amber-800 dark:bg-amber-950/50"
   707→   757→            }`}>
   708→   758→              <p className={`text-sm font-semibold ${
   709→   759→                isCorrect ? "text-emerald-700 dark:text-emerald-400" : "text-amber-700 dark:text-amber-400"
   710→   760→              }`}>
   711→   761→                {isCorrect ? "🎉 Correct! Well done!" : "❌ Not quite right."}
   712→   762→              </p>
   713→   763→              {!isCorrect && (
   714→   764→                <p className="mt-1 text-xs text-muted-foreground">
   715→   765→                  The correct answer is <strong>{String.fromCharCode(65 + correctIdx)}</strong>.
   716→   766→                </p>
   717→   767→              )}
   718→   768→            </div>
   719→   769→          )}
   720→   770→        </div>
   721→   771→      )}
   722→   772→    </div>
   723→   773→  );
   724→   774→}
