   496→
   497→  // ============================================
   498→  // GENERATE SLIDES — Streaming HTML for all slides
   499→  // ============================================
   500→
   501→  const handleGenerateSlides = useCallback(
   502→    async (lessonId: string) => {
   503→      const lesson = outlineLessons.find((l) => l.id === lessonId);
   504→      if (!lesson) return;
   505→      if (lesson.slides.length === 0) {
   506→        toast.error("No slides to generate");
   507→        return;
   508→      }
   509→
   510→      // Initialize per-slide gen states
   511→      const initialStates: Record<string, SlideGenState> = {};
   512→      lesson.slides.forEach((s) => {
   513→        if (s.slideId) {
   514→          initialStates[s.slideId] = { status: "pending" };
   515→        }
   516→      });
   517→      setSlideGenStates(initialStates);
   518→      setGeneratingLessonId(lessonId);
   519→      setCurrentGenSlideId(null);
   520→      setStreamingHtml("");
   521→      setGenProgress({ current: 0, total: lesson.slides.length });
   522→      setExpandedOutlineLessonId(lessonId);
   523→
   524→      // Close modal if open
   525→      setModalOpen(false);
   526→
   527→      const abort = new AbortController();
   528→      abortGenRef.current = abort;
   529→
   530→      try {
   531→        const res = await fetch("/api/lessons/generate-slides", {
   532→          method: "POST",
   533→          headers: { "Content-Type": "application/json" },
   534→          body: JSON.stringify({
   535→            lessonId,
   536→            language: lesson.language,
   537→          }),
   538→          signal: abort.signal,
   539→        });
   540→
   541→        if (!res.ok || !res.body) {
   542→          toast.error("Failed to start slide generation");
   543→          setGeneratingLessonId(null);
   544→          return;
   545→        }
   546→
   547→        const reader = res.body.getReader();
   548→        const decoder = new TextDecoder();
   549→        let buffer = "";
   550→        let slidesCompleted = 0;
   551→        let rawHtmlAccum = "";
   552→
   553→        while (true) {
   554→          const { done, value } = await reader.read();
   555→          if (done) break;
   556→          buffer += decoder.decode(value, { stream: true });
   557→
   558→          const lines = buffer.split("\n");
   559→          buffer = lines.pop() || "";
   560→
   561→          for (const line of lines) {
   562→            if (!line.startsWith("data: ")) continue;
   563→            try {
   564→              const data = JSON.parse(line.slice(6));
   565→
   566→              // Top-level error event (no slideId) — e.g. "Lesson not found"
   567→              if (data.error && !data.slideId) {
   568→                toast.error(data.error);
   569→                break; // Stop processing
   570→              }
   571→
   572→              if (data.slideId && data.slideTitle && data.totalSlides !== undefined) {
   573→                // slide_start event
   574→                setCurrentGenSlideId(data.slideId);
   575→                setSlideGenStates((prev) => ({
   576→                  ...prev,
   577→                  [data.slideId]: { status: "generating" },
   578→                }));
   579→                rawHtmlAccum = "";
   580→                setGenProgress({ current: data.slideIndex + 1, total: data.totalSlides });
   581→              }
   582→
   583→              if (data.slideId && data.html) {
   584→                // chunk event
   585→                rawHtmlAccum += data.html;
   586→                setStreamingHtml(buildLivePreviewDoc(rawHtmlAccum, data.slideId));
   587→              }
   588→
   589→              if (data.slideId && data.htmlBody) {
   590→                // slide_complete event
   591→                slidesCompleted++;
   592→                setSlideGenStates((prev) => ({
   593→                  ...prev,
   594→                  [data.slideId]: { status: "complete", htmlBody: data.htmlBody },
   595→                }));
   596→                setOutlineLessons((prev) =>
   597→                  prev.map((ol) =>
   598→                    ol.id === lessonId ? { ...ol, allReady: slidesCompleted === ol.slides.length } : ol
   599→                  )
   600→                );
   601→              }
   602→
   603→              if (data.slideId && data.error) {
   604→                // slide_error event
   605→                slidesCompleted++;
   606→                setSlideGenStates((prev) => ({
   607→                  ...prev,
   608→                  [data.slideId]: { status: "error", error: data.error },
