     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef } from "react";
     4→import {
     5→  ArrowLeft,
     6→  Plus,
     7→  Trash2,
     8→  FileText,
     9→  GripVertical,
    10→  ChevronUp,
    11→  ChevronDown,
    12→  Loader2,
    13→  ImageIcon,
    14→  X,
    15→  FileUp,
    16→  Sparkles,
    17→  Eye,
    18→} from "lucide-react";
    19→import { Button } from "@/components/ui/button";
    20→import { Input } from "@/components/ui/input";
    21→import { Textarea } from "@/components/ui/textarea";
    22→import { Label } from "@/components/ui/label";
    23→import { Card, CardContent } from "@/components/ui/card";
    24→import { Badge } from "@/components/ui/badge";
    25→import { Separator } from "@/components/ui/separator";
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from "@/components/ui/select";
    33→import { RadioGroup, RadioGroupItem } from "@/components/ui/radio-group";
    34→import {
    35→  Dialog,
    36→  DialogContent,
    37→  DialogHeader,
    38→  DialogTitle,
    39→  DialogDescription,
    40→  DialogFooter,
    41→} from "@/components/ui/dialog";
    42→import { useNavigationStore, useUserStore, useCourseStore } from "@/stores/lms-store";
    43→import type { CategoryItem } from "@/types/lms";
    44→import { toast } from "sonner";
    45→
    46→// ============================================
    47→// Types
    48→// ============================================
    49→
    50→interface LessonDraft {
    51→  id: string;
    52→  title: string;
    53→  totalPages: number;
    54→  htmlBody: string;
    55→  language: string;
    56→}
    57→
    58→// ============================================
    59→// Create Course Page Component
    60→// ============================================
    61→
    62→const MAX_LESSONS = 10;
    63→const MAX_TITLE_LENGTH = 100;
    64→const MAX_DESC_LENGTH = 3000;
    65→const MAX_COURSE_DESC_LENGTH = 500;
    66→
    67→export function CreateCoursePage() {
    68→  const { goBack } = useNavigationStore();
    69→  const { currentUserId } = useUserStore();
    70→  const { createPrompt, setCreatePrompt } = useCourseStore();
    71→
    72→  // ---- Form state ----
    73→  const [title, setTitle] = useState("");
    74→  const [description, setDescription] = useState("");
    75→  const [categoryId, setCategoryId] = useState("");
    76→  const [language, setLanguage] = useState("english");
    77→  const [coverImage, setCoverImage] = useState("");
    78→  const [coverPreview, setCoverPreview] = useState<string | null>(null);
    79→  const [lessons, setLessons] = useState<LessonDraft[]>([]);
    80→
    81→  // ---- UI state ----
    82→  const [categories, setCategories] = useState<CategoryItem[]>([]);
    83→  const [loading, setLoading] = useState(false);
    84→  const [saving, setSaving] = useState(false);
    85→  const [generating, setGenerating] = useState(false);
    86→  const [modalOpen, setModalOpen] = useState(false);
    87→  const [urlInputOpen, setUrlInputOpen] = useState(false);
    88→  const [imageUrl, setImageUrl] = useState("");
    89→
    90→  // ---- Modal state ----
    91→  const [lessonName, setLessonName] = useState("");
    92→  const [lessonPrompt, setLessonPrompt] = useState("");
    93→  const [lessonLanguage, setLessonLanguage] = useState("english");
    94→  const [lessonPdfName, setLessonPdfName] = useState("");
    95→
    96→  // ---- Live streaming state ----
    97→  const [streamingHtml, setStreamingHtml] = useState("");
    98→  const [showStreamPreview, setShowStreamPreview] = useState(false);
    99→   const streamPreviewRef = useRef<HTMLDivElement>(null);
   100→
   101→  // ---- Expanded lesson preview ----
   102→  const [expandedLessonId, setExpandedLessonId] = useState<string | null>(null);
   103→
   104→  // ---- Pre-fill title from hero prompt ----
   105→  useEffect(() => {
   106→    if (createPrompt) {
   107→      setTitle(createPrompt);
   108→      setCreatePrompt("");
   109→    }
   110→  }, [createPrompt, setCreatePrompt]);
   111→
   112→  // ---- Refs ----
   113→  const fileInputRef = useRef<HTMLInputElement>(null);
   114→  const lessonFileInputRef = useRef<HTMLInputElement>(null);
   115→  const abortControllerRef = useRef<AbortController | null>(null);
   116→
   117→  // ---- Fetch categories ----
   118→  useEffect(() => {
   119→    async function fetchCategories() {
   120→      setLoading(true);
   121→      try {
   122→        const res = await fetch("/api/categories");
   123→        const json = await res.json();
   124→        if (json.success) {
   125→          setCategories(json.data);
   126→        }
   127→      } catch {
   128→        toast.error("Failed to load categories");
   129→      } finally {
   130→        setLoading(false);
   131→      }
   132→    }
   133→    fetchCategories();
   134→  }, []);
   135→
   136→  // ---- Cover image handlers ----
   137→  const handleCoverUpload = () => {
   138→    setUrlInputOpen(true);
   139→  };
   140→
   141→  const handleUrlConfirm = () => {
   142→    if (imageUrl.trim()) {
   143→      setCoverImage(imageUrl.trim());
   144→      setCoverPreview(imageUrl.trim());
   145→      setUrlInputOpen(false);
   146→      setImageUrl("");
   147→    }
   148→  };
   149→
   150→  const handleRemoveCover = () => {
   151→    setCoverImage("");
   152→    setCoverPreview(null);
   153→  };
   154→
   155→  // ---- Lesson management ----
   156→  const handleOpenModal = () => {
   157→    setLessonName("");
   158→    setLessonPrompt("");
   159→    setLessonLanguage(language);
   160→    setLessonPdfName("");
   161→    setStreamingHtml("");
   162→    setShowStreamPreview(false);
   163→    setModalOpen(true);
   164→  };
   165→
   166→  const handleGenerateLesson = useCallback(async () => {
   167→    if (!lessonName.trim()) {
   168→      toast.error("Lesson name is required");
   169→      return;
   170→    }
   171→    if (!lessonPrompt.trim()) {
   172→      toast.error("Content prompt is required");
   173→      return;
   174→    }
   175→    if (lessons.length >= MAX_LESSONS) {
   176→      toast.error(`Maximum ${MAX_LESSONS} lessons allowed`);
   177→      return;
   178→    }
   179→
   180→    setGenerating(true);
   181→    setStreamingHtml("");
   182→    setShowStreamPreview(true);
   183→
   184→    const abort = new AbortController();
   185→    abortControllerRef.current = abort;
   186→
   187→    try {
   188→      const res = await fetch("/api/generate-slide-html", {
   189→        method: "POST",
   190→        headers: { "Content-Type": "application/json" },
   191→        body: JSON.stringify({
   192→          slideTitle: lessonName.trim(),
   193→          prompt: lessonPrompt.trim(),
   194→          language: lessonLanguage,
   195→        }),
   196→        signal: abort.signal,
   197→      });
   198→
   199→      if (!res.ok || !res.body) {
   200→        toast.error("Failed to start generation");
   201→        setGenerating(false);
   202→        setShowStreamPreview(false);
   203→        return;
   204→      }
   205→
   206→      // Read SSE stream with live preview
   207→      const reader = res.body.getReader();
   208→      const decoder = new TextDecoder();
   209→      let buffer = "";
   210→      let htmlBody = "";
   211→      let slideTitle = lessonName.trim();
   212→      let rawHtml = ""; // accumulating raw HTML fragments for live preview
   213→
   214→      while (true) {
   215→        const { done, value } = await reader.read();
   216→        if (done) break;
   217→        buffer += decoder.decode(value, { stream: true });
   218→
   219→        const lines = buffer.split("\n");
   220→        buffer = lines.pop() || "";
   221→
   222→        for (const line of lines) {
   223→          if (!line.startsWith("data: ")) continue;
   224→          try {
   225→            const data = JSON.parse(line.slice(6));
   226→            if (data.htmlBody) {
   227→              htmlBody = data.htmlBody;
   228→            }
   229→            if (data.slideTitle) {
   230→              slideTitle = data.slideTitle;
   231→            }
   232→            if (data.html) {
   233→              rawHtml += data.html;
   234→              // Build live preview from accumulated raw HTML
   235→              const previewDoc = buildLivePreviewDoc(rawHtml, slideTitle);
   236→              setStreamingHtml(previewDoc);
   237→              // Auto-scroll preview into view
   238→              streamPreviewRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
   239→            }
   240→          } catch {
   241→            // skip malformed JSON
   242→          }
   243→        }
   244→      }
   245→
   246→      if (!htmlBody) {
   247→        toast.error("Generation returned empty content");
   248→        setGenerating(false);
   249→        setShowStreamPreview(false);
   250→        return;
   251→      }
   252→
   253→      const newLesson: LessonDraft = {
   254→        id: `sec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   255→        title: slideTitle,
   256→        totalPages: 1,
   257→        htmlBody,
   258→        language: lessonLanguage,
   259→      };
   260→
   261→      setLessons((prev) => [...prev, newLesson]);
   262→      setModalOpen(false);
   263→      setShowStreamPreview(false);
   264→      setStreamingHtml("");
   265→      toast.success(`Lesson \"${slideTitle}\" generated successfully`);
   266→    } catch (err) {
   267→      if (err instanceof Error && err.name === "AbortError") return;
   268→      toast.error("Failed to generate lesson. Please try again.");
   269→    } finally {
   270→      setGenerating(false);
   271→      abortControllerRef.current = null;
   272→    }
   273→  }, [lessonName, lessonPrompt, lessonLanguage, lessons.length]);
   274→
   275→  const handleCancelGeneration = () => {
   276→    abortControllerRef.current?.abort();
   277→    setGenerating(false);
   278→    setShowStreamPreview(false);
   279→    setStreamingHtml("");
   280→  };
   281→
   282→  const handleDeleteLesson = (id: string) => {
   283→    setLessons((prev) => prev.filter((s) => s.id !== id));
   284→    if (expandedLessonId === id) setExpandedLessonId(null);
   285→  };
   286→
   287→  const handleMoveLesson = (index: number, direction: "up" | "down") => {
   288→    const newLessons = [...lessons];
   289→    const targetIndex = direction === "up" ? index - 1 : index + 1;
   290→    if (targetIndex < 0 || targetIndex >= newLessons.length) return;
   291→    [newLessons[index], newLessons[targetIndex]] = [
   292→      newLessons[targetIndex],
   293→      newLessons[index],
   294→    ];
   295→    setLessons(newLessons);
   296→  };
   297→
   298→  // ---- Form submission ----
   299→  const handleSave = async () => {
   300→    if (!title.trim()) {
   301→      toast.error("Course title is required");
   302→      return;
   303→    }
   304→
   305→    // Check if any lesson is missing htmlBody
   306→    const missingContent = lessons.filter((s) => !s.htmlBody);
   307→    if (missingContent.length > 0) {
   308→      toast.error(`${missingContent.length} lesson(s) have no content yet. Generate content for all lessons first.`);
   309→      return;
   310→    }
   311→
   312→    setSaving(true);
   313→    try {
   314→      const payload = {
   315→        title: title.trim(),
   316→        description: description.trim() || null,
   317→        categoryId: categoryId || null,
   318→        language,
   319→        creatorId: currentUserId,
   320→        coverImage: coverImage || null,
   321→        lessons: lessons.map((sec, index) => ({
   322→          title: sec.title,
   323→          htmlBody: sec.htmlBody,
   324→          totalPages: sec.totalPages,
   325→          order: index,
   326→        })),
   327→      };
   328→
   329→      const res = await fetch("/api/courses", {
   330→        method: "POST",
   331→        headers: { "Content-Type": "application/json" },
   332→        body: JSON.stringify(payload),
   333→      });
   334→
   335→      const json = await res.json();
   336→
   337→      if (json.success) {
   338→        toast.success("Course created successfully!");
   339→        goBack();
   340→      } else {
   341→        toast.error(json.error || "Failed to create course");
   342→      }
   343→    } catch {
   344→      toast.error("Failed to create course. Please try again.");
   345→    } finally {
   346→      setSaving(false);
   347→    }
   348→  };
   349→
   350→  // ---- Lesson PDF upload simulation ----
   351→  const handleLessonPdfUpload = () => {
   352→    lessonFileInputRef.current?.click();
   353→  };
   354→
   355→  const handleLessonFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   356→    const file = e.target.files?.[0];
   357→    if (file) {
   358→      setLessonPdfName(file.name);
   359→      if (!lessonPrompt.trim()) {
   360→        const nameWithoutExt = file.name.replace(/\.pdf$/i, "");
   361→        setLessonPrompt(
   362→          `Generate course content based on the uploaded PDF: ${nameWithoutExt}`
   363→        );
   364→        if (!lessonName.trim()) {
   365→          setLessonName(nameWithoutExt);
   366→        }
   367→      }
   368→    }
   369→  };
   370→
   371→  // ============================================
   372→  // Render
   373→  // ============================================
   374→
   375→  return (
   376→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   377→      {/* ---- Breadcrumbs ---- */}
   378→      <div className="mb-6 flex items-center gap-3">
   379→        <button
   380→          onClick={goBack}
   381→          className="flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-foreground"
   382→        >
   383→          <ArrowLeft className="h-4 w-4" />
   384→          Back
   385→        </button>
   386→        <Separator orientation="vertical" className="h-4" />
   387→        <h1 className="text-lg font-semibold text-foreground">Create Course</h1>
   388→      </div>
   389→
   390→      {/* ---- Two-Column Layout ---- */}
   391→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-5">
   392→        {/* ======== Left Panel: Course Metadata (3 cols) ======== */}
   393→        <div className="lg:col-span-3">
   394→          <Card className="border-border/50">
   395→            <CardContent className="space-y-6 p-6">
   396→              {/* Cover Image Upload */}
   397→              <div className="space-y-2">
   398→                <Label className="text-sm font-medium">Cover Image</Label>
   399→                {coverPreview ? (
   400→                  <div className="relative overflow-hidden rounded-lg border border-border">
   401→                    <img
   402→                      src={coverPreview}
   403→                      alt="Course cover"
   404→                      className="h-48 w-full object-cover"
   405→                      onError={() => setCoverPreview(null)}
   406→                    />
   407→                    <button
   408→                      onClick={handleRemoveCover}
   409→                      className="absolute top-2 right-2 flex h-7 w-7 items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80"
   410→                    >
   411→                      <X className="h-3.5 w-3.5" />
   412→                    </button>
   413→                  </div>
   414→                ) : (
   415→                  <button
   416→                    type="button"
   417→                    onClick={handleCoverUpload}
   418→                    className="flex h-48 w-full flex-col items-center justify-center gap-3 rounded-lg border-2 border-dashed border-border/70 bg-muted/30 transition-colors hover:border-primary/50 hover:bg-muted/50"
   419→                  >
   420→                    <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted">
   421→                      <ImageIcon className="h-5 w-5 text-muted-foreground" />
   422→                    </div>
   423→                    <div className="text-center">
   424→                      <p className="text-sm font-medium text-foreground">
   425→                        Upload cover image
   426→                      </p>
   427→                      <p className="mt-0.5 text-xs text-muted-foreground">
   428→                        Recommended 1920×1080
   429→                      </p>
   430→                    </div>
   431→                  </button>
   432→                )}
   433→
   434→                {/* URL Input */}
   435→                {urlInputOpen && (
   436→                  <div className="flex items-center gap-2">
   437→                    <Input
   438→                      placeholder="Paste image URL here..."
   439→                      value={imageUrl}
   440→                      onChange={(e) => setImageUrl(e.target.value)}
   441→                      onKeyDown={(e) => e.key === "Enter" && handleUrlConfirm()}
   442→                      className="flex-1"
   443→                    />
   444→                    <Button size="sm" onClick={handleUrlConfirm}>
   445→                      Apply
   446→                    </Button>
   447→                    <Button
   448→                      size="sm"
   449→                      variant="ghost"
   450→                      onClick={() => {
   451→                        setUrlInputOpen(false);
   452→                        setImageUrl("");
   453→                      }}
   454→                    >
   455→                      Cancel
   456→                    </Button>
   457→                  </div>
   458→                )}
   459→                <input
   460→                  ref={fileInputRef}
   461→                  type="file"
   462→                  accept="image/*"
   463→                  className="hidden"
   464→                />
   465→              </div>
   466→
   467→              <Separator />
   468→
   469→              {/* Course Title */}
   470→              <div className="space-y-2">
   471→                <div className="flex items-center justify-between">
   472→                  <Label htmlFor="course-title" className="text-sm font-medium">
   473→                    Course Title <span className="text-destructive">*</span>
   474→                  </Label>
   475→                  <span className="text-xs text-muted-foreground">
   476→                    {title.length}/{MAX_TITLE_LENGTH}
   477→                  </span>
   478→                </div>
   479→                <Input
   480→                  id="course-title"
   481→                  placeholder="Enter course title"
   482→                  value={title}
   483→                  onChange={(e) => {
   484→                    if (e.target.value.length <= MAX_TITLE_LENGTH) {
   485→                      setTitle(e.target.value);
   486→                    }
   487→                  }}
   488→                  className="h-10"
   489→                />
   490→              </div>
   491→
   492→              {/* Course Description */}
   493→              <div className="space-y-2">
   494→                <div className="flex items-center justify-between">
   495→                  <Label
   496→                    htmlFor="course-description"
   497→                    className="text-sm font-medium"
   498→                  >
   499→                    Description
   500→                  </Label>
   501→                  <span className="text-xs text-muted-foreground">
   502→                    {description.length}/{MAX_COURSE_DESC_LENGTH}
   503→                  </span>
   504→                </div>
   505→                <Textarea
   506→                  id="course-description"
   507→                  placeholder="Describe what this course is about..."
   508→                  value={description}
   509→                  onChange={(e) => {
   510→                    if (e.target.value.length <= MAX_COURSE_DESC_LENGTH) {
   511→                      setDescription(e.target.value);
   512→                    }
   513→                  }}
   514→                  rows={4}
   515→                  className="resize-none"
   516→                />
   517→              </div>
   518→
   519→              <Separator />
   520→
   521→              {/* Category */}
   522→              <div className="space-y-2">
   523→                <Label className="text-sm font-medium">Category</Label>
   524→                <Select value={categoryId} onValueChange={setCategoryId}>
   525→                  <SelectTrigger className="w-full h-10">
   526→                    <SelectValue placeholder="Select a category" />
   527→                  </SelectTrigger>
   528→                  <SelectContent>
   529→                    {categories.map((cat) => (
   530→                      <SelectItem key={cat.id} value={cat.id}>
   531→                        {cat.name}
   532→                      </SelectItem>
   533→                    ))}
   534→                  </SelectContent>
   535→                </Select>
   536→              </div>
   537→
   538→              {/* Language */}
   539→              <div className="space-y-3">
   540→                <Label className="text-sm font-medium">Language</Label>
   541→                <RadioGroup
   542→                  value={language}
   543→                  onValueChange={setLanguage}
   544→                  className="flex gap-6"
   545→                >
   546→                  <div className="flex items-center gap-2">
   547→                    <RadioGroupItem value="chinese" id="lang-zh" />
   548→                    <Label
   549→                      htmlFor="lang-zh"
   550→                      className="cursor-pointer text-sm font-normal"
   551→                    >
   552→                      Chinese
   553→                    </Label>
   554→                  </div>
   555→                  <div className="flex items-center gap-2">
   556→                    <RadioGroupItem value="english" id="lang-en" />
   557→                    <Label
   558→                      htmlFor="lang-en"
   559→                      className="cursor-pointer text-sm font-normal"
   560→                    >
   561→                      English
   562→                    </Label>
   563→                  </div>
   564→                </RadioGroup>
   565→              </div>
   566→
   567→              <Separator />
   568→
   569→              {/* Save Button */}
   570→              <Button
   571→                onClick={handleSave}
   572→                disabled={saving || !title.trim()}
   573→                className="w-full h-10"
   574→              >
   575→                {saving ? (
   576→                  <>
   577→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   578→                    Saving...
   579→                  </>
   580→                ) : (
   581→                  "Save Course"
   582→                )}
   583→              </Button>
   584→            </CardContent>
   585→          </Card>
   586→        </div>
   587→
   588→        {/* ======== Right Panel: Lessons (2 cols) ======== */}
   589→        <div className="lg:col-span-2">
   590→          <Card className="border-border/50">
   591→            <CardContent className="p-6">
   592→              {/* Header */}
   593→              <div className="mb-4 flex items-center justify-between">
   594→                <div className="flex items-center gap-2">
   595→                  <h2 className="text-base font-semibold text-foreground">
   596→                    Lessons
   597→                  </h2>
   598→                  <Badge
   599→                    variant="secondary"
   600→                    className="text-xs font-normal"
   601→                  >
   602→                    {lessons.length}/{MAX_LESSONS}
   603→                  </Badge>
   604→                </div>
   605→                <div className="flex items-center gap-2">
   606→                  <Button
   607→                    size="sm"
   608→                    variant="outline"
   609→                    onClick={handleOpenModal}
   610→                    disabled={lessons.length >= MAX_LESSONS}
   611→                    className="gap-1.5"
   612→                  >
   613→                    <Plus className="h-3.5 w-3.5" />
   614→                    Add
   615→                  </Button>
   616→                </div>
   617→              </div>
   618→
   619→              {/* Lesson List */}
   620→              {lessons.length === 0 ? (
   621→                <div className="flex flex-col items-center justify-center rounded-lg border border-dashed border-border/70 bg-muted/20 py-16">
   622→                  <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted">
   623→                    <FileText className="h-5 w-5 text-muted-foreground" />
   624→                  </div>
   625→                  <p className="mt-3 text-sm text-muted-foreground">
   626→                    No lessons yet
   627→                  </p>
   628→                  <p className="mt-1 text-xs text-muted-foreground/70">
   629→                    Click <span className="font-medium text-foreground">+ Add</span> to create a lesson, then use the AI generate button to create content.
   630→                  </p>
   631→                </div>
   632→              ) : (
   633→                <div className="space-y-2 max-h-[600px] overflow-y-auto pr-1">
   634→                  {lessons.map((lesson, index) => (

   640→                      expanded={expandedLessonId === lesson.id}
   641→                      onToggleExpand={() =>
   642→                        setExpandedLessonId((prev) =>
   643→                          prev === lesson.id ? null : lesson.id

   645→                      }
   646→                      onMoveUp={() => handleMoveLesson(index, "up")}
   647→                      onMoveDown={() => handleMoveLesson(index, "down")}

   651→                </div>
   652→              )}
   653→            </CardContent>
   654→          </Card>
   655→        </div>
   656→      </div>
   657→
   658→      {/* ======== Add Lesson Modal ======== */}
   659→      <Dialog open={modalOpen} onOpenChange={(open) => { if (!open && generating) return; setModalOpen(open); }}>
   660→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">

   678→                <span className="text-xs text-muted-foreground">
   679→                  {lessonName.length}/{MAX_TITLE_LENGTH}
   680→                </span>
   681→              </div>
   682→              <Input
   683→                placeholder="e.g., Introduction to Machine Learning"
   684→                value={lessonName}
   685→                onChange={(e) => {
   686→                  if (e.target.value.length <= MAX_TITLE_LENGTH) {
   687→                    setLessonName(e.target.value);
   688→                  }
   689→                }}
   690→                className="h-10"
   691→                disabled={generating}
   692→              />

   702→                  {lessonPrompt.length}/{MAX_DESC_LENGTH}
   703→                </span>
   704→              </div>
   705→              <Textarea
   706→                placeholder="Describe the content you want to generate..."
   707→                value={lessonPrompt}
   708→                onChange={(e) => {
   709→                  if (e.target.value.length <= MAX_DESC_LENGTH) {

   720→            <div className="space-y-2">
   721→              <Label className="text-sm font-medium">PDF Courseware</Label>
   722→              <button
   723→                type="button"
   724→                onClick={handleLessonPdfUpload}
   725→                disabled={generating}
   726→                className="flex w-full items-center justify-center gap-2 rounded-lg border border-dashed border-border/70 bg-muted/20 px-4 py-5 text-sm text-muted-foreground transition-colors hover:border-primary/50 hover:bg-muted/30 disabled:opacity-50"
   727→              >
   728→                <FileUp className="h-4 w-4" />
   729→                {lessonPdfName ? (
   730→                  <span className="text-foreground font-medium">
   731→                    {lessonPdfName}
   732→                  </span>

   737→              <input
   738→                ref={lessonFileInputRef}
   739→                type="file"
   740→                accept=".pdf"
   741→                className="hidden"
   742→                onChange={handleLessonFileChange}
   743→              />
   744→            </div>
   745→
   746→            {/* Language */}
   747→            <div className="space-y-3">
   748→              <Label className="text-sm font-medium">

   750→              </Label>
   751→              <RadioGroup
   752→                value={lessonLanguage}
   753→                onValueChange={setLessonLanguage}
   754→                className="flex gap-6"
   755→                disabled={generating}

   758→                  <RadioGroupItem
   759→                    value="chinese"
   760→                    id="sec-lang-zh"
   761→                  />
   762→                  <Label
   763→                    htmlFor="sec-lang-zh"

   798→                    </div>

   800→                </div>

   826→                <Button
   827→                  variant="ghost"

   830→                  Cancel
   831→                </Button>
   832→                <Button
   833→                  onClick={handleGenerateLesson}
   834→                  disabled={!lessonName.trim() || !lessonPrompt.trim()}
   835→                  className="gap-2"
   836→                >
   837→                  <Sparkles className="h-4 w-4" />
   838→                  Generate
   839→                </Button>
   840→              </>
   841→            )}
   842→          </DialogFooter>
   843→        </DialogContent>
   844→      </Dialog>
   845→    </div>
   846→  );

   864→function LessonCard({
   865→  lesson,
   866→  index,
   867→  totalCount,
   868→  expanded,
   869→  onToggleExpand,
   870→  onMoveUp,
   871→  onMoveDown,

   924→              <>
   925→                <span className="inline-flex items-center gap-1 rounded-full bg-emerald-100 dark:bg-emerald-900/30 px-1.5 py-0.5 text-[10px] font-medium text-emerald-700 dark:text-emerald-400">
   926→                  <span className="inline-block h-1.5 w-1.5 rounded-full bg-emerald-500" />

   928→                </span>

   930→                <span>{lesson.language === "chinese" ? "中文" : "English"}</span>

   932→            ) : (
   933→              <span className="rounded-full bg-amber-100 dark:bg-amber-900/30 px-1.5 py-0.5 text-[10px] font-medium text-amber-700 dark:text-amber-400">
   934→                No content
   935→              </span>
   936→            )}
   937→          </div>
   938→        </div>
   939→
   940→        {/* Preview Toggle */}

   943→            onClick={onToggleExpand}

   945→              expanded
   946→                ? "bg-primary/10 text-primary"
   947→                : "text-muted-foreground/50 hover:text-muted-foreground hover:bg-muted"
   948→            }`}
   949→            title={expanded ? "Hide preview" : "Show preview"}
   950→          >
   951→            <Eye className="h-3.5 w-3.5" />
   952→          </button>
   953→        )}
   954→
   955→        {/* Delete Button */}
   956→        <button
   957→          onClick={onDelete}
   958→          className="flex h-7 w-7 shrink-0 items-center justify-center rounded-md text-muted-foreground/50 opacity-0 transition-all hover:bg-destructive/10 hover:text-destructive group-hover:opacity-100"

   960→        >
   961→          <Trash2 className="h-3.5 w-3.5" />
   962→        </button>

   971→              sandbox="allow-same-origin"
   972→              className="w-full border-0"
   973→              style={{ aspectRatio: "16/9" }}

   975→            />
   976→          </div>
   977→        </div>
   978→      )}
   979→    </div>
   980→  );
   981→}

   987→/** Build a live preview document from raw streaming HTML fragments */
   988→function buildLivePreviewDoc(rawHtml: string, title: string): string {
   989→  return `<!DOCTYPE html>
   990→<html lang="en">
   991→<head>
   992→<meta charset="UTF-8" />
   993→<meta name="viewport" content="width=device-width, initial-scale=1.0" />
   994→<title>${escapeAttr(title)}</title>
   995→<script src="https://cdn.tailwindcss.com"></script>
   996→<style>
   997→  body { margin: 0; padding: 0; font-family: system-ui, -apple-system, sans-serif; }
   998→  * { box-sizing: border-box; }
   999→  .streaming-cursor::after {
  1000→    content: '▊';
  1001→    animation: blink 0.7s infinite;
  1002→    color: currentColor;
  1003→    opacity: 0.7;
  1004→  }
  1005→  @keyframes blink { 0%, 50% { opacity: 0.7; } 51%, 100% { opacity: 0; } }
  1006→</style>
  1007→</head>
  1008→<body class="bg-white dark:bg-zinc-900 text-zinc-900 dark:text-zinc-100">
  1009→  <div class="streaming-cursor">${rawHtml}</div>
  1010→</body>
  1011→</html>`;
  1012→}
  1013→
  1014→function escapeAttr(str: string): string {
  1015→  return str
  1016→    .replace(/&/g, "&amp;")
  1017→    .replace(/</g, "&lt;")
  1018→    .replace(/>/g, "&gt;")
  1019→    .replace(/"/g, "&quot;");
  1020→}
