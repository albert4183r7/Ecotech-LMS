     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import {
     5→  ZoomIn,
     6→  ZoomOut,
     7→  Maximize2,
     8→  RotateCcw,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  X,
    12→  GraduationCap,
    13→  Loader2,
    14→  StickyNote,
    15→  Bookmark,
    16→  BookmarkCheck,
    17→  Trash2,
    18→  Send,
    19→  FileDown,
    20→} from "lucide-react";
    21→import { toast } from "sonner";
    22→import { Button } from "@/components/ui/button";
    23→import { Separator } from "@/components/ui/separator";
    24→import { Tooltip, TooltipTrigger, TooltipContent } from "@/components/ui/tooltip";
    25→import {
    26→  Sheet,
    27→  SheetContent,
    28→  SheetHeader,
    29→  SheetTitle,
    30→  SheetDescription,
    31→} from "@/components/ui/sheet";
    32→import { ScrollArea } from "@/components/ui/scroll-area";
    33→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    34→import type { SlideContent, ClassroomState } from "@/types/lms";
    35→import { StudyTimer } from "@/components/lms/study-timer";
    36→
    37→const MIN_ZOOM = 50;
    38→const MAX_ZOOM = 200;
    39→const ZOOM_STEP = 25;
    40→
    41→export function ClassroomPage() {
    42→  const { classroomState, goBack } = useNavigationStore();
    43→  const userId = useUserStore((s) => s.currentUserId);
    44→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    45→  const [zoom, setZoom] = useState(100);
    46→  const [loadingSlides, setLoadingSlides] = useState(false);
    47→  const [slideDirection, setSlideDirection] = useState<"forward" | "back">("forward");
    48→  const [isAnimating, setIsAnimating] = useState(false);
    49→  const hasFetchedRef = useRef(false);
    50→  const confettiShownRef = useRef(false);
    51→  const [showConfetti, setShowConfetti] = useState(false);
    52→
    53→  // ─── Notes Sidebar State ──────────────────────
    54→  interface Note {
    55→    id: string;
    56→    content: string;
    57→    slideNumber: number;
    58→    bookmarked: boolean;
    59→    createdAt?: string;
    60→  }
    61→  const [notes, setNotes] = useState<Note[]>([]);
    62→  const [notesSidebarOpen, setNotesSidebarOpen] = useState(false);
    63→  const [newNoteContent, setNewNoteContent] = useState("");
    64→  const [savingNote, setSavingNote] = useState(false);
    65→  const notesEndRef = useRef<HTMLDivElement>(null);
    66→
    67→  // ─── Keyboard Hint Fade ────────────────────────
    68→  const [showKeyboardHint, setShowKeyboardHint] = useState(true);
    69→  const [isMobile, setIsMobile] = useState(false);
    70→  const [downloadingPptx, setDownloadingPptx] = useState(false);
    71→
    72→  useEffect(() => {
    73→    const mq = window.matchMedia("(max-width: 1023px)");
    74→    setIsMobile(mq.matches);
    75→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
    76→    mq.addEventListener("change", handler);
    77→    return () => mq.removeEventListener("change", handler);
    78→  }, []);
    79→
    80→  // Keep a working copy of classroomState so we can mutate currentSlide
    81→  useEffect(() => {
    82→    if (classroomState) {
    83→      setLocalState({ ...classroomState });
    84→      setZoom(100);
    85→      hasFetchedRef.current = false;
    86→    }
    87→  }, [classroomState]);
    88→
    89→  // If slides were empty initially, try fetching them
    90→  useEffect(() => {
    91→    if (!localState || localState.slides.length > 0 || hasFetchedRef.current) return;
    92→    hasFetchedRef.current = true;
    93→    const fetchSlides = async () => {
    94→      setLoadingSlides(true);
    95→      try {
    96→        const res = await fetch(`/api/sections/${localState.sectionId}`);
    97→        if (!res.ok) return;
    98→        const json = await res.json();
    99→        if (json.success && json.data.content) {
   100→          const slides: SlideContent[] = Array.isArray(json.data.content)
   101→            ? json.data.content
   102→            : [];
   103→          setLocalState((prev) =>
   104→            prev
   105→              ? {
   106→                  ...prev,
   107→                  slides,
   108→                  totalPages: slides.length || prev.totalPages,
   109→                }
   110→              : prev
   111→          );
   112→        }
   113→      } catch {
   114→        // Silently fail
   115→      } finally {
   116→        setLoadingSlides(false);
   117→      }
   118→    };
   119→    fetchSlides();
   120→  }, [localState?.sectionId, localState?.slides.length]);
   121→
   122→  /** Navigate slides with transition animation */
   123→  const goToSlide = useCallback(
   124→    (index: number) => {
   125→      if (!localState || isAnimating) return;
   126→      const next = Math.max(0, Math.min(index, localState.totalPages - 1));
   127→      if (next === localState.currentSlide) return;
   128→      setSlideDirection(next > localState.currentSlide ? "forward" : "back");
   129→      setIsAnimating(true);
   130→      // Small delay for exit animation, then update slide
   131→      setTimeout(() => {
   132→        setLocalState((prev) =>
   133→          prev ? { ...prev, currentSlide: next } : prev
   134→        );
   135→        setIsAnimating(false);
   136→      }, 150);
   137→    },
   138→    [localState, isAnimating]
   139→  );
   140→
   141→  /** Persist progress to API when slide changes */
   142→  useEffect(() => {
   143→    if (!localState || !userId) return;
   144→    const saveProgress = async () => {
   145→      try {
   146→        // Get enrollment ID from the API
   147→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   148→        const enrollJson = await enrollRes.json();
   149→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   150→          const enrollment = enrollJson.data.find(
   151→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   152→          );
   153→          if (enrollment) {
   154→            const completed = localState.currentSlide >= localState.totalPages - 1;
   155→            await fetch("/api/progress", {
   156→              method: "POST",
   157→              headers: { "Content-Type": "application/json" },
   158→              body: JSON.stringify({
   159→                enrollmentId: enrollment.id,
   160→                sectionId: localState.sectionId,
   161→                currentPage: localState.currentSlide + 1,
   162→                completed,
   163→              }),
   164→            });
   165→          }
   166→        }
   167→      } catch {
   168→        // Progress saving is best-effort
   169→      }
   170→    };
   171→    // Debounce to avoid excessive API calls
   172→    const timer = setTimeout(saveProgress, 500);
   173→    return () => clearTimeout(timer);
   174→  }, [localState?.currentSlide]);
   175→
   176→  const goPrev = () => goToSlide((localState?.currentSlide ?? 0) - 1);
   177→  const goNext = () => goToSlide((localState?.currentSlide ?? 0) + 1);
   178→
   179→  // ─── Keyboard shortcuts ──────────────────────────────────────
   180→  useEffect(() => {
   181→    function handleKeyDown(e: KeyboardEvent) {
   182→      const tag = (e.target as HTMLElement)?.tagName;
   183→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   184→
   185→      switch (e.key) {
   186→        case "Escape":
   187→          e.preventDefault();
   188→          goBack();
   189→          break;
   190→        case "ArrowLeft":
   191→          e.preventDefault();
   192→          goPrev();
   193→          break;
   194→        case "ArrowRight":
   195→          e.preventDefault();
   196→          goNext();
   197→          break;
   198→        case " ":
   199→          e.preventDefault();
   200→          goNext();
   201→          break;
   202→      }
   203→    }
   204→
   205→    document.addEventListener("keydown", handleKeyDown);
   206→
   207→    // Listen for custom event from KeyboardShortcuts component
   208→    function handleCustomNextSlide() {
   209→      goNext();
   210→    }
   211→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   212→
   213→    return () => {
   214→      document.removeEventListener("keydown", handleKeyDown);
   215→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   216→    };
   217→  }, [goBack, goPrev, goNext]);
   218→
   219→  /** Zoom controls */
   220→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   221→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   222→  const zoomFit = () => setZoom(100);
   223→
   224→  // ─── Confetti on section completion ───────────
   225→  useEffect(() => {
   226→    if (!localState) return;
   227→    const isOnLastSlide = localState.currentSlide === localState.totalPages - 1;
   228→    if (isOnLastSlide && !confettiShownRef.current) {
   229→      confettiShownRef.current = true;
   230→      setShowConfetti(true);
   231→      toast.success("🎉 You completed the course! Great job!");
   232→      // Auto-cleanup after 3.5s (animation lasts ~3s + buffer)
   233→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   234→      return () => clearTimeout(timer);
   235→    }
   236→  }, [localState?.currentSlide]);
   237→
   238→  // Reset confetti flag when classroom state changes (new section)
   239→  useEffect(() => {
   240→    confettiShownRef.current = false;
   241→    setShowConfetti(false);
   242→  }, [classroomState?.sectionId]);
   243→
   244→  // ─── Keyboard Hint Auto-Fade ────────────────────
   245→  useEffect(() => {
   246→    setShowKeyboardHint(true);
   247→    const timer = setTimeout(() => setShowKeyboardHint(false), 5000);
   248→    return () => clearTimeout(timer);
   249→  }, [localState?.sectionId]);
   250→
   251→  // ─── Notes CRUD ─────────────────────────────────
   252→  const fetchNotes = useCallback(async () => {
   253→    if (!localState || !userId) return;
   254→    try {
   255→      const res = await fetch(
   256→        `/api/notes?userId=${userId}&courseId=${localState.courseId}&sectionId=${localState.sectionId}`
   257→      );
   258→      if (!res.ok) return;
   259→      const json = await res.json();
   260→      if (json.success && Array.isArray(json.data)) {
   261→        setNotes(json.data);
   262→      }
   263→    } catch {
   264→      // Silently fail
   265→    }
   266→  }, [localState, userId]);
   267→
   268→  // Fetch notes when section changes or sidebar opens
   269→  useEffect(() => {
   270→    if (notesSidebarOpen) fetchNotes();
   271→  }, [notesSidebarOpen, fetchNotes, localState?.sectionId]);
   272→
   273→  const createNote = useCallback(async () => {
   274→    if (!localState || !userId || !newNoteContent.trim()) return;
   275→    setSavingNote(true);
   276→    try {
   277→      const res = await fetch("/api/notes", {
   278→        method: "POST",
   279→        headers: { "Content-Type": "application/json" },
   280→        body: JSON.stringify({
   281→          userId,
   282→          courseId: localState.courseId,
   283→          sectionId: localState.sectionId,
   284→          content: newNoteContent.trim(),
   285→          slideNumber: localState.currentSlide + 1,
   286→        }),
   287→      });
   288→      if (!res.ok) return;
   289→      const json = await res.json();
   290→      if (json.success && json.data) {
   291→        setNotes((prev) => [...prev, json.data]);
   292→        setNewNoteContent("");
   293→        setTimeout(() => notesEndRef.current?.scrollIntoView({ behavior: "smooth" }), 50);
   294→      }
   295→    } catch {
   296→      toast.error("Failed to save note");
   297→    } finally {
   298→      setSavingNote(false);
   299→    }
   300→  }, [localState, userId, newNoteContent]);
   301→
   302→  const deleteNote = useCallback(async (noteId: string) => {
   303→    try {
   304→      const res = await fetch(`/api/notes?id=${noteId}`, { method: "DELETE" });
   305→      if (!res.ok) return;
   306→      const json = await res.json();
   307→      if (json.success) {
   308→        setNotes((prev) => prev.filter((n) => n.id !== noteId));
   309→      }
   310→    } catch {
   311→      // Silently fail
   312→    }
   313→  }, []);
   314→
   315→  const toggleBookmark = useCallback((noteId: string) => {
   316→    setNotes((prev) =>
   317→      prev.map((n) =>
   318→        n.id === noteId ? { ...n, bookmarked: !n.bookmarked } : n
   319→      )
   320→    );
   321→  }, []);
   322→
   323→  // ─── Download as PPTX ─────────────────────────────
   324→  const handleDownloadPptx = useCallback(async () => {
   325→    if (!localState || localState.slides.length === 0 || downloadingPptx) return;
   326→    setDownloadingPptx(true);
   327→    try {
   328→      const res = await fetch('/api/generate-pptx', {
   329→        method: 'POST',
   330→        headers: { 'Content-Type': 'application/json' },
   331→        body: JSON.stringify({
   332→          slides: localState.slides,
   333→          courseName: localState.courseTitle,
   334→        }),
   335→      });
   336→      if (!res.ok) {
   337→        toast.error('Failed to generate PPT');
   338→        return;
   339→      }
   340→      const blob = await res.blob();
   341→      const url = URL.createObjectURL(blob);
   342→      const a = document.createElement('a');
   343→      a.href = url;
   344→      const safeName = localState.courseTitle
   345→        .replace(/[^a-zA-Z0-9\s-]/g, '')
   346→        .replace(/\s+/g, '-')
   347→        .toLowerCase()
   348→        .slice(0, 60);
   349→      a.download = `${safeName}.pptx`;
   350→      document.body.appendChild(a);
   351→      a.click();
   352→      a.remove();
   353→      URL.revokeObjectURL(url);
   354→      toast.success('PPT downloaded successfully!');
   355→    } catch {
   356→      toast.error('Failed to download PPT');
   357→    } finally {
   358→      setDownloadingPptx(false);
   359→    }
   360→  }, [localState, downloadingPptx]);
   361→
   362→  // ─── No state guard ─────────────────────────────
   363→  if (!localState) {
   364→    return (
   365→      <div className="flex h-screen items-center justify-center bg-muted/30">
   366→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   367→      </div>
   368→    );
   369→  }
   370→
   371→  const currentSlide =
   372→    localState.slides[localState.currentSlide] ?? null;
   373→  const isFirst = localState.currentSlide === 0;
   374→  const isLast = localState.currentSlide >= localState.totalPages - 1;
   375→  const progressPercent = localState.totalPages > 1
   376→    ? Math.round(((localState.currentSlide + 1) / localState.totalPages) * 100)
   377→    : 100;
   378→
   379→  return (
   380→    <div className="flex h-screen flex-col bg-muted/30 relative">
   381→      {/* ─── Mobile Notes Sheet ──────────────────── */}
   382→      <Sheet open={notesSidebarOpen && isMobile} onOpenChange={(open) => { if (!open) setNotesSidebarOpen(false); }}>
   383→        <SheetContent side="right" className="w-full sm:max-w-sm p-0">
   384→          <SheetHeader className="px-4 pt-4 pb-0">
   385→            <SheetTitle className="flex items-center gap-2 text-base">
   386→              <StickyNote className="h-4 w-4" />
   387→              Notes
   388→            </SheetTitle>
   389→            <SheetDescription>
   390→              Notes for slide {localState.currentSlide + 1} of {localState.totalPages}
   391→            </SheetDescription>
   392→          </SheetHeader>
   393→          <NotesSidebarContent
   394→            notes={notes}
   395→            currentSlide={localState.currentSlide}
   396→            newNoteContent={newNoteContent}
   397→            savingNote={savingNote}
   398→            onContentChange={setNewNoteContent}
   399→            onCreate={createNote}
   400→            onDelete={deleteNote}
   401→            onToggleBookmark={toggleBookmark}
   402→            endRef={notesEndRef}
   403→          />
   404→        </SheetContent>
   405→      </Sheet>
   406→      {/* ─── Confetti Celebration Overlay ─────── */}
   407→      {showConfetti && <ConfettiCelebration />}
   408→      {/* ─── Top Progress Bar ─────────────────────── */}
   409→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   410→        <div
   411→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   412→          style={{ width: `${progressPercent}%` }}
   413→        />
   414→      </div>
   415→
   416→      {/* ─── Top Bar ──────────────────────────────── */}
   417→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   418→        {/* Logo, Section Title, Course Title */}
   419→        <div className="flex items-center gap-3 min-w-0">
   420→          <div className="flex items-center gap-2 shrink-0">
   421→            <img src="/ecotech-logo.png" alt="Ecotech" className="h-7 w-7 rounded-md object-contain" />
   422→            <img src="/ecotech-name.png" alt="Ecotech" className="h-5 w-auto hidden sm:inline object-contain" />
   423→          </div>
   424→          <Separator orientation="vertical" className="h-5" />
   425→          <div className="min-w-0">
   426→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   427→              {localState.sectionTitle}
   428→            </h2>
   429→            <p className="text-xs text-muted-foreground truncate leading-tight">
   430→              {localState.courseTitle}
   431→            </p>
   432→          </div>
   433→        </div>
   434→
   435→        {/* Slide Counter - Enhanced */}
   436→        <span className="shrink-0 inline-flex items-center gap-1.5 rounded-full bg-primary/10 px-3 py-1 text-sm text-primary tabular-nums">
   437→          <span className="font-semibold">Slide {localState.currentSlide + 1}</span>
   438→          <span className="text-primary/40 font-normal">of</span>
   439→          <span className="font-semibold">{localState.totalPages}</span>
   440→        </span>
   441→      </header>
   442→
   443→      {/* ─── Main Layout: Slide + Desktop Notes Sidebar ── */}
   444→      <div className="flex flex-1 overflow-hidden">
   445→      {/* ─── Slide Content Area ────────────────────── */}
   446→      <div className="flex-1 overflow-auto flex flex-col items-center py-8 px-4 sm:px-6">
   447→        <div
   448→          className={`
   449→            bg-card rounded-2xl shadow-lg border w-full max-w-3xl
   450→            ring-1 ring-black/5
   451→            dark:ring-white/5
   452→            paper-texture
   453→            ${isAnimating
   454→              ? (slideDirection === "forward" ? "slide-exit" : "opacity-0 -translate-x-2 transition-all duration-150")
   455→              : "slide-enter"
   456→            }
   457→          `}
   458→          style={{
   459→            transform: `scale(${zoom / 100})`,
   460→            transformOrigin: "top center",
   461→          }}
   462→        >
   463→          {loadingSlides ? (
   464→            <div className="flex items-center justify-center py-32">
   465→              <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   466→            </div>
   467→          ) : currentSlide ? (
   468→            <div className="p-8 sm:p-12 max-w-2xl mx-auto">
   469→              <SlideRenderer slide={currentSlide} />
   470→            </div>
   471→          ) : (
   472→            <div className="flex items-center justify-center py-32 text-muted-foreground">
   473→              <p>No slide content available.</p>
   474→            </div>
   475→          )}
   476→        </div>
   477→
   478→        {/* ─── Slide Progress Dots ────────────────── */}
   479→        <div className="slide-progress-indicator mt-6 shrink-0">
   480→          {Array.from({ length: localState.totalPages }, (_, i) => {
   481→            const isCompleted = i < localState.currentSlide;
   482→            const isActive = i === localState.currentSlide;
   483→            return (
   484→              <button
   485→                key={i}
   486→                className={`dot ${isActive ? "active" : ""} ${isCompleted ? "completed" : ""}`}
   487→                onClick={() => goToSlide(i)}
   488→                aria-label={`Go to slide ${i + 1}`}
   489→              />
   490→            );
   491→          })}
   492→        </div>
   493→      </div>
   494→
   495→      {/* ─── Desktop Notes Sidebar (slide-in panel) ── */}
   496→      <aside
   497→        className={`hidden lg:flex flex-col shrink-0 border-l bg-card transition-all duration-300 ease-in-out overflow-hidden ${
   498→          notesSidebarOpen ? "w-80 opacity-100" : "w-0 opacity-0 border-l-0"
   499→        }`}
   500→      >
   501→        <div className="flex flex-col h-full w-80">
   502→          {/* Sidebar Header */}
   503→          <div className="flex items-center justify-between px-4 py-3 border-b">
   504→            <div className="flex items-center gap-2">
   505→              <StickyNote className="h-4 w-4 text-primary" />
   506→              <span className="text-sm font-semibold">Notes</span>
   507→            </div>
   508→            <Button
   509→              variant="ghost"
   510→              size="icon"
   511→              className="h-7 w-7"
   512→              onClick={() => setNotesSidebarOpen(false)}
   513→            >
   514→              <X className="h-4 w-4" />
   515→            </Button>
   516→          </div>
   517→          <NotesSidebarContent
   518→            notes={notes}
   519→            currentSlide={localState.currentSlide}
   520→            newNoteContent={newNoteContent}
   521→            savingNote={savingNote}
   522→            onContentChange={setNewNoteContent}
   523→            onCreate={createNote}
   524→            onDelete={deleteNote}
   525→            onToggleBookmark={toggleBookmark}
   526→            endRef={notesEndRef}
   527→          />
   528→        </div>
   529→      </aside>
   530→      </div>
   531→
   532→      {/* ─── Keyboard Shortcuts Hint (auto-fades) ── */}
   533→      <div
   534→        className={`shrink-0 frosted-glass border-t px-4 py-1.5 text-center transition-opacity duration-700 ${
   535→          showKeyboardHint ? "opacity-100" : "opacity-0 pointer-events-none h-0 py-0 overflow-hidden"
   536→        }`}
   537→      >
   538→        <p className="text-xs text-muted-foreground/70 tracking-wide">
   539→          ← → Navigate&nbsp;&nbsp;|&nbsp;&nbsp;Space: Next&nbsp;&nbsp;|&nbsp;&nbsp;Esc: Exit
   540→        </p>
   541→      </div>
   542→
   543→      {/* ─── Bottom Controls (Frosted Glass) ───────── */}
   544→      <footer className="shrink-0 border-t frosted-glass">
   545→        <div className="mx-auto flex h-16 max-w-5xl items-center justify-between px-4 sm:px-6">
   546→          {/* Zoom Controls */}
   547→          <div className="flex items-center gap-1">
   548→            <Tooltip>
   549→              <TooltipTrigger asChild>
   550→                <Button
   551→                  variant="ghost"
   552→                  size="icon"
   553→                  className="h-8 w-8"
   554→                  onClick={zoomOut}
   555→                  disabled={zoom <= MIN_ZOOM}
   556→                >

   829→interface ConfettiParticle {
   830→  id: number;
   831→  left: number;
   832→  color: string;
   833→  shape: ParticleShape;
   834→  size: number;

   842→  const particles = useMemo<ConfettiParticle[]>(() => {
   843→    const result: ConfettiParticle[] = [];
   844→    const shapes: ParticleShape[] = ["circle", "rectangle", "triangle"];
   845→    const count = 80;
   846→
   847→    for (let i = 0; i < count; i++) {
   848→      const shape = shapes[i % 3 === 0 ? 0 : i % 3 === 1 ? 1 : 2];
   849→      result.push({
   850→        id: i,
   851→        left: Math.random() * 100,
   852→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   853→        shape,
   854→        size: Math.floor(Math.random() * 8) + 6, // 6-14px
   855→        fallDuration: `${(Math.random() * 1.5 + 2).toFixed(2)}s`, // 2-3.5s
   856→        fallDelay: `${(Math.random() * 0.8).toFixed(2)}s`, // 0-0.8s

   858→        spin: `${Math.floor(Math.random() * 720 + 360)}deg`, // 360-1080deg

   862→  }, []);

   890→              style={{
