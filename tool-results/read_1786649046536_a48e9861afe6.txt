    57→
    58→function CircularProgress({ percentage }: { percentage: number }) {
    59→  const radius = 70;
    60→  const stroke = 10;
    61→  const normalizedRadius = radius - stroke / 2;
    62→  const circumference = normalizedRadius * 2 * Math.PI;
    63→  const strokeDashoffset =
    64→    circumference - (percentage / 100) * circumference;
    65→
    66→  return (
    67→    <div className="relative inline-flex items-center justify-center">
    68→      <svg
    69→        width={radius * 2}
    70→        height={radius * 2}
    71→        className="-rotate-90"
    72→      >
    73→        {/* Background track */}
    74→        <circle
    75→          cx={radius}
    76→          cy={radius}
    77→          r={normalizedRadius}
    78→          fill="none"
    79→          stroke="currentColor"
    80→          strokeWidth={stroke}
    81→          className="text-muted/30"
    82→        />
    83→        {/* Progress arc with gradient stroke */}
    84→        <circle
    85→          cx={radius}
    86→          cy={radius}
    87→          r={normalizedRadius}
    88→          fill="none"
    89→          stroke="url(#progress-gradient)"
    90→          strokeWidth={stroke}
    91→          strokeLinecap="round"
    92→          strokeDasharray={circumference}
    93→          strokeDashoffset={strokeDashoffset}
    94→          className="transition-all duration-700 ease-out"
    95→        />
    96→        <defs>
    97→          <linearGradient
    98→            id="progress-gradient"
    99→            x1="0%"
   100→            y1="0%"
   101→            x2="100%"
   102→            y2="100%"
   103→          >
   104→            <stop offset="0%" stopColor="#06b6d4" />
   105→            <stop offset="50%" stopColor="#0891b2" />
   106→            <stop offset="100%" stopColor="#0d9488" />
   107→          </linearGradient>
   108→        </defs>
   109→      </svg>
   110→      <div className="absolute flex flex-col items-center justify-center">
   111→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   112→          {percentage}%
   113→        </span>
   114→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   115→          completed
   116→        </span>
   117→      </div>
   118→    </div>
   119→  );
   120→}

   392→
   393→  const fetchProfile = useCallback(async () => {
   394→    setLoading(true);
   395→    try {
   396→      const res = await fetch(`/api/user/${currentUserId}`);
   397→      const json = await res.json();
   398→      if (json.success) {
   399→        setProfile(json.data);
   400→      }
   401→    } catch {
   402→      /* silent */
   403→    } finally {
   404→      setLoading(false);
   405→    }
   406→  }, [currentUserId]);
   407→
   408→  // Fetch first completed course for certificate
   409→  useEffect(() => {
   410→    async function fetchCompletedCourse() {
   411→      try {
   412→        const res = await fetch('/api/enrollments?userId=' + currentUserId);
   413→        const json = await res.json();
   414→        if (json.success && json.data) {
   415→          const completed = json.data.find(
   416→            (e: { status: string; course: { title: string }; completedAt: string | null }) =>

   425→        /* silent */
   426→      }
   427→    }
   428→    fetchCompletedCourse();
   429→  }, [currentUserId]);
   430→
   431→  useEffect(() => {
   432→    fetchProfile();

   434→
   435→  if (loading) {
   436→    return <ProfileSkeleton />;

   438→
   439→  if (!profile) {
   440→    return (
   441→      <div className="flex items-center justify-center py-20">

   443→          Unable to load profile

   447→  }
   448→

   450→    .split(" ")
   451→    .map((w) => w.charAt(0).toUpperCase())
