     1→"use client";
     2→
     3→import { useCallback, useEffect, useState } from "react";
     4→import {
     5→  Star,
     6→  Heart,
     7→  Share2,
     8→  BookOpen,
     9→  Users,
    10→  Calendar,
    11→  ChevronLeft,
    12→  AlertTriangle,
    13→  FileText,
    14→  Loader2,
    15→  CheckCircle2,
    16→  Clock,
    17→  PlayCircle,
    18→  FileDown,
    19→} from "lucide-react";
    20→import { Button } from "@/components/ui/button";
    21→import { Badge } from "@/components/ui/badge";
    22→import { Separator } from "@/components/ui/separator";
    23→import { Skeleton } from "@/components/ui/skeleton";
    24→import {
    25→  Breadcrumb,
    26→  BreadcrumbItem,
    27→  BreadcrumbLink,
    28→  BreadcrumbList,
    29→  BreadcrumbPage,
    30→  BreadcrumbSeparator,
    31→} from "@/components/ui/breadcrumb";
    32→import {
    33→  Accordion,
    34→  AccordionContent,
    35→  AccordionItem,
    36→  AccordionTrigger,
    37→} from "@/components/ui/accordion";
    38→import { Progress } from "@/components/ui/progress";
    39→import { toast } from "sonner";
    40→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    41→import { DiscussionPanel } from "@/components/lms/discussion-panel";
    42→import { ProgressTimeline } from "@/components/lms/progress-timeline";
    43→import { StarRating } from "@/components/lms/star-rating";
    44→import type { CourseItem, SectionItem, ClassroomState, SlideContent } from "@/types/lms";
    45→
    46→type SectionProgress = {
    47→  sectionId: string;
    48→  currentPage: number;
    49→  completed: boolean;
    50→};
    51→
    52→type CourseDetailData = CourseItem & {
    53→  isEnrolled: boolean;
    54→  isFavorited: boolean;
    55→  enrollmentsCount?: number;
    56→  favoritesCount?: number;
    57→};
    58→
    59→export function CourseDetailPage() {
    60→  const { selectedCourseId, goBack, openClassroom, navigateTo } =
    61→    useNavigationStore();
    62→  const userId = useUserStore((s) => s.currentUserId);
    63→
    64→  const [course, setCourse] = useState<CourseDetailData | null>(null);
    65→  const [loading, setLoading] = useState(true);
    66→  const [error, setError] = useState<string | null>(null);
    67→  const [enrolling, setEnrolling] = useState(false);
    68→  const [togglingFav, setTogglingFav] = useState(false);
    69→  const [imgError, setImgError] = useState(false);
    70→  const [sectionProgress, setSectionProgress] = useState<SectionProgress[]>([]);
    71→  const [userRating, setUserRating] = useState<number | null>(null);
    72→  const [ratingCount, setRatingCount] = useState(0);
    73→  const [downloadingPptx, setDownloadingPptx] = useState(false);
    74→
    75→  /** Fetch course detail from API */
    76→  const fetchCourse = useCallback(async () => {
    77→    if (!selectedCourseId) return;
    78→    setLoading(true);
    79→    setError(null);
    80→    try {
    81→      const res = await fetch(
    82→        `/api/courses/${selectedCourseId}?userId=${userId}`
    83→      );
    84→      if (!res.ok) throw new Error("Failed to load course");
    85→      const json = await res.json();
    86→      if (json.success) {
    87→        setCourse(json.data);
    88→      } else {
    89→        throw new Error(json.error || "Unknown error");
    90→      }
    91→    } catch (err) {
    92→      setError(err instanceof Error ? err.message : "Failed to load course");
    93→    } finally {
    94→      setLoading(false);
    95→    }
    96→  }, [selectedCourseId, userId]);
    97→
    98→  useEffect(() => {
    99→    fetchCourse();
   100→  }, [fetchCourse]);
   101→
   102→  /** Fetch rating data */
   103→  const fetchRatingData = useCallback(async () => {
   104→    if (!selectedCourseId) return;
   105→    try {
   106→      const res = await fetch(
   107→        `/api/ratings?courseId=${selectedCourseId}&userId=${userId}`
   108→      );
   109→      if (res.ok) {
   110→        const json = await res.json();
   111→        if (json.success) {
   112→          setUserRating(json.data.userRating);
   113→          setRatingCount(json.data.count);
   114→          setCourse((prev) =>
   115→            prev ? { ...prev, rating: json.data.average } : prev
   116→          );
   117→        }
   118→      }
   119→    } catch {
   120→      // Best-effort
   121→    }
   122→  }, [selectedCourseId, userId]);
   123→
   124→  useEffect(() => {
   125→    fetchRatingData();
   126→  }, [fetchRatingData]);
   127→
   128→  /** Fetch section progress if enrolled */
   129→  const fetchSectionProgress = useCallback(async () => {
   130→    if (!userId || !course?.isEnrolled) return;
   131→    try {
   132→      const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   133→      const enrollJson = await enrollRes.json();
   134→      if (enrollJson.success && Array.isArray(enrollJson.data)) {
   135→        const enrollment = enrollJson.data.find(
   136→          (e: Record<string, unknown>) => e.courseId === selectedCourseId
   137→        );
   138→        if (enrollment) {
   139→          const progRes = await fetch(`/api/progress?enrollmentId=${enrollment.id}`);
   140→          const progJson = await progRes.json();
   141→          if (progJson.success && Array.isArray(progJson.data)) {
   142→            const mapped: SectionProgress[] = progJson.data.map((p: Record<string, unknown>) => ({
   143→              sectionId: (p.section as Record<string, unknown>)?.id ?? p.sectionId,
   144→              currentPage: (p.currentPage as number) || 0,
   145→              completed: (p.completed as boolean) || false,
   146→            }));
   147→            setSectionProgress(mapped);
   148→          }
   149→        }
   150→      }
   151→    } catch {
   152→      // Progress fetch is best-effort
   153→    }
   154→  }, [userId, course?.isEnrolled, selectedCourseId]);
   155→
   156→  useEffect(() => {
   157→    fetchSectionProgress();
   158→  }, [fetchSectionProgress]);
   159→
   160→  /** Handle enrollment */
   161→  const handleEnroll = async () => {
   162→    if (!selectedCourseId || enrolling) return;
   163→    setEnrolling(true);
   164→    try {
   165→      const res = await fetch("/api/enrollments", {
   166→        method: "POST",
   167→        headers: { "Content-Type": "application/json" },
   168→        body: JSON.stringify({ userId, courseId: selectedCourseId }),
   169→      });
   170→      if (res.ok) {
   171→        setCourse((prev) =>
   172→          prev
   173→            ? { ...prev, isEnrolled: true, studentCount: prev.studentCount + 1 }
   174→            : prev
   175→        );
   176→        toast.success("You're enrolled! Let's start learning.");
   177→      } else {
   178→        toast.error("Failed to enroll. Please try again.");
   179→      }
   180→    } catch {
   181→      toast.error("Network error. Please try again.");
   182→    } finally {
   183→      setEnrolling(false);
   184→    }
   185→  };
   186→
   187→  /** Handle favorite toggle */
   188→  const handleToggleFavorite = async (e: React.MouseEvent) => {
   189→    e.stopPropagation();
   190→    if (!selectedCourseId || togglingFav) return;
   191→    setTogglingFav(true);
   192→    try {
   193→      const res = await fetch("/api/favorites", {
   194→        method: "POST",
   195→        headers: { "Content-Type": "application/json" },
   196→        body: JSON.stringify({ userId, courseId: selectedCourseId }),
   197→      });
   198→      if (res.ok) {
   199→        const json = await res.json();
   200→        setCourse((prev) =>
   201→          prev
   202→            ? { ...prev, isFavorited: json.favorited }
   203→            : prev
   204→        );
   205→      } else {
   206→        toast.error("Failed to update favorite.");
   207→      }
   208→    } catch {
   209→      toast.error("Network error. Please try again.");
   210→    } finally {
   211→      setTogglingFav(false);
   212→    }
   213→  };
   214→
   215→  /** Download entire course as PPTX */
   216→  const handleDownloadCoursePptx = useCallback(async () => {
   217→    if (!course || downloadingPptx) return;
   218→    setDownloadingPptx(true);
   219→    try {
   220→      // Fetch all sections' content
   221→      const allSlides: SlideContent[] = [];
   222→      for (const section of course.sections) {
   223→        try {
   224→          const res = await fetch(`/api/sections/${section.id}`);
   225→          if (!res.ok) continue;
   226→          const json = await res.json();
   227→          if (json.success && json.data.content && Array.isArray(json.data.content)) {
   228→            allSlides.push(...json.data.content);
   229→          }
   230→        } catch {
   231→          // Skip sections that fail to load
   232→        }
   233→      }
   234→      if (allSlides.length === 0) {
   235→        toast.error('No slide content available for download.');
   236→        return;
   237→      }
   238→      const res = await fetch('/api/generate-pptx', {
   239→        method: 'POST',
   240→        headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify({ slides: allSlides, courseName: course.title }),
   242→      });
   243→      if (!res.ok) {
   244→        toast.error('Failed to generate PPT');
   245→        return;
   246→      }
   247→      const blob = await res.blob();
   248→      const url = URL.createObjectURL(blob);
   249→      const a = document.createElement('a');
   250→      a.href = url;
   251→      const safeName = course.title
   252→        .replace(/[^a-zA-Z0-9\s-]/g, '')
   253→        .replace(/\s+/g, '-')
   254→        .toLowerCase()
   255→        .slice(0, 60);
   256→      a.download = `${safeName}.pptx`;
   257→      document.body.appendChild(a);
   258→      a.click();
   259→      a.remove();
   260→      URL.revokeObjectURL(url);
   261→      toast.success('Course PPT downloaded successfully!');
   262→    } catch {
   263→      toast.error('Failed to download course PPT.');
   264→    } finally {
   265→      setDownloadingPptx(false);
   266→    }
   267→  }, [course, downloadingPptx]);
   268→
   269→  /** Handle share button */
   270→  const handleShare = async () => {
   271→    if (navigator.share) {
   272→      try {
   273→        await navigator.share({ title: course?.title, url: window.location.href });
   274→      } catch {
   275→        // User cancelled share
   276→      }
   277→    } else {
   278→      await navigator.clipboard.writeText(window.location.href);
   279→    }
   280→  };
   281→
   282→  /** Handle section click → open classroom */
   283→  const handleSectionClick = async (section: SectionItem) => {
   284→    if (!course) return;
   285→    try {
   286→      const res = await fetch(`/api/sections/${section.id}`);
   287→      if (!res.ok) return;
   288→      const json = await res.json();
   289→      if (json.success && json.data.content) {
   290→        const slides: SlideContent[] = Array.isArray(json.data.content)
   291→          ? json.data.content
   292→          : [];
   293→        const classroomState: ClassroomState = {
   294→          courseId: course.id,
   295→          courseTitle: course.title,
   296→          sectionId: section.id,
   297→          sectionTitle: section.title,
   298→          slides,
   299→          currentSlide: 0,
   300→          totalPages: slides.length || section.totalPages,
   301→        };
   302→        openClassroom(classroomState);
   303→      }
   304→    } catch {
   305→      // Silently fail
   306→    }
   307→  };
   308→
   309→  /** Render star rating */
   310→  const renderStars = (rating: number) => {
   311→    const stars = [];
   312→    for (let i = 1; i <= 5; i++) {
   313→      stars.push(
   314→        <Star
   315→          key={i}
   316→          className={`h-4 w-4 ${
   317→            i <= Math.round(rating)
   318→              ? "fill-amber-400 text-amber-400"
   319→              : "text-muted-foreground/30"
   320→          }`}
   321→        />
   322→      );
   323→    }
   324→    return stars;
   325→  };
   326→
   327→  /** Generate gradient fallback for cover image */
   328→  const getGradientStyle = (): React.CSSProperties => {
   329→    const colors = [
   330→      "linear-gradient(135deg, #0891b2 0%, #06b6d4 100%)",
   331→      "linear-gradient(135deg, #0d9488 0%, #14b8a6 100%)",
   332→      "linear-gradient(135deg, #059669 0%, #10b981 100%)",
   333→      "linear-gradient(135deg, #d97706 0%, #f59e0b 100%)",
   334→      "linear-gradient(135deg, #dc2626 0%, #ef4444 100%)",
   335→      "linear-gradient(135deg, #0891b2 0%, #0d9488 100%)",
   336→    ];
   337→    const index = course ? course.title.charCodeAt(0) % colors.length : 0;
   338→    return { background: colors[index] };
   339→  };
   340→
   341→  /** Format date */
   342→  const formatDate = (dateStr: string) => {
   343→    const d = new Date(dateStr);
   344→    return d.toLocaleDateString("en-US", {
   345→      year: "numeric",
   346→      month: "short",
   347→      day: "numeric",
   348→    });
   349→  };
   350→
   351→  // ─── Loading Skeleton ─────────────────────────────────
   352→  if (loading) {
   353→    return (
   354→      <div className="mx-auto max-w-5xl px-4 py-6 sm:px-6 lg:px-8">
   355→        {/* Hero skeleton (full-width banner) */}
   356→        <Skeleton className="w-full aspect-[21/9] sm:aspect-[3/1] rounded-2xl mb-6" />
   357→
   358→        {/* Description skeleton */}
   359→        <div className="flex items-start gap-3 mb-4">
   360→          <Skeleton className="h-5 w-16 rounded-full" />
   361→          <Skeleton className="h-4 w-2/3" />
   362→        </div>
   363→
   364→        {/* Action bar skeleton */}
   365→        <div className="flex items-center gap-3 mb-8">
   366→          <Skeleton className="h-10 w-10 rounded-lg" />
   367→          <Skeleton className="h-10 w-10 rounded-lg" />
   368→          <Skeleton className="ml-auto h-12 w-44 rounded-lg" />
   369→        </div>
   370→
   371→        {/* Curriculum skeleton */}
   372→        <Skeleton className="h-6 w-32 mb-4" />
   373→        <div className="space-y-3">
   374→          {[1, 2, 3].map((i) => (
   375→            <Skeleton key={i} className="h-16 w-full rounded-lg" />
   376→          ))}
   377→        </div>
   378→      </div>
   379→    );
   380→  }
   381→
   382→  // ─── Error State ───────────────────────────────────────
   383→  if (error || !course) {
   384→    return (
   385→      <div className="mx-auto max-w-5xl px-4 py-16 sm:px-6 lg:px-8 text-center">
   386→        <p className="text-muted-foreground text-lg mb-4">{error || "Course not found"}</p>
   387→        <Button variant="outline" onClick={goBack}>
   388→          <ChevronLeft className="h-4 w-4" />
   389→          Go Back
   390→        </Button>
   391→      </div>
   392→    );
   393→  }
   394→
   395→  const totalLessons = course.sections.reduce((sum, s) => sum + s.totalPages, 0);
   396→  const estimatedMinutes = Math.max(5, Math.round(totalLessons * 1.5));
   397→
   398→  /** Get progress info for a section */
   399→  const getSectionProgress = (sectionId: string) => {
   400→    return sectionProgress.find((p) => p.sectionId === sectionId);
   401→  };
   402→
   403→  /** Get section status badge */
   404→  const getSectionStatus = (section: SectionItem) => {
   405→    if (!course.isEnrolled) return null;
   406→    const prog = getSectionProgress(section.id);
   407→    if (!prog) return "not-started";
   408→    if (prog.completed) return "completed";
   409→    return "in-progress";
   410→  };
   411→
   412→  return (
   413→    <main className="mx-auto max-w-5xl px-4 py-6 sm:px-6 lg:px-8">
   414→      {/* ─── Course Hero (Full-Width Image + Overlay) ── */}
   415→      <div className="relative w-full aspect-[21/9] sm:aspect-[3/1] rounded-2xl overflow-hidden mb-6">
   416→        {/* Cover Image / Gradient Fallback */}
   417→        <div className="absolute inset-0">
   418→          {!imgError && course.coverImage ? (
   419→            <img
   420→              src={course.coverImage}
   421→              alt={course.title}
   422→              className="h-full w-full object-cover"
   423→              onError={() => setImgError(true)}
   424→            />
   425→          ) : (
   426→            <div
   427→              className="h-full w-full"
   428→              style={getGradientStyle()}
   429→            />
   430→          )}
   431→        </div>
   432→
   433→        {/* Gradient Overlay (bottom → transparent) */}
   434→        <div className="absolute inset-0 bg-gradient-to-t from-black/70 via-black/30 to-transparent" />
   435→
   436→        {/* Content overlayed on image */}
   437→        <div className="relative h-full flex flex-col justify-end p-5 sm:p-8">
   438→          {/* Breadcrumb overlayed on image */}
   439→          <Breadcrumb className="mb-3 [&_ol]:flex-nowrap">
   440→            <BreadcrumbList className="text-white/80">
   441→              <BreadcrumbItem>
   442→                <BreadcrumbLink
   443→                  className="cursor-pointer hover:text-white transition-colors text-white/80 hover:text-white"
   444→                  onClick={() => navigateTo("home")}
   445→                >
   446→                  Home
   447→                </BreadcrumbLink>
   448→              </BreadcrumbItem>
   449→              <BreadcrumbSeparator className="text-white/50" />
   450→              <BreadcrumbItem>
   451→                <BreadcrumbLink
   452→                  className="cursor-pointer hover:text-white transition-colors text-white/80 hover:text-white"
   453→                  onClick={() => navigateTo("courses")}
   454→                >
   455→                  All Courses
   456→                </BreadcrumbLink>
   457→              </BreadcrumbItem>
   458→              <BreadcrumbSeparator className="text-white/50" />
   459→              <BreadcrumbItem>
   460→                <BreadcrumbPage className="font-medium truncate max-w-[180px] sm:max-w-[280px] text-white">
   461→                  {course.title}
   462→                </BreadcrumbPage>
   463→              </BreadcrumbItem>
   464→            </BreadcrumbList>
   465→          </Breadcrumb>
   466→
   467→          {/* Title on image */}
   468→          <h1 className="text-2xl sm:text-3xl lg:text-4xl font-bold text-white leading-tight mb-3 [text-shadow:0_2px_8px_rgba(0,0,0,0.3)]">
   469→            {course.title}
   470→          </h1>
   471→
   472→          {/* Rating + Stats on image */}
   473→          <div className="flex flex-wrap items-center gap-x-4 gap-y-2 text-sm text-white/90">
   474→            {/* Rating with prominent number */}
   475→            <div className="flex items-center gap-1.5">
   476→              <span className="text-xl sm:text-2xl font-bold text-white [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   477→                {course.rating.toFixed(1)}
   478→              </span>
   479→              <div className="flex items-center gap-0.5">
   480→                {renderStars(course.rating)}
   481→              </div>
   482→              {ratingCount > 0 && (
   483→                <span className="text-xs text-white/60 ml-0.5">
   484→                  ({ratingCount})
   485→                </span>
   486→              )}
   487→            </div>
   488→
   489→            <span className="w-px h-4 bg-white/30" />
   490→
   491→            {/* Students count */}
   492→            <span className="inline-flex items-center gap-1.5 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   493→              <Users className="h-4 w-4" />
   494→              <span className="font-medium">{course.studentCount.toLocaleString()}</span>
   495→              <span className="text-white/70">students</span>
   496→            </span>
   497→
   498→            <span className="w-px h-4 bg-white/30" />
   499→
   500→            {/* Lesson count */}
   501→            <span className="inline-flex items-center gap-1.5 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   502→              <BookOpen className="h-4 w-4" />
   503→              <span className="font-medium">{totalLessons}</span>
   504→              <span className="text-white/70">lessons</span>
   505→            </span>
   506→
   507→            <span className="w-px h-4 bg-white/30" />
   508→
   509→            {/* Estimated duration */}
   510→            <span className="inline-flex items-center gap-1.5 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   511→              <Clock className="h-4 w-4" />
   512→              <span className="font-medium">~{estimatedMinutes} min</span>
   513→              <span className="text-white/70">estimated</span>
   514→            </span>
   515→
   516→            {/* Updated date */}
   517→            <span className="inline-flex items-center gap-1.5 text-white/70 [text-shadow:0_1px_4px_rgba(0,0,0,0.3)]">
   518→              <Calendar className="h-4 w-4" />
   519→              {formatDate(course.updatedAt)}
   520→            </span>
   521→          </div>
   522→        </div>
   523→      </div>
   524→
   525→      {/* ─── Description + Language ─────────────────── */}
   526→      <div className="flex items-start gap-3 mb-2">
   527→        {course.language && (
   528→          <Badge variant="secondary" className="shrink-0 mt-0.5">
   529→            {course.language}
   530→          </Badge>
   531→        )}
   532→        {course.description && (
   533→          <p className="text-sm text-muted-foreground leading-relaxed">
   534→            {course.description}
   535→          </p>
   536→        )}
   537→      </div>
   538→
   539→      {/* ─── Interactive Star Rating ─────────────── */}
   540→      {userId && (
   541→        <div className="mb-4">
   542→          <StarRating
   543→            courseId={course.id}
   544→            userId={userId}
   545→            currentRating={userRating}
   546→            averageRating={course.rating}
   547→            ratingCount={ratingCount}
   548→            onRate={(data) => {
   549→              setUserRating(data.score);
   550→              setRatingCount(data.count);
   551→              setCourse((prev) =>
   552→                prev ? { ...prev, rating: data.average } : prev
   553→              );
   554→            }}
   555→          />
   556→        </div>
   557→      )}
   558→
   559→      {/* ─── Action Bar ──────────────────────────── */}
   560→      <Separator className="my-4" />
   561→      <div className="flex flex-wrap items-center gap-3 mb-8">
   562→        {/* Grouped: Favorite + Share */}
   563→        <div className="flex items-center gap-2">
   564→          <Button
   565→            variant="outline"
   566→            size="icon"
   567→            className="h-10 w-10"
   568→            onClick={handleToggleFavorite}
   569→            disabled={togglingFav}
   570→            aria-label={course.isFavorited ? "Remove from favorites" : "Add to favorites"}
   571→          >
   572→            <Heart

   578→            />
   579→          </Button>

   581→            variant="outline"
   582→            size="icon"
   583→            className="h-10 w-10"
   584→            onClick={handleShare}

   591→        {/* Download as PPT - Outline Button */}

   708→                            </span>
   709→                          )}
   710→                          {course.isEnrolled && status === "not-started" && (
   711→                            <span className="text-xs text-muted-foreground">
   712→                              0/{totalP} pages
   713→                            </span>
   714→                          )}
   715→                        </div>
   716→                        {/* Mini progress bar if enrolled */}
   717→                        {course.isEnrolled && (
   718→                          <Progress
   719→                            value={progressPct}
   720→                            className="mt-1.5 h-1"
   721→                          />
   722→                        )}
   723→                      </div>
   724→                    </div>
   725→                  </AccordionTrigger>
   726→                  <AccordionContent className="pb-4">
   727→                    <p className="text-sm text-muted-foreground pl-11">
   728→                      Click &quot;{section.title}&quot; to open the lesson viewer and start learning.
   729→                    </p>
   730→                  </AccordionContent>
   731→                </AccordionItem>
   732→              );
   733→            })}
   734→          </Accordion>
   735→        )}
   736→      </section>
   737→
   738→      {/* ─── Progress Timeline ──────────────────── */}
   739→      {course.isEnrolled && (
   740→        <section className="mt-8">
   741→          <ProgressTimeline
   742→            courseId={course.id}
   743→            courseTitle={course.title}
   744→            userId={userId}
   745→            isEnrolled={course.isEnrolled}
   746→            sections={course.sections}
   747→            openClassroom={openClassroom}
   748→            course={course}
   749→          />
   750→        </section>
   751→      )}
   752→
   753→      <section className="mt-8">
   754→        <DiscussionPanel
   755→          courseId={course.id}
   756→          userId={userId}
   757→        />
   758→      </section>
   759→      {/* ─── Browser Warning ─────────────────────── */}
   760→      <div className="mt-8 flex items-start gap-3 rounded-lg border border-amber-200 bg-amber-50 p-3 text-sm text-amber-800 dark:border-amber-800/60 dark:bg-amber-950/40 dark:text-amber-200">
   761→        <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400" />
   762→        <p>
   763→          For the best learning experience, we recommend using the{" "}
   764→          <span className="font-semibold">Google Chrome</span> browser.
