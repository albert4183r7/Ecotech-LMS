    71→    71→function CircularProgress({ percentage }: { percentage: number }) {
    72→    72→  const radius = 70;
    73→    73→  const stroke = 10;
    74→    74→  const normalizedRadius = radius - stroke / 2;
    75→    75→  const circumference = normalizedRadius * 2 * Math.PI;
    76→    76→  const strokeDashoffset =
    77→    77→    circumference - (percentage / 100) * circumference;
    78→    78→
    79→    79→  return (
    80→    80→    <div className="relative inline-flex items-center justify-center">
    81→    81→      <svg
    82→    82→        width={radius * 2}
    83→    83→        height={radius * 2}
    84→    84→        className="-rotate-90"
    85→    85→      >
    86→    86→        {/* Background track */}
    87→    87→        <circle
    88→    88→          cx={radius}
    89→    89→          cy={radius}
    90→    90→          r={normalizedRadius}
    91→    91→          fill="none"
    92→    92→          stroke="currentColor"
    93→    93→          strokeWidth={stroke}
    94→    94→          className="text-muted/30"
    95→    95→        />
    96→    96→        {/* Progress arc with gradient stroke */}
    97→    97→        <circle
    98→    98→          cx={radius}
    99→    99→          cy={radius}
   100→   100→          r={normalizedRadius}
   101→   101→          fill="none"
   102→   102→          stroke="url(#progress-gradient)"
   103→   103→          strokeWidth={stroke}
   104→   104→          strokeLinecap="round"
   105→   105→          strokeDasharray={circumference}
   106→   106→          strokeDashoffset={strokeDashoffset}
   107→   107→          className="transition-all duration-700 ease-out"
   108→   108→        />
   109→   109→        <defs>
   110→   110→          <linearGradient
   111→   111→            id="progress-gradient"
   112→   112→            x1="0%"
   113→   113→            y1="0%"
   114→   114→            x2="100%"
   115→   115→            y2="100%"
   116→   116→          >
   117→   117→            <stop offset="0%" stopColor="#06b6d4" />

   136→   136→/*  Animated Counter Hook                                              */
   137→   137→/* ------------------------------------------------------------------ */
   138→   138→
   139→   139→function useAnimatedCounter(target: number, duration = 800) {
   140→   140→  const [count, setCount] = useState(target);
   141→   141→  const frameRef = useRef<number | null>(null);
   142→   142→
   143→   143→  useEffect(() => {
   144→   144→    if (target === 0) return;
   145→   145→    let start = 0;
   146→   146→    const startTime = performance.now();
   147→   147→    const step = (now: number) => {
   148→   148→      const elapsed = now - startTime;
   149→   149→      const progress = Math.min(elapsed / duration, 1);
   150→   150→      const eased = 1 - Math.pow(1 - progress, 3);
   151→   151→      start = Math.round(eased * target);
   152→   152→      setCount(start);
   153→   153→      if (progress < 1) {
   154→   154→        frameRef.current = requestAnimationFrame(step);
   155→   155→      }
   156→   156→    };
   157→   157→    frameRef.current = requestAnimationFrame(step);
   158→   158→    return () => {
   159→   159→      if (frameRef.current) cancelAnimationFrame(frameRef.current);
   160→   160→    };
   161→   161→  }, [target, duration]);
   162→   162→
   163→   163→  return count;
   164→   164→}

   262→   262→
   263→   263→function StreakCalendar({ streak, bestStreak, userId }: { streak: number; bestStreak: number; userId: string }) {
   264→   264→  const [activityMap, setActivityMap] = useState<Map<string, number>>(new Map());
   265→   265→  const [days, setDays] = useState<Array<{ date: string; day: string; isToday: boolean }>>([]);
   266→   266→  const [maxMinutes, setMaxMinutes] = useState(1);
   267→   267→
   268→   268→  useEffect(() => {
   269→   269→    async function fetchActivity() {
   270→   270→      try {
   271→   271→        const res = await fetch(`/api/activity?userId=${userId}&weeks=5`);
   272→   272→        const json = await res.json();
   273→   273→        if (json.success) {
   274→   274→          // Build 30-day grid from the response
   275→   275→          const allDays: ActivityDayEntry[] = [];
   276→   276→          const today = new Date();
   277→   277→          today.setHours(0, 0, 0, 0);
   278→   278→          const start = new Date(today);
   279→   279→          start.setDate(start.getDate() - 29);
   280→   280→
   281→   281→          const resMap = new Map<string, number>();
   282→   282→          for (const d of (json.data as Activity30Data).dailyData) {
   283→   283→            resMap.set(d.date, d.minutes);
   284→   284→          }
   285→   285→
   286→   286→          for (let i = 0; i < 30; i++) {
   287→   287→            const d = new Date(start);
   288→   288→            d.setDate(d.getDate() + i);
   289→   289→            const dateStr = d.toISOString().split('T')[0];
   290→   290→            allDays.push({
   291→   291→              date: dateStr,
   292→   292→              day: d.toLocaleDateString('en-US', { weekday: 'narrow' }),
   293→   293→              isToday: dateStr === today.toISOString().split('T')[0],
   294→   294→            });
   295→   295→          }
   296→   296→
   297→   297→          setDays(allDays);
   298→   298→          setActivityMap(resMap);
   299→   299→          const max = Math.max(...Array.from(resMap.values()), 1);
   300→   300→          setMaxMinutes(max);
   301→   301→        }
   302→   302→      } catch { /* silent */ }
   303→   303→    }
   304→   304→    fetchActivity();
   305→   305→  }, [userId]);
   306→   306→
   307→   307→  function getIntensity(date: string): number {
   308→   308→    const mins = activityMap.get(date) || 0;
   309→   309→    if (mins === 0) return 0;
   310→   310→    return Math.min(1, mins / maxMinutes);
   311→   311→  }
   312→   312→
   313→   313→  function getCellColor(intensity: number): string {
   314→   314→    if (intensity === 0) return 'bg-muted/40';
   315→   315→    if (intensity <= 0.25) return 'bg-cyan-300/40 dark:bg-cyan-700/40';
   316→   316→    if (intensity <= 0.5) return 'bg-cyan-400/60 dark:bg-cyan-600/50';
   317→   317→    if (intensity <= 0.75) return 'bg-cyan-500/70 dark:bg-cyan-500/60';
   318→   318→    return 'bg-cyan-600 dark:bg-cyan-400';
   319→   319→  }
   320→   320→
   321→   321→  return (
   322→   322→    <Card className="border-border/50">
   323→   323→      <CardHeader className="pb-3">
   324→   324→        <div className="flex items-center justify-between">
   325→   325→          <CardTitle className="flex items-center gap-2 text-base font-semibold">
   326→   326→            <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-gradient-to-br from-orange-500 to-amber-500">
   327→   327→              <Flame className="h-4 w-4 text-white" />
   328→   328→            </div>
   329→   329→            Learning Streak
   330→   330→          </CardTitle>
   331→   331→          <div className="flex items-center gap-3">
   332→   332→            <div className="flex items-center gap-1.5 text-sm font-bold text-orange-600 dark:text-orange-400">
   333→   333→              <span className="text-base">🔥</span>
   334→   334→              <span className="count-up">{streak}</span>
   335→   335→              <span className="text-xs font-medium text-muted-foreground">days</span>
   336→   336→            </div>
   337→   337→          </div>
   338→   338→        </div>
   339→   339→      </CardHeader>
   340→   340→      <CardContent className="space-y-4">
   341→   341→        {/* Streak stats row */}
   342→   342→        <div className="flex items-center justify-between">
   343→   343→          <div className="flex items-center gap-4 text-xs text-muted-foreground">
   344→   344→            <span className="flex items-center gap-1.5">
   345→   345→              <Flame className="h-3.5 w-3.5" />
   346→   346→              Current: <span className="font-semibold text-foreground">{streak} days</span>
   347→   347→            </span>
   348→   348→            <span className="flex items-center gap-1.5">
   349→   349→              <Trophy className="h-3.5 w-3.5" />
   350→   350→              Best: <span className="font-semibold text-foreground">{bestStreak} days</span>
   351→   351→            </span>
   352→   352→          </div>
   353→   353→        </div>
   354→   354→
   355→   355→        {/* 30-day grid - 6 columns x 5 rows */}
   356→   356→        <div>
   357→   357→          <p className="mb-2 text-[11px] font-medium text-muted-foreground">Last 30 days</p>
   358→   358→          <div className="grid grid-cols-6 gap-1.5 sm:grid-cols-10">
   359→   359→            {days.map((d) => {
   360→   360→              const intensity = getIntensity(d.date);
   361→   361→              return (
   362→   362→                <div
   363→   363→                  key={d.date}
   364→   364→                  title={`${d.date}: ${activityMap.get(d.date) || 0} min`}
   365→   365→                  className={`relative aspect-square rounded-sm transition-all duration-200 hover:scale-125 hover:ring-1 hover:ring-primary/30 ${getCellColor(intensity)} ${d.isToday ? 'ring-1 ring-primary' : ''}`}
   366→   366→                />
   367→   367→              );
   368→   368→            })}
   369→   369→          </div>
   370→   370→          {/* Intensity legend */}
   371→   371→          <div className="mt-2 flex items-center justify-end gap-1 text-[10px] text-muted-foreground">
   372→   372→            <span>Less</span>
   373→   373→            <div className="h-2.5 w-2.5 rounded-sm bg-muted/40" />
   374→   374→            <div className="h-2.5 w-2.5 rounded-sm bg-cyan-300/40 dark:bg-cyan-700/40" />
   375→   375→            <div className="h-2.5 w-2.5 rounded-sm bg-cyan-400/60 dark:bg-cyan-600/50" />
   376→   376→            <div className="h-2.5 w-2.5 rounded-sm bg-cyan-500/70 dark:bg-cyan-500/60" />
   377→   377→            <div className="h-2.5 w-2.5 rounded-sm bg-cyan-600 dark:bg-cyan-400" />
   378→   378→            <span>More</span>
   379→   379→          </div>
   380→   380→        </div>
   381→   381→      </CardContent>
   382→   382→    </Card>
