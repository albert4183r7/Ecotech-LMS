   124→/** Format date string to readable format */
   125→function formatDate(dateStr: string): string {
   126→  return new Date(dateStr).toLocaleDateString("en-US", {
   127→    month: "short",
   128→    day: "numeric",
   129→    year: "numeric",
   130→  });
   131→}
   132→
   133→/** Estimate remaining time based on sections left */
   134→function estimateRemainingTime(progress: number, totalSections: number): string {
   135→  const remaining = Math.ceil(totalSections * (1 - progress / 100));
   136→  if (remaining <= 0) return "Almost done!";
   137→  const hours = remaining * 0.5; // ~30 min per section
   138→  if (hours < 1) return `~${Math.ceil(hours * 60)} min left`;
   139→  if (hours < 4) return `~${Math.ceil(hours * 10) / 10} hrs left`;
   140→  return `~${Math.ceil(hours)} hrs left`;
   141→}
   142→
   143→/** Format relative time for "last accessed" */
   144→function formatLastAccessed(dateStr: string): string {
   145→  const now = Date.now();
   146→  const diff = now - new Date(dateStr).getTime();
   147→  const mins = Math.floor(diff / 60000);
   148→  if (mins < 1) return "Just now";
   149→  if (mins < 60) return `${mins}m ago`;
   150→  const hrs = Math.floor(mins / 60);
   151→  if (hrs < 24) return `${hrs}h ago`;
   152→  const days = Math.floor(hrs / 24);
   153→  if (days < 7) return `${days}d ago`;
   154→  return formatDate(dateStr);
   155→}

   207→
   208→function CircularMiniProgress({ percentage }: { percentage: number }) {
   209→  const size = 40;
   210→  const stroke = 3.5;
   211→  const r = (size - stroke) / 2;
   212→  const circ = r * 2 * Math.PI;
   213→  const offset = circ - (percentage / 100) * circ;
   214→
   215→  return (
   216→    <div className="relative inline-flex shrink-0 items-center justify-center">
   217→      <svg width={size} height={size} className="-rotate-90">
   218→        <circle
   219→          cx={size / 2}
   220→          cy={size / 2}
   221→          r={r}
   222→          fill="none"
   223→          stroke="currentColor"
   224→          strokeWidth={stroke}
   225→          className="text-muted/30"
   226→        />
   227→        <circle
   228→          cx={size / 2}
   229→          cy={size / 2}
   230→          r={r}
   231→          fill="none"
   232→          stroke="url(#mini-grad)"
   233→          strokeWidth={stroke}
   234→          strokeLinecap="round"
   235→          strokeDasharray={circ}
   236→          strokeDashoffset={offset}
   237→          className="transition-all duration-500 ease-out"
   238→        />
   239→        <defs>
   240→          <linearGradient id="mini-grad" x1="0%" y1="0%" x2="100%" y2="0%">
   241→            <stop offset="0%" stopColor="#0891b2" />
   242→            <stop offset="100%" stopColor="#14b8a6" />
   243→          </linearGradient>
   244→        </defs>
   245→      </svg>
   246→      <span className="absolute text-[9px] font-bold text-foreground">
   247→        {percentage}%
   248→      </span>
   249→    </div>
   250→  );
   251→}
