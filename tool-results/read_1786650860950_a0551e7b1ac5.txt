    67→function CircularProgress({ percentage }: { percentage: number }) {
    68→  const radius = 70;
    69→  const stroke = 10;
    70→  const normalizedRadius = radius - stroke / 2;
    71→  const circumference = normalizedRadius * 2 * Math.PI;
    72→  const strokeDashoffset =
    73→    circumference - (percentage / 100) * circumference;
    74→
    75→  return (
    76→    <div className="relative inline-flex items-center justify-center">
    77→      <svg
    78→        width={radius * 2}
    79→        height={radius * 2}
    80→        className="-rotate-90"
    81→      >
    82→        {/* Background track */}
    83→        <circle
    84→          cx={radius}
    85→          cy={radius}
    86→          r={normalizedRadius}
    87→          fill="none"
    88→          stroke="currentColor"
    89→          strokeWidth={stroke}
    90→          className="text-muted/30"
    91→        />
    92→        {/* Progress arc with gradient stroke */}
    93→        <circle
    94→          cx={radius}
    95→          cy={radius}
    96→          r={normalizedRadius}
    97→          fill="none"
    98→          stroke="url(#progress-gradient)"
    99→          strokeWidth={stroke}
   100→          strokeLinecap="round"
   101→          strokeDasharray={circumference}
   102→          strokeDashoffset={strokeDashoffset}
   103→          className="transition-all duration-700 ease-out"
   104→        />
   105→        <defs>
   106→          <linearGradient
   107→            id="progress-gradient"
   108→            x1="0%"
   109→            y1="0%"
   110→            x2="100%"
   111→            y2="100%"
   112→          >
   113→            <stop offset="0%" stopColor="#06b6d4" />
   114→            <stop offset="50%" stopColor="#0891b2" />
   115→            <stop offset="100%" stopColor="#0d9488" />
   116→          </linearGradient>
   117→        </defs>
   118→      </svg>
   119→      <div className="absolute flex flex-col items-center justify-center">
   120→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   121→          {percentage}%
   122→        </span>
   123→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   124→          completed
   125→        </span>
   126→      </div>
   127→    </div>
   128→  );
   129→}

   131→/* ------------------------------------------------------------------ */
   132→/*  Animated Counter Hook                                              */
   133→/* ------------------------------------------------------------------ */
   134→
   135→function useAnimatedCounter(target: number, duration = 800) {
   136→  const [count, setCount] = useState(target);
   137→
   138→  useEffect(() => {
   139→    if (target === 0) return;
   140→    let start = 0;
   141→    const startTime = performance.now();
   142→    const step = (now: number) => {
   143→      const elapsed = now - startTime;
   144→      const progress = Math.min(elapsed / duration, 1);
   145→      const eased = 1 - Math.pow(1 - progress, 3);
   146→      start = Math.round(eased * target);
   147→      setCount(start);
   148→      if (progress < 1) {
   149→        frameRef.current = requestAnimationFrame(step);
   150→      }
   151→    };
   152→    frameRef.current = requestAnimationFrame(step);
   153→    return () => cancelAnimationFrame(frameRef.current);
   154→  }, [target, duration]);
   155→
   156→  return count;
   157→}

   256→function StreakCalendar({ streak, bestStreak }: { streak: number; bestStreak: number }) {
   257→  const [activityMap, setActivityMap] = useState<Map<string, number>>(new Map());
   258→  const [days, setDays] = useState<Array<{ date: string; day: string; isToday: boolean }>>([]);
   259→  const [maxMinutes, setMaxMinutes] = useState(1);
   260→
   261→  useEffect(() => {
   262→    async function fetchActivity() {
   263→      try {
   264→        const res = await fetch('/api/activity?userId=user_demo_001&weeks=5');
   265→        const json = await res.json();
   266→        if (json.success) {
   267→          // Build 30-day grid from the response
   268→          const allDays: ActivityDayEntry[] = [];
   269→          const today = new Date();
   270→          today.setHours(0, 0, 0, 0);
   271→          const start = new Date(today);
   272→          start.setDate(start.getDate() - 29);
   273→
   274→          const resMap = new Map<string, number>();
   275→          for (const d of (json.data as Activity30Data).dailyData) {
   276→            resMap.set(d.date, d.minutes);
   277→          }
   278→
   279→          for (let i = 0; i < 30; i++) {
   280→            const d = new Date(start);
   281→            d.setDate(d.getDate() + i);
   282→            const dateStr = d.toISOString().split('T')[0];
   283→            allDays.push({
   284→              date: dateStr,
   285→              day: d.toLocaleDateString('en-US', { weekday: 'narrow' }),
   286→              isToday: dateStr === today.toISOString().split('T')[0],
   287→            });
   288→          }
   289→
   290→          setDays(allDays);
   291→          setActivityMap(resMap);
   292→          const max = Math.max(...Array.from(resMap.values()), 1);
   293→          setMaxMinutes(max);
   294→        }
   295→      } catch { /* silent */ }
   296→    }
   297→    fetchActivity();
   298→  }, []);
   299→
   300→  function getIntensity(date: string): number {
   301→    const mins = activityMap.get(date) || 0;
   302→    if (mins === 0) return 0;
   303→    return Math.min(1, mins / maxMinutes);
   304→  }
   305→
   306→  function getCellColor(intensity: number): string {
   307→    if (intensity === 0) return 'bg-muted/40';
   308→    if (intensity <= 0.25) return 'bg-cyan-300/40 dark:bg-cyan-700/40';
   309→    if (intensity <= 0.5) return 'bg-cyan-400/60 dark:bg-cyan-600/50';
   310→    if (intensity <= 0.75) return 'bg-cyan-500/70 dark:bg-cyan-500/60';
   311→    return 'bg-cyan-600 dark:bg-cyan-400';
   312→  }
   313→
   314→  return (
   315→    <Card className="border-border/50">
   316→      <CardHeader className="pb-3">
   317→        <div className="flex items-center justify-between">
   318→          <CardTitle className="flex items-center gap-2 text-base font-semibold">
   319→            <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-gradient-to-br from-orange-500 to-amber-500">
   320→              <Flame className="h-4 w-4 text-white" />
   321→            </div>
   322→            Learning Streak
   323→          </CardTitle>
   324→          <div className="flex items-center gap-3">
   325→            <div className="flex items-center gap-1.5 text-sm font-bold text-orange-600 dark:text-orange-400">
   326→              <span className="text-base">🔥</span>
   327→              <span className="count-up">{streak}</span>
   328→              <span className="text-xs font-medium text-muted-foreground">days</span>
   329→            </div>
   330→          </div>
   331→        </div>
   332→      </CardHeader>
   333→      <CardContent className="space-y-4">
   334→        {/* Streak stats row */}
   335→        <div className="flex items-center justify-between">
   336→          <div className="flex items-center gap-4 text-xs text-muted-foreground">

   342→              <Trophy className="h-3.5 w-3.5" />
