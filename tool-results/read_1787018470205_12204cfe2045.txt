    35→    35→
    36→    36→const MIN_ZOOM = 50;
    37→    37→const MAX_ZOOM = 200;
    38→    38→const ZOOM_STEP = 25;
    39→    39→
    40→    40→export function ClassroomPage() {
    41→    41→  const { classroomState, goBack } = useNavigationStore();
    42→    42→  const userId = useUserStore((s) => s.currentUserId);
    43→    43→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    44→    44→  const [zoom, setZoom] = useState(100);
    45→    45→  const [navigating, setNavigating] = useState(false);
    46→    46→  const confettiShownRef = useRef(false);
    47→    47→  const [showConfetti, setShowConfetti] = useState(false);
    48→    48→
    49→    49→  // ─── Notes Sidebar State ──────────────────────
    50→    50→  interface Note {
    51→    51→    id: string;
    52→    52→    content: string;
    53→    53→    slideNumber: number;
    54→    54→    bookmarked: boolean;
    55→    55→    createdAt?: string;
    56→    56→  }
    57→    57→  const [notes, setNotes] = useState<Note[]>([]);
    58→    58→  const [notesSidebarOpen, setNotesSidebarOpen] = useState(false);
    59→    59→  const [newNoteContent, setNewNoteContent] = useState("");
    60→    60→  const [savingNote, setSavingNote] = useState(false);
    61→    61→  const notesEndRef = useRef<HTMLDivElement>(null);
    62→    62→
    63→    63→  // ─── Keyboard Hint Fade ────────────────────────
    64→    64→  const [showKeyboardHint, setShowKeyboardHint] = useState(true);
    65→    65→  const [isMobile, setIsMobile] = useState(false);
    66→    66→  const [downloadingPptx, setDownloadingPptx] = useState(false);
    67→    67→
    68→    68→  useEffect(() => {
    69→    69→    const mq = window.matchMedia("(max-width: 1023px)");
    70→    70→    setIsMobile(mq.matches);
    71→    71→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
    72→    72→    mq.addEventListener("change", handler);
    73→    73→    return () => mq.removeEventListener("change", handler);
    74→    74→  }, []);
    75→    75→
    76→    76→  // Keep a working copy of classroomState
    77→    77→  useEffect(() => {
    78→    78→    if (classroomState) {
    79→    79→      setLocalState({ ...classroomState });
    80→    80→      setZoom(100);
    81→    81→      confettiShownRef.current = false;
    82→    82→      setShowConfetti(false);
    83→    83→    }
    84→    84→  }, [classroomState]);
    85→    85→
    86→    86→  // ─── Section Navigation ─────────────────────────
    87→    87→  const goToSection = useCallback(
    88→    88→    async (index: number) => {
    89→    89→      if (!localState || index < 0 || index >= localState.allSectionIds.length) return;
    90→    90→      const sectionId = localState.allSectionIds[index];
    91→    91→      try {
    92→    92→        setNavigating(true);
    93→    93→        const res = await fetch(`/api/sections/${sectionId}`);
    94→    94→        if (!res.ok) return;
    95→    95→        const json = await res.json();
    96→    96→        const htmlBody =
    97→    97→          json.success && json.data.htmlBody
    98→    98→            ? json.data.htmlBody
    99→    99→            : '<div class="flex items-center justify-center h-full"><p class="text-gray-500">No content available.</p></div>';
   100→   100→        const sectionTitle =
   101→   101→          json.success && json.data.title
   102→   102→            ? json.data.title
   103→   103→            : `Section ${index + 1}`;
   104→   104→        setLocalState((prev) =>
   105→   105→          prev
   106→   106→            ? {
   107→   107→                ...prev,
   108→   108→                sectionId,
   109→   109→                sectionTitle,
   110→   110→                htmlBody,
   111→   111→                currentSectionIndex: index,
   112→   112→              }
   113→   113→            : prev
   114→   114→        );
   115→   115→      } catch {
   116→   116→        /* silently fail */
   117→   117→      } finally {
   118→   118→        setNavigating(false);
   119→   119→      }
   120→   120→    },
   121→   121→    [localState]
   122→   122→  );
   123→   123→
   124→   124→  /** Mark section as completed and save progress */
   125→   125→  const markSectionCompleted = useCallback(
   126→   126→    async (sectionId: string) => {
   127→   127→      if (!userId || !localState) return;
   128→   128→      try {
   129→   129→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   130→   130→        const enrollJson = await enrollRes.json();
   131→   131→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   132→   132→          const enrollment = enrollJson.data.find(
   133→   133→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   134→   134→          );
   135→   135→          if (enrollment) {
   136→   136→            await fetch("/api/progress", {
   137→   137→              method: "POST",
   138→   138→              headers: { "Content-Type": "application/json" },
   139→   139→              body: JSON.stringify({
   140→   140→                enrollmentId: enrollment.id,
   141→   141→                sectionId,
   142→   142→                currentPage: 1,
   143→   143→                completed: true,
   144→   144→              }),
   145→   145→            });
   146→   146→          }
   147→   147→        }
   148→   148→      } catch {
   149→   149→        /* best-effort */
   150→   150→      }
   151→   151→    },
   152→   152→    [userId, localState]
   153→   153→  );
   154→   154→
   155→   155→  // Save progress when section changes
   156→   156→  useEffect(() => {
   157→   157→    if (!localState || !userId) return;
   158→   158→    const timer = setTimeout(() => {
   159→   159→      markSectionCompleted(localState.sectionId);
   160→   160→    }, 500);
   161→   161→    return () => clearTimeout(timer);
   162→   162→  }, [localState?.currentSectionIndex, userId, markSectionCompleted]);
   163→   163→
   164→   164→  const goPrev = () => {
   165→   165→    if (!localState) return;
   166→   166→    goToSection(localState.currentSectionIndex - 1);
   167→   167→  };
   168→   168→  const goNext = () => {
   169→   169→    if (!localState) return;
   170→   170→    goToSection(localState.currentSectionIndex + 1);
   171→   171→  };
   172→   172→
   173→   173→  // ─── Keyboard shortcuts ──────────────────────────────────────
   174→   174→  useEffect(() => {
   175→   175→    function handleKeyDown(e: KeyboardEvent) {
   176→   176→      const tag = (e.target as HTMLElement)?.tagName;
   177→   177→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   178→   178→
   179→   179→      switch (e.key) {
   180→   180→        case "Escape":
   181→   181→          e.preventDefault();
   182→   182→          goBack();
   183→   183→          break;
   184→   184→        case "ArrowLeft":
   185→   185→          e.preventDefault();
   186→   186→          goPrev();
   187→   187→          break;
   188→   188→        case "ArrowRight":
   189→   189→          e.preventDefault();
   190→   190→          goNext();
   191→   191→          break;
   192→   192→        case " ":
   193→   193→          e.preventDefault();
   194→   194→          goNext();
   195→   195→          break;
   196→   196→      }
   197→   197→    }
   198→   198→
   199→   199→    document.addEventListener("keydown", handleKeyDown);
   200→   200→
   201→   201→    // Listen for custom event from KeyboardShortcuts component
   202→   202→    function handleCustomNextSlide() {
   203→   203→      goNext();
   204→   204→    }
   205→   205→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   206→   206→
   207→   207→    return () => {
   208→   208→      document.removeEventListener("keydown", handleKeyDown);
   209→   209→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   210→   210→    };
   211→   211→  }, [goBack, goPrev, goNext]);
   212→   212→
   213→   213→  /** Zoom controls */
   214→   214→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   215→   215→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   216→   216→  const zoomFit = () => setZoom(100);
   217→   217→
   218→   218→  // ─── Confetti on last section ─────────────────
   219→   219→  useEffect(() => {
   220→   220→    if (!localState) return;
   221→   221→    const totalSections = localState.allSectionIds.length;
   222→   222→    const isOnLastSection =
   223→   223→      totalSections > 0 &&
   224→   224→      localState.currentSectionIndex === totalSections - 1;
   225→   225→    if (isOnLastSection && !confettiShownRef.current) {
   226→   226→      confettiShownRef.current = true;
   227→   227→      setShowConfetti(true);
   228→   228→      toast.success("🎉 You completed the course! Great job!");
   229→   229→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   230→   230→      return () => clearTimeout(timer);
   231→   231→    }
   232→   232→  }, [localState?.currentSectionIndex]);
   233→   233→
   234→   234→  // Reset confetti flag when classroom state changes (new course)
   235→   235→  useEffect(() => {
   236→   236→    confettiShownRef.current = false;
   237→   237→    setShowConfetti(false);
   238→   238→  }, [classroomState?.courseId]);
   239→   239→
   240→   240→  // ─── Keyboard Hint Auto-Fade ────────────────────
   241→   241→  useEffect(() => {
   242→   242→    setShowKeyboardHint(true);
   243→   243→    const timer = setTimeout(() => setShowKeyboardHint(false), 5000);
   244→   244→    return () => clearTimeout(timer);
   245→   245→  }, [localState?.sectionId]);
   246→   246→
   247→   247→  // ─── Notes CRUD ─────────────────────────────────
   248→   248→  const fetchNotes = useCallback(async () => {
   249→   249→    if (!localState || !userId) return;
   250→   250→    try {
   251→   251→      const res = await fetch(
   252→   252→        `/api/notes?userId=${userId}&courseId=${localState.courseId}&sectionId=${localState.sectionId}`
   253→   253→      );
   254→   254→      if (!res.ok) return;
   255→   255→      const json = await res.json();
   256→   256→      if (json.success && Array.isArray(json.data)) {
   257→   257→        setNotes(json.data);
   258→   258→      }
   259→   259→    } catch {
   260→   260→      // Silently fail
   261→   261→    }
   262→   262→  }, [localState, userId]);
   263→   263→
   264→   264→  // Fetch notes when section changes or sidebar opens
   265→   265→  useEffect(() => {
   266→   266→    if (notesSidebarOpen) fetchNotes();
   267→   267→  }, [notesSidebarOpen, fetchNotes, localState?.sectionId]);
   268→   268→
   269→   269→  const createNote = useCallback(async () => {
   270→   270→    if (!localState || !userId || !newNoteContent.trim()) return;
   271→   271→    setSavingNote(true);
   272→   272→    try {
   273→   273→      const res = await fetch("/api/notes", {
   274→   274→        method: "POST",
   275→   275→        headers: { "Content-Type": "application/json" },
   276→   276→        body: JSON.stringify({
   277→   277→          userId,
   278→   278→          courseId: localState.courseId,
   279→   279→          sectionId: localState.sectionId,
   280→   280→          content: newNoteContent.trim(),
   281→   281→          slideNumber: localState.currentSectionIndex + 1,
   282→   282→        }),
   283→   283→      });
   284→   284→      if (!res.ok) return;
   285→   285→      const json = await res.json();
   286→   286→      if (json.success && json.data) {
   287→   287→        setNotes((prev) => [...prev, json.data]);
   288→   288→        setNewNoteContent("");
   289→   289→        setTimeout(
   290→   290→          () => notesEndRef.current?.scrollIntoView({ behavior: "smooth" }),
   291→   291→          50
   292→   292→        );
   293→   293→      }
   294→   294→    } catch {
   295→   295→      toast.error("Failed to save note");
   296→   296→    } finally {
   297→   297→      setSavingNote(false);
   298→   298→    }
   299→   299→  }, [localState, userId, newNoteContent]);
   300→   300→
   301→   301→  const deleteNote = useCallback(async (noteId: string) => {
   302→   302→    try {
   303→   303→      const res = await fetch(`/api/notes?id=${noteId}`, {
   304→   304→        method: "DELETE",
   305→   305→      });
   306→   306→      if (!res.ok) return;
   307→   307→      const json = await res.json();
   308→   308→      if (json.success) {
   309→   309→        setNotes((prev) => prev.filter((n) => n.id !== noteId));
   310→   310→      }
   311→   311→    } catch {
   312→   312→      // Silently fail
   313→   313→    }
   314→   314→  }, []);
   315→   315→
   316→   316→  const toggleBookmark = useCallback((noteId: string) => {
   317→   317→    setNotes((prev) =>
   318→   318→      prev.map((n) =>
   319→   319→        n.id === noteId ? { ...n, bookmarked: !n.bookmarked } : n
   320→   320→      )
   321→   321→    );
   322→   322→  }, []);
   323→   323→
   324→   324→  // ─── Download as PPTX ─────────────────────────────
   325→   325→  const handleDownloadPptx = useCallback(async () => {
   326→   326→    if (!localState || downloadingPptx) return;
   327→   327→    setDownloadingPptx(true);
   328→   328→    try {
   329→   329→      const res = await fetch("/api/generate-pptx", {
   330→   330→        method: "POST",
   331→   331→        headers: { "Content-Type": "application/json" },
   332→   332→        body: JSON.stringify({
   333→   333→          courseId: localState.courseId,
   334→   334→          courseName: localState.courseTitle,
   335→   335→        }),
   336→   336→      });
   337→   337→      if (!res.ok) {
   338→   338→        toast.error("Failed to generate PPT");
   339→   339→        return;
   340→   340→      }
   341→   341→      const blob = await res.blob();
   342→   342→      const url = URL.createObjectURL(blob);
   343→   343→      const a = document.createElement("a");
   344→   344→      a.href = url;
   345→   345→      const safeName = localState.courseTitle
   346→   346→        .replace(/[^a-zA-Z0-9\s-]/g, "")
   347→   347→        .replace(/\s+/g, "-")
   348→   348→        .toLowerCase()
   349→   349→        .slice(0, 60);
   350→   350→      a.download = `${safeName}.pptx`;
   351→   351→      document.body.appendChild(a);
   352→   352→      a.click();
   353→   353→      a.remove();
   354→   354→      URL.revokeObjectURL(url);
   355→   355→      toast.success("PPT downloaded successfully!");
   356→   356→    } catch {
   357→   357→      toast.error("Failed to download PPT");
   358→   358→    } finally {
   359→   359→      setDownloadingPptx(false);
   360→   360→    }
   361→   361→  }, [localState, downloadingPptx]);
   362→   362→
   363→   363→  // ─── No state guard ─────────────────────────────
   364→   364→  if (!localState) {
   365→   365→    return (
   366→   366→      <div className="flex h-screen items-center justify-center bg-muted/30">
   367→   367→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   368→   368→      </div>
   369→   369→    );
   370→   370→  }
   371→   371→
   372→   372→  const totalSections = localState.allSectionIds.length;
   373→   373→  const currentIdx = localState.currentSectionIndex;
   374→   374→  const isFirst = currentIdx === 0;
   375→   375→  const isLast = totalSections > 0 && currentIdx >= totalSections - 1;
   376→   376→  const progressPercent =
   377→   377→    totalSections > 1
   378→   378→      ? Math.round(((currentIdx + 1) / totalSections) * 100)
   379→   379→      : 100;
   380→   380→
   381→   381→  return (
   382→   382→    <div className="flex h-screen flex-col bg-muted/30 relative">
   383→   383→      {/* ─── Mobile Notes Sheet ──────────────────── */}
   384→   384→      <Sheet
   385→   385→        open={notesSidebarOpen && isMobile}
   386→   386→        onOpenChange={(open) => {
   387→   387→          if (!open) setNotesSidebarOpen(false);
   388→   388→        }}
   389→   389→      >
   390→   390→        <SheetContent side="right" className="w-full sm:max-w-sm p-0">
   391→   391→          <SheetHeader className="px-4 pt-4 pb-0">
   392→   392→            <SheetTitle className="flex items-center gap-2 text-base">
   393→   393→              <StickyNote className="h-4 w-4" />
   394→   394→              Notes
   395→   395→            </SheetTitle>
   396→   396→            <SheetDescription>
   397→   397→              Notes for section {currentIdx + 1} of {totalSections}
   398→   398→            </SheetDescription>
   399→   399→          </SheetHeader>
   400→   400→          <NotesSidebarContent
   401→   401→            notes={notes}
   402→   402→            currentSectionIndex={currentIdx}
   403→   403→            newNoteContent={newNoteContent}
   404→   404→            savingNote={savingNote}
   405→   405→            onContentChange={setNewNoteContent}
   406→   406→            onCreate={createNote}
   407→   407→            onDelete={deleteNote}
   408→   408→            onToggleBookmark={toggleBookmark}
   409→   409→            endRef={notesEndRef}
   410→   410→          />
   411→   411→        </SheetContent>
   412→   412→      </Sheet>
   413→   413→
   414→   414→      {/* ─── Confetti Celebration Overlay ─────── */}
   415→   415→      {showConfetti && <ConfettiCelebration />}
   416→   416→
   417→   417→      {/* ─── Top Progress Bar ─────────────────────── */}
   418→   418→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   419→   419→        <div
   420→   420→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   421→   421→          style={{ width: `${progressPercent}%` }}
   422→   422→        />
   423→   423→      </div>
   424→   424→
   425→   425→      {/* ─── Top Bar ──────────────────────────────── */}
   426→   426→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   427→   427→        {/* Logo, Section Title, Course Title */}
   428→   428→        <div className="flex items-center gap-3 min-w-0">
   429→   429→          <div className="flex items-center gap-2 shrink-0">
   430→   430→            <img
   431→   431→              src="/ecotech-logo.png"
   432→   432→              alt="Ecotech"
   433→   433→              className="h-7 w-7 rounded-md object-contain"
   434→   434→            />
   435→   435→            <img
   436→   436→              src="/ecotech-name.png"
   437→   437→              alt="Ecotech"
   438→   438→              className="h-5 w-auto hidden sm:inline object-contain"
   439→   439→            />
   440→   440→          </div>
   441→   441→          <Separator orientation="vertical" className="h-5" />
   442→   442→          <div className="min-w-0">
   443→   443→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   444→   444→              {localState.sectionTitle}
   445→   445→            </h2>
   446→   446→            <p className="text-xs text-muted-foreground truncate leading-tight">
   447→   447→              {localState.courseTitle}
   448→   448→            </p>
   449→   449→          </div>
   450→   450→        </div>
   451→   451→
   452→   452→        {/* Section Counter */}
   453→   453→        <span className="shrink-0 inline-flex items-center gap-1.5 rounded-full bg-primary/10 px-3 py-1 text-sm text-primary tabular-nums">
   454→   454→          <span className="font-semibold">
   455→   455→            Section {currentIdx + 1}
   456→   456→          </span>
   457→   457→          <span className="text-primary/40 font-normal">of</span>
   458→   458→          <span className="font-semibold">{totalSections}</span>
   459→   459→        </span>
   460→   460→      </header>
   461→   461→
   462→   462→      {/* ─── Main Layout: Content + Desktop Notes Sidebar ── */}
   463→   463→      <div className="flex flex-1 overflow-hidden">
   464→   464→        {/* ─── Section Content Area ────────────────── */}
   465→   465→        <div className="flex-1 overflow-auto flex flex-col items-center py-8 px-4 sm:px-6">
   466→   466→          <div
   467→   467→            className="bg-card rounded-2xl shadow-lg border w-full max-w-3xl ring-1 ring-black/5 dark:ring-white/5 paper-texture overflow-hidden"
   468→   468→            style={{
   469→   469→              transform: `scale(${zoom / 100})`,
   470→   470→              transformOrigin: "top center",
   471→   471→            }}
   472→   472→          >
   473→   473→            {navigating ? (
   474→   474→              <div className="flex items-center justify-center py-32">
   475→   475→                <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   476→   476→              </div>
   477→   477→            ) : (
   478→   478→              <iframe
   479→   479→                srcDoc={localState.htmlBody || ""}
   480→   480→                sandbox="allow-same-origin"
   481→   481→                className="w-full rounded-lg border-0"
   482→   482→                style={{ aspectRatio: "16/9" }}
   483→   483→                title={`${localState.sectionTitle || "Slide"} content`}
   484→   484→              />
   485→   485→            )}
