   130→/** Format relative time for notifications */
   131→function formatRelativeTime(dateStr: string): string {
   132→  const now = new Date();
   133→  const date = new Date(dateStr);
   134→  const diffMs = now.getTime() - date.getTime();
   135→  const diffMins = Math.floor(diffMs / 60000);
   136→  const diffHours = Math.floor(diffMs / 3600000);
   137→  const diffDays = Math.floor(diffMs / 86400000);
   138→
   139→  if (diffMins < 1) return "Just now";
   140→  if (diffMins < 60) return `${diffMins}m ago`;
   141→  if (diffHours < 24) return `${diffHours}h ago`;
   142→  if (diffDays < 7) return `${diffDays}d ago`;
   143→  return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   144→}

   146→/** Group notifications by date category: Today, Yesterday, Earlier */
   147→function groupNotificationsByDate(items: NotificationItem[]) {
   148→  const now = new Date();
   149→  const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   150→  const yesterdayStart = todayStart - 86400000;
   151→
   152→  const today: NotificationItem[] = [];
   153→  const yesterday: NotificationItem[] = [];
   154→  const earlier: NotificationItem[] = [];
   155→
   156→  for (const item of items) {
   157→    const t = new Date(item.createdAt).getTime();
   158→    if (t >= todayStart) today.push(item);
   159→    else if (t >= yesterdayStart) yesterday.push(item);
   160→    else earlier.push(item);
   161→  }
   162→
   163→  return { today, yesterday, earlier };
   164→}

   182→      </p>
   183→      {items.map((notif) => (
   184→        <button
   185→          key={notif.id}
   186→          type="button"
   187→          onClick={() => onNotificationClick(notif)}
   188→          className={cn(
   189→            "group flex w-full items-start gap-3 rounded-lg px-3 py-3 text-left transition-colors notif-appear",
   190→            notif.read
   191→              ? "hover:bg-muted/50"
   192→              : "bg-primary/5 hover:bg-primary/8"
   193→          )}

   198→              "absolute left-1 top-3 bottom-3 w-0.5 rounded-full",
   199→              NotificationBorderAccent({ type: notif.type })

   201→          )}

   203→            "mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-lg transition-colors relative",
   204→            NotificationIconBg({ type: notif.type, read: notif.read })
   205→          )}>
   206→            <NotificationIcon type={notif.type} />
   207→          </div>
   208→          <div className="min-w-0 flex-1">
   209→            <div className="flex items-start justify-between gap-2">
   210→              <p className={cn(
   211→                "text-sm leading-snug",
   212→                notif.read
   213→                  ? "text-muted-foreground font-medium"
   214→                  : "text-foreground font-semibold"
   215→              )}>
   216→                {notif.title}
   217→              </p>
   218→              {!notif.read && (
   219→                <div className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary" />

   225→            <p className="mt-1 text-[10px] text-muted-foreground/60">
   226→              {formatRelativeTime(notif.createdAt)}
   227→            </p>
   228→          </div>
   229→        </button>
   230→      ))}
   231→    </div>
   232→  );
   233→}

   253→/** Parse link string into view + id for navigation */
   254→function parseNotificationLink(link: string | null): { view: ViewName; courseId?: string } | null {
   255→  if (!link) return null;
   256→  if (link === "profile") return { view: "profile" };
   257→  if (link.startsWith("course-detail:")) {
   258→    const courseId = link.replace("course-detail:", "");
   259→    return { view: "course-detail", courseId };
   260→  }
   261→  return null;
   262→}
