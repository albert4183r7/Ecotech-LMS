   184→   184→/* ------------------------------------------------------------------ */
   185→   185→
   186→   186→function CircularMiniProgress({ percentage }: { percentage: number }) {
   187→   187→  const size = 40;
   188→   188→  const stroke = 3.5;
   189→   189→  const r = (size - stroke) / 2;
   190→   190→  const circ = r * 2 * Math.PI;
   191→   191→  const offset = circ - (percentage / 100) * circ;
   192→   192→
   193→   193→  return (
   194→   194→    <div className="relative inline-flex shrink-0 items-center justify-center">
   195→   195→      <svg width={size} height={size} className="-rotate-90">
   196→   196→        <circle
   197→   197→          cx={size / 2}
   198→   198→          cy={size / 2}
   199→   199→          r={r}
   200→   200→          fill="none"
   201→   201→          stroke="currentColor"
   202→   202→          strokeWidth={stroke}
   203→   203→          className="text-muted/30"
   204→   204→        />
   205→   205→        <circle
   206→   206→          cx={size / 2}
   207→   207→          cy={size / 2}
   208→   208→          r={r}
   209→   209→          fill="none"
   210→   210→          stroke="url(#mini-grad)"
   211→   211→          strokeWidth={stroke}
   212→   212→          strokeLinecap="round"
   213→   213→          strokeDasharray={circ}
   214→   214→          strokeDashoffset={offset}
   215→   215→          className="transition-all duration-500 ease-out"
   216→   216→        />
   217→   217→        <defs>
   218→   218→          <linearGradient id="mini-grad" x1="0%" y1="0%" x2="100%" y2="0%">
   219→   219→            <stop offset="0%" stopColor="#0891b2" />
   220→   220→            <stop offset="100%" stopColor="#14b8a6" />
   221→   221→          </linearGradient>
   222→   222→        </defs>
   223→   223→      </svg>
   224→   224→      <span className="absolute text-[9px] font-bold text-foreground">
   225→   225→        {percentage}%
   226→   226→      </span>
   227→   227→    </div>
   228→   228→  );
   229→   229→}
