     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→import {
     5→  ZoomIn,
     6→  ZoomOut,
     7→  Maximize2,
     8→  RotateCcw,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  X,
    12→  GraduationCap,
    13→  Loader2,
    14→} from "lucide-react";
    15→import { toast } from "sonner";
    16→import { Button } from "@/components/ui/button";
    17→import { Separator } from "@/components/ui/separator";
    18→import { Tooltip, TooltipTrigger, TooltipContent } from "@/components/ui/tooltip";
    19→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    20→import type { SlideContent, ClassroomState } from "@/types/lms";
    21→
    22→const MIN_ZOOM = 50;
    23→const MAX_ZOOM = 200;
    24→const ZOOM_STEP = 25;
    25→
    26→export function ClassroomPage() {
    27→  const { classroomState, goBack } = useNavigationStore();
    28→  const userId = useUserStore((s) => s.currentUserId);
    29→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    30→  const [zoom, setZoom] = useState(100);
    31→  const [loadingSlides, setLoadingSlides] = useState(false);
    32→  const [slideDirection, setSlideDirection] = useState<"forward" | "back">("forward");
    33→  const [isAnimating, setIsAnimating] = useState(false);
    34→  const hasFetchedRef = useRef(false);
    35→  const confettiShownRef = useRef(false);
    36→  const [showConfetti, setShowConfetti] = useState(false);
    37→
    38→  // Keep a working copy of classroomState so we can mutate currentSlide
    39→  useEffect(() => {
    40→    if (classroomState) {
    41→      setLocalState({ ...classroomState });
    42→      setZoom(100);
    43→      hasFetchedRef.current = false;
    44→    }
    45→  }, [classroomState]);
    46→
    47→  // If slides were empty initially, try fetching them
    48→  useEffect(() => {
    49→    if (!localState || localState.slides.length > 0 || hasFetchedRef.current) return;
    50→    hasFetchedRef.current = true;
    51→    const fetchSlides = async () => {
    52→      setLoadingSlides(true);
    53→      try {
    54→        const res = await fetch(`/api/sections/${localState.sectionId}`);
    55→        if (!res.ok) return;
    56→        const json = await res.json();
    57→        if (json.success && json.data.content) {
    58→          const slides: SlideContent[] = Array.isArray(json.data.content)
    59→            ? json.data.content
    60→            : [];
    61→          setLocalState((prev) =>
    62→            prev
    63→              ? {
    64→                  ...prev,
    65→                  slides,
    66→                  totalPages: slides.length || prev.totalPages,
    67→                }
    68→              : prev
    69→          );
    70→        }
    71→      } catch {
    72→        // Silently fail
    73→      } finally {
    74→        setLoadingSlides(false);
    75→      }
    76→    };
    77→    fetchSlides();
    78→  }, [localState?.sectionId, localState?.slides.length]);
    79→
    80→  /** Navigate slides with transition animation */
    81→  const goToSlide = useCallback(
    82→    (index: number) => {
    83→      if (!localState || isAnimating) return;
    84→      const next = Math.max(0, Math.min(index, localState.totalPages - 1));
    85→      if (next === localState.currentSlide) return;
    86→      setSlideDirection(next > localState.currentSlide ? "forward" : "back");
    87→      setIsAnimating(true);
    88→      // Small delay for exit animation, then update slide
    89→      setTimeout(() => {
    90→        setLocalState((prev) =>
    91→          prev ? { ...prev, currentSlide: next } : prev
    92→        );
    93→        setIsAnimating(false);
    94→      }, 150);
    95→    },
    96→    [localState, isAnimating]
    97→  );
    98→
    99→  /** Persist progress to API when slide changes */
   100→  useEffect(() => {
   101→    if (!localState || !userId) return;
   102→    const saveProgress = async () => {
   103→      try {
   104→        // Get enrollment ID from the API
   105→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   106→        const enrollJson = await enrollRes.json();
   107→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   108→          const enrollment = enrollJson.data.find(
   109→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   110→          );
   111→          if (enrollment) {
   112→            const completed = localState.currentSlide >= localState.totalPages - 1;
   113→            await fetch("/api/progress", {
   114→              method: "POST",
   115→              headers: { "Content-Type": "application/json" },
   116→              body: JSON.stringify({
   117→                enrollmentId: enrollment.id,
   118→                sectionId: localState.sectionId,
   119→                currentPage: localState.currentSlide + 1,
   120→                completed,
   121→              }),
   122→            });
   123→          }
   124→        }
   125→      } catch {
   126→        // Progress saving is best-effort
   127→      }
   128→    };
   129→    // Debounce to avoid excessive API calls
   130→    const timer = setTimeout(saveProgress, 500);
   131→    return () => clearTimeout(timer);
   132→  }, [localState?.currentSlide]);
   133→
   134→  const goPrev = () => goToSlide((localState?.currentSlide ?? 0) - 1);
   135→  const goNext = () => goToSlide((localState?.currentSlide ?? 0) + 1);
   136→
   137→  // ─── Keyboard shortcuts ──────────────────────────────────────
   138→  useEffect(() => {
   139→    function handleKeyDown(e: KeyboardEvent) {
   140→      const tag = (e.target as HTMLElement)?.tagName;
   141→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   142→
   143→      switch (e.key) {
   144→        case "Escape":
   145→          e.preventDefault();
   146→          goBack();
   147→          break;
   148→        case "ArrowLeft":
   149→          e.preventDefault();
   150→          goPrev();
   151→          break;
   152→        case "ArrowRight":
   153→          e.preventDefault();
   154→          goNext();
   155→          break;
   156→        case " ":
   157→          e.preventDefault();
   158→          goNext();
   159→          break;
   160→      }
   161→    }
   162→
   163→    document.addEventListener("keydown", handleKeyDown);
   164→
   165→    // Listen for custom event from KeyboardShortcuts component
   166→    function handleCustomNextSlide() {
   167→      goNext();
   168→    }
   169→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   170→
   171→    return () => {
   172→      document.removeEventListener("keydown", handleKeyDown);
   173→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   174→    };
   175→  }, [goBack, goPrev, goNext]);
   176→
   177→  /** Zoom controls */
   178→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   179→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   180→  const zoomFit = () => setZoom(100);
   181→
   182→  // ─── Confetti on section completion ───────────
   183→  useEffect(() => {
   184→    if (!localState) return;
   185→    const isOnLastSlide = localState.currentSlide === localState.totalPages - 1;
   186→    if (isOnLastSlide && !confettiShownRef.current) {
   187→      confettiShownRef.current = true;
   188→      setShowConfetti(true);
   189→      toast.success("🎉 You completed the course! Great job!");
   190→      // Auto-cleanup after 3.5s (animation lasts ~3s + buffer)
   191→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   192→      return () => clearTimeout(timer);
   193→    }
   194→  }, [localState?.currentSlide]);
   195→
   196→  // Reset confetti flag when classroom state changes (new section)
   197→  useEffect(() => {
   198→    confettiShownRef.current = false;
   199→    setShowConfetti(false);
   200→  }, [classroomState?.sectionId]);
   201→
   202→  // ─── No state guard ─────────────────────────────
   203→  if (!localState) {
   204→    return (
   205→      <div className="flex h-screen items-center justify-center bg-muted/30">
   206→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   207→      </div>
   208→    );
   209→  }
   210→
   211→  const currentSlide =
   212→    localState.slides[localState.currentSlide] ?? null;
   213→  const isFirst = localState.currentSlide === 0;
   214→  const isLast = localState.currentSlide >= localState.totalPages - 1;
   215→  const progressPercent = localState.totalPages > 1
   216→    ? Math.round(((localState.currentSlide + 1) / localState.totalPages) * 100)
   217→    : 100;
   218→
   219→  return (
   220→    <div className="flex h-screen flex-col bg-muted/30 relative">
   221→      {/* ─── Confetti Celebration Overlay ─────── */}
   222→      {showConfetti && <ConfettiCelebration />}
   223→      {/* ─── Top Progress Bar ─────────────────────── */}
   224→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   225→        <div
   226→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   227→          style={{ width: `${progressPercent}%` }}
   228→        />
   229→      </div>
   230→
   231→      {/* ─── Top Bar ──────────────────────────────── */}
   232→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   233→        {/* Logo, Section Title, Course Title */}
   234→        <div className="flex items-center gap-3 min-w-0">
   235→          <div className="flex items-center gap-2 shrink-0">
   236→            <div className="flex h-7 w-7 items-center justify-center rounded-md bg-primary">
   237→              <GraduationCap className="h-4 w-4 text-primary-foreground" />
   238→            </div>
   239→            <span className="hidden sm:inline text-sm font-bold text-primary">
   240→              OpenClass
   241→            </span>
   242→          </div>
   243→          <Separator orientation="vertical" className="h-5" />
   244→          <div className="min-w-0">
   245→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   246→              {localState.sectionTitle}
   247→            </h2>
   248→            <p className="text-xs text-muted-foreground truncate leading-tight">
   249→              {localState.courseTitle}
   250→            </p>
   251→          </div>
   252→        </div>
   253→
   254→        {/* Slide Pagination - Pill Badge */}
   255→        <span className="shrink-0 inline-flex items-center gap-1 rounded-full bg-primary/10 px-3 py-1 text-sm font-semibold text-primary tabular-nums">
   256→          {localState.currentSlide + 1}
   257→          <span className="text-primary/40 font-normal">/</span>
   258→          {localState.totalPages}
   259→        </span>
   260→      </header>
   261→
   262→      {/* ─── Slide Content Area ────────────────────── */}
   263→      <div className="flex-1 overflow-auto flex items-start justify-center py-8 px-4 sm:px-6">
   264→        <div
   265→          className={`
   266→            bg-card rounded-2xl shadow-lg border w-full max-w-3xl
   267→            ring-1 ring-black/5
   268→            dark:ring-white/5
   269→            paper-texture
   270→            ${isAnimating
   271→              ? (slideDirection === "forward" ? "slide-exit" : "opacity-0 -translate-x-2 transition-all duration-150")
   272→              : "slide-enter"
   273→            }
   274→          `}
   275→          style={{
   276→            transform: `scale(${zoom / 100})`,
   277→            transformOrigin: "top center",
   278→          }}
   279→        >
   280→          {loadingSlides ? (
   281→            <div className="flex items-center justify-center py-32">
   282→              <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   283→            </div>
   284→          ) : currentSlide ? (
   285→            <div className="p-8 sm:p-12 max-w-2xl mx-auto">
   286→              <SlideRenderer slide={currentSlide} />
   287→            </div>
   288→          ) : (
   289→            <div className="flex items-center justify-center py-32 text-muted-foreground">
   290→              <p>No slide content available.</p>
   291→            </div>
   292→          )}
   293→        </div>
   294→      </div>
   295→
   296→      {/* ─── Bottom Controls (Frosted Glass) ───────── */}
   297→      <footer className="shrink-0 border-t frosted-glass">
   298→        <div className="mx-auto flex h-16 max-w-5xl items-center justify-between px-4 sm:px-6">
   299→          {/* Zoom Controls */}
   300→          <div className="flex items-center gap-1">
   301→            <Tooltip>
   302→              <TooltipTrigger asChild>
   303→                <Button
   304→                  variant="ghost"
   305→                  size="icon"
   306→                  className="h-8 w-8"
   307→                  onClick={zoomOut}
   308→                  disabled={zoom <= MIN_ZOOM}
   309→                >
   310→                  <ZoomOut className="h-4 w-4" />
   311→                </Button>
   312→              </TooltipTrigger>
   313→              <TooltipContent>Zoom Out</TooltipContent>
   314→            </Tooltip>
   315→
   316→            <span className="w-12 text-center text-xs font-medium text-muted-foreground tabular-nums">
   317→              {zoom}%
   318→            </span>
   319→
   320→            <Tooltip>
   321→              <TooltipTrigger asChild>
   322→                <Button
   323→                  variant="ghost"
   324→                  size="icon"
   325→                  className="h-8 w-8"
   326→                  onClick={zoomIn}
   327→                  disabled={zoom >= MAX_ZOOM}
   328→                >
   329→                  <ZoomIn className="h-4 w-4" />
   330→                </Button>
   331→              </TooltipTrigger>
   332→              <TooltipContent>Zoom In</TooltipContent>
   333→            </Tooltip>
   334→
   335→            <Tooltip>
   336→              <TooltipTrigger asChild>
   337→                <Button
   338→                  variant="ghost"
   339→                  size="icon"
   340→                  className="h-8 w-8"
   341→                  onClick={zoomFit}
   342→                >
   343→                  <Maximize2 className="h-4 w-4" />
   344→                </Button>
   345→              </TooltipTrigger>
   346→              <TooltipContent>Fit Width</TooltipContent>
   347→            </Tooltip>
   348→
   349→            <Tooltip>
   350→              <TooltipTrigger asChild>
   351→                <Button
   352→                  variant="ghost"
   353→                  size="icon"
   354→                  className="h-8 w-8"
   355→                  onClick={zoomFit}
   356→                >
   357→                  <RotateCcw className="h-4 w-4" />
   358→                </Button>
   359→              </TooltipTrigger>
   360→              <TooltipContent>Reset Zoom</TooltipContent>
   361→            </Tooltip>
   362→          </div>
   363→
   364→          {/* Navigation Buttons with labels on desktop */}
   365→          <div className="flex items-center gap-2">
   366→            <Button
   367→              variant="outline"
   368→              size="sm"
   369→              className={`gap-1.5 transition-opacity ${isFirst ? "opacity-40 cursor-not-allowed" : ""}`}
   370→              onClick={goPrev}
   371→              disabled={isFirst}
   372→            >
   373→              <ChevronLeft className="h-4 w-4" />
   374→              <span className="hidden sm:inline">Previous</span>
   375→            </Button>
   376→
   377→            <Button
   378→              variant="outline"
   379→              size="sm"
   380→              className={`gap-1.5 transition-opacity ${isLast ? "opacity-40 cursor-not-allowed" : ""}`}
   381→              onClick={goNext}
   382→              disabled={isLast}
   383→            >
   384→              <span className="hidden sm:inline">Next</span>
   385→              <ChevronRight className="h-4 w-4" />
   386→            </Button>
   387→          </div>
   388→
   389→          {/* Close + Keyboard Hint */}
   390→          <div className="flex items-center gap-3">
   391→            <span className="hidden lg:inline text-xs text-muted-foreground">
   392→              ← → navigate, ESC exit
   393→            </span>
   394→            <Button
   395→              variant="ghost"
   396→              size="sm"
   397→              className="gap-1.5 text-muted-foreground hover:text-foreground"
   398→              onClick={goBack}
   399→            >
   400→              <X className="h-4 w-4" />
   401→              <span className="hidden sm:inline">Close</span>
   402→            </Button>
   403→          </div>
   404→        </div>
   405→      </footer>
   406→    </div>
   407→  );
   408→}

   410→// ============================================
   411→// Confetti Celebration Component

   413→
   414→/** LMS color palette for confetti particles */
   415→const CONFETTI_COLORS = [
   416→  "#3882f6", // blue
   417→  "#14b8a6", // teal
   418→  "#22d3ee", // cyan
   419→  "#10b981", // emerald
   420→  "#f59e0b", // amber
   421→  "#6366f1", // indigo accent
   422→  "#8b5cf6", // violet accent
   423→  "#ec4899", // pink accent
   424→];
   425→
   426→type ParticleShape = "circle" | "rectangle" | "triangle";
   427→
   428→interface ConfettiParticle {
   429→  id: number;
   430→  left: number;
   431→  color: string;
   432→  shape: ParticleShape;
   433→  size: number;
   434→  fallDuration: string;
   435→  fallDelay: string;
   436→  drift: string;
   437→  spin: string;
   438→}
   439→
   440→function ConfettiCelebration() {
   441→  const particles = useMemo<ConfettiParticle[]>(() => {
   442→    const result: ConfettiParticle[] = [];
   443→    const shapes: ParticleShape[] = ["circle", "rectangle", "triangle"];
   444→    const count = 80;
   445→
   446→    for (let i = 0; i < count; i++) {
   447→      const shape = shapes[i % 3 === 0 ? 0 : i % 3 === 1 ? 1 : 2];
   448→      result.push({
   449→        id: i,
   450→        left: Math.random() * 100,
   451→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   452→        shape,
   453→        size: Math.floor(Math.random() * 8) + 6, // 6-14px
   454→        fallDuration: `${(Math.random() * 1.5 + 2).toFixed(2)}s`, // 2-3.5s
   455→        fallDelay: `${(Math.random() * 0.8).toFixed(2)}s`, // 0-0.8s
   456→        drift: `${(Math.random() * 120 - 60).toFixed(0)}px`, // -60 to +60px
   457→        spin: `${Math.floor(Math.random() * 720 + 360)}deg`, // 360-1080deg
   458→      });
   459→    }
   460→    return result;
   461→  }, []);
   462→
   463→  return (
   464→    <>
   465→      {/* Confetti particles layer */}
   466→      <div className="confetti-container" aria-hidden="true">
   467→        {particles.map((p) => {
   468→          if (p.shape === "triangle") {
   469→            return (
   470→              <div
   471→                key={p.id}
   472→                className="confetti-particle confetti-triangle"
   473→                style={{
   474→                  left: `${p.left}%`,
   475→                  "--confetti-color": p.color,
   476→                  "--tri-size": `${p.size}px`,
   477→                  "--fall-duration": p.fallDuration,
   478→                  "--fall-delay": p.fallDelay,
   479→                  "--drift": p.drift,
   480→                  "--spin": p.spin,
   481→                } as React.CSSProperties}
   482→              />
   483→            );
   484→          }
   485→          return (
   486→            <div
   487→              key={p.id}
   488→              className={`confetti-particle ${p.shape === "circle" ? "confetti-circle" : "confetti-rectangle"}`}
   489→              style={{
   490→                left: `${p.left}%`,
   491→                width: p.shape === "rectangle" ? `${p.size * 1.4}px` : `${p.size}px`,
   492→                height: `${p.size}px`,
   493→                backgroundColor: p.color,
   494→                "--fall-duration": p.fallDuration,
   495→                "--fall-delay": p.fallDelay,
   496→                "--drift": p.drift,
   497→                "--spin": p.spin,
   498→              } as React.CSSProperties}
   499→            />
   500→          );
   501→        })}
   502→      </div>
   503→
   504→      {/* Congratulations message */}
   505→      <div className="confetti-message" role="status" aria-label="Congratulations!">
   506→        <div className="confetti-message-text flex flex-col items-center gap-3">
   507→          <div className="flex h-16 w-16 items-center justify-center rounded-2xl bg-white/90 dark:bg-zinc-800/90 shadow-lg">
   508→            <span className="text-4xl">🎉</span>
   509→          </div>
   510→          <div className="rounded-2xl bg-white/90 dark:bg-zinc-800/90 px-8 py-5 shadow-xl text-center">
   511→            <h2 className="text-2xl sm:text-3xl font-bold text-foreground">
   512→              🎉 Congratulations!
   513→            </h2>
   514→            <p className="mt-2 text-sm text-muted-foreground">
   515→              You&apos;ve completed this section!
   516→            </p>
   517→          </div>
   518→        </div>
   519→      </div>
   520→    </>
   521→  );
   522→}

   524→// ============================================
   525→// Slide Content Renderer
   526→// ============================================
   527→
   528→function SlideRenderer({ slide }: { slide: SlideContent }) {
   529→  switch (slide.type) {
   530→    case "title":
   531→      return <TitleSlide slide={slide} />;
   532→    case "content":
   533→      return <ContentSlide slide={slide} />;
   534→    case "table":
   535→      return <TableSlide slide={slide} />;
   536→    case "list":
   537→      return <ListSlide slide={slide} />;
   538→    case "code":
   539→      return <CodeSlide slide={slide} />;
   540→    case "quiz":
   541→      return <QuizSlide slide={slide} />;
   542→    default:
   543→      return <FallbackSlide slide={slide} />;
   544→  }
   545→}
   546→
   547→// ─── Title Slide ──────────────────────────────────
   548→function TitleSlide({ slide }: { slide: SlideContent }) {
   549→  return (
   550→    <div className="flex min-h-[300px] flex-col items-center justify-center text-center">
   551→      <div className="mb-6 flex h-16 w-16 items-center justify-center rounded-2xl bg-gradient-to-br from-primary/10 to-accent/10">
   552→        <GraduationCap className="h-8 w-8 text-primary" />
   553→      </div>
   554→      <h1 className="text-3xl sm:text-4xl font-bold text-foreground mb-4 leading-tight">
   555→        {slide.title}
   556→      </h1>
   557→      {slide.subtitle && (
   558→        <p className="text-lg text-muted-foreground max-w-xl leading-relaxed">
   559→          {slide.subtitle}
   560→        </p>
   561→      )}
   562→    </div>
   563→  );
   564→}
   565→
   566→// ─── Content Slide ────────────────────────────────
   567→function ContentSlide({ slide }: { slide: SlideContent }) {
   568→  return (
   569→    <div>
   570→      <div className="mb-6 flex items-center gap-3">
   571→        <div className="h-1 w-8 rounded-full bg-gradient-to-r from-primary to-accent" />
   572→        <h2 className="text-2xl font-bold text-foreground">{slide.title}</h2>
   573→      </div>
   574→      {slide.items && slide.items.length > 0 && (
   575→        <div className="space-y-5">

   703→// ─── Quiz Slide (Interactive) ────────────────────
   704→function QuizSlide({ slide }: { slide: SlideContent }) {
   705→  const [selectedIdx, setSelectedIdx] = useState<number | null>(null);
   706→  const hasSubmitted = selectedIdx !== null;
   707→
   708→  // Determine correct answer: look for item with icon === "check" or first item
   709→  const correctIdx = slide.items?.findIndex((item) => item.icon === "check") ?? 0;
   710→  const isCorrect = selectedIdx === correctIdx;
   711→
   712→  return (
   713→    <div>
   714→      <div className="mb-6 flex items-center gap-3">
   715→        <div className="h-1 w-8 rounded-full bg-gradient-to-r from-primary to-accent" />
   716→        <h2 className="text-2xl font-bold text-foreground">{slide.title}</h2>
   717→      </div>
   718→      {slide.items && slide.items.length > 0 && (
   719→        <div className="space-y-3">
   720→          {slide.items.map((item, i) => {
   721→            const letter = String.fromCharCode(65 + i);
   722→            const isSelected = selectedIdx === i;
   723→            let borderColor = "border-border hover:border-primary/30 hover:bg-primary/5";
   724→            if (hasSubmitted) {
   725→              if (i === correctIdx) borderColor = "border-emerald-500 bg-emerald-500/10";
   726→              else if (isSelected && !isCorrect) borderColor = "border-red-400 bg-red-400/10";
   727→              else borderColor = "border-border opacity-60";
   728→            } else if (isSelected) {
   729→              borderColor = "border-primary bg-primary/10";
   730→            }
   731→
   732→            return (
   733→              <button
   734→                key={i}
   735→                type="button"
   736→                className={`w-full rounded-lg border p-4 text-left transition-all duration-200 ${borderColor}`}
   737→                onClick={() => {
   738→                  if (!hasSubmitted) setSelectedIdx(i);
   739→                }}
   740→                disabled={hasSubmitted}
   741→              >
   742→                <div className="flex items-start gap-3">
   743→                  <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold ${
   744→                    hasSubmitted && i === correctIdx
   745→                      ? "bg-emerald-500 text-white"
   746→                      : hasSubmitted && isSelected && !isCorrect
   747→                        ? "bg-red-400 text-white"
   748→                        : "bg-primary/10 text-primary"
   749→                  }`}>
   750→                    {hasSubmitted && i === correctIdx ? "✓" : hasSubmitted && isSelected && !isCorrect ? "✗" : letter}
   751→                  </span>
   752→                  <div className="flex-1">
   753→                    <p className="text-sm font-medium text-foreground">
   754→                      {item.heading || item.text}
   755→                    </p>
   756→                    {item.heading && item.text !== item.heading && (
   757→                      <p className="text-xs text-muted-foreground mt-1">{item.text}</p>
   758→                    )}
   759→                  </div>
   760→                </div>
   761→              </button>
   762→            );
   763→          })}
   764→
   765→          {/* Feedback after selection */}
   766→          {hasSubmitted && (
   767→            <div className={`mt-4 rounded-lg border p-4 ${
   768→              isCorrect
   769→                ? "border-emerald-200 bg-emerald-50 dark:border-emerald-800 dark:bg-emerald-950/50"
   770→                : "border-amber-200 bg-amber-50 dark:border-amber-800 dark:bg-amber-950/50"
   771→            }`}>
   772→              <p className={`text-sm font-semibold ${
   773→                isCorrect ? "text-emerald-700 dark:text-emerald-400" : "text-amber-700 dark:text-amber-400"
   774→              }`}>
   775→                {isCorrect ? "🎉 Correct! Well done!" : "❌ Not quite right."}
   776→              </p>
   777→              {!isCorrect && (
   778→                <p className="mt-1 text-xs text-muted-foreground">
   779→                  The correct answer is <strong>{String.fromCharCode(65 + correctIdx)}</strong>.
   780→                </p>
   781→              )}
   782→            </div>
   783→          )}
   784→        </div>
   785→      )}
   786→    </div>
   787→  );
   788→}
