     1→     1→"use client";
     2→     2→
     3→     3→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     4→     4→import {
     5→     5→  ZoomIn,
     6→     6→  ZoomOut,
     7→     7→  Maximize2,
     8→     8→  RotateCcw,
     9→     9→  ChevronLeft,
    10→    10→  ChevronRight,
    11→    11→  X,
    12→    12→  GraduationCap,
    13→    13→  Loader2,
    14→    14→} from "lucide-react";
    15→    15→import { toast } from "sonner";
    16→    16→import { Button } from "@/components/ui/button";
    17→    17→import { Separator } from "@/components/ui/separator";
    18→    18→import { Tooltip, TooltipTrigger, TooltipContent } from "@/components/ui/tooltip";
    19→    19→import { useNavigationStore, useUserStore } from "@/stores/lms-store";
    20→    20→import type { SlideContent, ClassroomState } from "@/types/lms";
    21→    21→
    22→    22→const MIN_ZOOM = 50;
    23→    23→const MAX_ZOOM = 200;
    24→    24→const ZOOM_STEP = 25;
    25→    25→
    26→    26→export function ClassroomPage() {
    27→    27→  const { classroomState, goBack } = useNavigationStore();
    28→    28→  const userId = useUserStore((s) => s.currentUserId);
    29→    29→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    30→    30→  const [zoom, setZoom] = useState(100);
    31→    31→  const [loadingSlides, setLoadingSlides] = useState(false);
    32→    32→  const [slideDirection, setSlideDirection] = useState<"forward" | "back">("forward");
    33→    33→  const [isAnimating, setIsAnimating] = useState(false);
    34→    34→  const hasFetchedRef = useRef(false);
    35→    35→  const confettiShownRef = useRef(false);
    36→    36→  const [showConfetti, setShowConfetti] = useState(false);
    37→    37→
    38→    38→  // Keep a working copy of classroomState so we can mutate currentSlide
    39→    39→  useEffect(() => {
    40→    40→    if (classroomState) {
    41→    41→      setLocalState({ ...classroomState });
    42→    42→      setZoom(100);
    43→    43→      hasFetchedRef.current = false;
    44→    44→    }
    45→    45→  }, [classroomState]);
    46→    46→
    47→    47→  // If slides were empty initially, try fetching them
    48→    48→  useEffect(() => {
    49→    49→    if (!localState || localState.slides.length > 0 || hasFetchedRef.current) return;
    50→    50→    hasFetchedRef.current = true;
    51→    51→    const fetchSlides = async () => {
    52→    52→      setLoadingSlides(true);
    53→    53→      try {
    54→    54→        const res = await fetch(`/api/sections/${localState.sectionId}`);
    55→    55→        if (!res.ok) return;
    56→    56→        const json = await res.json();
    57→    57→        if (json.success && json.data.content) {
    58→    58→          const slides: SlideContent[] = Array.isArray(json.data.content)
    59→    59→            ? json.data.content
    60→    60→            : [];
    61→    61→          setLocalState((prev) =>
    62→    62→            prev
    63→    63→              ? {
    64→    64→                  ...prev,
    65→    65→                  slides,
    66→    66→                  totalPages: slides.length || prev.totalPages,
    67→    67→                }
    68→    68→              : prev
    69→    69→          );
    70→    70→        }
    71→    71→      } catch {
    72→    72→        // Silently fail
    73→    73→      } finally {
    74→    74→        setLoadingSlides(false);
    75→    75→      }
    76→    76→    };
    77→    77→    fetchSlides();
    78→    78→  }, [localState?.sectionId, localState?.slides.length]);
    79→    79→
    80→    80→  /** Navigate slides with transition animation */
    81→    81→  const goToSlide = useCallback(
    82→    82→    (index: number) => {
    83→    83→      if (!localState || isAnimating) return;
    84→    84→      const next = Math.max(0, Math.min(index, localState.totalPages - 1));
    85→    85→      if (next === localState.currentSlide) return;
    86→    86→      setSlideDirection(next > localState.currentSlide ? "forward" : "back");
    87→    87→      setIsAnimating(true);
    88→    88→      // Small delay for exit animation, then update slide
    89→    89→      setTimeout(() => {
    90→    90→        setLocalState((prev) =>
    91→    91→          prev ? { ...prev, currentSlide: next } : prev
    92→    92→        );
    93→    93→        setIsAnimating(false);
    94→    94→      }, 150);
    95→    95→    },
    96→    96→    [localState, isAnimating]
    97→    97→  );
    98→    98→
    99→    99→  /** Persist progress to API when slide changes */
   100→   100→  useEffect(() => {
   101→   101→    if (!localState || !userId) return;
   102→   102→    const saveProgress = async () => {
   103→   103→      try {
   104→   104→        // Get enrollment ID from the API
   105→   105→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   106→   106→        const enrollJson = await enrollRes.json();
   107→   107→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   108→   108→          const enrollment = enrollJson.data.find(
   109→   109→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   110→   110→          );
   111→   111→          if (enrollment) {
   112→   112→            const completed = localState.currentSlide >= localState.totalPages - 1;
   113→   113→            await fetch("/api/progress", {
   114→   114→              method: "POST",
   115→   115→              headers: { "Content-Type": "application/json" },
   116→   116→              body: JSON.stringify({
   117→   117→                enrollmentId: enrollment.id,
   118→   118→                sectionId: localState.sectionId,
   119→   119→                currentPage: localState.currentSlide + 1,
   120→   120→                completed,
   121→   121→              }),
   122→   122→            });
   123→   123→          }
   124→   124→        }
   125→   125→      } catch {
   126→   126→        // Progress saving is best-effort
   127→   127→      }
   128→   128→    };
   129→   129→    // Debounce to avoid excessive API calls
   130→   130→    const timer = setTimeout(saveProgress, 500);
   131→   131→    return () => clearTimeout(timer);
   132→   132→  }, [localState?.currentSlide]);
   133→   133→
   134→   134→  const goPrev = () => goToSlide((localState?.currentSlide ?? 0) - 1);
   135→   135→  const goNext = () => goToSlide((localState?.currentSlide ?? 0) + 1);
   136→   136→
   137→   137→  // ─── Keyboard shortcuts ──────────────────────────────────────
   138→   138→  useEffect(() => {
   139→   139→    function handleKeyDown(e: KeyboardEvent) {
   140→   140→      const tag = (e.target as HTMLElement)?.tagName;
   141→   141→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   142→   142→
   143→   143→      switch (e.key) {
   144→   144→        case "Escape":
   145→   145→          e.preventDefault();
   146→   146→          goBack();
   147→   147→          break;
   148→   148→        case "ArrowLeft":
   149→   149→          e.preventDefault();
   150→   150→          goPrev();
   151→   151→          break;
   152→   152→        case "ArrowRight":
   153→   153→          e.preventDefault();
   154→   154→          goNext();
   155→   155→          break;
   156→   156→        case " ":
   157→   157→          e.preventDefault();
   158→   158→          goNext();
   159→   159→          break;
   160→   160→      }
   161→   161→    }
   162→   162→
   163→   163→    document.addEventListener("keydown", handleKeyDown);
   164→   164→
   165→   165→    // Listen for custom event from KeyboardShortcuts component
   166→   166→    function handleCustomNextSlide() {
   167→   167→      goNext();
   168→   168→    }
   169→   169→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   170→   170→
   171→   171→    return () => {
   172→   172→      document.removeEventListener("keydown", handleKeyDown);
   173→   173→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   174→   174→    };
   175→   175→  }, [goBack, goPrev, goNext]);
   176→   176→
   177→   177→  /** Zoom controls */
   178→   178→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   179→   179→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   180→   180→  const zoomFit = () => setZoom(100);
   181→   181→
   182→   182→  // ─── Confetti on section completion ───────────
   183→   183→  useEffect(() => {
   184→   184→    if (!localState) return;
   185→   185→    const isOnLastSlide = localState.currentSlide === localState.totalPages - 1;
   186→   186→    if (isOnLastSlide && !confettiShownRef.current) {
   187→   187→      confettiShownRef.current = true;
   188→   188→      setShowConfetti(true);
   189→   189→      toast.success("🎉 You completed the course! Great job!");
   190→   190→      // Auto-cleanup after 3.5s (animation lasts ~3s + buffer)
   191→   191→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   192→   192→      return () => clearTimeout(timer);
   193→   193→    }
   194→   194→  }, [localState?.currentSlide]);
   195→   195→
   196→   196→  // Reset confetti flag when classroom state changes (new section)
   197→   197→  useEffect(() => {
   198→   198→    confettiShownRef.current = false;
   199→   199→    setShowConfetti(false);
   200→   200→  }, [classroomState?.sectionId]);
   201→   201→
   202→   202→  // ─── No state guard ─────────────────────────────
   203→   203→  if (!localState) {
   204→   204→    return (
   205→   205→      <div className="flex h-screen items-center justify-center bg-muted/30">
   206→   206→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   207→   207→      </div>
   208→   208→    );
   209→   209→  }
   210→   210→
   211→   211→  const currentSlide =
   212→   212→    localState.slides[localState.currentSlide] ?? null;
   213→   213→  const isFirst = localState.currentSlide === 0;
   214→   214→  const isLast = localState.currentSlide >= localState.totalPages - 1;
   215→   215→  const progressPercent = localState.totalPages > 1
   216→   216→    ? Math.round(((localState.currentSlide + 1) / localState.totalPages) * 100)
   217→   217→    : 100;
   218→   218→
   219→   219→  return (
   220→   220→    <div className="flex h-screen flex-col bg-muted/30 relative">
   221→   221→      {/* ─── Confetti Celebration Overlay ─────── */}
   222→   222→      {showConfetti && <ConfettiCelebration />}
   223→   223→      {/* ─── Top Progress Bar ─────────────────────── */}
   224→   224→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   225→   225→        <div
   226→   226→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   227→   227→          style={{ width: `${progressPercent}%` }}
   228→   228→        />
   229→   229→      </div>
   230→   230→
   231→   231→      {/* ─── Top Bar ──────────────────────────────── */}
   232→   232→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   233→   233→        {/* Logo, Section Title, Course Title */}
   234→   234→        <div className="flex items-center gap-3 min-w-0">
   235→   235→          <div className="flex items-center gap-2 shrink-0">
   236→   236→            <div className="flex h-7 w-7 items-center justify-center rounded-md bg-primary">
   237→   237→              <GraduationCap className="h-4 w-4 text-primary-foreground" />
   238→   238→            </div>
   239→   239→            <span className="hidden sm:inline text-sm font-bold text-primary">
   240→   240→              OpenClass
   241→   241→            </span>
   242→   242→          </div>
   243→   243→          <Separator orientation="vertical" className="h-5" />
   244→   244→          <div className="min-w-0">
   245→   245→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   246→   246→              {localState.sectionTitle}
   247→   247→            </h2>
   248→   248→            <p className="text-xs text-muted-foreground truncate leading-tight">
   249→   249→              {localState.courseTitle}
   250→   250→            </p>
   251→   251→          </div>
   252→   252→        </div>
   253→   253→
   254→   254→        {/* Slide Pagination - Pill Badge */}
   255→   255→        <span className="shrink-0 inline-flex items-center gap-1 rounded-full bg-primary/10 px-3 py-1 text-sm font-semibold text-primary tabular-nums">
   256→   256→          {localState.currentSlide + 1}
   257→   257→          <span className="text-primary/40 font-normal">/</span>
   258→   258→          {localState.totalPages}
   259→   259→        </span>
   260→   260→      </header>
   261→   261→
   262→   262→      {/* ─── Slide Content Area ────────────────────── */}
   263→   263→      <div className="flex-1 overflow-auto flex items-start justify-center py-8 px-4 sm:px-6">
   264→   264→        <div
   265→   265→          className={`
   266→   266→            bg-card rounded-2xl shadow-lg border w-full max-w-3xl
   267→   267→            ring-1 ring-black/5
   268→   268→            dark:ring-white/5
   269→   269→            paper-texture
   270→   270→            ${isAnimating
   271→   271→              ? (slideDirection === "forward" ? "slide-exit" : "opacity-0 -translate-x-2 transition-all duration-150")
   272→   272→              : "slide-enter"
   273→   273→            }
   274→   274→          `}
   275→   275→          style={{
   276→   276→            transform: `scale(${zoom / 100})`,
   277→   277→            transformOrigin: "top center",
   278→   278→          }}
   279→   279→        >
   280→   280→          {loadingSlides ? (
   281→   281→            <div className="flex items-center justify-center py-32">
   282→   282→              <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   283→   283→            </div>
   284→   284→          ) : currentSlide ? (
   285→   285→            <div className="p-8 sm:p-12 max-w-2xl mx-auto">
   286→   286→              <SlideRenderer slide={currentSlide} />
   287→   287→            </div>
   288→   288→          ) : (
   289→   289→            <div className="flex items-center justify-center py-32 text-muted-foreground">
   290→   290→              <p>No slide content available.</p>
   291→   291→            </div>
   292→   292→          )}
   293→   293→        </div>
   294→   294→      </div>
   295→   295→
   296→   296→      {/* ─── Bottom Controls (Frosted Glass) ───────── */}
   297→   297→      <footer className="shrink-0 border-t frosted-glass">
   298→   298→        <div className="mx-auto flex h-16 max-w-5xl items-center justify-between px-4 sm:px-6">
   299→   299→          {/* Zoom Controls */}
   300→   300→          <div className="flex items-center gap-1">
   301→   301→            <Tooltip>
   302→   302→              <TooltipTrigger asChild>
   303→   303→                <Button
   304→   304→                  variant="ghost"
   305→   305→                  size="icon"
   306→   306→                  className="h-8 w-8"
   307→   307→                  onClick={zoomOut}
   308→   308→                  disabled={zoom <= MIN_ZOOM}
   309→   309→                >
   310→   310→                  <ZoomOut className="h-4 w-4" />
   311→   311→                </Button>
   312→   312→              </TooltipTrigger>
   313→   313→              <TooltipContent>Zoom Out</TooltipContent>
   314→   314→            </Tooltip>
   315→   315→
   316→   316→            <span className="w-12 text-center text-xs font-medium text-muted-foreground tabular-nums">
   317→   317→              {zoom}%
   318→   318→            </span>
   319→   319→
   320→   320→            <Tooltip>
   321→   321→              <TooltipTrigger asChild>
   322→   322→                <Button
   323→   323→                  variant="ghost"
   324→   324→                  size="icon"
   325→   325→                  className="h-8 w-8"
   326→   326→                  onClick={zoomIn}
   327→   327→                  disabled={zoom >= MAX_ZOOM}
   328→   328→                >
   329→   329→                  <ZoomIn className="h-4 w-4" />
   330→   330→                </Button>
   331→   331→              </TooltipTrigger>
   332→   332→              <TooltipContent>Zoom In</TooltipContent>
   333→   333→            </Tooltip>
   334→   334→
   335→   335→            <Tooltip>
   336→   336→              <TooltipTrigger asChild>
   337→   337→                <Button
   338→   338→                  variant="ghost"
   339→   339→                  size="icon"
   340→   340→                  className="h-8 w-8"
   341→   341→                  onClick={zoomFit}
   342→   342→                >
   343→   343→                  <Maximize2 className="h-4 w-4" />
   344→   344→                </Button>
   345→   345→              </TooltipTrigger>
   346→   346→              <TooltipContent>Fit Width</TooltipContent>
   347→   347→            </Tooltip>
   348→   348→
   349→   349→            <Tooltip>
   350→   350→              <TooltipTrigger asChild>
   351→   351→                <Button
   352→   352→                  variant="ghost"
   353→   353→                  size="icon"
   354→   354→                  className="h-8 w-8"
   355→   355→                  onClick={zoomFit}
   356→   356→                >
   357→   357→                  <RotateCcw className="h-4 w-4" />
   358→   358→                </Button>
   359→   359→              </TooltipTrigger>
   360→   360→              <TooltipContent>Reset Zoom</TooltipContent>
   361→   361→            </Tooltip>
   362→   362→          </div>
   363→   363→
   364→   364→          {/* Navigation Buttons with labels on desktop */}
   365→   365→          <div className="flex items-center gap-2">
   366→   366→            <Button
   367→   367→              variant="outline"
   368→   368→              size="sm"
   369→   369→              className={`gap-1.5 transition-opacity ${isFirst ? "opacity-40 cursor-not-allowed" : ""}`}
   370→   370→              onClick={goPrev}
   371→   371→              disabled={isFirst}
   372→   372→            >
   373→   373→              <ChevronLeft className="h-4 w-4" />
   374→   374→              <span className="hidden sm:inline">Previous</span>
   375→   375→            </Button>
   376→   376→

   378→   378→              variant="outline"
   379→   379→              size="sm"
   380→   380→              className={`gap-1.5 transition-opacity ${isLast ? "opacity-40 cursor-not-allowed" : ""}`}
   381→   381→              onClick={goNext}
   382→   382→              disabled={isLast}
   383→   383→            >

   385→   385→              <ChevronRight className="h-4 w-4" />

   389→   389→          {/* Close + Keyboard Hint */}
   390→   390→          <div className="flex items-center gap-3">

   392→   392→              ← → navigate, ESC exit
   393→   393→            </span>
   394→   394→            <Button
   395→   395→              variant="ghost"
   396→   396→              size="sm"
   397→   397→              className="gap-1.5 text-muted-foreground hover:text-foreground"
   398→   398→              onClick={goBack}

   401→   401→              <span className="hidden sm:inline">Close</span>

   403→   403→          </div>
   404→   404→        </div>
   405→   405→      </footer>
   406→   406→    </div>
   407→   407→  );
   408→   408→}

   411→   411→// Confetti Celebration Component

   413→   413→
   414→   414→/** LMS color palette for confetti particles */
   415→   415→const CONFETTI_COLORS = [
   416→   416→  "#3882f6", // blue
   417→   417→  "#14b8a6", // teal
   418→   418→  "#22d3ee", // cyan
   419→   419→  "#10b981", // emerald
   420→   420→  "#f59e0b", // amber
   421→   421→  "#6366f1", // indigo accent
   422→   422→  "#8b5cf6", // violet accent
   423→   423→  "#ec4899", // pink accent
   424→   424→];
   425→   425→
   426→   426→type ParticleShape = "circle" | "rectangle" | "triangle";
   427→   427→
   428→   428→interface ConfettiParticle {
   429→   429→  id: number;
   430→   430→  left: number;
   431→   431→  color: string;
   432→   432→  shape: ParticleShape;
   433→   433→  size: number;
   434→   434→  fallDuration: string;
   435→   435→  fallDelay: string;
   436→   436→  drift: string;
   437→   437→  spin: string;
   438→   438→}
   439→   439→
   440→   440→function ConfettiCelebration() {
   441→   441→  const particles = useMemo<ConfettiParticle[]>(() => {
   442→   442→    const result: ConfettiParticle[] = [];
   443→   443→    const shapes: ParticleShape[] = ["circle", "rectangle", "triangle"];
   444→   444→    const count = 80;
   445→   445→
   446→   446→    for (let i = 0; i < count; i++) {
   447→   447→      const shape = shapes[i % 3 === 0 ? 0 : i % 3 === 1 ? 1 : 2];
   448→   448→      result.push({
   449→   449→        id: i,
   450→   450→        left: Math.random() * 100,
   451→   451→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   452→   452→        shape,
   453→   453→        size: Math.floor(Math.random() * 8) + 6, // 6-14px
   454→   454→        fallDuration: `${(Math.random() * 1.5 + 2).toFixed(2)}s`, // 2-3.5s
   455→   455→        fallDelay: `${(Math.random() * 0.8).toFixed(2)}s`, // 0-0.8s
   456→   456→        drift: `${(Math.random() * 120 - 60).toFixed(0)}px`, // -60 to +60px
   457→   457→        spin: `${Math.floor(Math.random() * 720 + 360)}deg`, // 360-1080deg
   458→   458→      });
   459→   459→    }
   460→   460→    return result;
   461→   461→  }, []);
   462→   462→
   463→   463→  return (
   464→   464→    <>
   465→   465→      {/* Confetti particles layer */}
   466→   466→      <div className="confetti-container" aria-hidden="true">
   467→   467→        {particles.map((p) => {
   468→   468→          if (p.shape === "triangle") {
   469→   469→            return (
   470→   470→              <div
   471→   471→                key={p.id}
   472→   472→                className="confetti-particle confetti-triangle"
   473→   473→                style={{
   474→   474→                  left: `${p.left}%`,
   475→   475→                  "--confetti-color": p.color,
   476→   476→                  "--tri-size": `${p.size}px`,
   477→   477→                  "--fall-duration": p.fallDuration,
   478→   478→                  "--fall-delay": p.fallDelay,
   479→   479→                  "--drift": p.drift,

   704→   704→function QuizSlide({ slide }: { slide: SlideContent }) {
   705→   705→  const [selectedIdx, setSelectedIdx] = useState<number | null>(null);
   706→   706→  const hasSubmitted = selectedIdx !== null;
   707→   707→
   708→   708→  // Determine correct answer: look for item with icon === "check" or first item
   709→   709→  const correctIdx = slide.items?.findIndex((item) => item.icon === "check") ?? 0;
   710→   710→  const isCorrect = selectedIdx === correctIdx;
   711→   711→
   712→   712→  return (
   713→   713→    <div>
   714→   714→      <div className="mb-6 flex items-center gap-3">
   715→   715→        <div className="h-1 w-8 rounded-full bg-gradient-to-r from-primary to-accent" />
   716→   716→        <h2 className="text-2xl font-bold text-foreground">{slide.title}</h2>
   717→   717→      </div>
   718→   718→      {slide.items && slide.items.length > 0 && (
   719→   719→        <div className="space-y-3">
   720→   720→          {slide.items.map((item, i) => {
   721→   721→            const letter = String.fromCharCode(65 + i);
   722→   722→            const isSelected = selectedIdx === i;
   723→   723→            let borderColor = "border-border hover:border-primary/30 hover:bg-primary/5";
   724→   724→            if (hasSubmitted) {
   725→   725→              if (i === correctIdx) borderColor = "border-emerald-500 bg-emerald-500/10";
   726→   726→              else if (isSelected && !isCorrect) borderColor = "border-red-400 bg-red-400/10";
   727→   727→              else borderColor = "border-border opacity-60";
   728→   728→            } else if (isSelected) {
   729→   729→              borderColor = "border-primary bg-primary/10";
   730→   730→            }
   731→   731→
   732→   732→            return (
   733→   733→              <button
   734→   734→                key={i}
   735→   735→                type="button"
   736→   736→                className={`w-full rounded-lg border p-4 text-left transition-all duration-200 ${borderColor}`}
   737→   737→                onClick={() => {
   738→   738→                  if (!hasSubmitted) setSelectedIdx(i);
   739→   739→                }}
   740→   740→                disabled={hasSubmitted}
   741→   741→              >
   742→   742→                <div className="flex items-start gap-3">
   743→   743→                  <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold ${
   744→   744→                    hasSubmitted && i === correctIdx
   745→   745→                      ? "bg-emerald-500 text-white"
   746→   746→                      : hasSubmitted && isSelected && !isCorrect
   747→   747→                        ? "bg-red-400 text-white"
   748→   748→                        : "bg-primary/10 text-primary"
   749→   749→                  }`}>
   750→   750→                    {hasSubmitted && i === correctIdx ? "✓" : hasSubmitted && isSelected && !isCorrect ? "✗" : letter}
   751→   751→                  </span>
   752→   752→                  <div className="flex-1">
   753→   753→                    <p className="text-sm font-medium text-foreground">
   754→   754→                      {item.heading || item.text}
   755→   755→                    </p>
   756→   756→                    {item.heading && item.text !== item.heading && (
   757→   757→                      <p className="text-xs text-muted-foreground mt-1">{item.text}</p>
   758→   758→                    )}
   759→   759→                  </div>
   760→   760→                </div>
   761→   761→              </button>
   762→   762→            );
   763→   763→          })}
   764→   764→
   765→   765→          {/* Feedback after selection */}
   766→   766→          {hasSubmitted && (
   767→   767→            <div className={`mt-4 rounded-lg border p-4 ${
   768→   768→              isCorrect
   769→   769→                ? "border-emerald-200 bg-emerald-50 dark:border-emerald-800 dark:bg-emerald-950/50"
   770→   770→                : "border-amber-200 bg-amber-50 dark:border-amber-800 dark:bg-amber-950/50"
   771→   771→            }`}>
   772→   772→              <p className={`text-sm font-semibold ${
   773→   773→                isCorrect ? "text-emerald-700 dark:text-emerald-400" : "text-amber-700 dark:text-amber-400"
   774→   774→              }`}>
   775→   775→                {isCorrect ? "🎉 Correct! Well done!" : "❌ Not quite right."}
   776→   776→              </p>
   777→   777→              {!isCorrect && (
   778→   778→                <p className="mt-1 text-xs text-muted-foreground">
   779→   779→                  The correct answer is <strong>{String.fromCharCode(65 + correctIdx)}</strong>.
   780→   780→                </p>
   781→   781→              )}
   782→   782→            </div>
   783→   783→          )}
   784→   784→        </div>
   785→   785→      )}
   786→   786→    </div>
   787→   787→  );
   788→   788→}
