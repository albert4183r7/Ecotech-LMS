    56→    56→
    57→    57→function CircularProgress({ percentage }: { percentage: number }) {
    58→    58→  const radius = 70;
    59→    59→  const stroke = 10;
    60→    60→  const normalizedRadius = radius - stroke / 2;
    61→    61→  const circumference = normalizedRadius * 2 * Math.PI;
    62→    62→  const strokeDashoffset =
    63→    63→    circumference - (percentage / 100) * circumference;
    64→    64→
    65→    65→  return (
    66→    66→    <div className="relative inline-flex items-center justify-center">
    67→    67→      <svg
    68→    68→        width={radius * 2}
    69→    69→        height={radius * 2}
    70→    70→        className="-rotate-90"
    71→    71→      >
    72→    72→        {/* Background track */}
    73→    73→        <circle
    74→    74→          cx={radius}
    75→    75→          cy={radius}
    76→    76→          r={normalizedRadius}
    77→    77→          fill="none"
    78→    78→          stroke="currentColor"
    79→    79→          strokeWidth={stroke}
    80→    80→          className="text-muted/30"
    81→    81→        />
    82→    82→        {/* Progress arc with gradient stroke */}
    83→    83→        <circle
    84→    84→          cx={radius}
    85→    85→          cy={radius}
    86→    86→          r={normalizedRadius}
    87→    87→          fill="none"
    88→    88→          stroke="url(#progress-gradient)"
    89→    89→          strokeWidth={stroke}
    90→    90→          strokeLinecap="round"
    91→    91→          strokeDasharray={circumference}
    92→    92→          strokeDashoffset={strokeDashoffset}
    93→    93→          className="transition-all duration-700 ease-out"
    94→    94→        />
    95→    95→        <defs>
    96→    96→          <linearGradient
    97→    97→            id="progress-gradient"
    98→    98→            x1="0%"
    99→    99→            y1="0%"
   100→   100→            x2="100%"
   101→   101→            y2="100%"
   102→   102→          >
   103→   103→            <stop offset="0%" stopColor="#06b6d4" />
   104→   104→            <stop offset="50%" stopColor="#0891b2" />
   105→   105→            <stop offset="100%" stopColor="#0d9488" />
   106→   106→          </linearGradient>
   107→   107→        </defs>
   108→   108→      </svg>
   109→   109→      <div className="absolute flex flex-col items-center justify-center">
   110→   110→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   111→   111→          {percentage}%
   112→   112→        </span>
   113→   113→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   114→   114→          completed
   115→   115→        </span>
   116→   116→      </div>
   117→   117→    </div>
   118→   118→  );
   119→   119→}
