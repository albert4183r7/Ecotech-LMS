   421→   421→    async (lessonId: string) => {

   423→   423→      if (!lesson) return;

   434→   434→        }
   435→   435→      });

   437→   437→      setGeneratingLessonId(lessonId);

   440→   440→      setGenProgress({ current: 0, total: lesson.slides.length });

   450→   450→        const res = await fetch("/api/lessons/generate-slides", {

   452→   452→          headers: { "Content-Type": "application/json" },
   453→   453→          body: JSON.stringify({
   454→   454→            lessonId,
   455→   455→            language: lesson.language,
   456→   456→          }),
   457→   457→          signal: abort.signal,
   458→   458→        });

   460→   460→        if (!res.ok || !res.body) {
   461→   461→          toast.error("Failed to start slide generation");
   462→   462→          setGeneratingLessonId(null);
   463→   463→          return;
   464→   464→        }
   465→   465→
   466→   466→        const reader = res.body.getReader();
   467→   467→        const decoder = new TextDecoder();
   468→   468→        let buffer = "";
   469→   469→        let slidesCompleted = 0;
   470→   470→        let rawHtmlAccum = "";
   471→   471→
   472→   472→        while (true) {
   473→   473→          const { done, value } = await reader.read();
   474→   474→          if (done) break;
   475→   475→          buffer += decoder.decode(value, { stream: true });
   476→   476→
   477→   477→          const lines = buffer.split("\n");
   478→   478→          buffer = lines.pop() || "";
   479→   479→
   480→   480→          for (const line of lines) {
   481→   481→            if (!line.startsWith("data: ")) continue;
   482→   482→            try {
   483→   483→              const data = JSON.parse(line.slice(6));
   484→   484→
   485→   485→              if (data.slideId && data.slideTitle && data.totalSlides !== undefined) {
   486→   486→                // slide_start event
   487→   487→                setCurrentGenSlideId(data.slideId);
   488→   488→                setSlideGenStates((prev) => ({
   489→   489→                  ...prev,
   490→   490→                  [data.slideId]: { status: "generating" },
   491→   491→                }));
   492→   492→                rawHtmlAccum = "";
   493→   493→                setGenProgress({ current: data.slideIndex + 1, total: data.totalSlides });
   494→   494→              }
   495→   495→
   496→   496→              if (data.slideId && data.html) {
   497→   497→                // chunk event
   498→   498→                rawHtmlAccum += data.html;
   499→   499→                setStreamingHtml(buildLivePreviewDoc(rawHtmlAccum, data.slideId));
   500→   500→              }
   501→   501→
   502→   502→              if (data.slideId && data.htmlBody) {
   503→   503→                // slide_complete event
   504→   504→                slidesCompleted++;
