    56→
    57→function CircularProgress({ percentage }: { percentage: number }) {
    58→  const radius = 70;
    59→  const stroke = 10;
    60→  const normalizedRadius = radius - stroke / 2;
    61→  const circumference = normalizedRadius * 2 * Math.PI;
    62→  const strokeDashoffset =
    63→    circumference - (percentage / 100) * circumference;
    64→
    65→  return (
    66→    <div className="relative inline-flex items-center justify-center">
    67→      <svg
    68→        width={radius * 2}
    69→        height={radius * 2}
    70→        className="-rotate-90"
    71→      >
    72→        {/* Background track */}
    73→        <circle
    74→          cx={radius}
    75→          cy={radius}
    76→          r={normalizedRadius}
    77→          fill="none"
    78→          stroke="currentColor"
    79→          strokeWidth={stroke}
    80→          className="text-muted/30"
    81→        />
    82→        {/* Progress arc with gradient stroke */}
    83→        <circle
    84→          cx={radius}
    85→          cy={radius}
    86→          r={normalizedRadius}
    87→          fill="none"
    88→          stroke="url(#progress-gradient)"
    89→          strokeWidth={stroke}
    90→          strokeLinecap="round"
    91→          strokeDasharray={circumference}
    92→          strokeDashoffset={strokeDashoffset}
    93→          className="transition-all duration-700 ease-out"
    94→        />
    95→        <defs>
    96→          <linearGradient
    97→            id="progress-gradient"
    98→            x1="0%"
    99→            y1="0%"
   100→            x2="100%"
   101→            y2="100%"
   102→          >
   103→            <stop offset="0%" stopColor="#06b6d4" />
   104→            <stop offset="50%" stopColor="#0891b2" />
   105→            <stop offset="100%" stopColor="#0d9488" />
   106→          </linearGradient>
   107→        </defs>
   108→      </svg>
   109→      <div className="absolute flex flex-col items-center justify-center">
   110→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   111→          {percentage}%
   112→        </span>
   113→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   114→          completed
   115→        </span>
   116→      </div>
   117→    </div>
   118→  );
   119→}

   248→        /* silent */

   250→        setLoading(false);

   393→    setLoading(true);

   395→      const res = await fetch(`/api/user/${currentUserId}`);
   396→      const json = await res.json();
   397→      if (json.success) {
   398→        setProfile(json.data);
   399→      }
   400→    } catch {
   401→      /* silent */
   402→    } finally {
   403→      setLoading(false);
   404→    }
   405→  }, [currentUserId]);
   406→
   407→  // Fetch first completed course for certificate
   408→  useEffect(() => {
   409→    async function fetchCompletedCourse() {
   410→      try {
   411→        const res = await fetch('/api/enrollments?userId=' + currentUserId);

   415→            (e: { status: string; course: { title: string }; completedAt: string | null }) =>

   424→        /* silent */
   425→      }

   430→  useEffect(() => {
   431→    fetchProfile();

   434→  if (loading) {
   435→    return <ProfileSkeleton />;

   438→  if (!profile) {

   449→    .split(" ")
   450→    .map((w) => w.charAt(0).toUpperCase())
