    82→    82→  const [coverImage, setCoverImage] = useState("");
    83→    83→  const [coverPreview, setCoverPreview] = useState<string | null>(null);
    84→    84→  const [lessons, setLessons] = useState<LessonDraft[]>([]);
    85→    85→
    86→    86→  // ---- UI state ----
    87→    87→  const [categories, setCategories] = useState<CategoryItem[]>([]);
    88→    88→  const [loading, setLoading] = useState(false);
    89→    89→  const [saving, setSaving] = useState(false);
    90→    90→  const [generating, setGenerating] = useState(false);
    91→    91→  const [modalOpen, setModalOpen] = useState(false);
    92→    92→  const [urlInputOpen, setUrlInputOpen] = useState(false);
    93→    93→  const [imageUrl, setImageUrl] = useState("");
    94→    94→
    95→    95→  // ---- Modal state ----
    96→    96→  const [lessonName, setLessonName] = useState("");
    97→    97→  const [lessonPrompt, setLessonPrompt] = useState("");
    98→    98→  const [lessonLanguage, setLessonLanguage] = useState("english");
    99→    99→  const [lessonPdfName, setLessonPdfName] = useState("");
   100→   100→
   101→   101→  // ---- Live streaming state ----
   102→   102→  const [streamingHtml, setStreamingHtml] = useState("");
   103→   103→  const [showStreamPreview, setShowStreamPreview] = useState(false);
   104→   104→   const streamPreviewRef = useRef<HTMLDivElement>(null);
   105→   105→
   106→   106→  // ---- Outline generation state ----
   107→   107→  const [outlineGenerating, setOutlineGenerating] = useState(false);
   108→   108→  const [outlineModalOpen, setOutlineModalOpen] = useState(false);
   109→   109→  const [outlineTopic, setOutlineTopic] = useState("");
   110→   110→  const [outlineSections, setOutlineSections] = useState<OutlineSection[]>([]);
   111→   111→
   112→   112→  // ---- Expanded lesson preview ----
   113→   113→  const [expandedLessonId, setExpandedLessonId] = useState<string | null>(null);
   114→   114→
   115→   115→  // ---- Pre-fill title from hero prompt ----
   116→   116→  useEffect(() => {
   117→   117→    if (createPrompt) {
   118→   118→      setTitle(createPrompt);
   119→   119→      setCreatePrompt("");
   120→   120→    }
   121→   121→  }, [createPrompt, setCreatePrompt]);
   122→   122→
   123→   123→  // ---- Refs ----
   124→   124→  const fileInputRef = useRef<HTMLInputElement>(null);
   125→   125→  const lessonFileInputRef = useRef<HTMLInputElement>(null);
   126→   126→  const abortControllerRef = useRef<AbortController | null>(null);
   127→   127→
   128→   128→  // ---- Fetch categories ----
   129→   129→  useEffect(() => {
   130→   130→    async function fetchCategories() {
   131→   131→      setLoading(true);
   132→   132→      try {
   133→   133→        const res = await fetch("/api/categories");
   134→   134→        const json = await res.json();
   135→   135→        if (json.success) {
   136→   136→          setCategories(json.data);
   137→   137→        }
   138→   138→      } catch {
   139→   139→        toast.error("Failed to load categories");
   140→   140→      } finally {
   141→   141→        setLoading(false);
   142→   142→      }
   143→   143→    }
   144→   144→    fetchCategories();
   145→   145→  }, []);
   146→   146→
   147→   147→  // ---- Cover image handlers ----
   148→   148→  const handleCoverUpload = () => {
   149→   149→    setUrlInputOpen(true);
   150→   150→  };
   151→   151→
   152→   152→  const handleUrlConfirm = () => {
   153→   153→    if (imageUrl.trim()) {
   154→   154→      setCoverImage(imageUrl.trim());
   155→   155→      setCoverPreview(imageUrl.trim());
   156→   156→      setUrlInputOpen(false);
   157→   157→      setImageUrl("");
   158→   158→    }
   159→   159→  };
   160→   160→
   161→   161→  const handleRemoveCover = () => {
   162→   162→    setCoverImage("");
   163→   163→    setCoverPreview(null);
   164→   164→  };
   165→   165→
   166→   166→  // ---- Lesson management ----
   167→   167→  const handleOpenModal = () => {
   168→   168→    setLessonName("");
   169→   169→    setLessonPrompt("");
   170→   170→    setLessonLanguage(language);
   171→   171→    setLessonPdfName("");
   172→   172→    setStreamingHtml("");
   173→   173→    setShowStreamPreview(false);
   174→   174→    setModalOpen(true);
   175→   175→  };
   176→   176→
   177→   177→  const handleGenerateLesson = useCallback(async () => {
   178→   178→    if (!lessonName.trim()) {
   179→   179→      toast.error("Lesson name is required");
   180→   180→      return;
   181→   181→    }
   182→   182→    if (!lessonPrompt.trim()) {
   183→   183→      toast.error("Content prompt is required");
   184→   184→      return;
   185→   185→    }
   186→   186→    if (lessons.length >= MAX_LESSONS) {
   187→   187→      toast.error(`Maximum ${MAX_LESSONS} lessons allowed`);
   188→   188→      return;
   189→   189→    }
   190→   190→
   191→   191→    setGenerating(true);
   192→   192→    setStreamingHtml("");
   193→   193→    setShowStreamPreview(true);
   194→   194→
   195→   195→    const abort = new AbortController();
   196→   196→    abortControllerRef.current = abort;
   197→   197→
   198→   198→    try {
   199→   199→      const res = await fetch("/api/generate-slide-html", {
   200→   200→        method: "POST",
   201→   201→        headers: { "Content-Type": "application/json" },
   202→   202→        body: JSON.stringify({
   203→   203→          slideTitle: lessonName.trim(),
   204→   204→          prompt: lessonPrompt.trim(),
   205→   205→          language: lessonLanguage,
   206→   206→        }),
   207→   207→        signal: abort.signal,
   208→   208→      });
   209→   209→
   210→   210→      if (!res.ok || !res.body) {
   211→   211→        toast.error("Failed to start generation");
   212→   212→        setGenerating(false);
   213→   213→        setShowStreamPreview(false);
   214→   214→        return;
   215→   215→      }
   216→   216→
   217→   217→      // Read SSE stream with live preview
   218→   218→      const reader = res.body.getReader();
   219→   219→      const decoder = new TextDecoder();
   220→   220→      let buffer = "";
   221→   221→      let htmlBody = "";
   222→   222→      let slideTitle = lessonName.trim();
   223→   223→      let rawHtml = ""; // accumulating raw HTML fragments for live preview
   224→   224→
   225→   225→      while (true) {
   226→   226→        const { done, value } = await reader.read();

   228→   228→        buffer += decoder.decode(value, { stream: true });
   229→   229→
   230→   230→        const lines = buffer.split("\n");
   231→   231→        buffer = lines.pop() || "";
   232→   232→
   233→   233→        for (const line of lines) {
   234→   234→          if (!line.startsWith("data: ")) continue;
   235→   235→          try {
   236→   236→            const data = JSON.parse(line.slice(6));
   237→   237→            if (data.htmlBody) {
   238→   238→              htmlBody = data.htmlBody;
   239→   239→            }
   240→   240→            if (data.slideTitle) {
   241→   241→              slideTitle = data.slideTitle;
   242→   242→            }
   243→   243→            if (data.html) {
   244→   244→              rawHtml += data.html;
   245→   245→              // Build live preview from accumulated raw HTML
   246→   246→              const previewDoc = buildLivePreviewDoc(rawHtml, slideTitle);
   247→   247→              setStreamingHtml(previewDoc);
   248→   248→              // Auto-scroll preview into view
   249→   249→              streamPreviewRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
   250→   250→            }
   251→   251→          } catch {
   252→   252→            // skip malformed JSON
   253→   253→          }
   254→   254→        }
   255→   255→      }
   256→   256→
   257→   257→      if (!htmlBody) {
   258→   258→        toast.error("Generation returned empty content");
   259→   259→        setGenerating(false);
   260→   260→        setShowStreamPreview(false);
   261→   261→        return;
   262→   262→      }
   263→   263→
   264→   264→      const newLesson: LessonDraft = {
   265→   265→        id: `sec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   266→   266→        title: slideTitle,
   267→   267→        totalPages: 1,
   268→   268→        htmlBody,
   269→   269→        language: lessonLanguage,
   270→   270→      };
   271→   271→
   272→   272→      setLessons((prev) => [...prev, newLesson]);
   273→   273→      setModalOpen(false);
   274→   274→      setShowStreamPreview(false);
   275→   275→      setStreamingHtml("");
   276→   276→      toast.success(`Lesson \"${slideTitle}\" generated successfully`);
   277→   277→    } catch (err) {
   278→   278→      if (err instanceof Error && err.name === "AbortError") return;
   279→   279→      toast.error("Failed to generate lesson. Please try again.");
   280→   280→    } finally {
   281→   281→      setGenerating(false);
   282→   282→      abortControllerRef.current = null;
   283→   283→    }
   284→   284→  }, [lessonName, lessonPrompt, lessonLanguage, lessons.length]);
   285→   285→
   286→   286→  const handleCancelGeneration = () => {

   289→   289→    setShowStreamPreview(false);
   290→   290→    setStreamingHtml("");
   291→   291→  };
   292→   292→
   293→   293→  const handleDeleteLesson = (id: string) => {
   294→   294→    setLessons((prev) => prev.filter((s) => s.id !== id));
   295→   295→    if (expandedLessonId === id) setExpandedLessonId(null);
   296→   296→  };

   298→   298→  const handleMoveLesson = (index: number, direction: "up" | "down") => {
   299→   299→    const newLessons = [...lessons];
   300→   300→    const targetIndex = direction === "up" ? index - 1 : index + 1;
   301→   301→    if (targetIndex < 0 || targetIndex >= newLessons.length) return;
   302→   302→    [newLessons[index], newLessons[targetIndex]] = [
   303→   303→      newLessons[targetIndex],
   304→   304→      newLessons[index],
   305→   305→    ];
   306→   306→    setLessons(newLessons);
   307→   307→  };
   308→   308→
   309→   309→  // ---- Outline generation ----
   310→   310→  const handleOpenOutlineModal = () => {
   311→   311→    setOutlineTopic(title || "");
   312→   312→    setOutlineSections([]);
   313→   313→    setOutlineModalOpen(true);
   314→   314→  };
   315→   315→
   316→   316→  const handleGenerateOutline = useCallback(async () => {
   317→   317→    if (!outlineTopic.trim()) {
   318→   318→      toast.error("Please enter a course topic");
   319→   319→      return;
   320→   320→    }
   321→   321→
   322→   322→    setOutlineGenerating(true);
   323→   323→    try {
   324→   324→      const res = await fetch("/api/generate-outline", {
   325→   325→        method: "POST",
   326→   326→        headers: { "Content-Type": "application/json" },
   327→   327→        body: JSON.stringify({
   328→   328→          topic: outlineTopic.trim(),
   329→   329→          prompt: `Create a comprehensive course outline for: ${outlineTopic.trim()}. Include 4-6 lessons that progressively build understanding.`,
   330→   330→          language,
   331→   331→        }),
   332→   332→      });
   333→   333→
   334→   334→      const json = await res.json();
   335→   335→      if (json.success && json.data) {
   336→   336→        setOutlineSections(json.data.lessons || []);
   337→   337→        if (json.data.title && !title) {
   338→   338→          setTitle(json.data.title);
   339→   339→        }
   340→   340→        toast.success("Outline generated! Select lessons to generate.");
   341→   341→      } else {
   342→   342→        toast.error(json.error || "Failed to generate outline");
   343→   343→      }
   344→   344→    } catch {
   345→   345→      toast.error("Failed to generate outline. Please try again.");
   346→   346→    } finally {
   347→   347→      setOutlineGenerating(false);
   348→   348→    }
   349→   349→  }, [outlineTopic, language, title]);
   350→   350→
   351→   351→  const handleAddOutlineSections = () => {
   352→   352→    // Add outline lessons as drafts (without htmlBody yet)
   353→   353→    const newDrafts: LessonDraft[] = outlineSections.map((sec, i) => ({
   354→   354→      id: `sec_${Date.now()}_${i}_${Math.random().toString(36).slice(2, 8)}`,
   355→   355→      title: sec.title,
   356→   356→      totalPages: 1,
   357→   357→      htmlBody: "",
   358→   358→      language,
   359→   359→    }));
   360→   360→
   361→   361→    const total = lessons.length + newDrafts.length;
   362→   362→    if (total > MAX_LESSONS) {
   363→   363→      toast.error(`Can only add ${MAX_LESSONS - lessons.length} more lessons`);
   364→   364→      return;
   365→   365→    }
   366→   366→
   367→   367→    setLessons((prev) => [...prev, ...newDrafts]);
   368→   368→    setOutlineModalOpen(false);
   369→   369→    toast.success(`${newDrafts.length} lessons added. Click the generate button on each to create content.`);
   370→   370→  };
   371→   371→
   372→   372→  // ---- Form submission ----
   373→   373→  const handleSave = async () => {
   374→   374→    if (!title.trim()) {
   375→   375→      toast.error("Course title is required");
   376→   376→      return;
   377→   377→    }
   378→   378→
   379→   379→    // Check if any lesson is missing htmlBody
   380→   380→    const missingContent = lessons.filter((s) => !s.htmlBody);
   381→   381→    if (missingContent.length > 0) {
   382→   382→      toast.error(`${missingContent.length} lesson(s) have no content yet. Generate content for all lessons first.`);
   383→   383→      return;
   384→   384→    }
   385→   385→
   386→   386→    setSaving(true);
   387→   387→    try {
   388→   388→      const payload = {
   389→   389→        title: title.trim(),
   390→   390→        description: description.trim() || null,
   391→   391→        categoryId: categoryId || null,
   392→   392→        language,
   393→   393→        creatorId: currentUserId,
   394→   394→        coverImage: coverImage || null,
   395→   395→        lessons: lessons.map((sec, index) => ({
   396→   396→          title: sec.title,
   397→   397→          htmlBody: sec.htmlBody,
   398→   398→          totalPages: sec.totalPages,
   399→   399→          order: index,
   400→   400→        })),
   401→   401→      };
   402→   402→
   403→   403→      const res = await fetch("/api/courses", {
   404→   404→        method: "POST",
   405→   405→        headers: { "Content-Type": "application/json" },
   406→   406→        body: JSON.stringify(payload),
   407→   407→      });
   408→   408→
   409→   409→      const json = await res.json();
   410→   410→
   411→   411→      if (json.success) {
   412→   412→        toast.success("Course created successfully!");
   413→   413→        goBack();
   414→   414→      } else {
   415→   415→        toast.error(json.error || "Failed to create course");
   416→   416→      }
   417→   417→    } catch {
   418→   418→      toast.error("Failed to create course. Please try again.");
   419→   419→    } finally {
   420→   420→      setSaving(false);
   421→   421→    }
   422→   422→  };
   423→   423→
   424→   424→  // ---- Lesson PDF upload simulation ----
   425→   425→  const handleLessonPdfUpload = () => {
   426→   426→    lessonFileInputRef.current?.click();
   427→   427→  };
   428→   428→
   429→   429→  const handleLessonFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   430→   430→    const file = e.target.files?.[0];
   431→   431→    if (file) {
   432→   432→      setLessonPdfName(file.name);
   433→   433→      if (!lessonPrompt.trim()) {
   434→   434→        const nameWithoutExt = file.name.replace(/\.pdf$/i, "");
   435→   435→        setLessonPrompt(
   436→   436→          `Generate course content based on the uploaded PDF: ${nameWithoutExt}`
   437→   437→        );
   438→   438→        if (!lessonName.trim()) {
   439→   439→          setLessonName(nameWithoutExt);
   440→   440→        }
   441→   441→      }
   442→   442→    }
   443→   443→  };
   444→   444→
   445→   445→  // ============================================
   446→   446→  // Render
   447→   447→  // ============================================
   448→   448→
   449→   449→  return (
   450→   450→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   451→   451→      {/* ---- Breadcrumbs ---- */}
   452→   452→      <div className="mb-6 flex items-center gap-3">
   453→   453→        <button
   454→   454→          onClick={goBack}
   455→   455→          className="flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-foreground"
   456→   456→        >
   457→   457→          <ArrowLeft className="h-4 w-4" />
   458→   458→          Back
   459→   459→        </button>
   460→   460→        <Separator orientation="vertical" className="h-4" />
   461→   461→        <h1 className="text-lg font-semibold text-foreground">Create Course</h1>
   462→   462→      </div>
   463→   463→
   464→   464→      {/* ---- Two-Column Layout ---- */}
   465→   465→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-5">
   466→   466→        {/* ======== Left Panel: Course Metadata (3 cols) ======== */}
   467→   467→        <div className="lg:col-span-3">
   468→   468→          <Card className="border-border/50">
   469→   469→            <CardContent className="space-y-6 p-6">
   470→   470→              {/* Cover Image Upload */}
   471→   471→              <div className="space-y-2">
   472→   472→                <Label className="text-sm font-medium">Cover Image</Label>
   473→   473→                {coverPreview ? (
   474→   474→                  <div className="relative overflow-hidden rounded-lg border border-border">
   475→   475→                    <img
   476→   476→                      src={coverPreview}
   477→   477→                      alt="Course cover"
   478→   478→                      className="h-48 w-full object-cover"
   479→   479→                      onError={() => setCoverPreview(null)}
   480→   480→                    />
   481→   481→                    <button
   482→   482→                      onClick={handleRemoveCover}
   483→   483→                      className="absolute top-2 right-2 flex h-7 w-7 items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80"
   484→   484→                    >
   485→   485→                      <X className="h-3.5 w-3.5" />
   486→   486→                    </button>
   487→   487→                  </div>
   488→   488→                ) : (
   489→   489→                  <button
   490→   490→                    type="button"
   491→   491→                    onClick={handleCoverUpload}
   492→   492→                    className="flex h-48 w-full flex-col items-center justify-center gap-3 rounded-lg border-2 border-dashed border-border/70 bg-muted/30 transition-colors hover:border-primary/50 hover:bg-muted/50"
   493→   493→                  >
   494→   494→                    <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted">
   495→   495→                      <ImageIcon className="h-5 w-5 text-muted-foreground" />
   496→   496→                    </div>
   497→   497→                    <div className="text-center">
   498→   498→                      <p className="text-sm font-medium text-foreground">
   499→   499→                        Upload cover image
   500→   500→                      </p>
   501→   501→                      <p className="mt-0.5 text-xs text-muted-foreground">
   502→   502→                        Recommended 1920×1080
