   264→/*  Donut Chart                                                        */

   266→
   267→function DonutChart({
   268→  data,
   269→  total,
   270→}: {
   271→  data: { categoryName: string; categoryColor: string | null; courseCount: number }[];
   272→  total: number;
   273→}) {
   274→  // Build conic-gradient
   275→  const gradientParts: string[] = [];
   276→  let currentAngle = 0;
   277→
   278→  const DONUT_COLORS = [
   279→    "#14b8a6",
   280→    "#f59e0b",
   281→    "#ef4444",
   282→    "#8b5cf6",
   283→    "#ec4899",
   284→    "#06b6d4",
   285→    "#84cc16",
   286→    "#f97316",
   287→  ];
   288→
   289→  data.forEach((item, i) => {
   290→    const color = item.categoryColor || DONUT_COLORS[i % DONUT_COLORS.length];
   291→    const angle = total > 0 ? (item.courseCount / total) * 360 : 0;
   292→    gradientParts.push(`${color} ${currentAngle}deg ${currentAngle + angle}deg`);
   293→    currentAngle += angle;
   294→  });
   295→
   296→  const gradientStr =
   297→    gradientParts.length > 0
   298→      ? `conic-gradient(${gradientParts.join(", ")})`
   299→      : "conic-gradient(#e5e7eb 0deg 360deg)";
   300→
   301→  return (
   302→    <div className="flex flex-col items-center gap-4">
   303→      {/* Donut */}
   304→      <div
   305→        className="relative h-40 w-40 sm:h-48 sm:w-48 rounded-full stat-pop"
   306→        style={{
   307→          background: gradientStr,
   308→        }}
   309→      >
   310→        {/* Inner circle (donut hole) */}
   311→        <div className="absolute inset-0 m-auto h-[60%] w-[60%] rounded-full bg-card flex flex-col items-center justify-center shadow-inner">
   312→          <span className="text-2xl font-bold text-foreground">{total}</span>
   313→          <span className="text-[11px] text-muted-foreground">courses</span>
   314→        </div>
   315→      </div>
   316→
   317→      {/* Legend */}
   318→      <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 w-full max-w-[220px]">
   319→        {data.map((item, i) => {
   320→          const color =
   321→            item.categoryColor || DONUT_COLORS[i % DONUT_COLORS.length];
   322→          const pct = total > 0 ? Math.round((item.courseCount / total) * 100) : 0;
   323→          return (
   324→            <div key={item.categoryName} className="flex items-center gap-2">
   325→              <span
   326→                className="h-2.5 w-2.5 rounded-full shrink-0"
   327→                style={{ backgroundColor: color }}
   328→              />
   329→              <div className="min-w-0">
   330→                <span className="text-xs text-foreground truncate block">
   331→                  {item.categoryName}
   332→                </span>
   333→                <span className="text-[10px] text-muted-foreground">
   334→                  {pct}%
   335→                </span>
   336→              </div>
   337→            </div>
   338→          );
   339→        })}
   340→      </div>
   341→    </div>
   342→  );
   343→}

   346→/*  Weekly Heatmap                                                     */
   347→/* ------------------------------------------------------------------ */
   348→
   349→function WeeklyHeatmap({
   350→  userId,
   351→}: {
   352→  userId: string;
   353→}) {
   354→  const [heatmapData, setHeatmapData] = useState<number[][]>([]);
   355→  const [monthLabels, setMonthLabels] = useState<string[]>([]);
   356→
   357→  useEffect(() => {
   358→    async function fetchHeatmap() {
   359→      try {
   360→        const res = await fetch(`/api/activity?userId=${userId}`);
   361→        const json = await res.json();
   362→        if (json.success && json.data) {
   363→          // Use the activity data to build a heatmap
   364→          const activityMap: Record<string, number> = {};
   365→          const activities = json.data;
   366→          for (const act of activities) {
   367→            const dateStr = new Date(act.date || act.timestamp).toISOString().split("T")[0];
   368→            activityMap[dateStr] = (activityMap[dateStr] || 0) + 1;
   369→          }
   370→
   371→          // Build 12-week × 7-day grid
   372→          const grid: number[][] = [];
   373→          const today = new Date();
   374→          const months: string[] = [];
   375→          const seenMonths = new Set<string>();
   376→
   377→          // Start from 12 weeks ago, aligned to Sunday
   378→          const startDate = new Date(today);
   379→          startDate.setDate(startDate.getDate() - 83); // ~12 weeks
   380→          const dayOfWeek = startDate.getDay();
   381→          startDate.setDate(startDate.getDate() - dayOfWeek); // Align to Sunday
   382→
   383→          for (let week = 0; week < 12; week++) {
   384→            for (let day = 0; day < 7; day++) {
   385→              if (day === 0) grid.push([]);
   386→              const cellDate = new Date(startDate);
   387→              cellDate.setDate(cellDate.getDate() + week * 7 + day);
   388→              const dateStr = cellDate.toISOString().split("T")[0];
   389→              grid[week].push(activityMap[dateStr] || 0);
   390→
   391→              // Track month labels
   392→              if (day === 0) {
   393→                const monthName = cellDate.toLocaleString("en-US", {
   394→                  month: "short",
   395→                });
   396→                if (!seenMonths.has(monthName)) {
   397→                  seenMonths.add(monthName);
   398→                  months.push(monthName);
   399→                } else {
   400→                  months.push("");
   401→                }
   402→              }
   403→            }
   404→          }
   405→
   406→          setHeatmapData(grid);
   407→          setMonthLabels(months);
   408→        }
   409→      } catch {
   410→        // Generate fallback random data
   411→        const rand = seededRandom(42);
   412→        const grid: number[][] = [];
   413→        for (let week = 0; week < 12; week++) {
   414→          grid.push([]);
   415→          for (let day = 0; day < 7; day++) {
   416→            grid[week].push(Math.floor(rand() * 5));
   417→          }
   418→        }
   419→        setHeatmapData(grid);
   420→        setMonthLabels(["Jan", "", "Feb", "", "Mar", "", "Apr"]);
   421→      }
   422→    }
   423→    fetchHeatmap();
   424→  }, [userId]);
   425→
   426→  const dayLabels = ["Sun", "", "Tue", "", "Thu", "", "Sat"];
   427→
   428→  const getHeatColor = (level: number) => {
   429→    switch (level) {
   430→      case 0:
   431→        return "bg-muted";
   432→      case 1:
   433→        return "bg-emerald-200 dark:bg-emerald-900/50";
   434→      case 2:
   435→        return "bg-emerald-300 dark:bg-emerald-800/60";
   436→      case 3:
   437→        return "bg-emerald-400 dark:bg-emerald-700/70";
   438→      default:
   439→        return "bg-emerald-500 dark:bg-emerald-600";
   440→    }
   441→  };
   442→
   443→  if (heatmapData.length === 0) return null;
   444→
   445→  return (
   446→    <TooltipProvider delayDuration={200}>
   447→      <div className="space-y-2">
   448→        {/* Month labels */}
   449→        <div className="flex gap-0.5 pl-8">
   450→          {monthLabels.map((label, i) => (
   451→            <div
   452→              key={i}
   453→              className="flex-1 min-w-0"
   454→            >
   455→              <span className="text-[9px] text-muted-foreground truncate">
   456→                {label}
   457→              </span>
   458→            </div>
   459→          ))}
   460→        </div>
   461→
   462→        <div className="flex gap-1.5">
   463→          {/* Day labels */}
   464→          <div className="flex flex-col gap-0.5 shrink-0 pt-0">
   465→            {dayLabels.map((label, i) => (
   466→              <div key={i} className="h-[14px] flex items-center">
   467→                <span className="text-[9px] text-muted-foreground w-7 text-right pr-1.5">
   468→                  {label}
   469→                </span>
   470→              </div>
   471→            ))}
   472→          </div>
   473→
   474→          {/* Heatmap grid */}
   475→          <div className="flex gap-[3px] flex-1 overflow-x-auto">
   476→            {heatmapData.map((week, weekIdx) => (
   477→              <div key={weekIdx} className="flex flex-col gap-[3px]">
   478→                {week.map((level, dayIdx) => (
   479→                  <Tooltip key={dayIdx}>
   480→                    <TooltipTrigger asChild>
   481→                      <div
   482→                        className={cn(
   483→                          "h-[14px] w-[14px] rounded-[3px] transition-colors hover:ring-1 hover:ring-foreground/20",
   484→                          getHeatColor(level)
   485→                        )}
   486→                      />
   487→                    </TooltipTrigger>
   488→                    <TooltipContent side="top" className="text-xs">
   489→                      {level > 0
   490→                        ? `${level} activit${level === 1 ? "y" : "ies"}`
   491→                        : "No activity"}
   492→                    </TooltipContent>
   493→                  </Tooltip>
   494→                ))}
   495→              </div>
   496→            ))}
   497→          </div>
   498→        </div>
   499→
   500→        {/* Legend */}
   501→        <div className="flex items-center justify-end gap-1.5 text-[10px] text-muted-foreground">
   502→          <span>Less</span>
   503→          {[0, 1, 2, 3, 4].map((level) => (
   504→            <div
   505→              key={level}
   506→              className={cn("h-[10px] w-[10px] rounded-[2px]", getHeatColor(level))}
   507→            />
   508→          ))}
   509→          <span>More</span>
   510→        </div>
   511→      </div>
   512→    </TooltipProvider>
   513→  );
   514→}
