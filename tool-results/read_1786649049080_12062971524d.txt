    57→    57→
    58→    58→function CircularProgress({ percentage }: { percentage: number }) {
    59→    59→  const radius = 70;
    60→    60→  const stroke = 10;
    61→    61→  const normalizedRadius = radius - stroke / 2;
    62→    62→  const circumference = normalizedRadius * 2 * Math.PI;
    63→    63→  const strokeDashoffset =
    64→    64→    circumference - (percentage / 100) * circumference;
    65→    65→
    66→    66→  return (
    67→    67→    <div className="relative inline-flex items-center justify-center">
    68→    68→      <svg
    69→    69→        width={radius * 2}
    70→    70→        height={radius * 2}
    71→    71→        className="-rotate-90"
    72→    72→      >
    73→    73→        {/* Background track */}
    74→    74→        <circle
    75→    75→          cx={radius}
    76→    76→          cy={radius}
    77→    77→          r={normalizedRadius}
    78→    78→          fill="none"
    79→    79→          stroke="currentColor"
    80→    80→          strokeWidth={stroke}
    81→    81→          className="text-muted/30"
    82→    82→        />
    83→    83→        {/* Progress arc with gradient stroke */}
    84→    84→        <circle
    85→    85→          cx={radius}
    86→    86→          cy={radius}
    87→    87→          r={normalizedRadius}
    88→    88→          fill="none"
    89→    89→          stroke="url(#progress-gradient)"
    90→    90→          strokeWidth={stroke}
    91→    91→          strokeLinecap="round"
    92→    92→          strokeDasharray={circumference}
    93→    93→          strokeDashoffset={strokeDashoffset}
    94→    94→          className="transition-all duration-700 ease-out"
    95→    95→        />
    96→    96→        <defs>
    97→    97→          <linearGradient
    98→    98→            id="progress-gradient"
    99→    99→            x1="0%"
   100→   100→            y1="0%"
   101→   101→            x2="100%"
   102→   102→            y2="100%"
   103→   103→          >
   104→   104→            <stop offset="0%" stopColor="#06b6d4" />
   105→   105→            <stop offset="50%" stopColor="#0891b2" />
   106→   106→            <stop offset="100%" stopColor="#0d9488" />
   107→   107→          </linearGradient>
   108→   108→        </defs>
   109→   109→      </svg>
   110→   110→      <div className="absolute flex flex-col items-center justify-center">
   111→   111→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   112→   112→          {percentage}%
   113→   113→        </span>
   114→   114→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   115→   115→          completed
   116→   116→        </span>
   117→   117→      </div>
   118→   118→    </div>
   119→   119→  );
   120→   120→}
