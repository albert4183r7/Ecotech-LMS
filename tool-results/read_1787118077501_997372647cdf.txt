    62→    62→function timeAgo(dateStr: string): string {
    63→    63→  const now = new Date();
    64→    64→  const date = new Date(dateStr);
    65→    65→  const diffMs = now.getTime() - date.getTime();
    66→    66→  const diffSec = Math.floor(diffMs / 1000);
    67→    67→  const diffMin = Math.floor(diffSec / 60);
    68→    68→  const diffHr = Math.floor(diffMin / 60);
    69→    69→  const diffDay = Math.floor(diffHr / 24);
    70→    70→  const diffWeek = Math.floor(diffDay / 7);
    71→    71→  const diffMonth = Math.floor(diffDay / 30);
    72→    72→
    73→    73→  if (diffSec < 60) return "just now";
    74→    74→  if (diffMin < 60) return `${diffMin}m ago`;
    75→    75→  if (diffHr < 24) return `${diffHr}h ago`;
    76→    76→  if (diffDay < 7) return `${diffDay}d ago`;
    77→    77→  if (diffWeek < 5) return `${diffWeek}w ago`;
    78→    78→  return `${diffMonth}mo ago`;
    79→    79→}

    82→    82→/*  Seeded random for consistent sparkline data                        */
    83→    83→/* ------------------------------------------------------------------ */
    84→    84→
    85→    85→function seededRandom(seed: number) {
    86→    86→  let s = seed;
    87→    87→  return () => {
    88→    88→    s = (s * 16807 + 0) % 2147483647;
    89→    89→    return (s - 1) / 2147483646;
    90→    90→  };
    91→    91→}

    96→    96→
    97→    97→function SparklineChart({
    98→    98→  data,
    99→    99→  gradientFrom,
   100→   100→  gradientTo,
   101→   101→}: {
   102→   102→  data: number[];
   103→   103→  gradientFrom: string;
   104→   104→  gradientTo: string;
   105→   105→}) {
   106→   106→  const max = Math.max(...data, 1);
   107→   107→  return (
   108→   108→    <div className="flex items-end gap-[3px] h-8">
   109→   109→      {data.map((val, i) => {
   110→   110→        const height = Math.max(4, (val / max) * 100);
   111→   111→        return (
   112→   112→          <div
   113→   113→            key={i}
   114→   114→            className={cn(
   115→   115→              "w-[6px] rounded-full transition-all duration-500",
   116→   116→              `bg-gradient-to-t ${gradientFrom} ${gradientTo}`
   117→   117→            )}
   118→   118→            style={{
   119→   119→              height: `${height}%`,
   120→   120→              opacity: 0.5 + (i / data.length) * 0.5,
   121→   121→              animationDelay: `${i * 60}ms`,
   122→   122→            }}
   123→   123→          />
   124→   124→        );
   125→   125→      })}
   126→   126→    </div>
   127→   127→  );
   128→   128→}
