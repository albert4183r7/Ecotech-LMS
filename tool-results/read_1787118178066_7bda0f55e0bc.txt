   512→
   513→  // ============================================
   514→  // GENERATE SLIDES — Streaming HTML for all slides
   515→  // ============================================
   516→
   517→  const handleGenerateSlides = useCallback(
   518→    async (lessonId: string) => {
   519→      const lesson = outlineLessons.find((l) => l.id === lessonId);
   520→      if (!lesson) return;
   521→      if (lesson.slides.length === 0) {
   522→        toast.error("No slides to generate");
   523→        return;
   524→      }
   525→
   526→      // Initialize per-slide gen states
   527→      const initialStates: Record<string, SlideGenState> = {};
   528→      lesson.slides.forEach((s) => {
   529→        if (s.slideId) {
   530→          initialStates[s.slideId] = { status: "pending" };
   531→        }
   532→      });
   533→      setSlideGenStates(initialStates);
   534→      setGeneratingLessonId(lessonId);
   535→      setCurrentGenSlideId(null);
   536→      setStreamingHtml("");
   537→      setGenProgress({ current: 0, total: lesson.slides.length });
   538→      setExpandedOutlineLessonId(lessonId);
   539→
   540→      // Close modal if open
   541→      setModalOpen(false);
   542→
   543→      const abort = new AbortController();
   544→      abortGenRef.current = abort;
   545→
   546→      try {
   547→        const res = await fetch("/api/lessons/generate-slides", {
   548→          method: "POST",
   549→          headers: { "Content-Type": "application/json" },
   550→          body: JSON.stringify({
   551→            lessonId,
   552→            language: lesson.language,
   553→          }),
   554→          signal: abort.signal,
   555→        });
   556→
   557→        if (!res.ok || !res.body) {
   558→          toast.error("Failed to start slide generation");
   559→          setGeneratingLessonId(null);
   560→          return;
   561→        }
   562→
   563→        const reader = res.body.getReader();
   564→        const decoder = new TextDecoder();
   565→        let buffer = "";
   566→        let slidesCompleted = 0;
   567→        let rawHtmlAccum = "";
   568→
   569→        while (true) {
   570→          const { done, value } = await reader.read();
   571→          if (done) break;
   572→          buffer += decoder.decode(value, { stream: true });
   573→
   574→          const lines = buffer.split("\n");
   575→          buffer = lines.pop() || "";
   576→
   577→          for (const line of lines) {
   578→            if (!line.startsWith("data: ")) continue;
   579→            try {
   580→              const data = JSON.parse(line.slice(6));
   581→
   582→              // Top-level error event (no slideId) — e.g. "Lesson not found"
   583→              if (data.error && !data.slideId) {
   584→                toast.error(data.error);
   585→                break; // Stop processing
   586→              }
   587→
   588→              if (data.slideId && data.slideTitle && data.totalSlides !== undefined) {
   589→                // slide_start event
   590→                setCurrentGenSlideId(data.slideId);
   591→                setSlideGenStates((prev) => ({
   592→                  ...prev,
   593→                  [data.slideId]: { status: "generating" },
   594→                }));
   595→                rawHtmlAccum = "";
   596→                setGenProgress({ current: data.slideIndex + 1, total: data.totalSlides });
   597→              }
   598→
   599→              if (data.slideId && data.html) {
   600→                // chunk event
   601→                rawHtmlAccum += data.html;
   602→                setStreamingHtml(buildLivePreviewDoc(rawHtmlAccum, data.slideId));
   603→              }
   604→
   605→              if (data.slideId && data.htmlBody) {
   606→                // slide_complete event
   607→                slidesCompleted++;

   615→                  )
   616→                );
   617→              }
   618→
   619→              if (data.slideId && data.error) {
   620→                // slide_error event
   621→                slidesCompleted++;
   622→                setSlideGenStates((prev) => ({
   623→                  ...prev,
   624→                  [data.slideId]: { status: "error", error: data.error },
   625→                }));
   626→                toast.error(`Slide generation failed: ${data.error}`);

   630→                // all_complete event
   631→                const totalForLesson = outlineLessons.find((ol) => ol.id === data.lessonId)?.slides.length || 0;

   633→                  toast.success(`All ${data.slidesGenerated} slides generated!`);
   634→                } else {
   635→                  toast.warning(`${data.slidesGenerated} of ${totalForLesson} slides generated. Some slides failed.`);

   638→            } catch {
   639→              // skip malformed JSON
   640→            }
   641→          }
   642→        }

   654→        }
   655→        setGeneratingLessonId(null);
   656→        setCurrentGenSlideId(null);
   657→        abortGenRef.current = null;
   658→      }
   659→    },
   660→    [outlineLessons]
