   228→      const res = await fetch("/api/generate-slide-html", {
   229→        method: "POST",
   230→        headers: { "Content-Type": "application/json" },
   231→        body: JSON.stringify({
   232→          slideTitle: lessonName.trim(),

   242→        setShowStreamPreview(false);
   243→        return;
   244→      }

   247→      const reader = res.body.getReader();

   253→
   254→      while (true) {
   255→        const { done, value } = await reader.read();

   257→        buffer += decoder.decode(value, { stream: true });

   259→        const lines = buffer.split("\n");
   260→        buffer = lines.pop() || "";

   262→        for (const line of lines) {
   263→          if (!line.startsWith("data: ")) continue;
   264→          try {
   265→            const data = JSON.parse(line.slice(6));
   266→            if (data.htmlBody) {
   267→              htmlBody = data.htmlBody;
   268→            }
   269→            if (data.slideTitle) {
   270→              slideTitle = data.slideTitle;
   271→            }
   272→            if (data.html) {
   273→              rawHtml += data.html;
   274→              // Build live preview from accumulated raw HTML
   275→              const previewDoc = buildLivePreviewDoc(rawHtml, slideTitle);
   276→              setStreamingHtml(previewDoc);
   277→              // Auto-scroll preview into view

   279→            }
   280→          } catch {
   281→            // skip malformed JSON
   282→          }
   283→        }
   284→      }
   285→
   286→      if (!htmlBody) {
   287→        toast.error("Generation returned empty content");
   288→        setGenerating(false);
   289→        setShowStreamPreview(false);
   290→        return;
   291→      }
   292→
   293→      const newLesson: LessonDraft = {
   294→        id: `sec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   295→        title: slideTitle,
   296→        totalPages: 1,

   298→        language: lessonLanguage,
   299→      };

   303→      setShowStreamPreview(false);

   306→    } catch (err) {

   308→      toast.error("Failed to generate lesson. Please try again.");
   309→    } finally {

   560→          ol.id === editingOutlineLesson

   604→      newLessons[index],

   607→  };
   608→
   609→  // ---- Form submission ----
   610→  const handleSave = async () => {
   611→    if (!title.trim()) {
   612→      toast.error("Course title is required");
   613→      return;
   614→    }
