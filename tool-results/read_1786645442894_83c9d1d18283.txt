   182→/* ------------------------------------------------------------------ */
   183→/*  Circular Mini Progress (for course rows)                           */
   184→/* ------------------------------------------------------------------ */
   185→
   186→function CircularMiniProgress({ percentage }: { percentage: number }) {
   187→  const size = 40;
   188→  const stroke = 3.5;
   189→  const r = (size - stroke) / 2;
   190→  const circ = r * 2 * Math.PI;
   191→  const offset = circ - (percentage / 100) * circ;
   192→
   193→  return (
   194→    <div className="relative inline-flex shrink-0 items-center justify-center">
   195→      <svg width={size} height={size} className="-rotate-90">
   196→        <circle
   197→          cx={size / 2}
   198→          cy={size / 2}
   199→          r={r}
   200→          fill="none"
   201→          stroke="currentColor"
   202→          strokeWidth={stroke}
   203→          className="text-muted/30"
   204→        />
   205→        <circle
   206→          cx={size / 2}
   207→          cy={size / 2}
   208→          r={r}
   209→          fill="none"
   210→          stroke="url(#mini-grad)"
   211→          strokeWidth={stroke}
   212→          strokeLinecap="round"
   213→          strokeDasharray={circ}
   214→          strokeDashoffset={offset}
   215→          className="transition-all duration-500 ease-out"
   216→        />
   217→        <defs>
   218→          <linearGradient id="mini-grad" x1="0%" y1="0%" x2="100%" y2="0%">
   219→            <stop offset="0%" stopColor="#0891b2" />
   220→            <stop offset="100%" stopColor="#14b8a6" />
   221→          </linearGradient>
   222→        </defs>
   223→      </svg>
   224→      <span className="absolute text-[9px] font-bold text-foreground">
   225→        {percentage}%
   226→      </span>
   227→    </div>
   228→  );
   229→}

   605→export function MyLearningPage() {
   606→  const { tab, setTab, setEnrollments, setFavorites } = useMyLearningStore();
   607→  const { currentUserId } = useUserStore();
   608→
   609→  const [stats, setStats] = useState<UserStats | null>(null);
   610→  const [enrollments, setEnrollmentsState] = useState<EnrollmentItem[]>([]);
   611→  const [favorites, setFavoritesState] = useState<FavoriteItem[]>([]);
   612→  const [loadingStats, setLoadingStats] = useState(true);
   613→  const [loadingEnrollments, setLoadingEnrollments] = useState(true);
   614→  const [loadingFavorites, setLoadingFavorites] = useState(true);
   615→
   616→  const fetchStats = useCallback(async () => {
   617→    setLoadingStats(true);
   618→    try {
   619→      const res = await fetch(`/api/user/${currentUserId}`);
   620→      const json = await res.json();
   621→      if (json.success) {
   622→        setStats(json.data.stats);
   623→      }
   624→    } catch {
   625→      /* silent */
   626→    } finally {
   627→      setLoadingStats(false);
   628→    }
   629→  }, [currentUserId]);
   630→
   631→  const fetchEnrollments = useCallback(async () => {
   632→    setLoadingEnrollments(true);
   633→    try {
   634→      const res = await fetch(
   635→        `/api/enrollments?userId=${currentUserId}`
   636→      );
   637→      const json = await res.json();
   638→      if (json.success) {
   639→        setEnrollmentsState(json.data);
   640→        setEnrollments(
   641→          json.data.map((e: EnrollmentItem) => enrollmentToCourseItem(e))
   642→        );
   643→      }
   644→    } catch {
   645→      /* silent */
   646→    } finally {
   647→      setLoadingEnrollments(false);
   648→    }
   649→  }, [currentUserId, setEnrollments]);
   650→
   651→  const fetchFavorites = useCallback(async () => {
   652→    setLoadingFavorites(true);
   653→    try {
   654→      const res = await fetch(
   655→        `/api/favorites?userId=${currentUserId}`
   656→      );
   657→      const json = await res.json();
   658→      if (json.success) {
   659→        setFavoritesState(json.data);
   660→        setFavorites(
   661→          json.data.map((f: FavoriteItem) => favoriteToCourseItem(f))
   662→        );
   663→      }
   664→    } catch {
   665→      /* silent */
   666→    } finally {
   667→      setLoadingFavorites(false);
   668→    }
   669→  }, [currentUserId, setFavorites]);
   670→
   671→  useEffect(() => {
   672→    fetchStats();
   673→    fetchEnrollments();
   674→    fetchFavorites();
   675→  }, [fetchStats, fetchEnrollments, fetchFavorites]);
   676→
   677→  /* Derived lists */
   678→  const inProgressList = enrollments.filter(
   679→    (e) => e.status === "in_progress"
   680→  );
   681→  const completedList = enrollments.filter(
   682→    (e) => e.status === "completed"
   683→  );
   684→
   685→  /* Tab change handler */
   686→  const handleTabChange = (value: string) => {
   687→    setTab(value as MyLearningTab);
   688→  };
   689→
   690→  return (
   691→    <div className="space-y-6">
   692→      {/* Page Header */}
   693→      <div>
   694→        <h1 className="text-2xl font-bold text-foreground">My Learning</h1>
   695→        <p className="mt-1 text-sm text-muted-foreground">
   696→          Track your learning progress and manage your courses
   697→        </p>
   698→      </div>
   699→
   700→      {/* Stats Row */}
   701→      {loadingStats ? (
   702→        <StatsSkeleton />
   703→      ) : stats ? (
   704→        <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   705→          <StatCard
   706→            icon={BookOpen}
   707→            label="Total Courses"
   708→            value={stats.totalCourses}
   709→            gradient="bg-gradient-to-br from-blue-500 to-blue-700"
   710→            trend="up"
   711→            loading={false}
   712→          />
   713→          <StatCard
   714→            icon={Clock}
   715→            label="In Progress"
   716→            value={stats.inProgress}
   717→            gradient="bg-gradient-to-br from-teal-500 to-teal-700"
   718→            loading={false}
   719→          />
   720→          <StatCard
   721→            icon={CheckCircle}
   722→            label="Completed"
   723→            value={stats.completed}
   724→            gradient="bg-gradient-to-br from-emerald-500 to-emerald-700"
   725→            trend="up"
   726→            loading={false}
   727→          />
   728→          <StatCard
   729→            icon={TrendingUp}
   730→            label="Avg Progress"
   731→            value={`${stats.avgProgress}%`}
   732→            gradient="bg-gradient-to-br from-amber-500 to-amber-700"
   733→            trend="up"
   734→            loading={false}
   735→          />
   736→        </div>
   737→      ) : null}
   738→
   739→      {/* Continue Learning Widget */}
   740→      <ContinueLearningWidget
   741→        enrollment={
   742→          loadingEnrollments
   743→            ? undefined
   744→            : inProgressList.length > 0
   745→              ? inProgressList[0]
   746→              : null
   747→        }
   748→      />
   749→
   750→      {/* Tabs */}
   751→      <Tabs
   752→        value={tab}
   753→        onValueChange={handleTabChange}
   754→        className="w-full"
   755→      >
   756→        {/* Underline-style tab list with animated indicator */}
   757→        <TabsList className="relative h-auto w-full justify-start gap-6 rounded-none border-b border-border bg-transparent p-0">
   758→          <TabsTrigger
   759→            value="in-progress"
   760→            className="rounded-none border-b-2 border-transparent px-1 pb-3 pt-1 text-sm font-medium text-muted-foreground transition-all duration-300 data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:font-semibold data-[state=active]:shadow-none data-[state=active]:text-foreground"
   761→          >
   762→            <span className="flex items-center gap-1.5">
   763→              In Progress
   764→              {!loadingEnrollments && inProgressList.length > 0 && (
   765→                <Badge className="ml-0.5 h-5 min-w-5 rounded-full bg-primary/10 px-1.5 text-[10px] font-bold text-primary">
   766→                  {inProgressList.length}
   767→                </Badge>
   768→              )}
   769→            </span>
   770→          </TabsTrigger>
   771→          <TabsTrigger
   772→            value="completed"
   773→            className="rounded-none border-b-2 border-transparent px-1 pb-3 pt-1 text-sm font-medium text-muted-foreground transition-all duration-300 data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:font-semibold data-[state=active]:shadow-none data-[state=active]:text-foreground"
   774→          >
   775→            <span className="flex items-center gap-1.5">
   776→              Completed
   777→              {!loadingEnrollments && completedList.length > 0 && (
   778→                <Badge className="ml-0.5 h-5 min-w-5 rounded-full bg-primary/10 px-1.5 text-[10px] font-bold text-primary">
   779→                  {completedList.length}
   780→                </Badge>
   781→              )}
   782→            </span>
   783→          </TabsTrigger>
   784→          <TabsTrigger
   785→            value="favorites"
   786→            className="rounded-none border-b-2 border-transparent px-1 pb-3 pt-1 text-sm font-medium text-muted-foreground transition-all duration-300 data-[state=active]:border-primary data-[state=active]:bg-transparent data-[state=active]:font-semibold data-[state=active]:shadow-none data-[state=active]:text-foreground"
   787→          >
   788→            <span className="flex items-center gap-1.5">
   789→              Favorites
   790→              {!loadingFavorites && favorites.length > 0 && (
   791→                <Badge className="ml-0.5 h-5 min-w-5 rounded-full bg-primary/10 px-1.5 text-[10px] font-bold text-primary">
   792→                  {favorites.length}
   793→                </Badge>
   794→              )}
   795→            </span>
   796→          </TabsTrigger>
   797→        </TabsList>
   798→
   799→        {/* In Progress Tab */}
   800→        <TabsContent value="in-progress" className="mt-6">
   801→          {loadingEnrollments ? (
   802→            <ListSkeleton />
   803→          ) : inProgressList.length === 0 ? (
   804→            <EmptyState
   805→              icon={BookOpen}

   807→              description="Browse our course catalog and enroll in courses that interest you. Your active courses will appear here with progress tracking."
   808→            />
   809→          ) : (
   810→            <div className="space-y-3">
   811→              {inProgressList.map((enrollment) => (
   812→                <InProgressRow
   813→                  key={enrollment.id}
