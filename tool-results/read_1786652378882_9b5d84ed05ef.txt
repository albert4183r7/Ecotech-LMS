   121→/** Group notifications by date category: Today, Yesterday, Earlier */
   122→function groupNotificationsByDate(items: NotificationItem[]) {
   123→  const now = new Date();
   124→  const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   125→  const yesterdayStart = todayStart - 86400000;
   126→
   127→  const today: NotificationItem[] = [];
   128→  const yesterday: NotificationItem[] = [];
   129→  const earlier: NotificationItem[] = [];
   130→
   131→  for (const item of items) {
   132→    const t = new Date(item.createdAt).getTime();
   133→    if (t >= todayStart) today.push(item);
   134→    else if (t >= yesterdayStart) yesterday.push(item);
   135→    else earlier.push(item);
   136→  }
   137→
   138→  return { today, yesterday, earlier };
   139→}

   155→/** Format relative time for notifications */
   156→function formatRelativeTime(dateStr: string): string {
   157→  const now = new Date();
   158→  const date = new Date(dateStr);
   159→  const diffMs = now.getTime() - date.getTime();
   160→  const diffMins = Math.floor(diffMs / 60000);
   161→  const diffHours = Math.floor(diffMs / 3600000);
   162→  const diffDays = Math.floor(diffMs / 86400000);
   163→
   164→  if (diffMins < 1) return "Just now";
   165→  if (diffMins < 60) return `${diffMins}m ago`;
   166→  if (diffHours < 24) return `${diffHours}h ago`;
   167→  if (diffDays < 7) return `${diffDays}d ago`;
   168→  return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   169→}
