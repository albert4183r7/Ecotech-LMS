    95→
    96→export function CreateCoursePage() {
    97→  const { goBack } = useNavigationStore();
    98→  const { currentUserId } = useUserStore();

   100→
   101→  // ---- Course form state ----
   102→  const [title, setTitle] = useState("");
   103→  const [description, setDescription] = useState("");
   104→  const [categoryId, setCategoryId] = useState("");
   105→  const [language, setLanguage] = useState("english");
   106→  const [coverImage, setCoverImage] = useState("");
   107→  const [coverPreview, setCoverPreview] = useState<string | null>(null);
   108→  const [courseId, setCourseId] = useState<string | null>(null);
   109→
   110→  // ---- UI state ----
   111→  const [categories, setCategories] = useState<CategoryItem[]>([]);
   112→  const [loading, setLoading] = useState(false);
   113→  const [saving, setSaving] = useState(false);
   114→  const [modalOpen, setModalOpen] = useState(false);
   115→  const [urlInputOpen, setUrlInputOpen] = useState(false);
   116→  const [imageUrl, setImageUrl] = useState("");
   117→
   118→  // ---- Outline modal state ----
   119→  const [outlineTopic, setOutlineTopic] = useState("");
   120→  const [outlineSlideCount, setOutlineSlideCount] = useState(DEFAULT_SLIDE_COUNT);
   121→  const [outlineStyle, setOutlineStyle] = useState("professional");
   122→  const [outlineGenerating, setOutlineGenerating] = useState(false);
   123→  const [editingOutlineLesson, setEditingOutlineLesson] = useState<string | null>(null);
   124→  const [outlineEditingSlides, setOutlineEditingSlides] = useState<OutlineSlideDraft[]>([]);
   125→
   126→  // ---- Lessons list ----
   127→  const [outlineLessons, setOutlineLessons] = useState<OutlineLessonDraft[]>([]);
   128→  const [expandedOutlineLessonId, setExpandedOutlineLessonId] = useState<string | null>(null);
   129→
   130→  // ---- Slide generation state ----
   131→  const [generatingLessonId, setGeneratingLessonId] = useState<string | null>(null);
   132→  const [slideGenStates, setSlideGenStates] = useState<Record<string, SlideGenState>>({});
   133→  const [currentGenSlideId, setCurrentGenSlideId] = useState<string | null>(null);
   134→  const [streamingHtml, setStreamingHtml] = useState("");
   135→  const [genProgress, setGenProgress] = useState({ current: 0, total: 0 });
   136→  const abortGenRef = useRef<AbortController | null>(null);
   137→
   138→  // ---- Refs ----
   139→  const fileInputRef = useRef<HTMLInputElement>(null);
   140→
   141→  // ---- Pre-fill title from hero prompt ----
   142→  useEffect(() => {
   143→    if (createPrompt) {
   144→      setTitle(createPrompt);
   145→      setCreatePrompt("");
   146→    }
   147→  }, [createPrompt, setCreatePrompt]);
   148→
   149→  // ---- Fetch categories ----
   150→  useEffect(() => {
   151→    async function fetchCategories() {
   152→      setLoading(true);
   153→      try {
   154→        const res = await fetch("/api/categories");
   155→        const json = await res.json();
   156→        if (json.success) {
   157→          setCategories(json.data);
   158→        }
   159→      } catch {
   160→        toast.error("Failed to load categories");
   161→      } finally {
   162→        setLoading(false);
   163→      }
   164→    }
   165→    fetchCategories();
   166→  }, []);
   167→
   168→  // ---- Cover image handlers ----
   169→  const handleCoverUpload = () => {
   170→    setUrlInputOpen(true);
   171→  };
   172→
   173→  const handleUrlConfirm = () => {
   174→    if (imageUrl.trim()) {
   175→      setCoverImage(imageUrl.trim());
   176→      setCoverPreview(imageUrl.trim());
   177→      setUrlInputOpen(false);
   178→      setImageUrl("");
   179→    }
   180→  };
   181→
   182→  const handleRemoveCover = () => {
   183→    setCoverImage("");
   184→    setCoverPreview(null);
   185→  };
   186→
   187→  // ---- Auto-save course as draft (needed before outline generation) ----
   188→  const ensureCourseSaved = useCallback(async (): Promise<string | null> => {
   189→    if (courseId) return courseId;
   190→    if (!title.trim() || !currentUserId) {
   191→      toast.error("Please enter a course title first");
   192→      return null;
   193→    }
   194→    try {
   195→      const res = await fetch("/api/courses", {
   196→        method: "POST",
   197→        headers: { "Content-Type": "application/json" },
   198→        body: JSON.stringify({
   199→          title: title.trim(),
   200→          description: description.trim() || null,
   201→          categoryId: categoryId || null,
   202→          language,
   203→          creatorId: currentUserId,
   204→          coverImage: coverImage || null,
   205→        }),
   206→      });
   207→      const json = await res.json();
   208→      if (json.success && json.data?.id) {
   209→        setCourseId(json.data.id);
   210→        return json.data.id;
   211→      }
   212→      toast.error("Failed to save course draft");
   213→      return null;
   214→    } catch {
   215→      toast.error("Failed to save course draft");
   216→      return null;
   217→    }
   218→  }, [courseId, title, description, categoryId, language, currentUserId, coverImage]);
   219→
   220→  // ---- Open modal for new outline ----
   221→  const handleOpenModal = () => {
   222→    setOutlineTopic("");
   223→    setOutlineSlideCount(DEFAULT_SLIDE_COUNT);
   224→    setOutlineStyle("professional");
   225→    setOutlineGenerating(false);
   226→    setEditingOutlineLesson(null);
   227→    setOutlineEditingSlides([]);
   228→    setModalOpen(true);
   229→  };
   230→
   231→  // ---- Generate outline ----
   232→  const handleGenerateOutline = useCallback(async () => {
   233→    if (!outlineTopic.trim()) {
   234→      toast.error("Topic is required");
   235→      return;
   236→    }
   237→    if (outlineLessons.length >= MAX_LESSONS) {
   238→      toast.error(`Maximum ${MAX_LESSONS} lessons allowed`);
   239→      return;
   240→    }
   241→
   242→    const savedCourseId = await ensureCourseSaved();
   243→    if (!savedCourseId) return;
   244→
   245→    setOutlineGenerating(true);
   246→    try {
   247→      const res = await fetch("/api/lessons/generate-outline", {
   248→        method: "POST",
   249→        headers: { "Content-Type": "application/json" },
   250→        body: JSON.stringify({
   251→          courseId: savedCourseId,
   252→          topic: outlineTopic.trim(),
   253→          slideCount: outlineSlideCount,
   254→          style: outlineStyle,
   255→          language,
   256→        }),
   257→      });
   258→      const json = await res.json();
   259→
   260→      if (!json.success) {
   261→        toast.error(json.error || "Failed to generate outline");
   262→        setOutlineGenerating(false);
   263→        return;
   264→      }
   265→
   266→      const lessonData = json.data;
   267→      const parsedOutline = JSON.parse(lessonData.outlineJson || "{}");
   268→      const slides: OutlineSlideDraft[] = lessonData.slides.map(
   269→        (s: { id: string; title: string; order: number }, i: number) => ({
   270→          id: `local_${Date.now()}_${i}`,
   271→          slideId: s.id,
   272→          title: s.title,
   273→          outline: parsedOutline.slides?.[i]?.outline || "",
   274→          order: s.order,
   275→        })
   276→      );
   277→
   278→      const newLesson: OutlineLessonDraft = {
   279→        id: lessonData.id,
   280→        title: lessonData.title,
   281→        slides,
   282→        language,
   283→        style: outlineStyle,
   284→        topic: outlineTopic.trim(),
   285→      };
   286→
   287→      setOutlineLessons((prev) => [...prev, newLesson]);
   288→      setEditingOutlineLesson(lessonData.id);
   289→      setOutlineEditingSlides(slides);
   290→      setOutlineGenerating(false);
   291→      toast.success(`Outline generated: ${slides.length} slides`);
   292→    } catch {
   293→      toast.error("Failed to generate outline. Please try again.");
   294→      setOutlineGenerating(false);
   295→    }
   296→  }, [outlineTopic, outlineSlideCount, outlineStyle, language, ensureCourseSaved, outlineLessons.length]);
   297→
   298→  // ---- Update slide title (inline edit) ----
   299→  const handleUpdateSlideTitle = async (slideId: string, newTitle: string) => {
   300→    setOutlineEditingSlides((prev) =>
   301→      prev.map((s) => (s.slideId === slideId ? { ...s, title: newTitle } : s))
   302→    );
   303→    setOutlineLessons((prev) =>
   304→      prev.map((ol) =>
   305→        ol.id === editingOutlineLesson
   306→          ? {
   307→              ...ol,
   308→              slides: ol.slides.map((s) =>
   309→                s.slideId === slideId ? { ...s, title: newTitle } : s
   310→              ),
   311→            }
   312→          : ol
   313→      )
   314→    );
   315→    try {
   316→      await fetch(`/api/slides/${slideId}`, {
   317→        method: "PUT",
   318→        headers: { "Content-Type": "application/json" },
   319→        body: JSON.stringify({ title: newTitle }),
   320→      });
   321→    } catch {
   322→      // silent
   323→    }
   324→  };
   325→
   326→  // ---- Delete a slide from outline ----
   327→  const handleDeleteOutlineSlide = async (slideId: string, localId: string) => {
   328→    setOutlineEditingSlides((prev) => prev.filter((s) => s.id !== localId));
   329→    setOutlineLessons((prev) =>
   330→      prev.map((ol) =>
   331→        ol.id === editingOutlineLesson
   332→          ? { ...ol, slides: ol.slides.filter((s) => s.id !== localId) }
   333→          : ol
   334→      )
   335→    );
   336→    if (slideId) {
   337→      try {
   338→        await fetch(`/api/slides/${slideId}`, { method: "DELETE" });
   339→      } catch {
   340→        // silent
   341→      }
   342→    }
   343→  };
   344→
   345→  // ---- Add a blank slide to outline ----
   346→  const handleAddOutlineSlide = () => {
   347→    const newOrder = outlineEditingSlides.length;
   348→    const localId = `local_${Date.now()}_new`;
   349→    const newSlide: OutlineSlideDraft = {
   350→      id: localId,
   351→      slideId: null,
   352→      title: `Slide ${newOrder + 1}`,
   353→      outline: "",
   354→      order: newOrder,
   355→    };
   356→    setOutlineEditingSlides((prev) => [...prev, newSlide]);
   357→    setOutlineLessons((prev) =>
   358→      prev.map((ol) =>
   359→        ol.id === editingOutlineLesson
   360→          ? { ...ol, slides: [...ol.slides, newSlide] }
   361→          : ol
   362→      )
   363→    );
   364→  };
   365→
   366→  // ---- Regenerate outline ----
   367→  const handleRegenerateOutline = async () => {
   368→    if (!editingOutlineLesson) return;
   369→    setOutlineGenerating(true);
   370→    try {
   371→      const res = await fetch("/api/lessons/generate-outline", {
   372→        method: "POST",
   373→        headers: { "Content-Type": "application/json" },
   374→        body: JSON.stringify({
   375→          courseId,
   376→          topic: outlineTopic.trim(),
   377→          slideCount: outlineSlideCount,
   378→          style: outlineStyle,
   379→          language,
   380→          existingLessonId: editingOutlineLesson,
   381→        }),
   382→      });
   383→      const json = await res.json();
   384→      if (!json.success) {
   385→        toast.error(json.error || "Failed to regenerate");
   386→        setOutlineGenerating(false);
   387→        return;
   388→      }
   389→      const lessonData = json.data;
   390→      const parsedOutline = JSON.parse(lessonData.outlineJson || "{}");
   391→      const slides: OutlineSlideDraft[] = lessonData.slides.map(
   392→        (s: { id: string; title: string; order: number }, i: number) => ({
   393→          id: `local_${Date.now()}_${i}`,
   394→          slideId: s.id,
   395→          title: s.title,
   396→          outline: parsedOutline.slides?.[i]?.outline || "",
   397→          order: s.order,
   398→        })
   399→      );
   400→      setOutlineEditingSlides(slides);
   401→      setOutlineLessons((prev) =>
   402→        prev.map((ol) =>
   403→          ol.id === editingOutlineLesson
   404→            ? { ...ol, title: lessonData.title, slides }
   405→            : ol
   406→        )
   407→      );
   408→      setOutlineGenerating(false);
   409→      toast.success("Outline regenerated");
   410→    } catch {
   411→      toast.error("Failed to regenerate outline");
   412→      setOutlineGenerating(false);
   413→    }
   414→  };
   415→
   416→  // ============================================
   417→  // GENERATE SLIDES — Streaming HTML for all slides
   418→  // ============================================
   419→
   420→  const handleGenerateSlides = useCallback(
   421→    async (lessonId: string) => {
   422→      const lesson = outlineLessons.find((l) => l.id === lessonId);
   423→      if (!lesson) return;
   424→      if (lesson.slides.length === 0) {
   425→        toast.error("No slides to generate");
   426→        return;
   427→      }
   428→
   429→      // Initialize per-slide gen states
   430→      const initialStates: Record<string, SlideGenState> = {};
   431→      lesson.slides.forEach((s) => {
   432→        if (s.slideId) {
   433→          initialStates[s.slideId] = { status: "pending" };
   434→        }
   435→      });
   436→      setSlideGenStates(initialStates);
   437→      setGeneratingLessonId(lessonId);
   438→      setCurrentGenSlideId(null);
   439→      setStreamingHtml("");
   440→      setGenProgress({ current: 0, total: lesson.slides.length });
   441→      setExpandedOutlineLessonId(lessonId);
   442→
   443→      // Close modal if open
   444→      setModalOpen(false);
   445→
   446→      const abort = new AbortController();
   447→      abortGenRef.current = abort;
   448→
   449→      try {
   450→        const res = await fetch("/api/lessons/generate-slides", {
   451→          method: "POST",
   452→          headers: { "Content-Type": "application/json" },
   453→          body: JSON.stringify({
   454→            lessonId,
   455→            language: lesson.language,
   456→          }),
   457→          signal: abort.signal,
   458→        });
   459→
   460→        if (!res.ok || !res.body) {
   461→          toast.error("Failed to start slide generation");
   462→          setGeneratingLessonId(null);
   463→          return;
   464→        }
   465→
   466→        const reader = res.body.getReader();
   467→        const decoder = new TextDecoder();
   468→        let buffer = "";
   469→        let slidesCompleted = 0;
   470→        let rawHtmlAccum = "";
   471→
   472→        while (true) {
   473→          const { done, value } = await reader.read();
   474→          if (done) break;
   475→          buffer += decoder.decode(value, { stream: true });
   476→
   477→          const lines = buffer.split("\n");
   478→          buffer = lines.pop() || "";
   479→
   480→          for (const line of lines) {
   481→            if (!line.startsWith("data: ")) continue;
   482→            try {
   483→              const data = JSON.parse(line.slice(6));
   484→
   485→              if (data.slideId && data.slideTitle && data.totalSlides !== undefined) {
   486→                // slide_start event
   487→                setCurrentGenSlideId(data.slideId);
   488→                setSlideGenStates((prev) => ({
   489→                  ...prev,
   490→                  [data.slideId]: { status: "generating" },
   491→                }));
   492→                rawHtmlAccum = "";
   493→                setGenProgress({ current: data.slideIndex + 1, total: data.totalSlides });
   494→              }
   495→
   496→              if (data.slideId && data.html) {
   497→                // chunk event
   498→                rawHtmlAccum += data.html;
   499→                setStreamingHtml(buildLivePreviewDoc(rawHtmlAccum, data.slideId));
   500→              }
   501→
   502→              if (data.slideId && data.htmlBody) {
   503→                // slide_complete event
   504→                slidesCompleted++;
   505→                setSlideGenStates((prev) => ({
   506→                  ...prev,
   507→                  [data.slideId]: { status: "complete", htmlBody: data.htmlBody },
   508→                }));
   509→                setOutlineLessons((prev) =>
   510→                  prev.map((ol) =>
   511→                    ol.id === lessonId ? { ...ol, allReady: slidesCompleted === ol.slides.length } : ol
   512→                  )
   513→                );
   514→              }
   515→
   516→              if (data.slideId && data.error) {
   517→                // slide_error event
   518→                slidesCompleted++;
   519→                setSlideGenStates((prev) => ({
   520→                  ...prev,
   521→                  [data.slideId]: { status: "error", error: data.error },
   522→                }));
   523→              }
   524→
   525→              if (data.lessonId && data.slidesGenerated !== undefined) {
   526→                // all_complete event
   527→                toast.success(`All ${data.slidesGenerated} slides generated!`);
   528→              }
   529→            } catch {
   530→              // skip malformed JSON
   531→            }
   532→          }
   533→        }
   534→      } catch (err) {
   535→        if (err instanceof Error && err.name === "AbortError") {
   536→          toast.info("Generation cancelled");
   537→        } else {
   538→          toast.error("Failed to generate slides. Please try again.");
   539→        }
   540→      } finally {
   541→        setGeneratingLessonId(null);
   542→        setCurrentGenSlideId(null);
   543→        abortGenRef.current = null;
   544→      }
   545→    },
   546→    [outlineLessons]
   547→  );
   548→
   549→  const handleCancelGeneration = () => {
   550→    abortGenRef.current?.abort();
   551→  };
   552→
   553→  // ---- Delete lesson ----
   554→  const handleDeleteOutlineLesson = async (lessonId: string) => {
   555→    setOutlineLessons((prev) => prev.filter((ol) => ol.id !== lessonId));
   556→    if (expandedOutlineLessonId === lessonId) {
   557→      setExpandedOutlineLessonId(null);
   558→    }
   559→    try {
   560→      const res = await fetch(`/api/lessons/${lessonId}`, { method: "DELETE" });
   561→      const json = await res.json();
   562→      if (!json.success) toast.error("Failed to remove lesson");
   563→    } catch {
   564→      toast.error("Failed to remove lesson");
   565→    }
   566→  };
   567→
   568→  // ---- Form submission ----
   569→  const handleSave = async () => {
   570→    if (!title.trim()) {
   571→      toast.error("Course title is required");
   572→      return;
   573→    }
   574→
   575→    // Check if any lesson is still generating
   576→    if (generatingLessonId) {
   577→      toast.error("Please wait for slide generation to finish");
   578→      return;
   579→    }
   580→
   581→    setSaving(true);
   582→    try {
   583→      if (courseId) {
   584→        // Update existing course to published
   585→        const res = await fetch(`/api/courses/${courseId}`, {
   586→          method: "PUT",
   587→          headers: { "Content-Type": "application/json" },
   588→          body: JSON.stringify({
   589→            title: title.trim(),
   590→            description: description.trim() || null,
   591→            categoryId: categoryId || null,
   592→            language,
   593→            coverImage: coverImage || null,
   594→            status: "published",
   595→          }),
   596→        });
   597→        const json = await res.json();
   598→        if (json.success) {
   599→          toast.success("Course published successfully!");
   600→          goBack();
   601→        } else {
   602→          toast.error(json.error || "Failed to publish course");
   603→        }
   604→      } else {
   605→        toast.error("Please add at least one lesson first");
   606→      }
   607→    } catch {
   608→      toast.error("Failed to save course. Please try again.");
   609→    } finally {
   610→      setSaving(false);
   611→    }
   612→  };
   613→
   614→  // ============================================
   615→  // Render
   616→  // ============================================
   617→
   618→  return (
   619→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   620→      {/* ---- Breadcrumbs ---- */}
   621→      <div className="mb-6 flex items-center gap-3">

  1124→function OutlineLessonCard({

  1141→  const styleLabel = SLIDE_STYLES.find((s) => s.value === lesson.style)?.label || lesson.style;
  1142→  const hasReadySlides = lesson.slides.some((s) => s.slideId && slideGenStates[s.slideId]?.status === "complete");
  1143→  const allComplete = lesson.allReady || lesson.slides.every(
  1144→    (s) => !s.slideId || slideGenStates[s.slideId]?.status === "complete"
  1145→  );
  1146→  const completedCount = lesson.slides.filter(
  1147→    (s) => s.slideId && slideGenStates[s.slideId]?.status === "complete"
  1148→  ).length;
  1149→
  1150→  return (
  1151→    <div
  1152→      className={`group rounded-lg border transition-colors ${
  1153→        isGenerating
  1154→          ? "border-primary/50 bg-primary/[0.03]"
  1155→          : allComplete
  1156→            ? "border-emerald-300/60 bg-emerald-50/30 dark:bg-emerald-950/10"
  1157→            : hasReadySlides
  1158→              ? "border-border/60 bg-card hover:bg-accent/30"
  1159→              : "border-primary/30 bg-primary/[0.02] hover:bg-primary/[0.04]"
  1160→      }`}
  1161→    >
  1162→      <div className="flex items-center gap-3 p-3">

  1164→        <div

  1176→        {/* Content */}
  1177→        <div className="min-w-0 flex-1">

  1331→
  1332→          {/* Generate Slides button (only when not generating and not all complete) */}
  1333→          {!isGenerating && !allComplete && (
