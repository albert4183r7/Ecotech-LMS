    71→function CircularProgress({ percentage }: { percentage: number }) {
    72→  const radius = 70;
    73→  const stroke = 10;
    74→  const normalizedRadius = radius - stroke / 2;
    75→  const circumference = normalizedRadius * 2 * Math.PI;
    76→  const strokeDashoffset =
    77→    circumference - (percentage / 100) * circumference;
    78→
    79→  return (
    80→    <div className="relative inline-flex items-center justify-center">
    81→      <svg
    82→        width={radius * 2}
    83→        height={radius * 2}
    84→        className="-rotate-90"
    85→      >
    86→        {/* Background track */}
    87→        <circle
    88→          cx={radius}
    89→          cy={radius}
    90→          r={normalizedRadius}
    91→          fill="none"
    92→          stroke="currentColor"
    93→          strokeWidth={stroke}
    94→          className="text-muted/30"
    95→        />
    96→        {/* Progress arc with gradient stroke */}
    97→        <circle
    98→          cx={radius}
    99→          cy={radius}
   100→          r={normalizedRadius}
   101→          fill="none"
   102→          stroke="url(#progress-gradient)"
   103→          strokeWidth={stroke}
   104→          strokeLinecap="round"
   105→          strokeDasharray={circumference}
   106→          strokeDashoffset={strokeDashoffset}
   107→          className="transition-all duration-700 ease-out"
   108→        />
   109→        <defs>
   110→          <linearGradient
   111→            id="progress-gradient"
   112→            x1="0%"
   113→            y1="0%"
   114→            x2="100%"
   115→            y2="100%"
   116→          >
   117→            <stop offset="0%" stopColor="#06b6d4" />
   118→            <stop offset="50%" stopColor="#0891b2" />
   119→            <stop offset="100%" stopColor="#0d9488" />
   120→          </linearGradient>
   121→        </defs>
   122→      </svg>
   123→      <div className="absolute flex flex-col items-center justify-center">
   124→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   125→          {percentage}%
   126→        </span>
   127→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   128→          completed
   129→        </span>
   130→      </div>
   131→    </div>
   132→  );
   133→}

   135→/* ------------------------------------------------------------------ */
   136→/*  Animated Counter Hook                                              */
   137→/* ------------------------------------------------------------------ */
   138→
   139→function useAnimatedCounter(target: number, duration = 800) {
   140→  const [count, setCount] = useState(target);
   141→  const frameRef = useRef<number | null>(null);
   142→
   143→  useEffect(() => {
   144→    if (target === 0) return;
   145→    let start = 0;
   146→    const startTime = performance.now();
   147→    const step = (now: number) => {
   148→      const elapsed = now - startTime;
   149→      const progress = Math.min(elapsed / duration, 1);
   150→      const eased = 1 - Math.pow(1 - progress, 3);
   151→      start = Math.round(eased * target);
   152→      setCount(start);
   153→      if (progress < 1) {
   154→        frameRef.current = requestAnimationFrame(step);
   155→      }
   156→    };
   157→    frameRef.current = requestAnimationFrame(step);
   158→    return () => {
   159→      if (frameRef.current) cancelAnimationFrame(frameRef.current);
   160→    };
   161→  }, [target, duration]);
   162→
   163→  return count;
   164→}

   263→function StreakCalendar({ streak, bestStreak, userId }: { streak: number; bestStreak: number; userId: string }) {
   264→  const [activityMap, setActivityMap] = useState<Map<string, number>>(new Map());
   265→  const [days, setDays] = useState<Array<{ date: string; day: string; isToday: boolean }>>([]);
   266→  const [maxMinutes, setMaxMinutes] = useState(1);
   267→
   268→  useEffect(() => {
   269→    async function fetchActivity() {
   270→      try {
   271→        const res = await fetch(`/api/activity?userId=${userId}&weeks=5`);
   272→        const json = await res.json();
   273→        if (json.success) {
   274→          // Build 30-day grid from the response
   275→          const allDays: ActivityDayEntry[] = [];
   276→          const today = new Date();
   277→          today.setHours(0, 0, 0, 0);
   278→          const start = new Date(today);
   279→          start.setDate(start.getDate() - 29);
   280→
   281→          const resMap = new Map<string, number>();
   282→          for (const d of (json.data as Activity30Data).dailyData) {
   283→            resMap.set(d.date, d.minutes);
   284→          }
   285→
   286→          for (let i = 0; i < 30; i++) {
   287→            const d = new Date(start);
   288→            d.setDate(d.getDate() + i);
   289→            const dateStr = d.toISOString().split('T')[0];
   290→            allDays.push({
   291→              date: dateStr,
   292→              day: d.toLocaleDateString('en-US', { weekday: 'narrow' }),
   293→              isToday: dateStr === today.toISOString().split('T')[0],
   294→            });
   295→          }
   296→
   297→          setDays(allDays);
   298→          setActivityMap(resMap);
   299→          const max = Math.max(...Array.from(resMap.values()), 1);
   300→          setMaxMinutes(max);
   301→        }
   302→      } catch { /* silent */ }
   303→    }
   304→    fetchActivity();
   305→  }, [userId]);
   306→
   307→  function getIntensity(date: string): number {
   308→    const mins = activityMap.get(date) || 0;
   309→    if (mins === 0) return 0;
   310→    return Math.min(1, mins / maxMinutes);
   311→  }
   312→
   313→  function getCellColor(intensity: number): string {
   314→    if (intensity === 0) return 'bg-muted/40';
   315→    if (intensity <= 0.25) return 'bg-cyan-300/40 dark:bg-cyan-700/40';
   316→    if (intensity <= 0.5) return 'bg-cyan-400/60 dark:bg-cyan-600/50';
   317→    if (intensity <= 0.75) return 'bg-cyan-500/70 dark:bg-cyan-500/60';
   318→    return 'bg-cyan-600 dark:bg-cyan-400';
   319→  }
   320→
   321→  return (
   322→    <Card className="border-border/50">
   323→      <CardHeader className="pb-3">
   324→        <div className="flex items-center justify-between">
   325→          <CardTitle className="flex items-center gap-2 text-base font-semibold">
   326→            <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-gradient-to-br from-orange-500 to-amber-500">
   327→              <Flame className="h-4 w-4 text-white" />
   328→            </div>
   329→            Learning Streak
   330→          </CardTitle>
   331→          <div className="flex items-center gap-3">
   332→            <div className="flex items-center gap-1.5 text-sm font-bold text-orange-600 dark:text-orange-400">
   333→              <span className="text-base">🔥</span>
   334→              <span className="count-up">{streak}</span>
   335→              <span className="text-xs font-medium text-muted-foreground">days</span>
   336→            </div>
   337→          </div>
   338→        </div>
   339→      </CardHeader>
   340→      <CardContent className="space-y-4">
   341→        {/* Streak stats row */}
   342→        <div className="flex items-center justify-between">
   343→          <div className="flex items-center gap-4 text-xs text-muted-foreground">
   344→            <span className="flex items-center gap-1.5">
   345→              <Flame className="h-3.5 w-3.5" />
   346→              Current: <span className="font-semibold text-foreground">{streak} days</span>
   347→            </span>
   348→            <span className="flex items-center gap-1.5">
   349→              <Trophy className="h-3.5 w-3.5" />
   350→              Best: <span className="font-semibold text-foreground">{bestStreak} days</span>
   351→            </span>
   352→          </div>
   353→        </div>
   354→
   355→        {/* 30-day grid - 6 columns x 5 rows */}
   356→        <div>
   357→          <p className="mb-2 text-[11px] font-medium text-muted-foreground">Last 30 days</p>
   358→          <div className="grid grid-cols-6 gap-1.5 sm:grid-cols-10">
   359→            {days.map((d) => {
   360→              const intensity = getIntensity(d.date);
   361→              return (
   362→                <div
   363→                  key={d.date}
   364→                  title={`${d.date}: ${activityMap.get(d.date) || 0} min`}
   365→                  className={`relative aspect-square rounded-sm transition-all duration-200 hover:scale-125 hover:ring-1 hover:ring-primary/30 ${getCellColor(intensity)} ${d.isToday ? 'ring-1 ring-primary' : ''}`}
   366→                />
   367→              );
   368→            })}
   369→          </div>
   370→          {/* Intensity legend */}
   371→          <div className="mt-2 flex items-center justify-end gap-1 text-[10px] text-muted-foreground">
   372→            <span>Less</span>
   373→            <div className="h-2.5 w-2.5 rounded-sm bg-muted/40" />
   374→            <div className="h-2.5 w-2.5 rounded-sm bg-cyan-300/40 dark:bg-cyan-700/40" />
   375→            <div className="h-2.5 w-2.5 rounded-sm bg-cyan-400/60 dark:bg-cyan-600/50" />
