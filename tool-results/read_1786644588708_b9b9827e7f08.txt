   249→// ============================================
   250→// Create Course Page Component
   251→// ============================================
   252→
   253→const MAX_SECTIONS = 10;
   254→const MAX_TITLE_LENGTH = 100;
   255→const MAX_DESC_LENGTH = 3000;
   256→const MAX_COURSE_DESC_LENGTH = 500;
   257→
   258→export function CreateCoursePage() {
   259→  const { goBack } = useNavigationStore();
   260→  const { currentUserId } = useUserStore();
   261→
   262→  // ---- Form state ----
   263→  const [title, setTitle] = useState("");
   264→  const [description, setDescription] = useState("");
   265→  const [categoryId, setCategoryId] = useState("");
   266→  const [language, setLanguage] = useState("english");
   267→  const [coverImage, setCoverImage] = useState("");
   268→  const [coverPreview, setCoverPreview] = useState<string | null>(null);
   269→  const [sections, setSections] = useState<SectionDraft[]>([]);
   270→
   271→  // ---- UI state ----
   272→  const [categories, setCategories] = useState<CategoryItem[]>([]);
   273→  const [loading, setLoading] = useState(false);
   274→  const [saving, setSaving] = useState(false);
   275→  const [generating, setGenerating] = useState(false);
   276→  const [modalOpen, setModalOpen] = useState(false);
   277→  const [urlInputOpen, setUrlInputOpen] = useState(false);
   278→  const [imageUrl, setImageUrl] = useState("");
   279→
   280→  // ---- Modal state ----
   281→  const [sectionName, setSectionName] = useState("");
   282→  const [sectionPrompt, setSectionPrompt] = useState("");
   283→  const [sectionLanguage, setSectionLanguage] = useState("english");
   284→  const [sectionPdfName, setSectionPdfName] = useState("");
   285→
   286→  // ---- Refs ----
   287→  const fileInputRef = useRef<HTMLInputElement>(null);
   288→  const sectionFileInputRef = useRef<HTMLInputElement>(null);
   289→
   290→  // ---- Fetch categories ----
   291→  useEffect(() => {
   292→    async function fetchCategories() {
   293→      setLoading(true);
   294→      try {
   295→        const res = await fetch("/api/categories");
   296→        const json = await res.json();
   297→        if (json.success) {
   298→          setCategories(json.data);
   299→        }
   300→      } catch {
   301→        toast.error("Failed to load categories");
   302→      } finally {
   303→        setLoading(false);
   304→      }
   305→    }
   306→    fetchCategories();
   307→  }, []);
   308→
   309→  // ---- Cover image handlers ----
   310→  const handleCoverUpload = () => {
   311→    // In sandbox, simulate file selection
   312→    setUrlInputOpen(true);
   313→  };
   314→
   315→  const handleUrlConfirm = () => {
   316→    if (imageUrl.trim()) {
   317→      setCoverImage(imageUrl.trim());
   318→      setCoverPreview(imageUrl.trim());
   319→      setUrlInputOpen(false);
   320→      setImageUrl("");
   321→    }
   322→  };
   323→
   324→  const handleRemoveCover = () => {
   325→    setCoverImage("");
   326→    setCoverPreview(null);
   327→  };
   328→
   329→  // ---- Section management ----
   330→  const handleOpenModal = () => {
   331→    setSectionName("");
   332→    setSectionPrompt("");
   333→    setSectionLanguage(language);
   334→    setSectionPdfName("");
   335→    setModalOpen(true);
   336→  };
   337→
   338→  const handleGenerateSection = useCallback(() => {
   339→    if (!sectionName.trim()) {
   340→      toast.error("Section name is required");
   341→      return;
   342→    }
   343→    if (!sectionPrompt.trim()) {
   344→      toast.error("Course prompt is required");
   345→      return;
   346→    }
   347→    if (sections.length >= MAX_SECTIONS) {
   348→      toast.error(`Maximum ${MAX_SECTIONS} sections allowed`);
   349→      return;
   350→    }
   351→
   352→    setGenerating(true);
   353→
   354→    // Simulate generation delay
   355→    setTimeout(() => {
   356→      const slides = generatePlaceholderSlides(
   357→        sectionName.trim(),
   358→        sectionLanguage
   359→      );
   360→      const newSection: SectionDraft = {
   361→        id: `sec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
   362→        title: sectionName.trim(),
   363→        totalPages: slides.length,
   364→        content: JSON.stringify(slides),
   365→        language: sectionLanguage,
   366→      };
   367→
   368→      setSections((prev) => [...prev, newSection]);
   369→      setGenerating(false);
   370→      setModalOpen(false);
   371→      toast.success(`Section "${sectionName.trim()}" generated successfully`);
   372→    }, 1200);
   373→  }, [sectionName, sectionPrompt, sectionLanguage, sections.length]);
   374→
   375→  const handleDeleteSection = (id: string) => {
   376→    setSections((prev) => prev.filter((s) => s.id !== id));
   377→  };
   378→
   379→  const handleMoveSection = (index: number, direction: "up" | "down") => {
   380→    const newSections = [...sections];
   381→    const targetIndex = direction === "up" ? index - 1 : index + 1;
   382→    if (targetIndex < 0 || targetIndex >= newSections.length) return;
   383→    [newSections[index], newSections[targetIndex]] = [
   384→      newSections[targetIndex],
   385→      newSections[index],
   386→    ];
   387→    setSections(newSections);
   388→  };
   389→
   390→  // ---- Form submission ----
   391→  const handleSave = async () => {
   392→    if (!title.trim()) {
   393→      toast.error("Course title is required");
   394→      return;
   395→    }
   396→
   397→    setSaving(true);
   398→    try {
   399→      const payload = {
   400→        title: title.trim(),
   401→        description: description.trim() || null,
   402→        categoryId: categoryId || null,
   403→        language,
   404→        creatorId: currentUserId,
   405→        coverImage: coverImage || null,
   406→        sections: sections.map((sec, index) => ({
   407→          title: sec.title,
   408→          content: sec.content,
   409→          totalPages: sec.totalPages,
   410→          order: index,
   411→        })),
   412→      };
   413→
   414→      const res = await fetch("/api/courses", {
   415→        method: "POST",
   416→        headers: { "Content-Type": "application/json" },
   417→        body: JSON.stringify(payload),
   418→      });
   419→
   420→      const json = await res.json();
   421→
   422→      if (json.success) {
   423→        toast.success("Course created successfully!");
   424→        goBack();
   425→      } else {
   426→        toast.error(json.error || "Failed to create course");
   427→      }
   428→    } catch {
   429→      toast.error("Failed to create course. Please try again.");
   430→    } finally {
   431→      setSaving(false);
   432→    }
   433→  };
   434→
   435→  // ---- Section PDF upload simulation ----
   436→  const handleSectionPdfUpload = () => {
   437→    sectionFileInputRef.current?.click();
   438→  };
   439→
   440→  const handleSectionFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   441→    const file = e.target.files?.[0];
   442→    if (file) {
   443→      setSectionPdfName(file.name);
   444→      // If no prompt yet, use the filename
   445→      if (!sectionPrompt.trim()) {
   446→        const nameWithoutExt = file.name.replace(/\.pdf$/i, "");
   447→        setSectionPrompt(
   448→          `Generate course content based on the uploaded PDF: ${nameWithoutExt}`
   449→        );
   450→        if (!sectionName.trim()) {
   451→          setSectionName(nameWithoutExt);
   452→        }
   453→      }
   454→    }
   455→  };
   456→
   457→  // ============================================
   458→  // Render
   459→  // ============================================
   460→
   461→  return (
   462→    <div className="mx-auto w-full max-w-7xl px-4 py-6 sm:px-6 lg:px-8">
   463→      {/* ---- Breadcrumbs ---- */}
   464→      <div className="mb-6 flex items-center gap-3">
   465→        <button
   466→          onClick={goBack}
   467→          className="flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-foreground"
   468→        >
   469→          <ArrowLeft className="h-4 w-4" />
   470→          Back
   471→        </button>
   472→        <Separator orientation="vertical" className="h-4" />
   473→        <h1 className="text-lg font-semibold text-foreground">Create Course</h1>
   474→      </div>
   475→
   476→      {/* ---- Two-Column Layout ---- */}
   477→      <div className="grid grid-cols-1 gap-6 lg:grid-cols-5">
   478→        {/* ======== Left Panel: Course Metadata (3 cols) ======== */}
   479→        <div className="lg:col-span-3">
   480→          <Card className="border-border/50">
   481→            <CardContent className="space-y-6 p-6">
   482→              {/* Cover Image Upload */}
   483→              <div className="space-y-2">
   484→                <Label className="text-sm font-medium">Cover Image</Label>
   485→                {coverPreview ? (
   486→                  <div className="relative overflow-hidden rounded-lg border border-border">
   487→                    <img
   488→                      src={coverPreview}
   489→                      alt="Course cover"
   490→                      className="h-48 w-full object-cover"
   491→                      onError={() => setCoverPreview(null)}
   492→                    />
   493→                    <button
   494→                      onClick={handleRemoveCover}
   495→                      className="absolute top-2 right-2 flex h-7 w-7 items-center justify-center rounded-full bg-black/60 text-white transition-colors hover:bg-black/80"
   496→                    >
   497→                      <X className="h-3.5 w-3.5" />
   498→                    </button>
   499→                  </div>
   500→                ) : (
   501→                  <button
   502→                    type="button"
   503→                    onClick={handleCoverUpload}
   504→                    className="flex h-48 w-full flex-col items-center justify-center gap-3 rounded-lg border-2 border-dashed border-border/70 bg-muted/30 transition-colors hover:border-primary/50 hover:bg-muted/50"
   505→                  >
   506→                    <div className="flex h-12 w-12 items-center justify-center rounded-full bg-muted">
   507→                      <ImageIcon className="h-5 w-5 text-muted-foreground" />
   508→                    </div>
   509→                    <div className="text-center">
   510→                      <p className="text-sm font-medium text-foreground">
   511→                        Upload cover image
   512→                      </p>
   513→                      <p className="mt-0.5 text-xs text-muted-foreground">
   514→                        Recommended 1920x1080
   515→                      </p>
   516→                    </div>
   517→                  </button>
   518→                )}
   519→
   520→                {/* URL Input */}
   521→                {urlInputOpen && (
   522→                  <div className="flex items-center gap-2">
   523→                    <Input
   524→                      placeholder="Paste image URL here..."
   525→                      value={imageUrl}
   526→                      onChange={(e) => setImageUrl(e.target.value)}
   527→                      onKeyDown={(e) => e.key === "Enter" && handleUrlConfirm()}
   528→                      className="flex-1"
   529→                    />
   530→                    <Button size="sm" onClick={handleUrlConfirm}>
   531→                      Apply
   532→                    </Button>
   533→                    <Button
   534→                      size="sm"
   535→                      variant="ghost"
   536→                      onClick={() => {
   537→                        setUrlInputOpen(false);
   538→                        setImageUrl("");
   539→                      }}
   540→                    >
   541→                      Cancel
   542→                    </Button>
   543→                  </div>
   544→                )}
   545→                <input
   546→                  ref={fileInputRef}
   547→                  type="file"
   548→                  accept="image/*"
   549→                  className="hidden"
   550→                />
   551→              </div>
   552→
   553→              <Separator />
   554→
   555→              {/* Course Title */}
   556→              <div className="space-y-2">
   557→                <div className="flex items-center justify-between">
   558→                  <Label htmlFor="course-title" className="text-sm font-medium">
   559→                    Course Title <span className="text-destructive">*</span>
   560→                  </Label>
   561→                  <span className="text-xs text-muted-foreground">
   562→                    {title.length}/100
   563→                  </span>
   564→                </div>
   565→                <Input
   566→                  id="course-title"
   567→                  placeholder="Enter course title"
   568→                  value={title}
   569→                  onChange={(e) => {
   570→                    if (e.target.value.length <= MAX_TITLE_LENGTH) {
   571→                      setTitle(e.target.value);
   572→                    }
   573→                  }}
   574→                  className="h-10"
   575→                />
   576→              </div>
   577→
   578→              {/* Course Description */}
   579→              <div className="space-y-2">
   580→                <div className="flex items-center justify-between">
   581→                  <Label
   582→                    htmlFor="course-description"
   583→                    className="text-sm font-medium"
   584→                  >
   585→                    Description
   586→                  </Label>
   587→                  <span className="text-xs text-muted-foreground">
   588→                    {description.length}/{MAX_COURSE_DESC_LENGTH}
   589→                  </span>
   590→                </div>
   591→                <Textarea
   592→                  id="course-description"
   593→                  placeholder="Describe what this course is about..."
   594→                  value={description}
   595→                  onChange={(e) => {
   596→                    if (e.target.value.length <= MAX_COURSE_DESC_LENGTH) {
   597→                      setDescription(e.target.value);
   598→                    }
   599→                  }}
   600→                  rows={4}
   601→                  className="resize-none"
   602→                />
   603→              </div>
   604→
   605→              <Separator />
   606→
   607→              {/* Category */}
   608→              <div className="space-y-2">
   609→                <Label className="text-sm font-medium">Category</Label>
   610→                <Select value={categoryId} onValueChange={setCategoryId}>
   611→                  <SelectTrigger className="w-full h-10">
   612→                    <SelectValue placeholder="Select a category" />
   613→                  </SelectTrigger>
   614→                  <SelectContent>
   615→                    {categories.map((cat) => (
   616→                      <SelectItem key={cat.id} value={cat.id}>
   617→                        {cat.name}
   618→                      </SelectItem>
   619→                    ))}
   620→                  </SelectContent>
   621→                </Select>
   622→              </div>
   623→
   624→              {/* Language */}
   625→              <div className="space-y-3">
   626→                <Label className="text-sm font-medium">Language</Label>
   627→                <RadioGroup
   628→                  value={language}
   629→                  onValueChange={setLanguage}
   630→                  className="flex gap-6"
   631→                >
   632→                  <div className="flex items-center gap-2">
   633→                    <RadioGroupItem value="chinese" id="lang-zh" />
   634→                    <Label
   635→                      htmlFor="lang-zh"
   636→                      className="cursor-pointer text-sm font-normal"
   637→                    >
   638→                      Chinese
   639→                    </Label>
   640→                  </div>
   641→                  <div className="flex items-center gap-2">
   642→                    <RadioGroupItem value="english" id="lang-en" />
   643→                    <Label
   644→                      htmlFor="lang-en"
   645→                      className="cursor-pointer text-sm font-normal"

   875→                  <RadioGroupItem
   876→                    value="chinese"
   877→                    id="sec-lang-zh"
   878→                  />
   879→                  <Label
   880→                    htmlFor="sec-lang-zh"
