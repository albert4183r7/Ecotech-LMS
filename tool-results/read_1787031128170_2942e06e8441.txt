   102→/* ------------------------------------------------------------------ */
   103→/*  Seeded random for consistent sparkline data                        */
   104→/* ------------------------------------------------------------------ */
   105→
   106→function seededRandom(seed: number) {
   107→  let s = seed;
   108→  return () => {
   109→    s = (s * 16807 + 0) % 2147483647;
   110→    return (s - 1) / 2147483646;
   111→  };
   112→}

   114→/* ------------------------------------------------------------------ */

   116→/* ------------------------------------------------------------------ */
   117→
   118→function SparklineChart({
   119→  data,
   120→  gradientFrom,
   121→  gradientTo,
   122→}: {
   123→  data: number[];
   124→  gradientFrom: string;
   125→  gradientTo: string;
   126→}) {
   127→  const max = Math.max(...data, 1);
   128→  return (
   129→    <div className="flex items-end gap-[3px] h-8">
   130→      {data.map((val, i) => {
   131→        const height = Math.max(4, (val / max) * 100);
   132→        return (
   133→          <div
   134→            key={i}
   135→            className={cn(
   136→              "w-[6px] rounded-full transition-all duration-500",
   137→              `bg-gradient-to-t ${gradientFrom} ${gradientTo}`
   138→            )}
   139→            style={{
   140→              height: `${height}%`,
   141→              opacity: 0.5 + (i / data.length) * 0.5,
   142→              animationDelay: `${i * 60}ms`,
   143→            }}
   144→          />
   145→        );
   146→      })}
   147→    </div>
   148→  );
   149→}

   263→/* ------------------------------------------------------------------ */
   264→/*  Donut Chart                                                        */
   265→/* ------------------------------------------------------------------ */
   266→
   267→function DonutChart({
   268→  data,
   269→  total,
   270→}: {
   271→  data: { categoryName: string; categoryColor: string | null; courseCount: number }[];
   272→  total: number;
   273→}) {
   274→  // Build conic-gradient
   275→  const gradientParts: string[] = [];
   276→  let currentAngle = 0;
   277→
   278→  const DONUT_COLORS = [
   279→    "#14b8a6",
   280→    "#f59e0b",
   281→    "#ef4444",
   282→    "#8b5cf6",
   283→    "#ec4899",
   284→    "#06b6d4",
   285→    "#84cc16",
   286→    "#f97316",
   287→  ];
   288→
   289→  data.forEach((item, i) => {
   290→    const color = item.categoryColor || DONUT_COLORS[i % DONUT_COLORS.length];
   291→    const angle = total > 0 ? (item.courseCount / total) * 360 : 0;
   292→    gradientParts.push(`${color} ${currentAngle}deg ${currentAngle + angle}deg`);
   293→    currentAngle += angle;
   294→  });
   295→
   296→  const gradientStr =
   297→    gradientParts.length > 0
   298→      ? `conic-gradient(${gradientParts.join(", ")})`
   299→      : "conic-gradient(#e5e7eb 0deg 360deg)";
   300→
   301→  return (
   302→    <div className="flex flex-col items-center gap-4">
   303→      {/* Donut */}
   304→      <div
   305→        className="relative h-40 w-40 sm:h-48 sm:w-48 rounded-full stat-pop"
   306→        style={{
   307→          background: gradientStr,
   308→        }}
   309→      >
   310→        {/* Inner circle (donut hole) */}
   311→        <div className="absolute inset-0 m-auto h-[60%] w-[60%] rounded-full bg-card flex flex-col items-center justify-center shadow-inner">
   312→          <span className="text-2xl font-bold text-foreground">{total}</span>
   313→          <span className="text-[11px] text-muted-foreground">courses</span>
   314→        </div>
   315→      </div>
   316→
   317→      {/* Legend */}
   318→      <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 w-full max-w-[220px]">
   319→        {data.map((item, i) => {
   320→          const color =
   321→            item.categoryColor || DONUT_COLORS[i % DONUT_COLORS.length];
   322→          const pct = total > 0 ? Math.round((item.courseCount / total) * 100) : 0;
   323→          return (
   324→            <div key={item.categoryName} className="flex items-center gap-2">
   325→              <span
   326→                className="h-2.5 w-2.5 rounded-full shrink-0"
   327→                style={{ backgroundColor: color }}
   328→              />
   329→              <div className="min-w-0">
   330→                <span className="text-xs text-foreground truncate block">
   331→                  {item.categoryName}
   332→                </span>
   333→                <span className="text-[10px] text-muted-foreground">
   334→                  {pct}%
   335→                </span>
   336→              </div>
   337→            </div>
   338→          );
   339→        })}
   340→      </div>
   341→    </div>
   342→  );
   343→}

   345→/* ------------------------------------------------------------------ */
   346→/*  Weekly Heatmap                                                     */
   347→/* ------------------------------------------------------------------ */
   348→
   349→function WeeklyHeatmap({
   350→  userId,
   351→}: {
   352→  userId: string;
   353→}) {
   354→  const [heatmapData, setHeatmapData] = useState<number[][]>([]);
   355→  const [monthLabels, setMonthLabels] = useState<string[]>([]);
   356→
   357→  useEffect(() => {
   358→    async function fetchHeatmap() {
   359→      try {
   360→        const res = await fetch(`/api/activity?userId=${userId}`);
   361→        const json = await res.json();
   362→        if (json.success && json.data && json.data.dailyData) {
   363→          // Use the activity data to build a heatmap
   364→          const activityMap: Record<string, number> = {};
   365→          const activities = json.data.dailyData;
   366→          for (const act of activities) {
   367→            const dateStr = act.date;
   368→            // Use minutes to derive activity level
   369→            const minutes = act.minutes || 0;
   370→            // Keep the max minutes per day
   371→            const existing = activityMap[dateStr] || 0;
   372→            activityMap[dateStr] = Math.max(existing, minutes);
   373→          }
   374→
   375→          // Build 12-week × 7-day grid
   376→          const grid: number[][] = [];
   377→          const today = new Date();
   378→          const months: string[] = [];
   379→          const seenMonths = new Set<string>();
   380→
   381→          // Start from 12 weeks ago, aligned to Sunday
   382→          const startDate = new Date(today);
   383→          startDate.setDate(startDate.getDate() - 83); // ~12 weeks
   384→          const dayOfWeek = startDate.getDay();
   385→          startDate.setDate(startDate.getDate() - dayOfWeek); // Align to Sunday
   386→
   387→          for (let week = 0; week < 12; week++) {
   388→            for (let day = 0; day < 7; day++) {
   389→              if (day === 0) grid.push([]);
   390→              const cellDate = new Date(startDate);
   391→              cellDate.setDate(cellDate.getDate() + week * 7 + day);
   392→              const dateStr = cellDate.toISOString().split("T")[0];
   393→              grid[week].push(activityMap[dateStr] || 0);
   394→
   395→              // Convert minutes to 0-4 level for heatmap coloring
   396→              const mins = activityMap[dateStr] || 0;
   397→              if (mins === 0) {
   398→                // already 0
   399→              } else {
   400→                // Normalize: 0=none, 1=1-30min, 2=31-60min, 3=61-90min, 4=91+min
   401→                grid[week][day] = mins <= 30 ? 1 : mins <= 60 ? 2 : mins <= 90 ? 3 : 4;
   402→              }
   403→              if (day === 0) {
   404→                const monthName = cellDate.toLocaleString("en-US", {
   405→                  month: "short",
   406→                });
   407→                if (!seenMonths.has(monthName)) {
   408→                  seenMonths.add(monthName);
   409→                  months.push(monthName);
   410→                } else {
   411→                  months.push("");
   412→                }
   413→              }
   414→            }
   415→          }
   416→
   417→          setHeatmapData(grid);
   418→          setMonthLabels(months);
   419→        }
   420→      } catch {
   421→        // Generate fallback random data
   422→        const rand = seededRandom(42);
   423→        const grid: number[][] = [];
   424→        for (let week = 0; week < 12; week++) {
   425→          grid.push([]);
   426→          for (let day = 0; day < 7; day++) {
   427→            grid[week].push(Math.floor(rand() * 5));
   428→          }
   429→        }
   430→        setHeatmapData(grid);
   431→        setMonthLabels(["Jan", "", "Feb", "", "Mar", "", "Apr"]);
   432→      }
   433→    }
   434→    fetchHeatmap();
   435→  }, [userId]);
   436→
   437→  const dayLabels = ["Sun", "", "Tue", "", "Thu", "", "Sat"];
   438→
   439→  const getHeatColor = (level: number) => {
   440→    switch (level) {
   441→      case 0:
   442→        return "bg-muted";
   443→      case 1:
   444→        return "bg-emerald-200 dark:bg-emerald-900/50";
   445→      case 2:
   446→        return "bg-emerald-300 dark:bg-emerald-800/60";
   447→      case 3:
   448→        return "bg-emerald-400 dark:bg-emerald-700/70";
   449→      default:
   450→        return "bg-emerald-500 dark:bg-emerald-600";
   451→    }
   452→  };
   453→
   454→  if (heatmapData.length === 0) return null;
   455→
   456→  return (
   457→    <TooltipProvider delayDuration={200}>
   458→      <div className="space-y-2">
   459→        {/* Month labels */}
   460→        <div className="flex gap-0.5 pl-8">
   461→          {monthLabels.map((label, i) => (
   462→            <div
   463→              key={i}
   464→              className="flex-1 min-w-0"
   465→            >
   466→              <span className="text-[9px] text-muted-foreground truncate">
   467→                {label}
   468→              </span>
   469→            </div>
   470→          ))}
   471→        </div>
   472→
   473→        <div className="flex gap-1.5">
   474→          {/* Day labels */}
   475→          <div className="flex flex-col gap-0.5 shrink-0 pt-0">
   476→            {dayLabels.map((label, i) => (
   477→              <div key={i} className="h-[14px] flex items-center">
   478→                <span className="text-[9px] text-muted-foreground w-7 text-right pr-1.5">
   479→                  {label}
   480→                </span>
   481→              </div>
   482→            ))}
   483→          </div>
   484→
   485→          {/* Heatmap grid */}
   486→          <div className="flex gap-[3px] flex-1 overflow-x-auto">
   487→            {heatmapData.map((week, weekIdx) => (
   488→              <div key={weekIdx} className="flex flex-col gap-[3px]">
   489→                {week.map((level, dayIdx) => (
   490→                  <Tooltip key={dayIdx}>
   491→                    <TooltipTrigger asChild>
   492→                      <div
   493→                        className={cn(
   494→                          "h-[14px] w-[14px] rounded-[3px] transition-colors hover:ring-1 hover:ring-foreground/20",
   495→                          getHeatColor(level)
   496→                        )}
   497→                      />
   498→                    </TooltipTrigger>
   499→                    <TooltipContent side="top" className="text-xs">
   500→                      {level > 0
   501→                        ? `${level === 1 ? "<30" : level === 2 ? "31-60" : level === 3 ? "61-90" : "90+"} min`
   502→                        : "No activity"}
   503→                    </TooltipContent>
   504→                  </Tooltip>
   505→                ))}
   506→              </div>
   507→            ))}
   508→          </div>
   509→        </div>
   510→
   511→        {/* Legend */}
   512→        <div className="flex items-center justify-end gap-1.5 text-[10px] text-muted-foreground">
   513→          <span>Less</span>
   514→          {[0, 1, 2, 3, 4].map((level) => (
   515→            <div
   516→              key={level}
   517→              className={cn("h-[10px] w-[10px] rounded-[2px]", getHeatColor(level))}
   518→            />
   519→          ))}
   520→          <span>More</span>
   521→        </div>
   522→      </div>
   523→    </TooltipProvider>
   524→  );
   525→}
