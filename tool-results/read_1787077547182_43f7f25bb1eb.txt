    43→export function ClassroomPage() {
    44→  const { classroomState, goBack } = useNavigationStore();
    45→  const userId = useUserStore((s) => s.currentUserId);
    46→  const [localState, setLocalState] = useState<ClassroomState | null>(null);
    47→  const [zoom, setZoom] = useState(100);
    48→  const [navigating, setNavigating] = useState(false);
    49→  const confettiShownRef = useRef(false);
    50→  const [showConfetti, setShowConfetti] = useState(false);
    51→
    52→  // ─── Notes Sidebar State ──────────────────────
    53→  interface Note {
    54→    id: string;
    55→    content: string;
    56→    slideNumber: number;
    57→    bookmarked: boolean;
    58→    createdAt?: string;
    59→  }
    60→  const [notes, setNotes] = useState<Note[]>([]);
    61→  const [notesSidebarOpen, setNotesSidebarOpen] = useState(false);
    62→  const [newNoteContent, setNewNoteContent] = useState("");
    63→  const [savingNote, setSavingNote] = useState(false);
    64→  const notesEndRef = useRef<HTMLDivElement>(null);
    65→
    66→  // ─── Keyboard Hint Fade ────────────────────────
    67→  const [showKeyboardHint, setShowKeyboardHint] = useState(true);
    68→  const [isMobile, setIsMobile] = useState(false);
    69→  const [downloadingPptx, setDownloadingPptx] = useState(false);
    70→
    71→  // ─── AI Edit State ─────────────────────────────
    72→  const [showAiEdit, setShowAiEdit] = useState(false);
    73→  const [aiEditInstruction, setAiEditInstruction] = useState("");
    74→  const [aiEditLoading, setAiEditLoading] = useState(false);
    75→
    76→  // ─── Click-to-Edit State ──────────────────────
    77→  const iframeRef = useRef<HTMLIFrameElement>(null);
    78→  const currentSlideIdRef = useRef<string | null>(null);
    79→  const slideContextRef = useRef<string>("");
    80→  const [elementEdit, setElementEdit] = useState<{
    81→    show: boolean;
    82→    targetElement: Element | null;
    83→    outerHTML: string;
    84→    position: { x: number; y: number };
    85→    instruction: string;
    86→    loading: boolean;
    87→  }>({
    88→    show: false,
    89→    targetElement: null,
    90→    outerHTML: "",
    91→    position: { x: 0, y: 0 },
    92→    instruction: "",
    93→    loading: false,
    94→  });
    95→
    96→  useEffect(() => {
    97→    const mq = window.matchMedia("(max-width: 1023px)");
    98→    setIsMobile(mq.matches);
    99→    const handler = (e: MediaQueryListEvent) => setIsMobile(e.matches);
   100→    mq.addEventListener("change", handler);
   101→    return () => mq.removeEventListener("change", handler);
   102→  }, []);
   103→
   104→  // Keep a working copy of classroomState
   105→  useEffect(() => {
   106→    if (classroomState) {
   107→      setLocalState({ ...classroomState });
   108→      setZoom(100);
   109→      confettiShownRef.current = false;
   110→      setShowConfetti(false);
   111→    }
   112→  }, [classroomState]);
   113→
   114→  // ─── Lesson Navigation ─────────────────────────
   115→  const goToLesson = useCallback(
   116→    async (index: number) => {
   117→      if (!localState || index < 0 || index >= localState.allLessonIds.length) return;
   118→      const lessonId = localState.allLessonIds[index];
   119→      try {
   120→        setNavigating(true);
   121→        const res = await fetch(`/api/lessons/${lessonId}`);
   122→        if (!res.ok) return;
   123→        const json = await res.json();
   124→        const slideData =
   125→          json.success && json.data.slides?.length > 0
   126→            ? json.data.slides[0]
   127→            : null;
   128→        const htmlBody = slideData?.htmlBody
   129→            || '<div class="flex items-center justify-center h-full"><p class="text-gray-500">No content available.</p></div>';
   130→        const lessonTitle =
   131→          json.success && json.data.title
   132→            ? json.data.title
   133→            : `Lesson ${index + 1}`;
   134→        // Track slide ID for element-edit persistence
   135→        if (slideData?.id) currentSlideIdRef.current = slideData.id;
   136→        // Extract slide style context from outlineJson
   137→        if (json.data.outlineJson) {
   138→          try {
   139→            const outline = JSON.parse(json.data.outlineJson);
   140→            const style = outline.style || "";
   141→            const topic = outline.topic || lessonTitle;
   142→            slideContextRef.current = `This is a ${style ? style + "-style" : ""} slide about "${topic}". Keep edits visually consistent with this style.`;
   143→          } catch { /* ignore parse errors */ }
   144→        }
   145→        setLocalState((prev) =>
   146→          prev
   147→            ? {
   148→                ...prev,
   149→                lessonId,
   150→                lessonTitle,
   151→                htmlBody,
   152→                currentLessonIndex: index,
   153→              }
   154→            : prev
   155→        );
   156→      } catch {
   157→        /* silently fail */
   158→      } finally {
   159→        setNavigating(false);
   160→      }
   161→    },
   162→    [localState]
   163→  );
   164→
   165→  /** Mark lesson as completed and save progress */
   166→  const markLessonCompleted = useCallback(
   167→    async (lessonId: string) => {
   168→      if (!userId || !localState) return;
   169→      try {
   170→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   171→        const enrollJson = await enrollRes.json();
   172→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   173→          const enrollment = enrollJson.data.find(
   174→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   175→          );
   176→          if (enrollment) {
   177→            await fetch("/api/progress", {
   178→              method: "POST",
   179→              headers: { "Content-Type": "application/json" },
   180→              body: JSON.stringify({
   181→                enrollmentId: enrollment.id,
   182→                lessonId,
   183→                currentPage: 1,
   184→                completed: true,
   185→              }),
   186→            });
   187→          }
   188→        }
   189→      } catch {
   190→        /* best-effort */
   191→      }
   192→    },
   193→    [userId, localState]
   194→  );
   195→
   196→  // Save progress when lesson changes
   197→  useEffect(() => {
   198→    if (!localState || !userId) return;
   199→    const timer = setTimeout(() => {
   200→      markLessonCompleted(localState.lessonId);
   201→    }, 500);
   202→    return () => clearTimeout(timer);
   203→  }, [localState?.currentLessonIndex, userId, markLessonCompleted]);
   204→
   205→  const goPrev = () => {
   206→    if (!localState) return;
   207→    goToLesson(localState.currentLessonIndex - 1);
   208→  };
   209→  const goNext = () => {
   210→    if (!localState) return;
   211→    goToLesson(localState.currentLessonIndex + 1);
   212→  };
   213→
   214→  // ─── Keyboard shortcuts ──────────────────────────────────────
   215→  useEffect(() => {
   216→    function handleKeyDown(e: KeyboardEvent) {
   217→      const tag = (e.target as HTMLElement)?.tagName;
   218→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   219→
   220→      switch (e.key) {
   221→        case "Escape":
   222→          e.preventDefault();
   223→          goBack();
   224→          break;
   225→        case "ArrowLeft":
   226→          e.preventDefault();
   227→          goPrev();
   228→          break;
   229→        case "ArrowRight":
   230→          e.preventDefault();
   231→          goNext();
   232→          break;
   233→        case " ":
   234→          e.preventDefault();
   235→          goNext();
   236→          break;
   237→      }
   238→    }
   239→
   240→    document.addEventListener("keydown", handleKeyDown);
   241→
   242→    // Listen for custom event from KeyboardShortcuts component
   243→    function handleCustomNextSlide() {
   244→      goNext();
   245→    }
   246→    document.addEventListener("lms:next-slide", handleCustomNextSlide);
   247→
   248→    return () => {
   249→      document.removeEventListener("keydown", handleKeyDown);
   250→      document.removeEventListener("lms:next-slide", handleCustomNextSlide);
   251→    };
   252→  }, [goBack, goPrev, goNext]);
   253→
   254→  // ─── Iframe Click-to-Edit Listener ─────────────
   255→  // Attaches a click listener to the iframe's document body.
   256→  // With sandbox="allow-same-origin", the parent can access iframe.contentDocument directly.
   257→  // No sandbox change needed — direct DOM access is safe under allow-same-origin.
   258→  const handleIframeClick = useCallback(
   259→    (e: MouseEvent) => {
   260→      const target = e.target as HTMLElement;
   261→      if (!target || target.tagName === "BODY" || target.tagName === "HTML") return;
   262→
   263→      const iframe = iframeRef.current;
   264→      if (!iframe?.contentDocument || !iframe.contentWindow) return;
   265→
   266→      // Capture the element reference and its outerHTML
   267→      const outerHTML = target.outerHTML;
   268→
   269→      // Map click coordinates from iframe viewport to parent viewport
   270→      // accounting for zoom transform on the container
   271→      const iframeRect = iframe.getBoundingClientRect();
   272→      const iframeWidth = iframe.contentWindow.innerWidth;
   273→      const iframeHeight = iframe.contentWindow.innerHeight;
   274→      const scaleX = iframeRect.width / iframeWidth;
   275→      const scaleY = iframeRect.height / iframeHeight;
   276→      const pageX = iframeRect.left + e.clientX * scaleX;
   277→      const pageY = iframeRect.top + e.clientY * scaleY;
   278→
   279→      // Position toolbar slightly offset from click
   280→      const toolbarX = Math.min(pageX + 10, window.innerWidth - 300);
   281→      const toolbarY = Math.min(pageY - 10, window.innerHeight - 120);
   282→
   283→      setElementEdit({
   284→        show: true,
   285→        targetElement: target,
   286→        outerHTML,
   287→        position: { x: Math.max(10, toolbarX), y: Math.max(10, toolbarY) },
   288→        instruction: "",
   289→        loading: false,
   290→      });
   291→    },
   292→    []
   293→  );
   294→
   295→  const attachIframeClickListener = useCallback(() => {
   296→    const iframe = iframeRef.current;
   297→    if (!iframe || !iframe.contentDocument?.body) return;
   298→
   299→    const body = iframe.contentDocument.body;
   300→    // Remove any previously attached listener to avoid duplicates
   301→    body.removeEventListener("click", handleIframeClick);
   302→    body.addEventListener("click", handleIframeClick);
   303→  }, [handleIframeClick]);
   304→
   305→  // Re-attach click listener when htmlBody changes (new slide loaded)
   306→  useEffect(() => {
   307→    // Small delay to ensure iframe has rendered the new srcDoc
   308→    const timer = setTimeout(attachIframeClickListener, 300);
   309→    return () => clearTimeout(timer);
   310→  }, [localState?.htmlBody, attachIframeClickListener]);
   311→
   312→  // Close element edit toolbar on Escape
   313→  useEffect(() => {
   314→    if (!elementEdit.show) return;
   315→    function handleEsc(e: KeyboardEvent) {
   316→      if (e.key === "Escape") {
   317→        setElementEdit((prev) => ({ ...prev, show: false, instruction: "" }));
   318→      }
   319→    }
   320→    document.addEventListener("keydown", handleEsc);
   321→    return () => document.removeEventListener("keydown", handleEsc);
   322→  }, [elementEdit.show]);
   323→
   324→  /** Submit element edit: call API, replace in DOM, persist */
   325→  const handleElementEdit = useCallback(async () => {
   326→    if (
   327→      !elementEdit.targetElement ||
   328→      elementEdit.loading ||
   329→      !elementEdit.instruction.trim()
   330→    )
   331→      return;
   332→
   333→    setElementEdit((prev) => ({ ...prev, loading: true }));
   334→
   335→    try {
   336→      const res = await fetch("/api/slides/element-edit", {
   337→        method: "POST",
   338→        headers: { "Content-Type": "application/json" },
   339→        body: JSON.stringify({
   340→          elementHtml: elementEdit.outerHTML,
   341→          instruction: elementEdit.instruction.trim(),
   342→          slideContext: slideContextRef.current,
   343→        }),
   344→      });
   345→
   346→      if (!res.ok) {
   347→        const errJson = await res.json().catch(() => null);
   348→        toast.error(errJson?.error || "Element edit failed");
   349→        return;
   350→      }
   351→
   352→      const json = await res.json();
   353→      if (!json.success || !json.data?.replacementHtml) {
   354→        toast.error("Element edit returned no content");
   355→        return;
   356→      }
   357→
   358→      const replacementHtml = json.data.replacementHtml;
   359→
   360→      // Replace the element in the iframe DOM
   361→      const iframe = iframeRef.current;
   362→      if (iframe?.contentDocument && elementEdit.targetElement) {
   363→        // Verify the element is still in the DOM (user may have navigated)
   364→        if (iframe.contentDocument.contains(elementEdit.targetElement)) {
   365→          const tempDiv = iframe.contentDocument.createElement("div");
   366→          tempDiv.innerHTML = replacementHtml;
   367→          const newElement = tempDiv.firstElementChild;
   368→          if (newElement) {
   369→            elementEdit.targetElement.replaceWith(newElement);
   370→          }
   371→        }
   372→
   373→        // Rebuild full HTML from the updated iframe DOM
   374→        const bodyHtml = iframe.contentDocument.body.innerHTML;
   375→        const title = localState?.lessonTitle || "Slide";
   376→        const safeTitle = title
   377→          .replace(/&/g, "&amp;")
   378→          .replace(/</g, "&lt;")
   379→          .replace(/>/g, "&gt;")
   380→          .replace(/"/g, "&quot;");
   381→        const updatedHtmlBody =
   382→          `<!DOCTYPE html>\n<html lang="en">\n<head>\n  <meta charset="UTF-8" />\n  <meta name="viewport" content="width=device-width, initial-scale=1.0" />\n  <title>${safeTitle}</title>\n  <script src="https://cdn.tailwindcss.com"><\/script>\n  <style>\n    body { margin: 0; padding: 0; font-family: system-ui, -apple-system, sans-serif; }\n    * { box-sizing: border-box; }\n  </style>\n</head>\n<body class="bg-white dark:bg-zinc-900 text-zinc-900 dark:text-zinc-100">\n  ${bodyHtml}\n</body>\n</html>`;
   383→
   384→        // Update local state
   385→        setLocalState((prev) =>
   386→          prev ? { ...prev, htmlBody: updatedHtmlBody } : prev
   387→        );
   388→
   389→        // Persist to database
   390→        const slideId = currentSlideIdRef.current;
   391→        if (slideId) {
   392→          fetch(`/api/slides/${slideId}`, {
   393→            method: "PUT",
   394→            headers: { "Content-Type": "application/json" },
   395→            body: JSON.stringify({ htmlBody: updatedHtmlBody }),
   396→          }).catch(() => { /* best-effort */ });
   397→        }
   398→
   399→        // Re-attach click listener for the new DOM
   400→        setTimeout(attachIframeClickListener, 100);
   401→
   402→        toast.success("Element edited successfully");
   403→      }
   404→    } catch {
   405→      toast.error("Element edit request failed");
   406→    } finally {
   407→      setElementEdit({
   408→        show: false,
   409→        targetElement: null,
   410→        outerHTML: "",
   411→        position: { x: 0, y: 0 },
   412→        instruction: "",
   413→        loading: false,
   414→      });
   415→    }
   416→  }, [elementEdit, localState?.lessonTitle, attachIframeClickListener]);
   417→
   418→  /** Zoom controls */
   419→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   420→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   421→  const zoomFit = () => setZoom(100);
   422→
   423→  // ─── Confetti on last lesson ─────────────────
   424→  useEffect(() => {
   425→    if (!localState) return;
   426→    const totalLessons = localState.allLessonIds.length;
   427→    const isOnLastLesson =
   428→      totalLessons > 0 &&
   429→      localState.currentLessonIndex === totalLessons - 1;
   430→    if (isOnLastLesson && !confettiShownRef.current) {
   431→      confettiShownRef.current = true;
   432→      setShowConfetti(true);
   433→      toast.success("🎉 You completed the course! Great job!");
   434→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   435→      return () => clearTimeout(timer);
   436→    }
   437→  }, [localState?.currentLessonIndex]);
   438→
   439→  // Reset confetti flag when classroom state changes (new course)
   440→  useEffect(() => {
   441→    confettiShownRef.current = false;
   442→    setShowConfetti(false);
   443→  }, [classroomState?.courseId]);
   444→
   445→  // ─── Keyboard Hint Auto-Fade ────────────────────
   446→  useEffect(() => {
   447→    setShowKeyboardHint(true);
   448→    const timer = setTimeout(() => setShowKeyboardHint(false), 5000);
   449→    return () => clearTimeout(timer);
   450→  }, [localState?.lessonId]);
   451→
   452→  // ─── Notes CRUD ─────────────────────────────────
   453→  const fetchNotes = useCallback(async () => {
   454→    if (!localState || !userId) return;
   455→    try {
   456→      const res = await fetch(
   457→        `/api/notes?userId=${userId}&courseId=${localState.courseId}&lessonId=${localState.lessonId}`
   458→      );
   459→      if (!res.ok) return;
   460→      const json = await res.json();
   461→      if (json.success && Array.isArray(json.data)) {
   462→        setNotes(json.data);
   463→      }
   464→    } catch {
   465→      // Silently fail
   466→    }
   467→  }, [localState, userId]);
   468→
   469→  // Fetch notes when section changes or sidebar opens
   470→  useEffect(() => {
   471→    if (notesSidebarOpen) fetchNotes();
   472→  }, [notesSidebarOpen, fetchNotes, localState?.lessonId]);
   473→
   474→  const createNote = useCallback(async () => {
   475→    if (!localState || !userId || !newNoteContent.trim()) return;
   476→    setSavingNote(true);
   477→    try {
   478→      const res = await fetch("/api/notes", {
   479→        method: "POST",
   480→        headers: { "Content-Type": "application/json" },
   481→        body: JSON.stringify({
   482→          userId,
   483→          courseId: localState.courseId,
   484→          lessonId: localState.lessonId,
   485→          content: newNoteContent.trim(),
   486→          slideNumber: localState.currentLessonIndex + 1,
   487→        }),
   488→      });
   489→      if (!res.ok) return;
   490→      const json = await res.json();
   491→      if (json.success && json.data) {
   492→        setNotes((prev) => [...prev, json.data]);
   493→        setNewNoteContent("");
   494→        setTimeout(
   495→          () => notesEndRef.current?.scrollIntoView({ behavior: "smooth" }),
   496→          50
   497→        );
   498→      }
   499→    } catch {
   500→      toast.error("Failed to save note");
   501→    } finally {
   502→      setSavingNote(false);
   503→    }
   504→  }, [localState, userId, newNoteContent]);
   505→
   506→  const deleteNote = useCallback(async (noteId: string) => {
   507→    try {
   508→      const res = await fetch(`/api/notes?id=${noteId}`, {
   509→        method: "DELETE",
   510→      });
   511→      if (!res.ok) return;
   512→      const json = await res.json();
   513→      if (json.success) {
   514→        setNotes((prev) => prev.filter((n) => n.id !== noteId));
   515→      }
   516→    } catch {
   517→      // Silently fail
   518→    }
   519→  }, []);
   520→
   521→  const toggleBookmark = useCallback((noteId: string) => {
   522→    setNotes((prev) =>
   523→      prev.map((n) =>
   524→        n.id === noteId ? { ...n, bookmarked: !n.bookmarked } : n
   525→      )
   526→    );
   527→  }, []);
   528→
   529→  // ─── Download as PPTX ─────────────────────────────
   530→  const handleDownloadPptx = useCallback(async () => {
   531→    if (!localState || downloadingPptx) return;
   532→    setDownloadingPptx(true);
   533→    try {
   534→      const res = await fetch("/api/generate-pptx", {
   535→        method: "POST",
   536→        headers: { "Content-Type": "application/json" },
   537→        body: JSON.stringify({
   538→          courseId: localState.courseId,
   539→          courseName: localState.courseTitle,
   540→        }),
   541→      });
   542→      if (!res.ok) {
   543→        toast.error("Failed to generate PPT");
   544→        return;
   545→      }
   546→      const blob = await res.blob();
   547→      const url = URL.createObjectURL(blob);
   548→      const a = document.createElement("a");
   549→      a.href = url;
   550→      const safeName = localState.courseTitle
   551→        .replace(/[^a-zA-Z0-9\s-]/g, "")
   552→        .replace(/\s+/g, "-")
   553→        .toLowerCase()
   554→        .slice(0, 60);
   555→      a.download = `${safeName}.pptx`;
   556→      document.body.appendChild(a);
   557→      a.click();
   558→      a.remove();
   559→      URL.revokeObjectURL(url);
   560→      toast.success("PPT downloaded successfully!");
   561→    } catch {
   562→      toast.error("Failed to download PPT");
   563→    } finally {
   564→      setDownloadingPptx(false);
   565→    }
   566→  }, [localState, downloadingPptx]);
   567→
   568→  // ─── AI Inline Edit ──────────────────────────────
   569→  const handleAiEdit = useCallback(async () => {
   570→    if (!localState || aiEditLoading || !aiEditInstruction.trim()) return;
   571→    setAiEditLoading(true);
   572→    try {
   573→      const res = await fetch("/api/generate-slide-inline-edit", {
   574→        method: "POST",
   575→        headers: { "Content-Type": "application/json" },
   576→        body: JSON.stringify({
   577→          htmlBody: localState.htmlBody,
   578→          instruction: aiEditInstruction.trim(),
   579→          slideTitle: localState.lessonTitle,
   580→        }),
   581→      });
   582→      if (!res.ok) {
   583→        toast.error("AI edit failed");
