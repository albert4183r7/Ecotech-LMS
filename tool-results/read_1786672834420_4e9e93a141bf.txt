   105→  const colorMap: Record<string, string> = {
   106→    info: read ? "bg-blue-500/10" : "bg-blue-500/15",

   129→/** Format relative time for notifications */
   130→function formatRelativeTime(dateStr: string): string {
   131→  const now = new Date();
   132→  const date = new Date(dateStr);
   133→  const diffMs = now.getTime() - date.getTime();
   134→  const diffMins = Math.floor(diffMs / 60000);
   135→  const diffHours = Math.floor(diffMs / 3600000);
   136→  const diffDays = Math.floor(diffMs / 86400000);
   137→
   138→  if (diffMins < 1) return "Just now";
   139→  if (diffMins < 60) return `${diffMins}m ago`;
   140→  if (diffHours < 24) return `${diffHours}h ago`;
   141→  if (diffDays < 7) return `${diffDays}d ago`;
   142→  return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   143→}

   145→/** Group notifications by date category: Today, Yesterday, Earlier */
   146→function groupNotificationsByDate(items: NotificationItem[]) {
   147→  const now = new Date();
   148→  const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   149→  const yesterdayStart = todayStart - 86400000;
   150→
   151→  const today: NotificationItem[] = [];
   152→  const yesterday: NotificationItem[] = [];
   153→  const earlier: NotificationItem[] = [];
   154→
   155→  for (const item of items) {
   156→    const t = new Date(item.createdAt).getTime();
   157→    if (t >= todayStart) today.push(item);
   158→    else if (t >= yesterdayStart) yesterday.push(item);
   159→    else earlier.push(item);
   160→  }
   161→
   162→  return { today, yesterday, earlier };
   163→}

   165→/** Notification group with date header */
   166→function NotificationGroup({
   167→  label,
   168→  items,
   169→  onNotificationClick,
   170→}: {
   171→  label: string;
   172→  items: NotificationItem[];
   173→  onNotificationClick: (notif: NotificationItem) => void;
   174→}) {
   175→  if (items.length === 0) return null;
   176→
   177→  return (
   178→    <div>
   179→      <p className="px-3 pt-2 pb-1 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground/60">
   180→        {label}
   181→      </p>
   182→      {items.map((notif) => (
   183→        <button
   184→          key={notif.id}
   185→          type="button"
   186→          onClick={() => onNotificationClick(notif)}
   187→          className={cn(
   188→            "group flex w-full items-start gap-3 rounded-lg px-3 py-3 text-left transition-colors notif-appear",
   189→            notif.read
   190→              ? "hover:bg-muted/50"
   191→              : "bg-primary/5 hover:bg-primary/8"
   192→          )}
   193→        >
   194→          {/* Left border accent for unread */}
   195→          {!notif.read && (
   196→            <div className={cn(
   197→              "absolute left-1 top-3 bottom-3 w-0.5 rounded-full",
   198→              NotificationBorderAccent({ type: notif.type })
   199→            )} style={{ position: "absolute" }} />
   200→          )}
   201→          <div className={cn(
   202→            "mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-lg transition-colors relative",
   203→            NotificationIconBg({ type: notif.type, read: notif.read })
   204→          )}>
   205→            <NotificationIcon type={notif.type} />
   206→          </div>
   207→          <div className="min-w-0 flex-1">
   208→            <div className="flex items-start justify-between gap-2">
   209→              <p className={cn(
   210→                "text-sm leading-snug",
   211→                notif.read
   212→                  ? "text-muted-foreground font-medium"
   213→                  : "text-foreground font-semibold"
   214→              )}>
   215→                {notif.title}
   216→              </p>
   217→              {!notif.read && (
   218→                <div className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary" />
   219→              )}
   220→            </div>
   221→            <p className="mt-0.5 text-xs text-muted-foreground leading-relaxed line-clamp-2">
   222→              {notif.message}
   223→            </p>
   224→            <p className="mt-1 text-[10px] text-muted-foreground/60">
   225→              {formatRelativeTime(notif.createdAt)}
   226→            </p>
   227→          </div>
   228→        </button>
   229→      ))}
   230→    </div>
   231→  );
   232→}

   252→/** Parse link string into view + id for navigation */
   253→function parseNotificationLink(link: string | null): { view: ViewName; courseId?: string } | null {
   254→  if (!link) return null;
   255→  if (link === "profile") return { view: "profile" };
   256→  if (link.startsWith("course-detail:")) {
   257→    const courseId = link.replace("course-detail:", "");
   258→    return { view: "course-detail", courseId };
   259→  }
   260→  return null;
   261→}

   279→  const notifRef = useRef<HTMLDivElement>(null);
   280→
   281→  // Theme toggle state for cycling animation
   282→  const [themeIconRotating, setThemeIconRotating] = useState(false);
   283→
   284→  // Breadcrumb: derive current page label
   285→  const currentPageLabel = getViewLabel(currentView);
   286→
   287→  // ============================================
   288→  // Fetch notifications from API
   289→  // ============================================
   290→  useEffect(() => {
   291→    let cancelled = false;
   292→    async function fetchNotifications() {
   293→      setNotificationsLoading(true);
   294→      try {
   295→        const res = await fetch(`/api/notifications?userId=${encodeURIComponent(currentUserId)}`);
   296→        if (!res.ok) throw new Error("Failed to fetch");
   297→        const data = await res.json();
   298→        if (!cancelled) {
   299→          setNotifications(data.notifications || []);
   300→        }
   301→      } catch {
   302→        // Silently fail — notifications will just be empty
   303→      } finally {
   304→        if (!cancelled) {
   305→          setNotificationsLoading(false);
   306→        }
   307→      }
   308→    }
   309→    fetchNotifications();
   310→    return () => { cancelled = true; };
   311→  }, [currentUserId]);
   312→
   313→  // Close mobile menu and search on view change
   314→  useEffect(() => {
   315→    const handleViewCleanup = () => {
   316→      setMobileMenuOpen(false);
   317→      setSearchExpanded(false);
   318→    };
   319→    handleViewCleanup();
   320→  }, [currentView]);
   321→
   322→  // Collapse search when clicking outside
   323→  useEffect(() => {
   324→    if (!searchExpanded) return;

   326→      if (searchInputRef.current && !searchInputRef.current.parentElement?.contains(e.target as Node)) {
   327→        setSearchExpanded(false);
   328→      }
   329→    }
   330→    document.addEventListener("mousedown", handleClickOutside);
   331→    return () => document.removeEventListener("mousedown", handleClickOutside);
   332→  }, [searchExpanded]);
   333→
   334→  // Close notification panel on outside click
   335→  useEffect(() => {
   336→    function handleClickOutside(e: MouseEvent) {
   337→      if (notifRef.current && !notifRef.current.contains(e.target as Node)) {
   338→        setShowNotifications(false);
   339→      }
   340→    }
   341→    if (showNotifications) {
   342→      document.addEventListener("mousedown", handleClickOutside);
   343→    }
   344→    return () => document.removeEventListener("mousedown", handleClickOutside);
   345→  }, [showNotifications]);
   346→
   347→  /** Toggle notification panel */
   348→  const toggleNotifications = useCallback(() => {
   349→    if (!showNotifications) {
   350→      setBellAnimating(true);
   351→      setTimeout(() => setBellAnimating(false), 600);
   352→    }
   353→    setShowNotifications((prev) => !prev);
   354→  }, [showNotifications]);
   355→
   356→  /** Mark a single notification as read via API */
   357→  const handleMarkAsRead = useCallback(async (notif: NotificationItem) => {
   358→    // Navigate if link is present
   359→    const parsed = parseNotificationLink(notif.link);
   360→    if (parsed) {
   361→      if (parsed.courseId) {
   362→        openCourseDetail(parsed.courseId);
   363→      } else {
   364→        navigateTo(parsed.view);
   365→      }
   366→      setShowNotifications(false);
   367→    }
   368→
   369→    // Mark as read in API
   370→    if (!notif.read) {
   371→      try {
   372→        await fetch("/api/notifications", {
   373→          method: "PUT",
   374→          headers: { "Content-Type": "application/json" },
   375→          body: JSON.stringify({ notificationId: notif.id, read: true }),
   376→        });
   377→        setNotifications((prev) =>
   378→          prev.map((n) => (n.id === notif.id ? { ...n, read: true } : n))
   379→        );
   380→      } catch {
   381→        // Silently fail
   382→      }
   383→    }
   384→  }, [navigateTo, openCourseDetail]);
   385→
   386→  /** Mark all notifications as read via API */
   387→  const markAllAsRead = useCallback(async () => {
   388→    try {
   389→      await fetch("/api/notifications", {
   390→        method: "PUT",
   391→        headers: { "Content-Type": "application/json" },
   392→        body: JSON.stringify({ userId: currentUserId, readAll: true }),
   393→      });
   394→      setNotifications((prev) => prev.map((n) => ({ ...n, read: true })));
   395→    } catch {
   396→      // Silently fail
   397→    }
   398→  }, [currentUserId]);

   402→    setShowNotifications(false);
   403→    navigateTo("profile");
