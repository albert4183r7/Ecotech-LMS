     3→// ============================================================
     4→// Helper: build a full HTML+Tailwind slide document
     5→// ============================================================
     6→function makeSlideHtml(title: string, paragraphs: string[]): string {
     7→  const bodyHtml = paragraphs
     8→    .map((p) => `      <p class="text-slate-300 text-lg leading-relaxed">${p}</p>`)
     9→    .join('\n')
    10→  return `<!DOCTYPE html>
    11→<html lang="en">
    12→<head>
    13→<meta charset="UTF-8">
    14→<meta name="viewport" content="width=device-width, initial-scale=1.0">
    15→<script src="https://cdn.tailwindcss.com"></script>
    16→</head>
    17→<body class="bg-gradient-to-br from-slate-900 to-slate-800 text-white flex items-center justify-center min-h-screen m-0 p-0">
    18→  <div class="max-w-3xl w-full px-12">
    19→    <h1 class="text-3xl font-bold mb-6 bg-gradient-to-r from-teal-400 to-cyan-300 bg-clip-text text-transparent">${title}</h1>
    20→    <div class="space-y-4 text-slate-300 text-lg leading-relaxed">
    21→${bodyHtml}
    22→    </div>
    23→  </div>
    24→</body>
    25→</html>`
    26→}
