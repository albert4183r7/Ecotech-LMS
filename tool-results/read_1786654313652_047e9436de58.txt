    70→function CircularProgress({ percentage }: { percentage: number }) {
    71→  const radius = 70;
    72→  const stroke = 10;
    73→  const normalizedRadius = radius - stroke / 2;
    74→  const circumference = normalizedRadius * 2 * Math.PI;
    75→  const strokeDashoffset =
    76→    circumference - (percentage / 100) * circumference;
    77→
    78→  return (
    79→    <div className="relative inline-flex items-center justify-center">
    80→      <svg
    81→        width={radius * 2}
    82→        height={radius * 2}
    83→        className="-rotate-90"
    84→      >
    85→        {/* Background track */}
    86→        <circle
    87→          cx={radius}
    88→          cy={radius}
    89→          r={normalizedRadius}
    90→          fill="none"
    91→          stroke="currentColor"
    92→          strokeWidth={stroke}
    93→          className="text-muted/30"
    94→        />
    95→        {/* Progress arc with gradient stroke */}
    96→        <circle
    97→          cx={radius}
    98→          cy={radius}
    99→          r={normalizedRadius}
   100→          fill="none"
   101→          stroke="url(#progress-gradient)"
   102→          strokeWidth={stroke}
   103→          strokeLinecap="round"
   104→          strokeDasharray={circumference}
   105→          strokeDashoffset={strokeDashoffset}
   106→          className="transition-all duration-700 ease-out"
   107→        />
   108→        <defs>
   109→          <linearGradient
   110→            id="progress-gradient"
   111→            x1="0%"
   112→            y1="0%"
   113→            x2="100%"
   114→            y2="100%"
   115→          >
   116→            <stop offset="0%" stopColor="#06b6d4" />
   117→            <stop offset="50%" stopColor="#0891b2" />
   118→            <stop offset="100%" stopColor="#0d9488" />
   119→          </linearGradient>
   120→        </defs>
   121→      </svg>
   122→      <div className="absolute flex flex-col items-center justify-center">
   123→        <span className="text-3xl font-extrabold tracking-tight text-foreground">
   124→          {percentage}%
   125→        </span>
   126→        <span className="mt-0.5 text-[11px] font-medium text-muted-foreground">
   127→          completed
   128→        </span>
   129→      </div>
   130→    </div>
   131→  );
   132→}

   134→/* ------------------------------------------------------------------ */
   135→/*  Animated Counter Hook                                              */
   136→/* ------------------------------------------------------------------ */
   137→
   138→function useAnimatedCounter(target: number, duration = 800) {
   139→  const [count, setCount] = useState(target);
   140→
   141→  useEffect(() => {
   142→    if (target === 0) return;
   143→    let start = 0;
   144→    const startTime = performance.now();
   145→    const step = (now: number) => {
   146→      const elapsed = now - startTime;
   147→      const progress = Math.min(elapsed / duration, 1);
   148→      const eased = 1 - Math.pow(1 - progress, 3);
   149→      start = Math.round(eased * target);
   150→      setCount(start);
   151→      if (progress < 1) {
   152→        frameRef.current = requestAnimationFrame(step);
   153→      }
   154→    };
   155→    frameRef.current = requestAnimationFrame(step);
   156→    return () => cancelAnimationFrame(frameRef.current);
   157→  }, [target, duration]);
   158→
   159→  return count;
   160→}

   259→function StreakCalendar({ streak, bestStreak }: { streak: number; bestStreak: number }) {
   260→  const [activityMap, setActivityMap] = useState<Map<string, number>>(new Map());
   261→  const [days, setDays] = useState<Array<{ date: string; day: string; isToday: boolean }>>([]);
   262→  const [maxMinutes, setMaxMinutes] = useState(1);
   263→
   264→  useEffect(() => {
   265→    async function fetchActivity() {
   266→      try {
   267→        const res = await fetch('/api/activity?userId=user_demo_001&weeks=5');
   268→        const json = await res.json();
   269→        if (json.success) {
   270→          // Build 30-day grid from the response
   271→          const allDays: ActivityDayEntry[] = [];
   272→          const today = new Date();
   273→          today.setHours(0, 0, 0, 0);
   274→          const start = new Date(today);
   275→          start.setDate(start.getDate() - 29);
   276→
   277→          const resMap = new Map<string, number>();
   278→          for (const d of (json.data as Activity30Data).dailyData) {
   279→            resMap.set(d.date, d.minutes);
   280→          }
   281→
   282→          for (let i = 0; i < 30; i++) {
   283→            const d = new Date(start);
   284→            d.setDate(d.getDate() + i);
   285→            const dateStr = d.toISOString().split('T')[0];
   286→            allDays.push({
   287→              date: dateStr,
   288→              day: d.toLocaleDateString('en-US', { weekday: 'narrow' }),
   289→              isToday: dateStr === today.toISOString().split('T')[0],
   290→            });
   291→          }
   292→
   293→          setDays(allDays);
   294→          setActivityMap(resMap);
   295→          const max = Math.max(...Array.from(resMap.values()), 1);
   296→          setMaxMinutes(max);
   297→        }
   298→      } catch { /* silent */ }
   299→    }
   300→    fetchActivity();
   301→  }, []);
   302→
   303→  function getIntensity(date: string): number {
   304→    const mins = activityMap.get(date) || 0;
   305→    if (mins === 0) return 0;
   306→    return Math.min(1, mins / maxMinutes);
   307→  }
   308→
   309→  function getCellColor(intensity: number): string {
   310→    if (intensity === 0) return 'bg-muted/40';
   311→    if (intensity <= 0.25) return 'bg-cyan-300/40 dark:bg-cyan-700/40';
   312→    if (intensity <= 0.5) return 'bg-cyan-400/60 dark:bg-cyan-600/50';
   313→    if (intensity <= 0.75) return 'bg-cyan-500/70 dark:bg-cyan-500/60';
   314→    return 'bg-cyan-600 dark:bg-cyan-400';
   315→  }
   316→
   317→  return (
   318→    <Card className="border-border/50">
   319→      <CardHeader className="pb-3">
   320→        <div className="flex items-center justify-between">
   321→          <CardTitle className="flex items-center gap-2 text-base font-semibold">
   322→            <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-gradient-to-br from-orange-500 to-amber-500">
   323→              <Flame className="h-4 w-4 text-white" />
   324→            </div>
   325→            Learning Streak
   326→          </CardTitle>
   327→          <div className="flex items-center gap-3">
   328→            <div className="flex items-center gap-1.5 text-sm font-bold text-orange-600 dark:text-orange-400">
   329→              <span className="text-base">🔥</span>
   330→              <span className="count-up">{streak}</span>
   331→              <span className="text-xs font-medium text-muted-foreground">days</span>
   332→            </div>
   333→          </div>
   334→        </div>
   335→      </CardHeader>
   336→      <CardContent className="space-y-4">
   337→        {/* Streak stats row */}
   338→        <div className="flex items-center justify-between">
