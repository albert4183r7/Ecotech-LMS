   103→   103→/*  Seeded random for consistent sparkline data                        */
   104→   104→/* ------------------------------------------------------------------ */
   105→   105→
   106→   106→function seededRandom(seed: number) {
   107→   107→  let s = seed;
   108→   108→  return () => {
   109→   109→    s = (s * 16807 + 0) % 2147483647;
   110→   110→    return (s - 1) / 2147483646;
   111→   111→  };
   112→   112→}

   118→   118→function SparklineChart({
   119→   119→  data,
   120→   120→  gradientFrom,
   121→   121→  gradientTo,

   137→   137→              `bg-gradient-to-t ${gradientFrom} ${gradientTo}`
   138→   138→            )}
   139→   139→            style={{
   140→   140→              height: `${height}%`,
   141→   141→              opacity: 0.5 + (i / data.length) * 0.5,
   142→   142→              animationDelay: `${i * 60}ms`,
   143→   143→            }}
   144→   144→          />

   266→   266→
   267→   267→function DonutChart({
   268→   268→  data,
   269→   269→  total,
   270→   270→}: {
   271→   271→  data: { categoryName: string; categoryColor: string | null; courseCount: number }[];
   272→   272→  total: number;
   273→   273→}) {
   274→   274→  // Build conic-gradient
   275→   275→  const gradientParts: string[] = [];
   276→   276→  let currentAngle = 0;
   277→   277→
   278→   278→  const DONUT_COLORS = [
   279→   279→    "#14b8a6",
   280→   280→    "#f59e0b",
   281→   281→    "#ef4444",
   282→   282→    "#8b5cf6",
   283→   283→    "#ec4899",
   284→   284→    "#06b6d4",
   285→   285→    "#84cc16",
   286→   286→    "#f97316",
   287→   287→  ];
   288→   288→
   289→   289→  data.forEach((item, i) => {
   290→   290→    const color = item.categoryColor || DONUT_COLORS[i % DONUT_COLORS.length];
   291→   291→    const angle = total > 0 ? (item.courseCount / total) * 360 : 0;
   292→   292→    gradientParts.push(`${color} ${currentAngle}deg ${currentAngle + angle}deg`);
   293→   293→    currentAngle += angle;
   294→   294→  });
   295→   295→
   296→   296→  const gradientStr =
   297→   297→    gradientParts.length > 0
   298→   298→      ? `conic-gradient(${gradientParts.join(", ")})`
   299→   299→      : "conic-gradient(#e5e7eb 0deg 360deg)";
   300→   300→
   301→   301→  return (
   302→   302→    <div className="flex flex-col items-center gap-4">
   303→   303→      {/* Donut */}
   304→   304→      <div
   305→   305→        className="relative h-40 w-40 sm:h-48 sm:w-48 rounded-full stat-pop"
   306→   306→        style={{
   307→   307→          background: gradientStr,
   308→   308→        }}
   309→   309→      >
   310→   310→        {/* Inner circle (donut hole) */}
   311→   311→        <div className="absolute inset-0 m-auto h-[60%] w-[60%] rounded-full bg-card flex flex-col items-center justify-center shadow-inner">
   312→   312→          <span className="text-2xl font-bold text-foreground">{total}</span>
   313→   313→          <span className="text-[11px] text-muted-foreground">courses</span>
   314→   314→        </div>
   315→   315→      </div>
   316→   316→
   317→   317→      {/* Legend */}
   318→   318→      <div className="grid grid-cols-2 gap-x-4 gap-y-1.5 w-full max-w-[220px]">
   319→   319→        {data.map((item, i) => {
   320→   320→          const color =
   321→   321→            item.categoryColor || DONUT_COLORS[i % DONUT_COLORS.length];
   322→   322→          const pct = total > 0 ? Math.round((item.courseCount / total) * 100) : 0;
   323→   323→          return (
   324→   324→            <div key={item.categoryName} className="flex items-center gap-2">
   325→   325→              <span
   326→   326→                className="h-2.5 w-2.5 rounded-full shrink-0"
   327→   327→                style={{ backgroundColor: color }}
   328→   328→              />
   329→   329→              <div className="min-w-0">
   330→   330→                <span className="text-xs text-foreground truncate block">
   331→   331→                  {item.categoryName}
   332→   332→                </span>
   333→   333→                <span className="text-[10px] text-muted-foreground">
   334→   334→                  {pct}%
   335→   335→                </span>
   336→   336→              </div>
   337→   337→            </div>
   338→   338→          );
   339→   339→        })}
   340→   340→      </div>
   341→   341→    </div>
   342→   342→  );
   343→   343→}

   345→   345→/* ------------------------------------------------------------------ */
   346→   346→/*  Weekly Heatmap                                                     */
   347→   347→/* ------------------------------------------------------------------ */
   348→   348→
   349→   349→function WeeklyHeatmap({
   350→   350→  userId,
   351→   351→}: {
   352→   352→  userId: string;
   353→   353→}) {
   354→   354→  const [heatmapData, setHeatmapData] = useState<number[][]>([]);
   355→   355→  const [monthLabels, setMonthLabels] = useState<string[]>([]);
   356→   356→
   357→   357→  useEffect(() => {
   358→   358→    async function fetchHeatmap() {
   359→   359→      try {
   360→   360→        const res = await fetch(`/api/activity?userId=${userId}`);
   361→   361→        const json = await res.json();
   362→   362→        if (json.success && json.data && json.data.dailyData) {
   363→   363→          // Use the activity data to build a heatmap
   364→   364→          const activityMap: Record<string, number> = {};
   365→   365→          const activities = json.data.dailyData;
   366→   366→          for (const act of activities) {
   367→   367→            const dateStr = act.date;
   368→   368→            // Use minutes to derive activity level
   369→   369→            const minutes = act.minutes || 0;
   370→   370→            // Keep the max minutes per day
   371→   371→            const existing = activityMap[dateStr] || 0;
   372→   372→            activityMap[dateStr] = Math.max(existing, minutes);
   373→   373→          }
   374→   374→
   375→   375→          // Build 12-week × 7-day grid
   376→   376→          const grid: number[][] = [];
   377→   377→          const today = new Date();
   378→   378→          const months: string[] = [];
   379→   379→          const seenMonths = new Set<string>();
   380→   380→
   381→   381→          // Start from 12 weeks ago, aligned to Sunday
   382→   382→          const startDate = new Date(today);
   383→   383→          startDate.setDate(startDate.getDate() - 83); // ~12 weeks
   384→   384→          const dayOfWeek = startDate.getDay();
   385→   385→          startDate.setDate(startDate.getDate() - dayOfWeek); // Align to Sunday
   386→   386→
   387→   387→          for (let week = 0; week < 12; week++) {
   388→   388→            for (let day = 0; day < 7; day++) {
   389→   389→              if (day === 0) grid.push([]);
   390→   390→              const cellDate = new Date(startDate);
   391→   391→              cellDate.setDate(cellDate.getDate() + week * 7 + day);
   392→   392→              const dateStr = cellDate.toISOString().split("T")[0];
   393→   393→              grid[week].push(activityMap[dateStr] || 0);
   394→   394→
   395→   395→              // Convert minutes to 0-4 level for heatmap coloring
   396→   396→              const mins = activityMap[dateStr] || 0;
   397→   397→              if (mins === 0) {
   398→   398→                // already 0
   399→   399→              } else {
   400→   400→                // Normalize: 0=none, 1=1-30min, 2=31-60min, 3=61-90min, 4=91+min
   401→   401→                grid[week][day] = mins <= 30 ? 1 : mins <= 60 ? 2 : mins <= 90 ? 3 : 4;
   402→   402→              }
   403→   403→              if (day === 0) {
   404→   404→                const monthName = cellDate.toLocaleString("en-US", {
   405→   405→                  month: "short",
   406→   406→                });
   407→   407→                if (!seenMonths.has(monthName)) {
   408→   408→                  seenMonths.add(monthName);
   409→   409→                  months.push(monthName);
   410→   410→                } else {
   411→   411→                  months.push("");
   412→   412→                }
   413→   413→              }
   414→   414→            }
   415→   415→          }
   416→   416→
   417→   417→          setHeatmapData(grid);
   418→   418→          setMonthLabels(months);
   419→   419→        }
   420→   420→      } catch {
   421→   421→        // Generate fallback random data
   422→   422→        const rand = seededRandom(42);
   423→   423→        const grid: number[][] = [];
   424→   424→        for (let week = 0; week < 12; week++) {
   425→   425→          grid.push([]);
   426→   426→          for (let day = 0; day < 7; day++) {
   427→   427→            grid[week].push(Math.floor(rand() * 5));
   428→   428→          }
   429→   429→        }
   430→   430→        setHeatmapData(grid);
   431→   431→        setMonthLabels(["Jan", "", "Feb", "", "Mar", "", "Apr"]);
   432→   432→      }
   433→   433→    }
   434→   434→    fetchHeatmap();
   435→   435→  }, [userId]);
   436→   436→
   437→   437→  const dayLabels = ["Sun", "", "Tue", "", "Thu", "", "Sat"];
   438→   438→
   439→   439→  const getHeatColor = (level: number) => {
   440→   440→    switch (level) {
   441→   441→      case 0:
   442→   442→        return "bg-muted";
   443→   443→      case 1:
   444→   444→        return "bg-emerald-200 dark:bg-emerald-900/50";
   445→   445→      case 2:
   446→   446→        return "bg-emerald-300 dark:bg-emerald-800/60";
   447→   447→      case 3:
   448→   448→        return "bg-emerald-400 dark:bg-emerald-700/70";
   449→   449→      default:
   450→   450→        return "bg-emerald-500 dark:bg-emerald-600";
   451→   451→    }
   452→   452→  };
   453→   453→
   454→   454→  if (heatmapData.length === 0) return null;
   455→   455→
   456→   456→  return (
   457→   457→    <TooltipProvider delayDuration={200}>
   458→   458→      <div className="space-y-2">
   459→   459→        {/* Month labels */}
   460→   460→        <div className="flex gap-0.5 pl-8">
   461→   461→          {monthLabels.map((label, i) => (
   462→   462→            <div
   463→   463→              key={i}
   464→   464→              className="flex-1 min-w-0"
   465→   465→            >
   466→   466→              <span className="text-[9px] text-muted-foreground truncate">
   467→   467→                {label}
   468→   468→              </span>
   469→   469→            </div>
   470→   470→          ))}
   471→   471→        </div>
   472→   472→
   473→   473→        <div className="flex gap-1.5">
   474→   474→          {/* Day labels */}
   475→   475→          <div className="flex flex-col gap-0.5 shrink-0 pt-0">
   476→   476→            {dayLabels.map((label, i) => (
   477→   477→              <div key={i} className="h-[14px] flex items-center">
   478→   478→                <span className="text-[9px] text-muted-foreground w-7 text-right pr-1.5">
   479→   479→                  {label}
   480→   480→                </span>
   481→   481→              </div>
   482→   482→            ))}
   483→   483→          </div>
   484→   484→
   485→   485→          {/* Heatmap grid */}
   486→   486→          <div className="flex gap-[3px] flex-1 overflow-x-auto">
   487→   487→            {heatmapData.map((week, weekIdx) => (
   488→   488→              <div key={weekIdx} className="flex flex-col gap-[3px]">
   489→   489→                {week.map((level, dayIdx) => (
   490→   490→                  <Tooltip key={dayIdx}>
   491→   491→                    <TooltipTrigger asChild>
   492→   492→                      <div
   493→   493→                        className={cn(
   494→   494→                          "h-[14px] w-[14px] rounded-[3px] transition-colors hover:ring-1 hover:ring-foreground/20",
   495→   495→                          getHeatColor(level)
   496→   496→                        )}
   497→   497→                      />
   498→   498→                    </TooltipTrigger>
   499→   499→                    <TooltipContent side="top" className="text-xs">
   500→   500→                      {level > 0
   501→   501→                        ? `${level === 1 ? "<30" : level === 2 ? "31-60" : level === 3 ? "61-90" : "90+"} min`
   502→   502→                        : "No activity"}
   503→   503→                    </TooltipContent>
   504→   504→                  </Tooltip>
   505→   505→                ))}
   506→   506→              </div>
   507→   507→            ))}
   508→   508→          </div>
   509→   509→        </div>
   510→   510→
   511→   511→        {/* Legend */}
   512→   512→        <div className="flex items-center justify-end gap-1.5 text-[10px] text-muted-foreground">
   513→   513→          <span>Less</span>
   514→   514→          {[0, 1, 2, 3, 4].map((level) => (
   515→   515→            <div
   516→   516→              key={level}
   517→   517→              className={cn("h-[10px] w-[10px] rounded-[2px]", getHeatColor(level))}
   518→   518→            />
