    57→/** Count-up animation hook */
    58→function useCountUp(target: number, duration = 1200, enabled = true) {
    59→  const [value, setValue] = useState(0);
    60→  const ref = useRef<number | null>(null);
    61→
    62→  useEffect(() => {
    63→    if (!enabled) return;
    64→    const start = performance.now();
    65→    const animate = (now: number) => {
    66→      const elapsed = now - start;
    67→      const progress = Math.min(elapsed / duration, 1);
    68→      const eased = 1 - Math.pow(1 - progress, 3);
    69→      setValue(Math.round(eased * target));
    70→      if (progress < 1) ref.current = requestAnimationFrame(animate);
    71→    };
    72→    ref.current = requestAnimationFrame(animate);
    73→    return () => {
    74→      if (ref.current) cancelAnimationFrame(ref.current);
    75→    };
    76→  }, [target, duration, enabled]);
    77→
    78→  return value;
    79→}
