    51→    const fetchSlides = async () => {
    52→      setLoadingSlides(true);
    53→      try {
    54→        const res = await fetch(`/api/sections/${localState.sectionId}`);
    55→        if (!res.ok) return;
    56→        const json = await res.json();
    57→        if (json.success && json.data.content) {
    58→          const slides: SlideContent[] = Array.isArray(json.data.content)
    59→            ? json.data.content
    60→            : [];
    61→          setLocalState((prev) =>
    62→            prev
    63→              ? {
    64→                  ...prev,
    65→                  slides,
    66→                  totalPages: slides.length || prev.totalPages,
    67→                }
    68→              : prev
    69→          );
    70→        }
    71→      } catch {
    72→        // Silently fail
    73→      } finally {
    74→        setLoadingSlides(false);
    75→      }
    76→    };
    77→    fetchSlides();
    78→  }, [localState?.sectionId, localState?.slides.length]);
    79→
    80→  /** Navigate slides with transition animation */
    81→  const goToSlide = useCallback(
    82→    (index: number) => {
    83→      if (!localState || isAnimating) return;
    84→      const next = Math.max(0, Math.min(index, localState.totalPages - 1));
    85→      if (next === localState.currentSlide) return;
    86→      setSlideDirection(next > localState.currentSlide ? "forward" : "back");
    87→      setIsAnimating(true);
    88→      // Small delay for exit animation, then update slide
    89→      setTimeout(() => {
    90→        setLocalState((prev) =>
    91→          prev ? { ...prev, currentSlide: next } : prev
    92→        );
    93→        setIsAnimating(false);
    94→      }, 150);
    95→    },
    96→    [localState, isAnimating]
    97→  );
    98→
    99→  /** Persist progress to API when slide changes */
   100→  useEffect(() => {
   101→    if (!localState || !userId) return;
   102→    const saveProgress = async () => {
   103→      try {
   104→        // Get enrollment ID from the API
   105→        const enrollRes = await fetch(`/api/enrollments?userId=${userId}`);
   106→        const enrollJson = await enrollRes.json();
   107→        if (enrollJson.success && Array.isArray(enrollJson.data)) {
   108→          const enrollment = enrollJson.data.find(
   109→            (e: Record<string, unknown>) => e.courseId === localState.courseId
   110→          );
   111→          if (enrollment) {
   112→            const completed = localState.currentSlide >= localState.totalPages - 1;
   113→            await fetch("/api/progress", {
   114→              method: "POST",
   115→              headers: { "Content-Type": "application/json" },
   116→              body: JSON.stringify({
   117→                enrollmentId: enrollment.id,
   118→                sectionId: localState.sectionId,
   119→                currentPage: localState.currentSlide + 1,
   120→                completed,
   121→              }),
   122→            });
   123→          }
   124→        }
   125→      } catch {
   126→        // Progress saving is best-effort
   127→      }
   128→    };
   129→    // Debounce to avoid excessive API calls
   130→    const timer = setTimeout(saveProgress, 500);
   131→    return () => clearTimeout(timer);
   132→  }, [localState?.currentSlide]);
   133→
   134→  const goPrev = () => goToSlide((localState?.currentSlide ?? 0) - 1);
   135→  const goNext = () => goToSlide((localState?.currentSlide ?? 0) + 1);
   136→
   137→  // ─── Keyboard shortcuts ──────────────────────────────────────
   138→  useEffect(() => {
   139→    function handleKeyDown(e: KeyboardEvent) {
   140→      const tag = (e.target as HTMLElement)?.tagName;
   141→      if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return;
   142→
   143→      switch (e.key) {
   144→        case "Escape":
   145→          e.preventDefault();
   146→          goBack();
   147→          break;
   148→        case "ArrowLeft":
   149→          e.preventDefault();
   150→          goPrev();
   151→          break;
   152→        case "ArrowRight":
   153→          e.preventDefault();
   154→          goNext();
   155→          break;
   156→      }
   157→    }
   158→
   159→    document.addEventListener("keydown", handleKeyDown);
   160→    return () => document.removeEventListener("keydown", handleKeyDown);
   161→  }, [goBack, goPrev, goNext]);
   162→
   163→  /** Zoom controls */
   164→  const zoomIn = () => setZoom((z) => Math.min(z + ZOOM_STEP, MAX_ZOOM));
   165→  const zoomOut = () => setZoom((z) => Math.max(z - ZOOM_STEP, MIN_ZOOM));
   166→  const zoomFit = () => setZoom(100);
   167→
   168→  // ─── Confetti on section completion ───────────
   169→  useEffect(() => {
   170→    if (!localState) return;

   172→    if (isOnLastSlide && !confettiShownRef.current) {
   173→      confettiShownRef.current = true;
   174→      setShowConfetti(true);
   175→      toast.success("🎉 You completed the course! Great job!");
   176→      // Auto-cleanup after 3.5s (animation lasts ~3s + buffer)
   177→      const timer = setTimeout(() => setShowConfetti(false), 3500);
   178→      return () => clearTimeout(timer);
   179→    }
   180→  }, [localState?.currentSlide]);
   181→
   182→  // Reset confetti flag when classroom state changes (new section)
   183→  useEffect(() => {
   184→    confettiShownRef.current = false;
   185→    setShowConfetti(false);
   186→  }, [classroomState?.sectionId]);
   187→
   188→  // ─── No state guard ─────────────────────────────
   189→  if (!localState) {
   190→    return (
   191→      <div className="flex h-screen items-center justify-center bg-muted/30">
   192→        <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   193→      </div>
   194→    );
   195→  }
   196→
   197→  const currentSlide =
   198→    localState.slides[localState.currentSlide] ?? null;
   199→  const isFirst = localState.currentSlide === 0;
   200→  const isLast = localState.currentSlide >= localState.totalPages - 1;
   201→  const progressPercent = localState.totalPages > 1
   202→    ? Math.round(((localState.currentSlide + 1) / localState.totalPages) * 100)
   203→    : 100;
   204→
   205→  return (
   206→    <div className="flex h-screen flex-col bg-muted/30 relative">
   207→      {/* ─── Confetti Celebration Overlay ─────── */}
   208→      {showConfetti && <ConfettiCelebration />}
   209→      {/* ─── Top Progress Bar ─────────────────────── */}
   210→      <div className="shrink-0 h-1 w-full bg-muted overflow-hidden">
   211→        <div
   212→          className="h-full bg-gradient-to-r from-primary to-accent transition-all duration-500 ease-out"
   213→          style={{ width: `${progressPercent}%` }}
   214→        />
   215→      </div>
   216→
   217→      {/* ─── Top Bar ──────────────────────────────── */}
   218→      <header className="flex h-14 shrink-0 items-center justify-between border-b bg-card px-4 sm:px-6">
   219→        {/* Logo, Section Title, Course Title */}
   220→        <div className="flex items-center gap-3 min-w-0">
   221→          <div className="flex items-center gap-2 shrink-0">
   222→            <div className="flex h-7 w-7 items-center justify-center rounded-md bg-primary">
   223→              <GraduationCap className="h-4 w-4 text-primary-foreground" />
   224→            </div>
   225→            <span className="hidden sm:inline text-sm font-bold text-primary">
   226→              OpenClass
   227→            </span>
   228→          </div>
   229→          <Separator orientation="vertical" className="h-5" />
   230→          <div className="min-w-0">
   231→            <h2 className="text-sm font-medium text-foreground truncate leading-tight">
   232→              {localState.sectionTitle}
   233→            </h2>
   234→            <p className="text-xs text-muted-foreground truncate leading-tight">
   235→              {localState.courseTitle}
   236→            </p>
   237→          </div>
   238→        </div>
   239→
   240→        {/* Slide Pagination - Pill Badge */}
   241→        <span className="shrink-0 inline-flex items-center gap-1 rounded-full bg-primary/10 px-3 py-1 text-sm font-semibold text-primary tabular-nums">
   242→          {localState.currentSlide + 1}
   243→          <span className="text-primary/40 font-normal">/</span>
   244→          {localState.totalPages}
   245→        </span>
   246→      </header>
   247→
   248→      {/* ─── Slide Content Area ────────────────────── */}
   249→      <div className="flex-1 overflow-auto flex items-start justify-center py-8 px-4 sm:px-6">
   250→        <div

   252→            bg-card rounded-2xl shadow-lg border w-full max-w-3xl
   253→            ring-1 ring-black/5
   254→            dark:ring-white/5
   255→            paper-texture
   256→            ${isAnimating
   257→              ? (slideDirection === "forward" ? "slide-exit" : "opacity-0 -translate-x-2 transition-all duration-150")
   258→              : "slide-enter"
   259→            }
   260→          `}
   261→          style={{
   262→            transform: `scale(${zoom / 100})`,
   263→            transformOrigin: "top center",
   264→          }}
   265→        >
   266→          {loadingSlides ? (
   267→            <div className="flex items-center justify-center py-32">
   268→              <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   269→            </div>
   270→          ) : currentSlide ? (
   271→            <div className="p-8 sm:p-12 max-w-2xl mx-auto">
   272→              <SlideRenderer slide={currentSlide} />
   273→            </div>
   274→          ) : (
   275→            <div className="flex items-center justify-center py-32 text-muted-foreground">

   400→/** LMS color palette for confetti particles */
   401→const CONFETTI_COLORS = [
   402→  "#3882f6", // blue
   403→  "#14b8a6", // teal
   404→  "#22d3ee", // cyan
   405→  "#10b981", // emerald

   411→
   412→type ParticleShape = "circle" | "rectangle" | "triangle";
   413→
   414→interface ConfettiParticle {
   415→  id: number;
   416→  left: number;
   417→  color: string;
   418→  shape: ParticleShape;
   419→  size: number;
   420→  fallDuration: string;
   421→  fallDelay: string;
   422→  drift: string;
   423→  spin: string;
   424→}
   425→
   426→function ConfettiCelebration() {
   427→  const particles = useMemo<ConfettiParticle[]>(() => {
   428→    const result: ConfettiParticle[] = [];
   429→    const shapes: ParticleShape[] = ["circle", "rectangle", "triangle"];
   430→    const count = 80;
   431→
   432→    for (let i = 0; i < count; i++) {
   433→      const shape = shapes[i % 3 === 0 ? 0 : i % 3 === 1 ? 1 : 2];
   434→      result.push({
   435→        id: i,
   436→        left: Math.random() * 100,
   437→        color: CONFETTI_COLORS[i % CONFETTI_COLORS.length],
   438→        shape,
   439→        size: Math.floor(Math.random() * 8) + 6, // 6-14px
   440→        fallDuration: `${(Math.random() * 1.5 + 2).toFixed(2)}s`, // 2-3.5s
   441→        fallDelay: `${(Math.random() * 0.8).toFixed(2)}s`, // 0-0.8s
   442→        drift: `${(Math.random() * 120 - 60).toFixed(0)}px`, // -60 to +60px
   443→        spin: `${Math.floor(Math.random() * 720 + 360)}deg`, // 360-1080deg
   444→      });
   445→    }
   446→    return result;
   447→  }, []);
   448→
   449→  return (
   450→    <>
   451→      {/* Confetti particles layer */}
   452→      <div className="confetti-container" aria-hidden="true">
   453→        {particles.map((p) => {
   454→          if (p.shape === "triangle") {
   455→            return (
   456→              <div
   457→                key={p.id}
   458→                className="confetti-particle confetti-triangle"
   459→                style={{
   460→                  left: `${p.left}%`,
   461→                  "--confetti-color": p.color,
   462→                  "--tri-size": `${p.size}px`,
   463→                  "--fall-duration": p.fallDuration,
   464→                  "--fall-delay": p.fallDelay,
   465→                  "--drift": p.drift,
   466→                  "--spin": p.spin,
   467→                } as React.CSSProperties}
   468→              />
   469→            );
   470→          }
   471→          return (
   472→            <div
   473→              key={p.id}
   474→              className={`confetti-particle ${p.shape === "circle" ? "confetti-circle" : "confetti-rectangle"}`}
   475→              style={{
   476→                left: `${p.left}%`,
   477→                width: p.shape === "rectangle" ? `${p.size * 1.4}px` : `${p.size}px`,
   478→                height: `${p.size}px`,
   479→                backgroundColor: p.color,
   480→                "--fall-duration": p.fallDuration,
   481→                "--fall-delay": p.fallDelay,
   482→                "--drift": p.drift,
   483→                "--spin": p.spin,
   484→              } as React.CSSProperties}
   485→            />
   486→          );
   487→        })}
   488→      </div>
   489→
   490→      {/* Congratulations message */}
   491→      <div className="confetti-message" role="status" aria-label="Congratulations!">
   492→        <div className="confetti-message-text flex flex-col items-center gap-3">
   493→          <div className="flex h-16 w-16 items-center justify-center rounded-2xl bg-white/90 dark:bg-zinc-800/90 shadow-lg">
   494→            <span className="text-4xl">🎉</span>
   495→          </div>
   496→          <div className="rounded-2xl bg-white/90 dark:bg-zinc-800/90 px-8 py-5 shadow-xl text-center">
   497→            <h2 className="text-2xl sm:text-3xl font-bold text-foreground">
   498→              🎉 Congratulations!
   499→            </h2>
   500→            <p className="mt-2 text-sm text-muted-foreground">
   501→              You&apos;ve completed this section!
   502→            </p>
   503→          </div>
   504→        </div>
   505→      </div>
   506→    </>
   507→  );
   508→}

   629→      <div className="mb-6 flex items-center gap-3">
   630→        <div className="h-1 w-8 rounded-full bg-gradient-to-r from-primary to-accent" />
   631→        <h2 className="text-2xl font-bold text-foreground">{slide.title}</h2>
   632→      </div>
   633→      {slide.items && slide.items.length > 0 && (
   634→        <ul className="space-y-3">
   635→          {slide.items.map((item, i) => (
   636→            <li key={i} className="flex items-start gap-3 rounded-lg bg-muted/40 p-3 transition-colors hover:bg-muted/60">
   637→              <span className="mt-1 flex h-5 w-5 shrink-0 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary">

   639→              </span>
   640→              <div>
   641→                {item.heading && (
   642→                  <span className="font-semibold text-foreground text-sm">

   644→                  </span>
   645→                )}
   646→                <span className="text-sm text-muted-foreground leading-relaxed">
   647→                  {item.text}
   648→                </span>
   649→              </div>
   650→            </li>
   651→          ))}
   652→        </ul>

   689→// ─── Quiz Slide (Interactive) ────────────────────
   690→function QuizSlide({ slide }: { slide: SlideContent }) {
   691→  const [selectedIdx, setSelectedIdx] = useState<number | null>(null);
   692→  const hasSubmitted = selectedIdx !== null;
   693→
   694→  // Determine correct answer: look for item with icon === "check" or first item
   695→  const correctIdx = slide.items?.findIndex((item) => item.icon === "check") ?? 0;
   696→  const isCorrect = selectedIdx === correctIdx;
   697→
   698→  return (
   699→    <div>
   700→      <div className="mb-6 flex items-center gap-3">
   701→        <div className="h-1 w-8 rounded-full bg-gradient-to-r from-primary to-accent" />
   702→        <h2 className="text-2xl font-bold text-foreground">{slide.title}</h2>
   703→      </div>
   704→      {slide.items && slide.items.length > 0 && (
   705→        <div className="space-y-3">
   706→          {slide.items.map((item, i) => {
   707→            const letter = String.fromCharCode(65 + i);
   708→            const isSelected = selectedIdx === i;
   709→            let borderColor = "border-border hover:border-primary/30 hover:bg-primary/5";
   710→            if (hasSubmitted) {
   711→              if (i === correctIdx) borderColor = "border-emerald-500 bg-emerald-500/10";
   712→              else if (isSelected && !isCorrect) borderColor = "border-red-400 bg-red-400/10";
   713→              else borderColor = "border-border opacity-60";
   714→            } else if (isSelected) {
   715→              borderColor = "border-primary bg-primary/10";
   716→            }
   717→
   718→            return (
   719→              <button
   720→                key={i}
   721→                type="button"
   722→                className={`w-full rounded-lg border p-4 text-left transition-all duration-200 ${borderColor}`}
   723→                onClick={() => {
   724→                  if (!hasSubmitted) setSelectedIdx(i);
   725→                }}
   726→                disabled={hasSubmitted}
   727→              >
   728→                <div className="flex items-start gap-3">
   729→                  <span className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold ${
   730→                    hasSubmitted && i === correctIdx
   731→                      ? "bg-emerald-500 text-white"
   732→                      : hasSubmitted && isSelected && !isCorrect
   733→                        ? "bg-red-400 text-white"
   734→                        : "bg-primary/10 text-primary"
   735→                  }`}>
   736→                    {hasSubmitted && i === correctIdx ? "✓" : hasSubmitted && isSelected && !isCorrect ? "✗" : letter}
   737→                  </span>
   738→                  <div className="flex-1">
   739→                    <p className="text-sm font-medium text-foreground">
   740→                      {item.heading || item.text}
   741→                    </p>
   742→                    {item.heading && item.text !== item.heading && (
   743→                      <p className="text-xs text-muted-foreground mt-1">{item.text}</p>
   744→                    )}
   745→                  </div>
   746→                </div>
   747→              </button>
   748→            );
   749→          })}
   750→
   751→          {/* Feedback after selection */}
   752→          {hasSubmitted && (
   753→            <div className={`mt-4 rounded-lg border p-4 ${
   754→              isCorrect
   755→                ? "border-emerald-200 bg-emerald-50 dark:border-emerald-800 dark:bg-emerald-950/50"
   756→                : "border-amber-200 bg-amber-50 dark:border-amber-800 dark:bg-amber-950/50"
   757→            }`}>
   758→              <p className={`text-sm font-semibold ${
   759→                isCorrect ? "text-emerald-700 dark:text-emerald-400" : "text-amber-700 dark:text-amber-400"
   760→              }`}>
   761→                {isCorrect ? "🎉 Correct! Well done!" : "❌ Not quite right."}
   762→              </p>
   763→              {!isCorrect && (
   764→                <p className="mt-1 text-xs text-muted-foreground">
   765→                  The correct answer is <strong>{String.fromCharCode(65 + correctIdx)}</strong>.
   766→                </p>
   767→              )}
   768→            </div>
   769→          )}
   770→        </div>
   771→      )}
   772→    </div>
   773→  );
   774→}
