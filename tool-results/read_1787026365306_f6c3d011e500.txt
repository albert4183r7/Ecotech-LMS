    59→/** Count-up animation hook */
    60→function useCountUp(target: number, duration = 1200, enabled = true) {
    61→  const [value, setValue] = useState(0);
    62→  const ref = useRef<number | null>(null);
    63→
    64→  useEffect(() => {
    65→    if (!enabled) return;
    66→    const start = performance.now();
    67→    const animate = (now: number) => {
    68→      const elapsed = now - start;
    69→      const progress = Math.min(elapsed / duration, 1);
    70→      const eased = 1 - Math.pow(1 - progress, 3);
    71→      setValue(Math.round(eased * target));
    72→      if (progress < 1) ref.current = requestAnimationFrame(animate);
    73→    };
    74→    ref.current = requestAnimationFrame(animate);
    75→    return () => {
    76→      if (ref.current) cancelAnimationFrame(ref.current);
    77→    };
    78→  }, [target, duration, enabled]);
    79→
    80→  return value;
    81→}
