    63→    63→function timeAgo(dateStr: string): string {
    64→    64→  const now = new Date();
    65→    65→  const date = new Date(dateStr);
    66→    66→  const diffMs = now.getTime() - date.getTime();
    67→    67→  const diffSec = Math.floor(diffMs / 1000);
    68→    68→  const diffMin = Math.floor(diffSec / 60);
    69→    69→  const diffHr = Math.floor(diffMin / 60);
    70→    70→  const diffDay = Math.floor(diffHr / 24);
    71→    71→  const diffWeek = Math.floor(diffDay / 7);
    72→    72→  const diffMonth = Math.floor(diffDay / 30);
    73→    73→
    74→    74→  if (diffSec < 60) return "just now";
    75→    75→  if (diffMin < 60) return `${diffMin}m ago`;
    76→    76→  if (diffHr < 24) return `${diffHr}h ago`;
    77→    77→  if (diffDay < 7) return `${diffDay}d ago`;
    78→    78→  if (diffWeek < 5) return `${diffWeek}w ago`;
    79→    79→  return `${diffMonth}mo ago`;
    80→    80→}

    83→    83→/*  Seeded random for consistent sparkline data                        */
    84→    84→/* ------------------------------------------------------------------ */
    85→    85→
    86→    86→function seededRandom(seed: number) {
    87→    87→  let s = seed;
    88→    88→  return () => {
    89→    89→    s = (s * 16807 + 0) % 2147483647;
    90→    90→    return (s - 1) / 2147483646;
    91→    91→  };
    92→    92→}

    98→    98→function SparklineChart({
    99→    99→  data,
   100→   100→  gradientFrom,
   101→   101→  gradientTo,
   102→   102→}: {
   103→   103→  data: number[];
   104→   104→  gradientFrom: string;
   105→   105→  gradientTo: string;
   106→   106→}) {
   107→   107→  const max = Math.max(...data, 1);
   108→   108→  return (
   109→   109→    <div className="flex items-end gap-[3px] h-8">
   110→   110→      {data.map((val, i) => {
   111→   111→        const height = Math.max(4, (val / max) * 100);
   112→   112→        return (
   113→   113→          <div
   114→   114→            key={i}
   115→   115→            className={cn(
   116→   116→              "w-[6px] rounded-full transition-all duration-500",
   117→   117→              `bg-gradient-to-t ${gradientFrom} ${gradientTo}`
   118→   118→            )}
   119→   119→            style={{
   120→   120→              height: `${height}%`,
   121→   121→              opacity: 0.5 + (i / data.length) * 0.5,
   122→   122→              animationDelay: `${i * 60}ms`,
   123→   123→            }}
   124→   124→          />
   125→   125→        );
   126→   126→      })}
   127→   127→    </div>
   128→   128→  );
   129→   129→}
