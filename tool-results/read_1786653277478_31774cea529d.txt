   102→  const base = read ? "bg-muted/60" : "bg-primary/10";
   103→  const colorMap: Record<string, string> = {
   104→    info: read ? "bg-blue-500/10" : "bg-blue-500/15",
   105→    success: read ? "bg-emerald-500/10" : "bg-emerald-500/15",

   127→/** Format relative time for notifications */
   128→function formatRelativeTime(dateStr: string): string {
   129→  const now = new Date();
   130→  const date = new Date(dateStr);
   131→  const diffMs = now.getTime() - date.getTime();
   132→  const diffMins = Math.floor(diffMs / 60000);
   133→  const diffHours = Math.floor(diffMs / 3600000);
   134→  const diffDays = Math.floor(diffMs / 86400000);
   135→
   136→  if (diffMins < 1) return "Just now";
   137→  if (diffMins < 60) return `${diffMins}m ago`;
   138→  if (diffHours < 24) return `${diffHours}h ago`;
   139→  if (diffDays < 7) return `${diffDays}d ago`;
   140→  return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   141→}

   143→/** Group notifications by date category: Today, Yesterday, Earlier */
   144→function groupNotificationsByDate(items: NotificationItem[]) {
   145→  const now = new Date();
   146→  const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   147→  const yesterdayStart = todayStart - 86400000;
   148→
   149→  const today: NotificationItem[] = [];
   150→  const yesterday: NotificationItem[] = [];
   151→  const earlier: NotificationItem[] = [];
   152→
   153→  for (const item of items) {
   154→    const t = new Date(item.createdAt).getTime();
   155→    if (t >= todayStart) today.push(item);
   156→    else if (t >= yesterdayStart) yesterday.push(item);
   157→    else earlier.push(item);
   158→  }
   159→
   160→  return { today, yesterday, earlier };
   161→}

   163→/** Notification group with date header */
   164→function NotificationGroup({
   165→  label,
   166→  items,
   167→  onNotificationClick,
   168→}: {
   169→  label: string;
   170→  items: NotificationItem[];
   171→  onNotificationClick: (notif: NotificationItem) => void;
   172→}) {
   173→  if (items.length === 0) return null;
   174→
   175→  return (
   176→    <div>
   177→      <p className="px-3 pt-2 pb-1 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground/60">
   178→        {label}
   179→      </p>
   180→      {items.map((notif) => (
   181→        <button
   182→          key={notif.id}
   183→          type="button"
   184→          onClick={() => onNotificationClick(notif)}
   185→          className={cn(
   186→            "group flex w-full items-start gap-3 rounded-lg px-3 py-3 text-left transition-colors notif-appear",
   187→            notif.read
   188→              ? "hover:bg-muted/50"
   189→              : "bg-primary/5 hover:bg-primary/8"
   190→          )}
   191→        >
   192→          {/* Left border accent for unread */}
   193→          {!notif.read && (
   194→            <div className={cn(
   195→              "absolute left-1 top-3 bottom-3 w-0.5 rounded-full",
   196→              NotificationBorderAccent({ type: notif.type })
   197→            )} style={{ position: "absolute" }} />
   198→          )}
   199→          <div className={cn(
   200→            "mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-lg transition-colors relative",
   201→            NotificationIconBg({ type: notif.type, read: notif.read })
   202→          )}>
   203→            <NotificationIcon type={notif.type} />
   204→          </div>
   205→          <div className="min-w-0 flex-1">
   206→            <div className="flex items-start justify-between gap-2">
   207→              <p className={cn(
   208→                "text-sm leading-snug",
   209→                notif.read
   210→                  ? "text-muted-foreground font-medium"
   211→                  : "text-foreground font-semibold"
   212→              )}>
   213→                {notif.title}
   214→              </p>
   215→              {!notif.read && (
   216→                <div className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary" />
   217→              )}
   218→            </div>
   219→            <p className="mt-0.5 text-xs text-muted-foreground leading-relaxed line-clamp-2">
   220→              {notif.message}
   221→            </p>
   222→            <p className="mt-1 text-[10px] text-muted-foreground/60">
   223→              {formatRelativeTime(notif.createdAt)}
   224→            </p>
   225→          </div>
   226→        </button>
   227→      ))}
   228→    </div>
   229→  );
   230→}

   250→/** Parse link string into view + id for navigation */
   251→function parseNotificationLink(link: string | null): { view: ViewName; courseId?: string } | null {
   252→  if (!link) return null;
   253→  if (link === "profile") return { view: "profile" };
   254→  if (link.startsWith("course-detail:")) {
   255→    const courseId = link.replace("course-detail:", "");
   256→    return { view: "course-detail", courseId };
   257→  }
   258→  return null;
   259→}

   277→  const notifRef = useRef<HTMLDivElement>(null);

   279→  // Theme toggle state for cycling animation
   280→  const [themeIconRotating, setThemeIconRotating] = useState(false);
   281→
   282→  // Breadcrumb: derive current page label
   283→  const currentPageLabel = getViewLabel(currentView);
   284→
   285→  // ============================================
   286→  // Fetch notifications from API
   287→  // ============================================
   288→  useEffect(() => {
   289→    let cancelled = false;
   290→    async function fetchNotifications() {
   291→      setNotificationsLoading(true);
   292→      try {
   293→        const res = await fetch(`/api/notifications?userId=${encodeURIComponent(currentUserId)}`);
   294→        if (!res.ok) throw new Error("Failed to fetch");
   295→        const data = await res.json();
   296→        if (!cancelled) {
   297→          setNotifications(data.notifications || []);
   298→        }
   299→      } catch {
   300→        // Silently fail — notifications will just be empty
   301→      } finally {
   302→        if (!cancelled) {
   303→          setNotificationsLoading(false);
   304→        }
   305→      }
   306→    }
   307→    fetchNotifications();
   308→    return () => { cancelled = true; };
   309→  }, [currentUserId]);
   310→
   311→  // Close mobile menu and search on view change
   312→  useEffect(() => {
   313→    const handleViewCleanup = () => {
   314→      setMobileMenuOpen(false);
   315→      setSearchExpanded(false);
   316→    };
   317→    handleViewCleanup();
   318→  }, [currentView]);

   320→  // Collapse search when clicking outside

   322→    if (!searchExpanded) return;

   324→      if (searchInputRef.current && !searchInputRef.current.parentElement?.contains(e.target as Node)) {
   325→        setSearchExpanded(false);
   326→      }
   327→    }
   328→    document.addEventListener("mousedown", handleClickOutside);
   329→    return () => document.removeEventListener("mousedown", handleClickOutside);
   330→  }, [searchExpanded]);
   331→
   332→  // Close notification panel on outside click
   333→  useEffect(() => {

   335→      if (notifRef.current && !notifRef.current.contains(e.target as Node)) {
   336→        setShowNotifications(false);
   337→      }
   338→    }
   339→    if (showNotifications) {
   340→      document.addEventListener("mousedown", handleClickOutside);
   341→    }
   342→    return () => document.removeEventListener("mousedown", handleClickOutside);
   343→  }, [showNotifications]);
   344→
   345→  /** Toggle notification panel */
   346→  const toggleNotifications = useCallback(() => {
   347→    if (!showNotifications) {
   348→      setBellAnimating(true);
   349→      setTimeout(() => setBellAnimating(false), 600);
   350→    }
   351→    setShowNotifications((prev) => !prev);
   352→  }, [showNotifications]);
   353→
   354→  /** Mark a single notification as read via API */

   356→    // Navigate if link is present
   357→    const parsed = parseNotificationLink(notif.link);
   358→    if (parsed) {
   359→      if (parsed.courseId) {
   360→        openCourseDetail(parsed.courseId);

   362→        navigateTo(parsed.view);
   363→      }
   364→      setShowNotifications(false);
   365→    }
   366→
   367→    // Mark as read in API
   368→    if (!notif.read) {
   369→      try {
   370→        await fetch("/api/notifications", {
   371→          method: "PUT",
   372→          headers: { "Content-Type": "application/json" },
   373→          body: JSON.stringify({ notificationId: notif.id, read: true }),
   374→        });
   375→        setNotifications((prev) =>
   376→          prev.map((n) => (n.id === notif.id ? { ...n, read: true } : n))
   377→        );
   378→      } catch {
   379→        // Silently fail
   380→      }
   381→    }
   382→  }, [navigateTo, openCourseDetail]);
   383→
   384→  /** Mark all notifications as read via API */
   385→  const markAllAsRead = useCallback(async () => {
   386→    try {
   387→      await fetch("/api/notifications", {
   388→        method: "PUT",
   389→        headers: { "Content-Type": "application/json" },

   391→      });

   393→    } catch {
   394→      // Silently fail
   395→    }
