    62→function timeAgo(dateStr: string): string {
    63→  const now = new Date();
    64→  const date = new Date(dateStr);
    65→  const diffMs = now.getTime() - date.getTime();
    66→  const diffSec = Math.floor(diffMs / 1000);
    67→  const diffMin = Math.floor(diffSec / 60);
    68→  const diffHr = Math.floor(diffMin / 60);
    69→  const diffDay = Math.floor(diffHr / 24);
    70→  const diffWeek = Math.floor(diffDay / 7);
    71→  const diffMonth = Math.floor(diffDay / 30);
    72→
    73→  if (diffSec < 60) return "just now";
    74→  if (diffMin < 60) return `${diffMin}m ago`;
    75→  if (diffHr < 24) return `${diffHr}h ago`;
    76→  if (diffDay < 7) return `${diffDay}d ago`;
    77→  if (diffWeek < 5) return `${diffWeek}w ago`;
    78→  return `${diffMonth}mo ago`;
    79→}

    81→/* ------------------------------------------------------------------ */
    82→/*  Seeded random for consistent sparkline data                        */
    83→/* ------------------------------------------------------------------ */
    84→
    85→function seededRandom(seed: number) {
    86→  let s = seed;
    87→  return () => {
    88→    s = (s * 16807 + 0) % 2147483647;
    89→    return (s - 1) / 2147483646;
    90→  };
    91→}

    93→/* ------------------------------------------------------------------ */
    94→/*  Sparkline Mini Chart                                               */
    95→/* ------------------------------------------------------------------ */
    96→
    97→function SparklineChart({
    98→  data,
    99→  gradientFrom,
   100→  gradientTo,
   101→}: {
   102→  data: number[];
   103→  gradientFrom: string;
   104→  gradientTo: string;
   105→}) {
   106→  const max = Math.max(...data, 1);
   107→  return (
   108→    <div className="flex items-end gap-[3px] h-8">
   109→      {data.map((val, i) => {
   110→        const height = Math.max(4, (val / max) * 100);
   111→        return (
   112→          <div
   113→            key={i}
   114→            className={cn(
   115→              "w-[6px] rounded-full transition-all duration-500",
   116→              `bg-gradient-to-t ${gradientFrom} ${gradientTo}`
   117→            )}
   118→            style={{
   119→              height: `${height}%`,
   120→              opacity: 0.5 + (i / data.length) * 0.5,
   121→              animationDelay: `${i * 60}ms`,
   122→            }}
   123→          />
   124→        );
   125→      })}
   126→    </div>
   127→  );
   128→}

   134→interface StatCardProps {

   144→
   145→function StatCard({
   146→  icon,
   147→  label,
   148→  value,
   149→  gradient,
   150→  gradientFrom,
   151→  gradientTo,
   152→  delay,
   153→  seed,
   154→}: StatCardProps) {
   155→  const sparkData = useMemo(() => {
   156→    const rand = seededRandom(seed);
   157→    return Array.from({ length: 7 }, () => 20 + rand() * 80);
   158→  }, [seed]);

   160→  return (
   161→    <Card
   162→      className="glass-card stat-pop hover-lift overflow-hidden border-border/50 transition-shadow hover:shadow-lg"
   163→      style={{ animationDelay: `${delay}ms` }}
   164→    >
   165→      <CardContent className="p-4 sm:p-5">
   166→        <div className="flex items-center gap-3.5">
   167→          <div
   168→            className={cn(
   169→              "flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-gradient-to-br shadow-md transition-transform duration-200 hover:scale-110",
   170→              gradient
   171→            )}
   172→          >
   173→            {icon}
   174→          </div>
   175→          <div className="min-w-0 flex-1">
   176→            <p className="text-[11px] font-medium text-muted-foreground uppercase tracking-wider">
   177→              {label}
   178→            </p>
   179→            <p className="text-2xl font-bold tracking-tight text-foreground leading-tight">
   180→              {value}
   181→            </p>
   182→          </div>
   183→          <div className="shrink-0 opacity-60">
   184→            <SparklineChart
   185→              data={sparkData}
   186→              gradientFrom={gradientFrom}
   187→              gradientTo={gradientTo}

   674→            </Card>
   675→          </div>
   676→        </div>
   677→      ) : null}
   678→    </div>
   679→  );
