   262→  const [days, setDays] = useState<Array<{ date: string; day: string; isToday: boolean }>>([]);
   263→  const [maxMinutes, setMaxMinutes] = useState(1);
   264→
   265→  useEffect(() => {
   266→    async function fetchActivity() {
   267→      try {
   268→        const res = await fetch('/api/activity?userId=user_demo_001&weeks=5');
   269→        const json = await res.json();
   270→        if (json.success) {
   271→          // Build 30-day grid from the response
   272→          const allDays: ActivityDayEntry[] = [];
   273→          const today = new Date();
   274→          today.setHours(0, 0, 0, 0);
   275→          const start = new Date(today);
   276→          start.setDate(start.getDate() - 29);
   277→
   278→          const resMap = new Map<string, number>();
   279→          for (const d of (json.data as Activity30Data).dailyData) {
   280→            resMap.set(d.date, d.minutes);
   281→          }
   282→
   283→          for (let i = 0; i < 30; i++) {
   284→            const d = new Date(start);
   285→            d.setDate(d.getDate() + i);
   286→            const dateStr = d.toISOString().split('T')[0];
   287→            allDays.push({
   288→              date: dateStr,
   289→              day: d.toLocaleDateString('en-US', { weekday: 'narrow' }),
   290→              isToday: dateStr === today.toISOString().split('T')[0],
   291→            });
   292→          }
   293→
   294→          setDays(allDays);
   295→          setActivityMap(resMap);
   296→          const max = Math.max(...Array.from(resMap.values()), 1);
   297→          setMaxMinutes(max);

   299→      } catch { /* silent */ }
