   107→  const colorMap: Record<string, string> = {

   131→/** Format relative time for notifications */
   132→function formatRelativeTime(dateStr: string): string {
   133→  const now = new Date();
   134→  const date = new Date(dateStr);
   135→  const diffMs = now.getTime() - date.getTime();
   136→  const diffMins = Math.floor(diffMs / 60000);
   137→  const diffHours = Math.floor(diffMs / 3600000);
   138→  const diffDays = Math.floor(diffMs / 86400000);
   139→
   140→  if (diffMins < 1) return "Just now";
   141→  if (diffMins < 60) return `${diffMins}m ago`;
   142→  if (diffHours < 24) return `${diffHours}h ago`;
   143→  if (diffDays < 7) return `${diffDays}d ago`;
   144→  return date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   145→}

   147→/** Group notifications by date category: Today, Yesterday, Earlier */
   148→function groupNotificationsByDate(items: NotificationItem[]) {
   149→  const now = new Date();
   150→  const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   151→  const yesterdayStart = todayStart - 86400000;
   152→
   153→  const today: NotificationItem[] = [];
   154→  const yesterday: NotificationItem[] = [];
   155→  const earlier: NotificationItem[] = [];
   156→
   157→  for (const item of items) {
   158→    const t = new Date(item.createdAt).getTime();
   159→    if (t >= todayStart) today.push(item);
   160→    else if (t >= yesterdayStart) yesterday.push(item);
   161→    else earlier.push(item);
   162→  }
   163→
   164→  return { today, yesterday, earlier };
   165→}

   167→/** Notification group with date header */
   168→function NotificationGroup({
   169→  label,
   170→  items,
   171→  onNotificationClick,
   172→}: {
   173→  label: string;
   174→  items: NotificationItem[];
   175→  onNotificationClick: (notif: NotificationItem) => void;
   176→}) {
   177→  if (items.length === 0) return null;
   178→
   179→  return (
   180→    <div>
   181→      <p className="px-3 pt-2 pb-1 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground/60">
   182→        {label}
   183→      </p>
   184→      {items.map((notif) => (
   185→        <button
   186→          key={notif.id}
   187→          type="button"
   188→          onClick={() => onNotificationClick(notif)}
   189→          className={cn(
   190→            "group flex w-full items-start gap-3 rounded-lg px-3 py-3 text-left transition-colors notif-appear",
   191→            notif.read
   192→              ? "hover:bg-muted/50"
   193→              : "bg-primary/5 hover:bg-primary/8"
   194→          )}
   195→        >
   196→          {/* Left border accent for unread */}
   197→          {!notif.read && (
   198→            <div className={cn(
   199→              "absolute left-1 top-3 bottom-3 w-0.5 rounded-full",
   200→              NotificationBorderAccent({ type: notif.type })
   201→            )} style={{ position: "absolute" }} />
   202→          )}
   203→          <div className={cn(
   204→            "mt-0.5 flex h-8 w-8 shrink-0 items-center justify-center rounded-lg transition-colors relative",
   205→            NotificationIconBg({ type: notif.type, read: notif.read })
   206→          )}>
   207→            <NotificationIcon type={notif.type} />
   208→          </div>
   209→          <div className="min-w-0 flex-1">
   210→            <div className="flex items-start justify-between gap-2">
   211→              <p className={cn(
   212→                "text-sm leading-snug",
   213→                notif.read
   214→                  ? "text-muted-foreground font-medium"
   215→                  : "text-foreground font-semibold"
   216→              )}>
   217→                {notif.title}
   218→              </p>
   219→              {!notif.read && (
   220→                <div className="mt-1.5 h-2 w-2 shrink-0 rounded-full bg-primary" />
   221→              )}
   222→            </div>
   223→            <p className="mt-0.5 text-xs text-muted-foreground leading-relaxed line-clamp-2">
   224→              {notif.message}
   225→            </p>
   226→            <p className="mt-1 text-[10px] text-muted-foreground/60">
   227→              {formatRelativeTime(notif.createdAt)}
   228→            </p>
   229→          </div>
   230→        </button>
   231→      ))}
   232→    </div>
   233→  );
   234→}

   254→/** Parse link string into view + id for navigation */
   255→function parseNotificationLink(link: string | null): { view: ViewName; courseId?: string } | null {
   256→  if (!link) return null;
   257→  if (link === "profile") return { view: "profile" };
   258→  if (link.startsWith("course-detail:")) {
   259→    const courseId = link.replace("course-detail:", "");
   260→    return { view: "course-detail", courseId };
   261→  }
   262→  return null;
   263→}

   348→

   352→      setBellAnimating(true);

   376→          headers: { "Content-Type": "application/json" },

   380→          prev.map((n) => (n.id === notif.id ? { ...n, read: true } : n))

   383→        // Silently fail
   384→      }
